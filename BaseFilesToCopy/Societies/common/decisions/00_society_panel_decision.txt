# The way into the society panel.
#
# All this does is raise a flag. The window watches for it and fills itself when it appears -- see
# society_panel_window and society_panel_gather in common/scripted_guis/00_society_panel_guis.txt.
#
# =====================================================================================
#  Why a decision, when there is now a HUD tab
# =====================================================================================
#
# The tab exists -- Emit/GuiWriter.cs PatchHudTabs splices it into vanilla's gui/hud.gui under
# Intrigue -- and this decision stays anyway, for two reasons.
#
# It is the FALLBACK. The tab is the one piece of the society feature that cannot live in this
# folder, because it edits a vanilla file and only the generator can do that. So a
# `--static-only --societies` run ships the panel, this decision and everything else, and no tab:
# that mode never reaches GuiWriter. Without the decision that run would produce a panel with no
# door. gen_artifact_index_decision has the same arrangement for the same reason.
#
# And both routes set the SAME `society_panel_open` variable, so they cannot disagree about
# whether the panel is up -- the tab lights from it, the window's X clears it, and Escape clears
# it through the X. Two doors, one latch.
#
# Worth keeping recorded: the vanilla tabs call `ToggleGameViewData('intrigue_window', ...)` and
# those 43 view names are registered in the engine, with no data file defining them and a C++
# data context behind each. A mod cannot add one, and a tab naming an unknown view resolves to
# nothing silently. The society tab therefore drives the society_panel_toggle scripted_gui
# exactly as this decision drives society_panel_window.
#
# =====================================================================================
#  Only members
# =====================================================================================
#
# A non-member has no society to look at, and a decision that opens an empty window is worse than
# no decision because the player has to open it to find that out. This is also the fourth place
# membership gates something, after the rite's three -- which is the point of the prototype.

society_panel_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}

	desc = society_panel_decision_desc
	selection_tooltip = society_panel_decision_tooltip
	confirm_text = society_panel_decision_confirm

	# Sorted low so it sits with the other informational decisions rather than above the ones
	# that change the game state.
	sort_order = 10

	is_shown = {
		has_trait = society_member
	}

	is_valid = {
		always = yes
	}

	effect = {
		set_variable = {
			name = society_panel_open
			value = yes
		}
	}

	# Never. It opens a window, and an AI ruler taking it would spend its yearly decision slot
	# looking at something nobody is looking at.
	ai_will_do = {
		base = 0
	}

	ai_check_interval = 0
}
