# The Restorationists' decisions: the two that move the aim, and the ones members use every day.
#
#   restor_raise_the_standard_decision    the pretender, once per phase, at 50 support: claims, and
#                                         the sworn are called to arms (restor.0610)
#   restor_restore_the_crown_decision     the pretender, holding a quarter of the crown's land with 40
#                                         support: the society crowns them. -> on_title_gain -> triumph
#   restor_give_to_the_cause_decision     any member: 100 gold to the pretender, for favour
#   restor_reward_the_faithful_decision   the restored king: honour the order, once a decade
#
# Vanilla's own creation (51% of the land, 500 gold) still works and also triggers the triumph --
# the society's decision is the cheaper door its backing opens, not the only one.

restor_raise_the_standard_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_knight_kneeling.dds"
	}
	decision_group_type = major
	ai_check_interval = 12

	desc = restor_raise_the_standard_decision_desc
	selection_tooltip = restor_raise_the_standard_decision_tooltip

	is_shown = {
		restor_underground_trigger = yes
		restor_is_pretender_trigger = yes
		is_landed = yes
		NOT = { exists = global_var:restor_standard_raised }
	}

	is_valid_showing_failures_only = {
		is_adult = yes
		is_imprisoned = no
		is_at_war = no
	}

	is_valid = {
		custom_tooltip = {
			text = restor_raise_the_standard_support_tt
			global_var:restor_support >= restor_rising_support_value
		}
	}

	cost = {
		prestige = 200
	}

	effect = {
		set_global_variable = { name = restor_standard_raised value = yes }
		add_character_modifier = {
			modifier = restor_raised_the_standard_modifier
			years = 6
		}

		# Held by an usurper: a claim on the crown itself, and vanilla's claim war does the rest.
		# Broken into pieces: claims on every duchy of it outside the pretender's realm, one war each.
		if = {
			limit = { exists = global_var:restor_crown.holder }
			custom_tooltip = restor_raise_the_standard_crown_claim_tt
			hidden_effect = { add_pressed_claim = global_var:restor_crown }
		}
		else = {
			custom_tooltip = restor_raise_the_standard_duchy_claims_tt
			hidden_effect = {
				save_scope_as = restor_standard_bearer
				global_var:restor_crown = {
					every_in_de_jure_hierarchy = {
						limit = {
							tier = tier_duchy
							exists = holder
							holder = {
								NOT = { this = scope:restor_standard_bearer }
								NOT = { any_liege_or_above = { this = scope:restor_standard_bearer } }
							}
						}
						save_scope_as = restor_claimed_duchy
						scope:restor_standard_bearer = { add_pressed_claim = scope:restor_claimed_duchy }
					}
				}
			}
		}

		custom_tooltip = restor_raise_the_standard_call_tt
		save_scope_as = restor_standard_bearer
		hidden_effect = {
			restor_notify_sworn_effect = { EVENT = restor.0610 DAYS = 3 }
		}
	}

	ai_potential = {
		restor_is_pretender_trigger = yes
	}

	ai_will_do = {
		base = 100
	}
}

restor_restore_the_crown_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = major
	ai_check_interval = 6

	desc = restor_restore_the_crown_decision_desc
	selection_tooltip = restor_restore_the_crown_decision_tooltip

	is_shown = {
		restor_underground_trigger = yes
		restor_is_pretender_trigger = yes
		is_landed = yes
		NOT = { exists = global_var:restor_crown.holder }
	}

	is_valid_showing_failures_only = {
		is_adult = yes
		is_imprisoned = no
		is_at_war = no
		custom_tooltip = {
			text = restor_restore_the_crown_liege_tt
			OR = {
				is_independent_ruler = yes
				liege.primary_title = { tier > tier_kingdom }
				liege = { has_trait = restor_member }
			}
		}
	}

	is_valid = {
		custom_tooltip = {
			text = restor_restore_the_crown_land_tt
			restor_pretender_share_value >= restor_crowning_share_value
		}
		custom_tooltip = {
			text = restor_restore_the_crown_support_tt
			global_var:restor_support >= restor_rising_support_value
		}
	}

	cost = {
		gold = 150
		prestige = 250
	}

	effect = {
		custom_tooltip = restor_restore_the_crown_effect_tt
		hidden_effect = {
			create_title_and_vassal_change = {
				type = created
				save_scope_as = change
				add_claim_on_loss = no
			}
			global_var:restor_crown = {
				change_title_holder = {
					holder = root
					change = scope:change
				}
			}
			# A heir whose first county came from a sworn lord answers to that lord. A crown does
			# not answer to a duke: the same change makes them independent (vanilla's event-war
			# pattern, 00_event_war.txt).
			if = {
				limit = {
					exists = liege
					liege.primary_title = { tier <= tier_kingdom }
				}
				becomes_independent = { change = scope:change }
			}
			resolve_title_and_vassal_change = scope:change
		}
	}

	ai_potential = {
		restor_is_pretender_trigger = yes
	}

	ai_will_do = {
		base = 100
	}
}

restor_give_to_the_cause_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	ai_check_interval = 24

	desc = restor_give_to_the_cause_decision_desc
	selection_tooltip = restor_give_to_the_cause_decision_tooltip

	is_shown = {
		restor_underground_trigger = yes
		has_trait = restor_member
		NOT = { restor_is_pretender_trigger = yes }
		exists = global_var:restor_pretender
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		NOT = { has_character_flag = restor_gave_recently }
	}

	cost = {
		gold = 100
	}

	effect = {
		global_var:restor_pretender = { add_gold = 100 }
		change_global_variable = { name = restor_war_chest add = 100 }
		restor_favour_gain_effect = { VALUE = 30 }
		add_trait_xp = {
			trait = restor_member
			track = restor_rank
			value = 3
		}
		hidden_effect = {
			add_character_flag = { flag = restor_gave_recently years = 1 }
			# Read by the war_chest charge. A counter, not a flag, so two gifts in a charge's
			# three years count as two.
			if = {
				limit = { NOT = { has_variable = restor_given_to_cause } }
				set_variable = { name = restor_given_to_cause value = 0 }
			}
			change_variable = { name = restor_given_to_cause add = 100 }
			# A war-chest charge is paid the moment it is done, not at the next season's check.
			restor_charge_check_effect = yes
		}
	}

	ai_potential = {
		has_trait = restor_member
		gold >= 250
	}

	ai_will_do = {
		base = 10
		modifier = { add = 20 gold >= 500 }
		modifier = { add = 15 has_trait = generous }
		modifier = { add = -10 has_trait = greedy }
		modifier = {
			add = 40
			has_variable = restor_charge
			var:restor_charge = flag:war_chest
		}
	}
}

restor_reward_the_faithful_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	ai_check_interval = 60

	desc = restor_reward_the_faithful_decision_desc
	selection_tooltip = restor_reward_the_faithful_decision_tooltip

	is_shown = {
		restor_restored_trigger = yes
		has_title = global_var:restor_crown
	}

	is_valid_showing_failures_only = {
		NOT = { has_character_flag = restor_rewarded_recently }
	}

	cost = {
		gold = 200
	}

	effect = {
		add_character_flag = { flag = restor_rewarded_recently years = 10 }
		add_prestige = 150
		custom_tooltip = restor_reward_the_faithful_effect_tt
		hidden_effect = {
			every_in_global_list = {
				variable = restor_roster
				limit = {
					is_alive = yes
					NOT = { this = root }
				}
				add_opinion = {
					target = root
					modifier = restor_raised_before_the_sworn_opinion
				}
				restor_favour_gain_effect = { VALUE = 25 }
				add_trait_xp = {
					trait = restor_member
					track = restor_rank
					value = 5
				}
			}
		}
	}

	ai_will_do = {
		base = 50
	}
}
