# The questions every inversion-cult file asks. See README.txt, THE INVERSION CULT.
# Safe on a world with no cult: an unset global makes `exists` false before anything reads it.

cult_active_trigger = {
	exists = global_var:cult_host_faith
}

# Phase 1: hidden inside the host faith, rotting it from within.
cult_hidden_trigger = {
	cult_active_trigger = yes
	global_var:cult_phase = 1
}

# Phase 2: the church is hollow -- its head (or its holy places) are the cult's -- but nobody knows.
cult_hollow_trigger = {
	cult_active_trigger = yes
	global_var:cult_phase = 2
}

# Phase 3: unveiled. The host faith's doctrines are turned over and the cult is its open face.
cult_unveiled_trigger = {
	cult_active_trigger = yes
	global_var:cult_phase = 3
}

# Hidden or hollow: the cult is still a secret, and exposure still matters.
cult_secret_phase_trigger = {
	cult_active_trigger = yes
	global_var:cult_phase < 3
}

# ---- Who someone is to the cult -------------------------------------------------------------------

cult_is_member_trigger = {
	has_trait = cult_member
}

cult_is_hierophant_trigger = {
	exists = global_var:cult_hierophant
	this = global_var:cult_hierophant
}

cult_is_hunter_trigger = {
	exists = global_var:cult_hunter
	this = global_var:cult_hunter
}

# Of the faith the cult lives inside.
cult_of_the_host_trigger = {
	cult_active_trigger = yes
	faith = global_var:cult_host_faith
}

cult_rank_at_least_trigger = {
	has_trait_xp = {
		trait = cult_member
		track = cult_rank
		value >= $RANK$
	}
}

# May take up the Hierophant's place: alive and not dying (on_death still sees the dying as alive;
# vanilla flags them), and not the one leaving.
cult_can_succeed_trigger = {
	is_alive = yes
	NOT = { has_character_flag = about_to_die_but_not_dead_yet_flag }
	trigger_if = {
		limit = { exists = scope:cult_old_hierophant }
		NOT = { this = scope:cult_old_hierophant }
	}
}

# A fair approach: adult, free, of the host faith, not sworn to either society, not the hunter, not
# refused recently. (One society per character, as in CK2.)
cult_recruitable_trigger = {
	is_adult = yes
	is_imprisoned = no
	cult_of_the_host_trigger = yes
	NOT = { has_trait = cult_member }
	NOT = { has_trait = restor_member }
	NOT = { cult_is_hunter_trigger = yes }
	NOT = { has_character_flag = cult_refused_recently }
}

# The church is the cult's. With a head of faith: the head is sworn. Without one: sworn realms hold
# at least half the faith's holy sites. Evaluated yearly and when the head changes.
cult_church_captured_trigger = {
	cult_active_trigger = yes
	trigger_if = {
		limit = { exists = global_var:cult_host_faith.religious_head }
		global_var:cult_host_faith.religious_head = { has_trait = cult_member }
	}
	trigger_else = {
		cult_holy_site_share_value >= 50
	}
}

# The church slips away: the head is not sworn, or (no head) the sworn share of holy sites falls
# below 40. Ten points below the capture line, so one county changing hands on a faith with four holy
# sites does not flip the phase -- and notify every member -- back and forth.
cult_church_lost_trigger = {
	cult_active_trigger = yes
	trigger_if = {
		limit = { exists = global_var:cult_host_faith.religious_head }
		global_var:cult_host_faith.religious_head = { NOT = { has_trait = cult_member } }
	}
	trigger_else = {
		cult_holy_site_share_value < 40
	}
}

# A holy-site county in sworn hands: its holder sworn, or answering to somebody sworn.
cult_sworn_county_trigger = {
	exists = holder
	holder = {
		OR = {
			has_trait = cult_member
			any_liege_or_above = { has_trait = cult_member }
		}
	}
}

# Could hunt the cult: alive, of the host faith, not sworn, landed. The head of faith first;
# failing that the strongest zealous ruler of the faith.
cult_valid_hunter_trigger = {
	is_alive = yes
	NOT = { has_character_flag = about_to_die_but_not_dead_yet_flag }
	cult_of_the_host_trigger = yes
	NOT = { has_trait = cult_member }
	is_landed = yes
}

# The panel's roster: would scope:cult_viewer know THIS member's name? A Magister or above -- and
# the Hierophant -- knows the whole circle; below that you know the people around you and the
# Hierophant. Once unveiled, everyone knows everyone. Shared by cult_panel_gather's rebuild and its
# has-anything-changed check, so the two never disagree.
cult_panel_knows_trigger = {
	is_alive = yes
	has_trait = cult_member
	is_target_in_global_variable_list = { name = cult_roster target = this }
	OR = {
		cult_unveiled_trigger = yes
		scope:cult_viewer = { cult_rank_at_least_trigger = { RANK = 50 } }
		scope:cult_viewer = { cult_is_hierophant_trigger = yes }
		this = scope:cult_viewer
		cult_is_hierophant_trigger = yes
		is_courtier_of = scope:cult_viewer
		scope:cult_viewer = { is_courtier_of = prev }
		is_vassal_of = scope:cult_viewer
		scope:cult_viewer = { is_vassal_of = prev }
		is_close_or_extended_family_of = scope:cult_viewer
	}
}
