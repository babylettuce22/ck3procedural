# Helpers for the Black Mass and the Pilgrimage of Defilement (cult.0900-0999). See the two
# activity files for the design. Built on the Restorationists' 00_restor_gathering_effects.txt.

# =====================================================================================
#  The Profanation: which counties the chapel can reach
# =====================================================================================
#
# Root = the host. Builds three temporary lists:
#
#   cult_mass_hall   everyone at the Mass (the host included, and alone outside an activity)
#   cult_mass_own    untainted counties of the host faith held directly by someone at the Mass
#   cult_mass_near   untainted counties of the host faith next to land held by someone at the
#                    Mass, and held by nobody at it
#
# Temporary lists do not outlive the effect block that built them, so cult.0902 calls this in
# `immediate` to count and again inside the chosen option to act.
cult_mass_profane_lists_effect = {
	if = {
		limit = { exists = scope:activity }
		scope:activity = {
			every_attending_character = {
				add_to_list = cult_mass_hall
			}
		}
	}
	if = {
		limit = { NOT = { is_in_list = cult_mass_hall } }
		add_to_list = cult_mass_hall
	}

	if = {
		limit = { exists = global_var:cult_host_faith }
		every_in_list = {
			list = cult_mass_hall
			every_held_county = {
				limit = {
					faith = global_var:cult_host_faith
					NOT = { has_county_modifier = cult_county_tainted }
					NOT = { is_in_list = cult_mass_own }
				}
				add_to_list = cult_mass_own
			}
		}

		every_in_list = {
			list = cult_mass_hall
			every_held_county = {
				every_neighboring_county = {
					limit = {
						faith = global_var:cult_host_faith
						NOT = { has_county_modifier = cult_county_tainted }
						NOT = { is_in_list = cult_mass_own }
						NOT = { is_in_list = cult_mass_near }
						exists = holder
						holder = { NOT = { is_in_list = cult_mass_hall } }
					}
					add_to_list = cult_mass_near
				}
			}
		}
	}
}

# Saves the counts cult.0901/0902 read:
#   scope:cult_profane_n            how many counties the Mass can taint: one per landed member
#                                   present, at least one, at most five
#   scope:cult_profane_own_count    size of cult_mass_own
#   scope:cult_profane_near_count   size of cult_mass_near
# Call after cult_mass_profane_lists_effect, in the same block.
cult_mass_profane_counts_effect = {
	save_scope_value_as = {
		name = cult_profane_n
		value = {
			value = 0
			every_in_list = {
				list = cult_mass_hall
				limit = { is_landed = yes }
				add = 1
			}
			min = 1
			max = 5
		}
	}
	save_scope_value_as = {
		name = cult_profane_own_count
		value = list_size:cult_mass_own
	}
	save_scope_value_as = {
		name = cult_profane_near_count
		value = list_size:cult_mass_near
	}
}

# Taints up to scope:cult_profane_n counties of the list $LIST$ for ten years, richest first, and
# saves how many as scope:cult_tainted_now. The rot is recomputed at once.
cult_mass_taint_list_effect = {
	save_scope_value_as = {
		name = cult_tainted_now
		value = {
			value = list_size:$LIST$
			max = scope:cult_profane_n
		}
	}
	ordered_in_list = {
		list = $LIST$
		order_by = development_level
		max = scope:cult_profane_n
		check_range_bounds = no
		add_county_modifier = {
			modifier = cult_county_tainted
			years = 10
		}
	}
	cult_rot_refresh_effect = yes
}

# =====================================================================================
#  The Initiation: who can be raised, and who the host could bring
# =====================================================================================
#
# Root = the host. Saves, when there is one:
#   scope:cult_mass_candidate   a member at the Mass, not the host, below Hierarch
#   scope:cult_mass_recruit     a courtier of the host who could be sworn: recruitable, not zealous,
#                               and a sinner by the host faith's own lights if any courtier is
cult_mass_find_initiates_effect = {
	if = {
		limit = { exists = scope:activity }
		scope:activity = {
			random_attending_character = {
				limit = {
					NOT = { this = root }
					has_trait = cult_member
					NOT = { cult_rank_at_least_trigger = { RANK = 75 } }
				}
				save_scope_as = cult_mass_candidate
			}
		}
	}
	random_courtier = {
		limit = {
			cult_recruitable_trigger = yes
			NOT = { has_trait = zealous }
			cult_sinner_trigger = yes
		}
		save_scope_as = cult_mass_recruit
	}
	if = {
		limit = { NOT = { exists = scope:cult_mass_recruit } }
		random_courtier = {
			limit = {
				cult_recruitable_trigger = yes
				NOT = { has_trait = zealous }
			}
			save_scope_as = cult_mass_recruit
		}
	}
}

# The Invocation: dark power and rank XP to every member present, the host included, and each of
# them pays for it in the soul. The host alone when fired outside an activity.
cult_mass_invoke_effect = {
	if = {
		limit = { exists = scope:activity }
		scope:activity = {
			every_attending_character = {
				limit = { has_trait = cult_member }
				cult_power_gain_effect = { VALUE = $POWER$ }
				cult_rank_gain_effect = { VALUE = $XP$ }
				cult_corrupt_soul_effect = yes
			}
		}
	}
	else = {
		cult_power_gain_effect = { VALUE = $POWER$ }
		cult_rank_gain_effect = { VALUE = $XP$ }
		cult_corrupt_soul_effect = yes
	}
}

# Tells every guest (everyone attending but the host) what the host decided. $PURPOSE$ is a flag
# cult.0906 reads off the activity.
cult_mass_tell_guests_effect = {
	if = {
		limit = { exists = scope:activity }
		scope:activity = {
			set_variable = {
				name = cult_mass_purpose
				value = flag:$PURPOSE$
			}
			every_attending_character = {
				limit = { NOT = { this = scope:host } }
				trigger_event = {
					id = cult.0906
					days = 1
				}
			}
		}
	}
}

# The county the Pilgrimage went to, as scope:cult_defilement_county. Unset outside an activity.
cult_defilement_save_county_effect = {
	if = {
		limit = {
			exists = scope:activity
			exists = scope:activity.activity_location
		}
		scope:activity.activity_location.county = { save_scope_as = cult_defilement_county }
	}
}

# What the road cost, on top of the purpose's own exposure. Root = the host, in cult.0914, with the
# snapshot values scope:cult_hasty / scope:cult_clean and scope:cult_witness (if any). The witness
# roll ends in a toast either way, so the player always learns whether they talked.
cult_defilement_consequences_effect = {
	if = {
		limit = { exists = scope:cult_hasty }
		cult_exposure_gain_effect = { VALUE = 2 }
	}
	if = {
		limit = { NOT = { exists = scope:cult_clean } }
		cult_exposure_gain_effect = { VALUE = 3 }
		if = {
			limit = { exists = scope:cult_witness }
			random_list = {
				50 = {
					send_interface_toast = {
						type = event_toast_effect_bad
						title = cult.0914.witness_talks
						left_icon = root
						right_icon = scope:cult_witness
						cult_exposure_gain_effect = { VALUE = 3 }
						if = {
							limit = { any_secret = { secret_type = secret_cult_member } }
							custom_tooltip = cult.0914.witness_secret_tt
							hidden_effect = {
								random_secret = {
									limit = { secret_type = secret_cult_member }
									reveal_to = scope:cult_witness
								}
							}
						}
					}
				}
				50 = {
					send_interface_toast = {
						type = event_toast_effect_good
						title = cult.0914.witness_quiet
						left_icon = root
						right_icon = scope:cult_witness
						custom_tooltip = cult.0914.witness_quiet_tt
					}
				}
			}
		}
	}
}

# Tells the site's holder what was done to the shrine (cult.0916): a player, not the pilgrim, not
# sworn. $WHAT$ is defile or relic. Root = the host.
cult_defilement_tell_holder_effect = {
	if = {
		limit = {
			exists = scope:cult_defilement_county
			exists = scope:cult_defilement_county.holder
			scope:cult_defilement_county.holder = {
				is_ai = no
				NOT = { this = root }
				NOT = { has_trait = cult_member }
			}
		}
		save_scope_as = cult_defiler
		save_scope_value_as = {
			name = cult_defiled_what
			value = flag:$WHAT$
		}
		scope:cult_defilement_county.holder = {
			trigger_event = {
				id = cult.0916
				days = 5
			}
		}
	}
}
