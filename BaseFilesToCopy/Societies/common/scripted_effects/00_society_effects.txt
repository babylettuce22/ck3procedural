# Swearing somebody in, in one place.
#
# =====================================================================================
#  Why this exists
# =====================================================================================
#
# Membership is granted in five places -- the approach event's accept option, the branch of that
# event that reveals a recruiter who "was a member all along", and three debug events. Every one
# of them used to be a bare `add_trait = society_member`, and every one of them was therefore a
# place that could forget to do the rest of what joining means.
#
# It already had: the panel reads two character variables, and CK3 is strict about this in a way
# that is easy to miss --
#
#   [E][jomini_effect.cpp:1161]: Variable 'society_dark_power' is used but is never set.
#                               Setting it in an unused scripted trigger or effect does not count
#
# -- logged once per read, which for a GUI meter means once per redraw. Roughly 200 in a few
# minutes of having the panel open. A variable read before anything writes it is an error, not a
# zero, and the `|0` format spec in the .gui hides the symptom rather than the cause.
#
# So joining initialises the meters. One effect, called everywhere, and a sixth caller added later
# gets it for free.

society_swear_in_effect = {
	add_trait = society_member

	# `set_variable` and not `change_variable`: this runs at the moment of joining, so there is
	# nothing to add to, and change_variable against an unset variable is the same error again.
	#
	# Guarded so that swearing in somebody who is already sworn -- which society.9002 does on
	# purpose, and the debug events can do by accident -- does not wipe what they have earned.
	if = {
		limit = { NOT = { has_variable = society_dark_power } }

		set_variable = {
			name = society_dark_power
			value = 0
		}
	}

	if = {
		limit = { NOT = { has_variable = society_visibility } }

		set_variable = {
			name = society_visibility
			value = 0
		}
	}
}

# What a refusal costs, regardless of what the recruiter does about it.
#
# Called from BOTH options of society.0003 rather than from that event's immediate block, and the
# reason is visibility in the UI sense rather than the mechanic's: effects in an `immediate` block
# do not render anywhere, so a player would see the option tooltips promise a gold cost and a -10
# exposure while the +15 and the opinion hit arrived silently underneath. Duplicated onto both
# options they show up in both tooltips, and the arithmetic the player is shown is the arithmetic
# that happens.
#
# ROOT is the recruiter; scope:society_answerer is the person who refused.
society_oath_refused_effect = {
	add_opinion = {
		target = scope:society_answerer
		modifier = society_refused_the_oath_opinion
	}

	# Three times what acceptance costs. Somebody who joined has the same reasons as you to say
	# nothing; somebody who refused has none, and is still standing there.
	society_visibility_gain_effect = { VALUE = 15 }
}

# =====================================================================================
#  Exposure, told to the player
# =====================================================================================
#
# Every place the society raises or lowers Visibility goes through one of these two, and the
# reason is that `change_variable` is completely silent. It renders nothing on an option tooltip,
# nothing in an effect list, and nothing on screen -- so the player was being charged the single
# most important resource in the set by a mechanic they could only detect by watching a number on
# a panel they had to open themselves.
#
# Three things happen here, and all three are needed:
#
#   save_scope_value_as   so the tooltip can name the amount. This is why there is ONE tooltip key
#                         instead of one per value -- the loc reads [scope:society_visibility_delta]
#                         rather than having the number baked into the string.
#   custom_tooltip        the BEFORE. Renders on the option, so the player reads the cost while
#                         deciding rather than discovering it afterwards.
#   send_interface_toast  the AFTER. Renders once it has happened, and states the new TOTAL, which
#                         the tooltip cannot -- a delta is only meaningful against a running sum,
#                         and the sum is the thing that eventually gets you caught.
#
# The toast is inside a hidden_effect so it does not appear a second time in the tooltip. Without
# that, an option reads its own consequence twice and the player reasonably concludes it costs
# double.
society_visibility_gain_effect = {
	save_scope_value_as = {
		name = society_visibility_delta
		value = $VALUE$
	}

	change_variable = {
		name = society_visibility
		add = $VALUE$
	}

	custom_tooltip = society_visibility_rises_tt

	hidden_effect = {
		send_interface_toast = {
			type = event_toast_effect_bad
			title = society_visibility_toast
			left_icon = root
		}

		# Every change re-reads the thresholds. Here rather than at the call sites because there
		# are already eight of those and the next one added would be the one that forgot.
		#
		# Hidden: the check announces its own consequences through events 0200-0203, and
		# tooltipping "and you may also cross a line" onto every action that costs visibility
		# would say it far more often than it happens.
		society_visibility_check_effect = yes
	}
}

# The mirror. Kept as a separate effect rather than a negative VALUE so the tooltip can be worded
# as a reduction -- "rises by -10" is the kind of thing that ships.
society_visibility_loss_effect = {
	save_scope_value_as = {
		name = society_visibility_delta
		value = $VALUE$
	}

	change_variable = {
		name = society_visibility
		subtract = $VALUE$
	}

	custom_tooltip = society_visibility_falls_tt

	hidden_effect = {
		send_interface_toast = {
			type = event_toast_effect_good
			title = society_visibility_toast
			left_icon = root
		}

		# The other half of the pair above -- see the note in society_visibility_gain_effect.
		# Falling back below a line has to come off as readily as crossing it went on, or the
		# panel's number and the character sheet's modifier stop agreeing.
		society_visibility_check_effect = yes
	}
}

# =====================================================================================
#  Soul corruption
# =====================================================================================
#
# CK2's third meter, and the one that gave the devil-worshipper society its actual character.
# Dark Power was what you earned and Visibility was what it cost you politically. Corruption was
# what it cost you PERSONALLY: every use of the society rolled a chance to strip a virtue or add
# a vice, so a long-serving member slowly stopped being the person who joined.
#
# Ported from CK2's minor_soul_corruption_effect, keeping the two properties that made it work:
#
#   1. REMOVING A VIRTUE IS TWICE AS LIKELY AS ADDING A VICE. CK2 weighted these 10 against 5.
#      That asymmetry is why corruption reads as erosion rather than as possession -- you notice
#      you have stopped being kind long before you notice you have become cruel.
#
#   2. EACH CHANGE IS ONE-SHOT. CK2 flagged every individual transition so it could not repeat.
#      Without that, a character with no virtues left keeps rolling "lose a virtue", the meter
#      appears to do nothing, and the mechanic silently stops mattering exactly when it should
#      matter most. The `has_character_flag` guards below are that, and they are load-bearing.
#
# CK2's trait names map onto CK3's directly: kind -> compassionate, just -> just,
# cruel -> callous/sadistic, arbitrary -> arbitrary, deceitful -> deceitful. Where CK3 splits a
# CK2 trait into a mild and a severe form the mild one is used, because this is erosion.
#
# `random_list` weights are read as a proportion of the total, so a branch zeroed by its modifier
# hands its share to the others -- a character who has already lost every virtue rolls a vice
# with certainty rather than rolling nothing.
society_corrupt_soul_effect = {
	custom_tooltip = society_corruption_tt

	hidden_effect = {
		random_list = {
			# ---- Lose compassion ----
			10 = {
				trigger = {
					has_trait = compassionate
					NOT = { has_character_flag = society_corruption_lost_compassion }
				}

				add_character_flag = society_corruption_lost_compassion
				remove_trait = compassionate

				trigger_event = { id = errand.0100 days = 3 }
			}

			# ---- Lose justice ----
			10 = {
				trigger = {
					has_trait = just
					NOT = { has_character_flag = society_corruption_lost_just }
				}

				add_character_flag = society_corruption_lost_just
				remove_trait = just

				trigger_event = { id = errand.0100 days = 3 }
			}

			# ---- Lose generosity ----
			10 = {
				trigger = {
					has_trait = generous
					NOT = { has_character_flag = society_corruption_lost_generous }
				}

				add_character_flag = society_corruption_lost_generous
				remove_trait = generous

				trigger_event = { id = errand.0100 days = 3 }
			}

			# ---- Gain callousness ----
			5 = {
				trigger = {
					NOR = {
						has_trait = callous
						has_trait = sadistic
						has_trait = compassionate
					}
					NOT = { has_character_flag = society_corruption_gained_callous }
				}

				add_character_flag = society_corruption_gained_callous
				add_trait = callous

				trigger_event = { id = errand.0100 days = 3 }
			}

			# ---- Gain arbitrariness ----
			5 = {
				trigger = {
					NOR = {
						has_trait = arbitrary
						has_trait = just
					}
					NOT = { has_character_flag = society_corruption_gained_arbitrary }
				}

				add_character_flag = society_corruption_gained_arbitrary
				add_trait = arbitrary

				trigger_event = { id = errand.0100 days = 3 }
			}

			# ---- Gain deceit ----
			5 = {
				trigger = {
					NOR = {
						has_trait = deceitful
						has_trait = honest
					}
					NOT = { has_character_flag = society_corruption_gained_deceitful }
				}

				add_character_flag = society_corruption_gained_deceitful
				add_trait = deceitful

				trigger_event = { id = errand.0100 days = 3 }
			}

			# ---- Nothing, this time ----
			#
			# Deliberately present and deliberately small. A mechanic that fires every single time
			# is a tax, and a tax gets budgeted for; one that USUALLY fires is a risk, and a risk
			# gets thought about. It is also the only branch left once every other one is flagged
			# out, which is what stops a fully-corrupted character throwing a silent no-op.
			10 = {
				trigger = { always = yes }
			}
		}
	}
}

# =====================================================================================
#  What an errand pays
# =====================================================================================
#
# The CK2 triad in one call, so that no future errand outcome can pay one axis and forget the
# others. That was the discipline in CK2's own mission blocks -- currency, visibility and
# corruption always arrived together -- and it is the thing most likely to be lost the third time
# somebody adds an outcome event in a hurry.
#
# DARK_POWER is passed rather than baked because the quarry decides it. CK2's ladder was a flat
# 1:2:3 (100/200/300 currency); ours keeps the ratio and rescales to this set's economy, where
# the rite pays 60 and an abduction costs 100 --
#
#     unmissed  80    highborn  140    devout  200
#
# Visibility is passed for the same reason and runs the other way up the same ladder.
society_errand_payout_effect = {
	add_trait_xp = {
		trait = society_member
		track = society_rank
		value = 10
	}

	change_variable = {
		name = society_dark_power
		add = $DARK_POWER$
	}

	custom_tooltip = society_errand_paid_tt

	society_visibility_gain_effect = { VALUE = $VISIBILITY$ }

	society_corrupt_soul_effect = yes
}

# =====================================================================================
#  What the meter DOES: thresholds
# =====================================================================================
#
# Until this existed, society_visibility was a number that went up and was displayed. CK2's
# answer to the same problem is worth copying almost verbatim, because the interesting thing
# about it is what it does NOT do.
#
# There is no discovery roll. None. The meter is entirely deterministic -- it crosses a line and
# a modifier appears -- and the randomness lives somewhere else entirely, in the Court Chaplain's
# `action_hunt_heretics` job, which searches the realm for anyone already carrying the worse of
# the two modifiers. The number is legible and predictable; the danger is a person doing
# something. That separation is why the CK2 system reads as tense rather than arbitrary, and it
# is a better design than the discovery roll this was first going to be.
#
# The three lines are CK2's, unchanged:
#
#   15   under suspicion   -- a rumour, and a slow prestige bleed
#   25   highly suspect    -- worse, AND the liege is told there is something in their realm
#   40   self-reveal       -- only for independent rulers, who have no liege to hide behind
#
# =====================================================================================
#  Where this deliberately departs from CK2
# =====================================================================================
#
# CK2's modifiers are `duration = -1` and never come off except on exposure or leaving the
# society. Combined with its yearly decay that produces a permanent ratchet: cross 15 once and
# you wear it forever, even after the number falls back to 3.
#
# Ours re-evaluates in both directions, because we display the number. A panel that says
# Visibility 4 beside a character sheet that says Under Suspicion is a panel the player stops
# trusting, and the fix is one `else` rather than an explanation. Falling back below a line is
# also the reward for having lain low, and CK2's version quietly withholds it.
society_visibility_check_effect = {
	if = {
		# 40+, independent, and a ruler: nobody above you to be hidden from, so the fiction has
		# nowhere left to go. The flag makes it once-only -- a self-reveal that fired every year
		# would be a stuck event, not a consequence.
		limit = {
			var:society_visibility >= 40
			is_independent_ruler = yes
			NOT = { has_character_flag = society_self_revealed }
		}

		add_character_flag = society_self_revealed
		trigger_event = { id = society.0202 days = 1 }
	}

	if = {
		limit = { var:society_visibility >= 25 }

		if = {
			limit = { has_character_modifier = society_under_suspicion }
			remove_character_modifier = society_under_suspicion
		}

		if = {
			limit = { NOT = { has_character_modifier = society_highly_suspect } }

			add_character_modifier = society_highly_suspect

			trigger_event = { id = society.0201 days = 1 }

			# The liege is told, which is the moment the mechanic stops being between the player
			# and a number. CK2 notifies at this same threshold.
			#
			# The save has to happen HERE, on the member, before the scope changes to the liege --
			# inside `liege ?= { }` root is still the member but `this` is not, and the event needs
			# a name to put on the right-hand portrait. Saved scopes ride into a triggered event,
			# so 0203 reads it without deriving anything.
			save_scope_as = society_suspect

			liege ?= { trigger_event = { id = society.0203 days = 1 } }
		}
	}
	else_if = {
		limit = { var:society_visibility >= 15 }

		if = {
			limit = { has_character_modifier = society_highly_suspect }
			remove_character_modifier = society_highly_suspect
		}

		if = {
			limit = { NOT = { has_character_modifier = society_under_suspicion } }

			add_character_modifier = society_under_suspicion

			trigger_event = { id = society.0200 days = 1 }
		}

		society_secret_create_effect = yes
	}
	else = {
		# Back under 15. Quiet again, as far as anybody outside is concerned.
		if = {
			limit = { has_character_modifier = society_under_suspicion }
			remove_character_modifier = society_under_suspicion
		}

		if = {
			limit = { has_character_modifier = society_highly_suspect }
			remove_character_modifier = society_highly_suspect
		}

		society_secret_remove_effect = yes
	}
}

# =====================================================================================
#  The yearly decay
# =====================================================================================
#
# CK2's `decrease_visibility_effect`, weights and all. Called once a year from
# common/on_action/00_society_on_actions.txt.
#
# The shape is the point. It is not a flat subtraction -- it is a weighted draw of -1/-2/-3/-4/-5
# in which being good at hiding makes the bigger draws likelier. So intrigue finally has a job in
# this system, "lie low for a few years" becomes a real strategy with a variance to it rather than
# an arithmetic certainty, and two members who did the same things end up in different places.
#
# CK2's modifiers: x1.5 for schemer/deceitful/patient/paranoid, and x1.25 per intrigue breakpoint
# at 12, 14 and 16, stacking. Ours drops `schemer` -- CK3 has no such trait -- and keeps the other
# three, which all exist. The breakpoints are unchanged.
society_visibility_decay_effect = {
	if = {
		limit = { var:society_visibility >= 1 }

		random_list = {
			50 = {
				society_visibility_loss_effect = { VALUE = 1 }
			}

			20 = {
				modifier = {
					add = 10
					OR = {
						has_trait = deceitful
						has_trait = patient
						has_trait = paranoid
					}
				}
				modifier = { add = 5  intrigue >= 12 }
				modifier = { add = 5  intrigue >= 14 }
				modifier = { add = 5  intrigue >= 16 }

				society_visibility_loss_effect = { VALUE = 2 }
			}

			15 = {
				modifier = {
					add = 7
					OR = {
						has_trait = deceitful
						has_trait = patient
						has_trait = paranoid
					}
				}
				modifier = { add = 4  intrigue >= 12 }
				modifier = { add = 4  intrigue >= 14 }
				modifier = { add = 4  intrigue >= 16 }

				society_visibility_loss_effect = { VALUE = 3 }
			}

			10 = {
				modifier = {
					add = 5
					OR = {
						has_trait = deceitful
						has_trait = patient
						has_trait = paranoid
					}
				}
				modifier = { add = 3  intrigue >= 14 }
				modifier = { add = 3  intrigue >= 16 }

				society_visibility_loss_effect = { VALUE = 4 }
			}

			5 = {
				modifier = {
					add = 3
					OR = {
						has_trait = deceitful
						has_trait = patient
						has_trait = paranoid
					}
				}
				modifier = { add = 3  intrigue >= 16 }

				society_visibility_loss_effect = { VALUE = 5 }
			}
		}

		# Never below zero. `change_variable subtract` will happily go negative, and a negative
		# meter would read as a bug on the panel and break the band comparisons underneath it.
		if = {
			limit = { var:society_visibility < 0 }

			set_variable = {
				name = society_visibility
				value = 0
			}
		}
	}
}

# =====================================================================================
#  The secret, created and withdrawn by the meter
# =====================================================================================
#
# Called from society_visibility_check_effect, at the same two places the suspicion modifier is
# added and removed, so the secret and the mark can never disagree about whether a member is on
# the board. See common/secret_types/00_society_secrets.txt for why the meter creates a secret
# rather than being rolled against.
#
# ROOT is the member in both.

society_secret_create_effect = {
	# Guarded, because society_visibility_check_effect runs on every visibility change and this
	# branch is reached repeatedly while a member sits above 15. `add_secret` does not
	# de-duplicate -- a member who took four sacrifices in a decade would otherwise be carrying
	# four identical secrets, each separately discoverable, which turns one exposure into four.
	if = {
		limit = {
			has_trait = society_member
			NOT = { any_secret = { secret_type = secret_society_membership } }
		}

		add_secret = {
			type = secret_society_membership
		}
	}
}

# Decayed back under 15: quiet again, and there is nothing left for the Spymaster to find.
#
# Removing rather than leaving it is the same principle the modifiers follow -- we SHOW the number
# on the panel, so a member reading Exposure 4 must not still be carrying a findable secret. CK2
# could leave its mark on forever because CK2 never showed you the figure.
#
# The known-by list goes with it. That is deliberate and it is the generous reading: somebody who
# lay low long enough that the talk died down gets the talk actually dying down, rather than
# carrying a permanent record of the one year they were careless.
society_secret_remove_effect = {
	every_secret = {
		limit = { secret_type = secret_society_membership }
		remove_secret = yes
	}
}
