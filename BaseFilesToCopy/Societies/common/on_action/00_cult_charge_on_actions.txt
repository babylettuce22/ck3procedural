# The inversion cult's charges on the calendar. See 00_cult_charge_effects.txt.
#
#   cult_charge_pulse       listed in cult_member_yearly (00_cult_on_actions.txt): each member, once
#                           a year at a random time. Offers, checks, the AI's work, and a player's
#                           chance to act.
#   cult_charge_quarterly   players only, four times a year: checks an open charge so a player who
#                           has done the work is paid within the season, not at the next pulse.

cult_charge_pulse = {
	trigger = {
		cult_active_trigger = yes
		has_trait = cult_member
	}
	effect = {
		cult_charge_pulse_effect = yes
	}
}

quarterly_playable_pulse = {
	on_actions = { cult_charge_quarterly }
}

cult_charge_quarterly = {
	trigger = {
		cult_active_trigger = yes
		is_ai = no
		has_variable = cult_charge
	}
	effect = {
		cult_charge_check_effect = yes
	}
}
