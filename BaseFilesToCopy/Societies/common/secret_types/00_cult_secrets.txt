# Being sworn to the cult, as something that can be found out.
#
# Modelled on vanilla's secret_witch: whether it is a crime or a disgrace is the HOST faith's own
# answer, read from its witchcraft doctrine -- which the generator rolled per religion. A faith that
# already tolerates witchcraft finds the cult merely distasteful; one that burns witches burns these.
# Created at Exposure 15 by cult_exposure_check_effect, removed below it, and dissolved for everyone
# at the Unveiling.

secret_cult_member = {
	category = religious

	is_valid = {
		scope:secret_owner = {
			has_trait = cult_member
			cult_secret_phase_trigger = yes
		}
	}

	is_criminal = {
		exists = global_var:cult_host_faith
		global_var:cult_host_faith = {
			OR = {
				has_doctrine = doctrine_witchcraft_crime
				has_doctrine = doctrine_witchcraft_shunned
			}
		}
		scope:secret_owner = {
			OR = {
				faith = global_var:cult_host_faith
				any_liege_or_above = { faith = global_var:cult_host_faith }
			}
		}
	}

	is_shunned = {
		exists = global_var:cult_host_faith
		scope:secret_owner = {
			OR = {
				faith = global_var:cult_host_faith
				any_liege_or_above = { faith = global_var:cult_host_faith }
			}
		}
	}

	on_discover = {
		scope:discoverer = {
			if = {
				limit = {
					exists = global_var:cult_hunter
					OR = {
						this = global_var:cult_hunter
						any_liege_or_above = { this = global_var:cult_hunter }
					}
					NOT = { this = global_var:cult_hunter }
				}
				scope:secret_owner = { save_scope_as = cult_found_member }
				global_var:cult_hunter = {
					trigger_event = { id = cult.0305 days = 3 }
				}
			}
		}
	}

	on_expose = {
		save_scope_as = secret
		scope:secret_owner = {
			save_scope_as = cult_exposed_member
			trigger_event = { id = cult.0301 days = 2 }
		}
		if = {
			limit = { exists = global_var:cult_hunter }
			global_var:cult_hunter = {
				trigger_event = { id = cult.0306 days = 5 }
			}
		}
	}
}
