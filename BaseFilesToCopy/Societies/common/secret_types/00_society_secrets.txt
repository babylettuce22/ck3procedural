# The membership itself, as a thing that can be found out.
#
# =====================================================================================
#  Why this exists, and why it is not a discovery roll
# =====================================================================================
#
# Until now Visibility was a number that went up and never did anything. Three separate files
# carried a comment promising that "when the secret arrives this is what its discovery chance
# reads", which described a design where the meter is rolled against each year. That design was
# wrong, and CK2 -- which is the thing being ported -- does not do it either.
#
# In CK2 the meter is entirely deterministic: it crosses a line and a modifier appears. The danger
# is not a die. The danger is the Court Chaplain running `action_hunt_heretics`, a job, on a timer,
# looking for people already carrying the mark. Legible number, and the threat is a person doing
# something.
#
# CK3 has that job. It is the Spymaster's `task_find_secrets`, and it needs exactly one thing from
# us to work: a secret to find. So the division is
#
#   BEING FINDABLE   ours      Visibility >= 15 creates this secret. Under 15 there is nothing
#                              on the board and the Spymaster has nothing to trip over.
#   BEING FOUND      vanilla   task_find_secrets, its own timer, its own odds, its own councillor
#   WHAT ARRIVES     ours      on_expose, below
#
# The meter therefore stops being a discovery chance and becomes the thing that puts a member on
# the board at all -- which is the same job it did in CK2, and is why it can stay deterministic.
#
# =====================================================================================
#  What vanilla gives us for free, and the two places it does not
# =====================================================================================
#
# Free: discovery by the Spymaster, `expose_secret`, the known-by list, blackmail hooks, and the
# secret showing on the character sheet only to people who know it -- which recovers something the
# modifier could not do. CK2's `under_suspicion` carried `is_visible = { same_society_as = FROM }`
# so the mark was hidden from everyone but fellow members; CK3 has no per-viewer modifier
# visibility, but a SECRET has a known-by list natively. The fiction comes back here.
#
# Two places it does not, both verified rather than assumed:
#
#   1. `task_find_secrets_reveal_selection` is a `first_valid` list of per-type reveal events
#      ending in `spymaster_task.0359 #Fallback`, which always passes. On_actions APPEND, so an
#      entry of ours would sit after the fallback and never fire. A discovered membership shows
#      vanilla's generic "found a secret" text and there is no appending our way out of it. That
#      is why the drama below hangs off `on_expose` -- which is ours outright -- rather than off
#      the discovery.
#
#   2. `secret_type_is_blackmailable_simple_trigger` dispatches on a hardcoded type flag and ends
#      `trigger_else = { always = no }`, so some AI blackmail-seeking paths will not consider this
#      secret. Discovery is NOT affected: that path uses `is_blackmailable_secret_trigger`, which
#      is generic and keys off is_shunned/is_criminal, which is why those two matter below.
#
# Modelled on `secret_witch` (common/secret_types/00_secret_types.txt:548), the closest vanilla
# analogue and the only one whose shape is "a thing you are, which some faiths hang you for".

secret_society_membership = {
	category = religious

	# Never held by somebody who is not in the society. The trait is the fact; this is only the
	# fact being concealable.
	is_valid = {
		scope:secret_owner = {
			has_trait = society_member
		}
	}

	# ---- The two that decide whether anyone cares ----
	#
	# These are what make the secret findable at all -- `is_blackmailable_secret_trigger` reads
	# `is_shunned_or_criminal_for`, and a secret that is neither is invisible to the Spymaster no
	# matter how high the meter climbs.
	#
	# Keyed on witchcraft doctrine rather than on a flat "everyone hates this", because the host
	# faith should get a vote. A faith that shrugs at witchcraft shrugs at this too, and on such a
	# map the society is a curiosity rather than a death sentence -- which is the correct outcome
	# and the one the generated version will produce on its own, since doctrine_witchcraft is
	# already rolled per religion.
	is_shunned = {
		scope:secret_owner = {
			OR = {
				faith = { has_doctrine_parameter = witchcraft_shunned }
				any_liege_or_above = { faith = { has_doctrine_parameter = witchcraft_shunned } }
			}
		}
	}

	is_criminal = {
		scope:secret_owner = {
			OR = {
				faith = { has_doctrine_parameter = witchcraft_illegal }
				any_liege_or_above = { faith = { has_doctrine_parameter = witchcraft_illegal } }
			}
		}
	}

	# ---- What arrives ----
	#
	# The whole reason the drama lives here and not on the discovery: this block is ours, has no
	# first_valid competing with it, and fires however the secret came out -- Spymaster, an
	# expose_secret from a rival, or a member turning on their own.
	on_expose = {
		save_scope_as = secret

		# Vanilla's own notification first, so the player gets the engine's account of what
		# happened before ours adds to it.
		secret_exposed_notification_effect = yes

		scope:secret_owner = {
			save_scope_as = society_exposed

			trigger_event = { id = society.0300 days = 2 }
		}
	}
}
