# Being sworn, as something the usurper's people can find.
#
# Modelled on vanilla's secret_coup_plotter rather than secret_witch: this is a political secret with
# a victim, and the victim is whoever holds what the society means to take. It is CRIMINAL wherever
# the usurper is the owner's liege or above -- treason is judged by the one betrayed -- and merely
# SHUNNED where the owner answers to the usurper's allies... which CK3 cannot express, so outside the
# usurper's realm it is neither, and exposure there costs reputation through the events alone.
#
# Created by restor_exposure_check_effect at Exposure 15, removed below it, removed from everyone
# when the crown is restored. The finding is vanilla's: the spymaster's task_find_secrets.

secret_restor_member = {
	category = civil

	is_valid = {
		scope:secret_owner = {
			has_trait = restor_member
			restor_underground_trigger = yes
		}
	}

	is_criminal = {
		scope:secret_owner = {
			exists = global_var:restor_usurper
			OR = {
				this = global_var:restor_usurper
				any_liege_or_above = { this = global_var:restor_usurper }
			}
		}
	}

	is_shunned = {
		always = no
	}

	on_discover = {
		# The discoverer learns who, and so does the usurper if the discoverer serves them.
		scope:discoverer = {
			if = {
				limit = {
					exists = global_var:restor_usurper
					OR = {
						this = global_var:restor_usurper
						any_liege_or_above = { this = global_var:restor_usurper }
					}
					NOT = { this = global_var:restor_usurper }
				}
				scope:secret_owner = { save_scope_as = restor_found_member }
				global_var:restor_usurper = {
					trigger_event = { id = restor.0305 days = 3 }
				}
			}
		}
	}

	# Vanilla's secret_coup_plotter shape: no generic notification (it wants a victim this secret
	# does not have); the owner's own event says what happened.
	on_expose = {
		save_scope_as = secret
		scope:secret_owner = {
			save_scope_as = restor_exposed_member
			trigger_event = { id = restor.0301 days = 2 }
		}
		if = {
			limit = { exists = global_var:restor_usurper }
			global_var:restor_usurper = {
				trigger_event = { id = restor.0306 days = 5 }
			}
		}
	}
}
