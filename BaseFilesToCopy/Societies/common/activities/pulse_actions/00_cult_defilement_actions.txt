# Pulse actions of the Pilgrimage of Defilement (activity_cult_defilement). Small moments at the shrine
# and on the road there, each a line in the activity's log with one small effect. Listed in the
# activity's pulse_actions block.
#
# Root is the activity, so the society's meter changes go through hidden events on the character
# (cult.1190 exposure +1, cult.1191 dark power +2, cult.1192 exposure -1), each with a custom_tooltip
# for the log line. Everyone who can attend is a member (can_be_activity_guest), but each action
# still checks the trait before touching a meter.
#
#   cult_defilement_apa_backwards_prayer    dark power +2
#   cult_defilement_apa_kissed_the_step     piety
#   cult_defilement_apa_shared_bread        stress loss
#   cult_defilement_apa_watched_the_priests intrigue lifestyle
#   cult_defilement_apa_lay_brother         exposure +1        (secret phases)
#   cult_defilement_apa_devout_pilgrim      exposure -1        (secret phases, some exposure to lose)
#   cult_defilement_apa_old_mark            rank XP +2
#   cult_defilement_apa_hymn                diplomacy lifestyle
#   cult_defilement_apa_blistered_feet      stress gain
#   cult_defilement_apa_walked_together     the host and a guest like each other more

cult_defilement_apa_backwards_prayer = {
	icon = positive
	is_valid = {
		has_activity_type = activity_cult_defilement
		is_current_phase_active = yes
		any_attending_character = {
			location = scope:province
			has_trait = cult_member
		}
	}
	weight = { value = 1 }
	effect = {
		random_attending_character = {
			limit = {
				location = scope:province
				has_trait = cult_member
			}
			save_scope_as = first
		}
		add_activity_log_entry = {
			key = cult_defilement_apa_backwards_prayer
			tags = { pulse_action }
			character = scope:first
			custom_tooltip = cult_defilement_apa_power_tt
			scope:first = { trigger_event = cult.1191 }
		}
	}
}

cult_defilement_apa_walked_together = {
	icon = public
	is_valid = {
		has_activity_type = activity_cult_defilement
		is_current_phase_active = yes
		scope:host = { location = scope:province }
		any_attending_character = {
			location = scope:province
			NOT = { this = scope:host }
		}
	}
	weight = { value = 1 }
	effect = {
		scope:host = { save_scope_as = first }
		random_attending_character = {
			limit = {
				location = scope:province
				NOT = { this = scope:host }
			}
			save_scope_as = second
		}
		add_activity_log_entry = {
			key = cult_defilement_apa_walked_together
			tags = { pulse_action }
			character = scope:first
			target = scope:second
			scope:first = {
				add_opinion = {
					target = scope:second
					modifier = friendliness_opinion
					opinion = 10
				}
			}
			scope:second = {
				add_opinion = {
					target = scope:first
					modifier = friendliness_opinion
					opinion = 10
				}
			}
		}
	}
}
