# The Black Mass's pulse actions: small moments in the chapel, each a line in the activity window's
# log and a small effect. Listed in 00_cult_mass_activity.txt under pulse_actions.
#
# Root here is the ACTIVITY, not a character, so nothing below calls the cult's own meter effects
# (their toasts name `root`). Only plain vanilla effects: stress, dread, prestige, piety, opinion,
# lifestyle and rank XP, gold.
#
# Icons are the three files the folder really has (public, positive, negative): ck3-tiger does not
# read the folder's path_overrides.txt and flags every other name as a missing file.

cult_mass_black_candles = {
	icon = public
	is_valid = {
		is_current_phase_active = yes
		any_attending_character = { location = scope:province }
	}
	weight = { value = 1 }
	effect = {
		random_attending_character = {
			limit = { location = scope:province }
			save_scope_as = first
		}
		add_activity_log_entry = {
			key = cult_mass_black_candles
			tags = { pulse_action }
			character = scope:first
			scope:first = {
				add_stress = minor_stress_loss
			}
		}
	}
}

cult_mass_whispered_name = {
	icon = public
	is_valid = {
		is_current_phase_active = yes
		any_attending_character = {
			location = scope:province
			NOT = { has_trait = compassionate }
		}
	}
	weight = {
		value = 1
		if = {
			limit = {
				any_attending_character = {
					location = scope:province
					OR = {
						has_trait = vengeful
						has_trait = sadistic
						has_trait = callous
					}
				}
			}
			add = 0.5
		}
	}
	effect = {
		random_attending_character = {
			limit = {
				location = scope:province
				NOT = { has_trait = compassionate }
			}
			weight = {
				base = 1
				modifier = {
					add = 2
					OR = {
						has_trait = vengeful
						has_trait = sadistic
						has_trait = callous
					}
				}
			}
			save_scope_as = first
		}
		add_activity_log_entry = {
			key = cult_mass_whispered_name
			tags = { pulse_action }
			character = scope:first
			scope:first = {
				add_dread = minor_dread_gain
			}
		}
	}
}

cult_mass_shared_chalice = {
	icon = positive
	is_valid = {
		is_current_phase_active = yes
		any_attending_character = {
			location = scope:province
			count >= 2
		}
	}
	weight = { value = 1 }
	effect = {
		random_attending_character = {
			limit = { location = scope:province }
			save_scope_as = first
		}
		random_attending_character = {
			limit = {
				location = scope:province
				this != scope:first
			}
			save_scope_as = second
		}
		add_activity_log_entry = {
			key = cult_mass_shared_chalice
			tags = { pulse_action }
			character = scope:first
			target = scope:second
			scope:first = {
				add_opinion = {
					target = scope:second
					modifier = friendliness_opinion
					opinion = 10
				}
			}
			scope:second = {
				add_opinion = {
					target = scope:first
					modifier = friendliness_opinion
					opinion = 10
				}
			}
		}
	}
}
