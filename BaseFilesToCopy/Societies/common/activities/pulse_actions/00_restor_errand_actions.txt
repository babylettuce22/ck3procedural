# The Errand's pulse actions: small road moments for the activity log. Listed in
# 00_restor_errand_activity.txt under pulse_actions.
#
# Root = the activity; scope:host, scope:province (the Errand's county capital). scope:first and
# scope:second are the characters the log entry shows. Effects stay small and vanilla-sized, and
# none moves favour or exposure: those two toast the member every time they change, and a pulse
# that toasts is noise. Rank XP is silent, so the society's own mark here is a point of standing.
#
# Icons: only the three that exist as files in gfx/interface/icons/activity_pulse_actions/ (public,
# positive, negative). The others vanilla uses (friend, money, prestige...) resolve through that
# folder's path_overrides.txt, which ck3-tiger does not read, so it reports each as a missing file.

# A milestone with the old arms.
restor_errand_apa_old_milestone = {
	icon = positive
	is_valid = {
		is_current_phase_active = yes
	}
	weight = { value = 1 }
	effect = {
		scope:host = { save_scope_as = first }
		add_activity_log_entry = {
			key = restor_errand_apa_old_milestone
			tags = { pulse_action }
			character = scope:first
			scope:first = {
				add_stress = minor_stress_loss
			}
		}
	}
}

# The old songs, sung low.
restor_errand_apa_old_songs = {
	icon = public
	is_valid = {
		is_current_phase_active = yes
	}
	weight = { value = 1 }
	effect = {
		random_attending_character = {
			limit = { location = scope:province }
			save_scope_as = first
		}
		if = {
			limit = { NOT = { exists = scope:first } }
			scope:host = { save_scope_as = first }
		}
		add_activity_log_entry = {
			key = restor_errand_apa_old_songs
			tags = { pulse_action }
			character = scope:first
			scope:first = {
				add_stress = minor_stress_loss
			}
		}
	}
}

# Two of the Sworn, late by the fire.
restor_errand_apa_by_the_fire = {
	icon = positive
	is_valid = {
		is_current_phase_active = yes
		any_attending_character = {
			NOT = { this = scope:host }
			has_trait = restor_member
		}
	}
	weight = { value = 2 }
	effect = {
		scope:host = { save_scope_as = first }
		random_attending_character = {
			limit = {
				NOT = { this = scope:host }
				has_trait = restor_member
			}
			save_scope_as = second
		}
		add_activity_log_entry = {
			key = restor_errand_apa_by_the_fire
			tags = { pulse_action }
			character = scope:first
			target = scope:second
			scope:first = {
				add_opinion = {
					target = scope:second
					modifier = restor_life_good_company_opinion
				}
			}
			scope:second = {
				add_opinion = {
					target = scope:first
					modifier = restor_life_good_company_opinion
				}
			}
		}
	}
}

# A companion scouts a quieter road.
restor_errand_apa_scouted_ahead = {
	icon = public
	is_valid = {
		is_current_phase_active = yes
		any_attending_character = {
			NOT = { this = scope:host }
			location = scope:province
			OR = {
				martial >= 10
				prowess >= 10
			}
		}
	}
	weight = { value = 1 }
	effect = {
		random_attending_character = {
			limit = {
				NOT = { this = scope:host }
				location = scope:province
				OR = {
					martial >= 10
					prowess >= 10
				}
			}
			save_scope_as = first
		}
		scope:host = { save_scope_as = second }
		add_activity_log_entry = {
			key = restor_errand_apa_scouted_ahead
			tags = { pulse_action }
			character = scope:first
			target = scope:second
			scope:first = {
				add_martial_lifestyle_xp = minor_lifestyle_xp
			}
			scope:second = {
				add_opinion = {
					target = scope:first
					modifier = grateful_opinion
					opinion = 10
				}
			}
		}
	}
}
