# The Gathering's pulse actions: small moments in the hall, logged in the activity window.
# Listed under pulse_actions in 00_restor_gathering_activity.txt. Scopes as in
# game/common/activities/pulse_actions/_pulse_actions.info: root = scope:activity, scope:host,
# scope:province (the hall). scope:first / scope:second are shown in the log entry.
#
# Icons resolve through game/gfx/interface/icons/activity_pulse_actions/path_overrides.txt (friend,
# martial, intrigue, prestige, money, stress_loss, stress_gain live there, not as files beside it);
# ck3-tiger does not read that file and reports them missing. "stress" and "piety", which a few
# vanilla actions use, are in neither place and draw the default icon, so they are not used here.
#
# Everything here is small and silent: stress, opinion, lifestyle XP, a little standing. The meters
# that toast (favour) are left to the events -- a toast per pulse would bury the player.

# ---- A whispered toast ------------------------------------------------------------------------------
restor_gathering_pa_whispered_toast = {
	icon = stress_loss
	is_valid = {
		is_current_phase_active = yes
		any_attending_character = {
			location = scope:province
			has_trait = restor_member
		}
	}
	weight = {
		value = 1
	}
	effect = {
		random_attending_character = {
			limit = {
				location = scope:province
				has_trait = restor_member
			}
			save_scope_as = first
		}
		add_activity_log_entry = {
			key = restor_gathering_pa_whispered_toast
			tags = { pulse_action }
			character = scope:first
			scope:first = { add_stress = minor_stress_loss }
		}
	}
}

# ---- An old song -------------------------------------------------------------------------------------
restor_gathering_pa_old_song = {
	icon = public
	is_valid = {
		is_current_phase_active = yes
		any_attending_character = {
			count >= 2
			location = scope:province
			has_trait = restor_member
		}
	}
	weight = {
		value = 1
	}
	effect = {
		random_attending_character = {
			limit = {
				location = scope:province
				has_trait = restor_member
			}
			weight = {
				base = 1
				modifier = { add = 2 has_trait = gregarious }
			}
			save_scope_as = first
		}
		random_attending_character = {
			limit = {
				location = scope:province
				has_trait = restor_member
				this != scope:first
			}
			save_scope_as = second
		}
		add_activity_log_entry = {
			key = restor_gathering_pa_old_song
			tags = { pulse_action }
			character = scope:first
			target = scope:second
			scope:second = {
				add_opinion = {
					target = scope:first
					modifier = friendliness_opinion
					opinion = 5
				}
			}
		}
	}
}

# ---- The host checks the bars on the doors ------------------------------------------------------------
