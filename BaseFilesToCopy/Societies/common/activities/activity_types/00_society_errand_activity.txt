# The errand: going out to find the society something it needs.
#
# =====================================================================================
#  What this is a port OF
# =====================================================================================
#
# CK2's devil-worshipper MISSIONS. In CK2 those were quests -- a standing objective handed to a
# member ("sacrifice a holy person", "sacrifice a ruler", "corrupt a priest"), completed anywhere
# in the world with whatever tools you had, then cashed in. The reward shape was consistent and is
# the thing worth porting, because it is what made the society feel like an employer rather than a
# menu. Every completed mission paid out along three axes at once:
#
#   SOCIETY CURRENCY   minor / medium / major, in a flat 1:2:3 ratio
#   VISIBILITY         up, always, every time
#   SOUL CORRUPTION    a roll that strips a virtue or grants a vice
#
# That triad is the design. You are paid, you are more exposed than you were, and you are slightly
# less the person you were when you started. Nothing in CK2's society system let you take the
# first without the other two, and nothing here should either.
#
# =====================================================================================
#  Why an activity, and what that costs us
# =====================================================================================
#
# CK3 has no quest system. There is no "standing objective with a target" primitive to port onto,
# and building one would mean a variable-list of assignments, a progress checker on some pulse,
# and a UI to read it from -- a system, to deliver what is really a piece of pacing.
#
# An activity is the closest shipped equivalent and it is a good one: it is a thing you decide to
# do, travel to, spend weeks on, and come back from changed. CK3's hunt is the exact structural
# precedent the design is copied from -- see the phase block below.
#
# THE COST, stated plainly because it is the significant compromise in this file: CK2's missions
# targeted real, named characters. This does not. The quarry is generated by the event chain and
# is scenery -- an unnamed devout, an unnamed highborn -- the same way the stag in a CK3 hunt is
# not a character. Targeting a real person would need the errand to check reachability, travel,
# capture and the diplomatic range rules, which is most of a scheme rather than most of an
# activity.
#
# The path back to real targets, if it is ever wanted, is NOT to rewrite this: it is to let the
# outcome event pick a real character where a suitable one is in range and fall back to scenery
# where none is. The event chain is where that decision would live, and nothing in this file
# would need to change.
#
# =====================================================================================
#  What is deliberately NOT here
# =====================================================================================
#
# No rank gate. The errand is how a novice earns their way up -- gating it on Standing would mean
# the only route to Standing is the rite, which needs 50 Standing to convene. That is a closed
# loop with no entrance. The rite is the privilege; this is the work.
#
# No Dark Power cost. Same reason the sacrifice has none: this is a tap, not a drain.

activity_society_errand = {
	# Under the society's own planner header, beside the rite.
	activity_group_type = society

	# Sorted above the rite, because it is the one a new member can actually do.
	sort_order = 10

	# ---- Gate: membership, and nothing else ----
	#
	# The same shape as the rite's, minus the Standing requirement. See the note above on why
	# there isn't one.
	is_shown = {
		has_trait = society_member

		is_playable_character = yes

		trigger_if = {
			limit = { is_ai = yes }
			is_at_war = no
			NOT = { has_variable = conqueror }
			NOT = { has_realm_law = uncrowned }
		}
	}

	can_always_plan = no

	can_start_showing_failures_only = {
		is_available_adult = yes

		NOT = { is_activity_type_on_cooldown = activity_society_errand }

		custom_tooltip = {
			text = society_power_requires_membership_tt
			has_trait = society_member
		}
	}

	is_valid = {
		NOT = { has_variable = activity_invalidated }

		scope:host = {
			is_playable_character = yes
			is_incapable = no
			has_trait = society_member
		}
	}

	on_host_death = {
		scope:activity = {
			set_variable = { name = activity_invalidated }
		}
	}

	on_invalidated = {
		invalidation_imprisoned_effect = yes
		invalidation_incapable_effect = yes
	}

	###################
	# WHAT WE ARE LOOKING FOR
	###################

	# CK2's mission list, made a planning decision instead of an assignment.
	#
	# `options` and not an event on arrival, and this is the one place this set departs from the
	# note in the rite's README that called options the better mechanic and then didn't use them.
	# Here they ARE the better mechanic and for a specific reason: the choice of quarry decides
	# how far you travel, how long you are gone and what you are risking, so it has to be made
	# before the journey rather than after it. An errand whose purpose is revealed on arrival is
	# a trip you cannot plan.
	#
	# Read back later with:
	#     scope:activity = { has_activity_option = { category = society_errand_quarry option = X } }
	#
	# The three are CK2's own, minus the ones that were about CK2 systems we do not have.
	#
	# The quarry decides the PAY and nothing else. Exposure is decided by how carefully the errand
	# is run (errand.0010 / .0020), not by who is taken -- see the ladder note in the events file
	# for why. That is a deliberate split: the two decisions are made at different moments by
	# different reasoning, and collapsing them into one axis would make the planner choice the only
	# one that mattered.
	options = {
		society_errand_quarry = {
			# ---- Someone nobody will look for ----
			#
			# The default, and the lowest paid. It is not the SAFE one -- the corruption roll is
			# identical for all three quarries, and the exposure depends on how it is run -- it is
			# simply the one that asks least of you to arrange. Which is the CK2 lesson worth
			# keeping: the errand nobody notices is the errand you will run again, and again.
			errand_quarry_unmissed = {
				default = yes

				is_shown = { always = yes }

				ai_will_do = {
					add = {
						value = 50
						desc = "base"
					}

					if = {
						limit = { has_trait = craven }
						add = {
							value = 25
							desc = "craven"
						}
					}
				}
			}

			# ---- Someone with a name ----
			errand_quarry_highborn = {
				is_shown = { always = yes }

				ai_will_do = {
					add = {
						value = 25
						desc = "base"
					}

					if = {
						limit = { has_trait = ambitious }
						add = {
							value = 20
							desc = "ambitious"
						}
					}
				}
			}

			# ---- Someone who believes ----
			#
			# The best-paid and the most exposing, because a missing devout is looked for by an
			# institution rather than by a family. Zealots and the faithless both want this one,
			# for opposite reasons, and the society does not care which.
			errand_quarry_devout = {
				is_shown = { always = yes }

				ai_will_do = {
					add = {
						value = 20
						desc = "base"
					}

					if = {
						limit = { has_trait = cynical }
						add = {
							value = 25
							desc = "cynical"
						}
					}

					if = {
						limit = { has_trait = zealous }
						add = {
							value = -40
							desc = "zealous"
						}
					}
				}
			}
		}
	}

	###################
	# WHERE
	###################

	# Anywhere in the realm, unlike the rite's single capital. An errand is the one society
	# activity that is explicitly about LEAVING -- doing it at home is what the rite is for, and
	# the whole point of the quarry ladder is that better quarry lives further away.
	province_filter = realm
	ai_province_filter = realm
	is_single_location = yes
	max_province_icons = 5

	# ROOT here is the PROVINCE, not a title -- so `exists = holder` is wrong twice over: holder is
	# a landed-title link, and a province does not have one. `has_province_owner` is the
	# province-scope form and is all this needs; province_filter = realm has already done the
	# narrowing, and this only has to exclude the unheld ones.
	is_location_valid = {
		has_province_owner = yes
	}

	province_description = {
		desc = activity_society_errand_province_desc
	}

	###################
	# WHEN
	###################

	wait_time_before_start = { months = 1 }
	max_guest_arrival_delay_time = { months = 2 }

	# Long, because the pay is good and the exposure is permanent. A member who could run one of
	# these every spring would out-earn the rite without ever convening anybody.
	cooldown = { years = 3 }

	###################
	# THE CHAIN
	###################

	# Copied in shape from vanilla's hunt (common/activities/activity_types/hunt.txt:5138), which
	# is the closest thing CK3 ships to what CK2 called a mission: a staged spine of events on a
	# timer, each one a beat in a story that resolves at the end.
	#
	#   day  1   THE SETTING OUT   what we told people we were doing
	#   day 14   THE QUARRY        we find them
	#   day 21   THE COMPLICATION  something is wrong with the plan
	#   day 28   THE OUTCOME       the triad is paid here, and only here
	#
	# Vanilla's hunt spaces its four beats 0/21/28/35 across six weeks. This is four weeks and
	# 1/14/21/28, tighter because there is no party to narrate and nobody to talk to -- an errand
	# is mostly travelling and one bad evening.
	#
	# The `this = scope:host` guard is what makes the chain the HOST'S story. Guests get one
	# arrival beat and nothing else: they are along to carry things, and the decisions belong to
	# whoever is answering for the errand. That is also why the phase timer sits inside the same
	# guard -- the rite's file has a long note on why this block runs per attending character.
	phases = {
		society_errand_phase = {
			is_predefined = yes

			on_phase_active = {
				if = {
					limit = { this = scope:host }

					scope:activity = { progress_activity_phase_after = { weeks = 4 } }

					trigger_event = {
						id = errand.0001
						days = 1
					}

					trigger_event = {
						id = errand.0010
						days = 14
					}

					trigger_event = {
						id = errand.0020
						days = 21
					}

					trigger_event = {
						id = errand.0030
						days = 28
					}
				}
				else = {
					trigger_event = {
						id = errand.0002
						days = 2
					}
				}
			}
		}
	}

	###################
	# WHO
	###################

	host_intents = {
		default = society_errand_intent
		intents = { society_errand_intent }
	}

	guest_intents = {
		default = society_errand_intent
		intents = { society_errand_intent }
	}

	# Two, at most, and none is fine. An errand is not a procession. `allow_zero_guest_invites`
	# is the important one: the whole activity has to work for a lone member, because a new
	# member IS a lone member and this is the activity meant for them.
	max_guests = 2
	allow_zero_guest_invites = yes
	open_invite = no

	# Indexed, not a bare list. `rules = { some_rule }` parses as a loose value and ck3-tiger
	# rejects it -- the rite next door has the correct shape and it is numbered.
	guest_invite_rules = {
		rules = {
			1 = activity_invite_rule_society_court
		}
	}

	# The same veto as the rite. Nobody unsworn comes along on this of all things.
	can_be_activity_guest = {
		is_adult = yes
		is_healthy = yes
		has_trait = society_member
		in_diplomatic_range = scope:host
	}

	# ---- Payment ----
	#
	# NOT here. on_complete runs once per attending character and the errand's reward depends on
	# what happened in the chain, so paying here would pay everybody the same regardless of how
	# it went. The triad is paid in errand.0030 and nowhere else -- see the note there.
	#
	# Guests get their Standing here, because turning up for somebody else's errand is worth
	# something and is not contingent on the outcome.
	on_complete = {
		if = {
			limit = { this != scope:host }

			add_trait_xp = {
				trait = society_member
				track = society_rank
				value = 4
			}
		}
	}

	###################
	# AI
	###################

	ai_will_do = {
		value = 20

		if = {
			limit = { has_trait = sadistic }
			add = 20
		}

		if = {
			limit = { has_trait = cynical }
			add = 10
		}
	}

	ai_check_interval_by_tier = {
		barony = 120
		county = 120
		duchy = 120
		kingdom = 120
		empire = 120
		hegemony = 120
	}

	ai_will_select_province = {
		value = 1
	}

	# `environment` and `ambience` are not optional beside `texture`. Without them CK3 logs
	# "Field 'background' not set", which reads like a missing texture and is not.
	background = {
		texture = "gfx/interface/illustrations/event_scenes/dungeon.dds"
		environment = "environment_event_dungeon"
		ambience = "event:/SFX/Ambience/2DAmbience/Interface/dungeon"
	}

	window_characters = {
		host = {
			camera = camera_body

			effect = {
				scope:host = { save_scope_as = selected_character }
			}
		}
	}
}
