# Who the rite's guest list is allowed to offer.
#
# ---- What a guest invite rule actually is ----
#
# Not a filter. A BUILDER. The planner shows one tab per rule the activity declares, and the
# tab's contents are whatever the rule's `effect` block put in the `characters` list -- so a
# rule cannot narrow the guest list, it can only produce one. That is why the trait test
# appears here as a `limit` on each iterator rather than as a condition somewhere central:
# every route into the list has to carry it, because each route is its own gathering.
#
# It is also why this file is only half of the answer. A rule that adds nobody but members
# still leaves the OTHER routes into an activity -- events, effects, a future rule someone
# adds without reading this -- unguarded. The hard gate is `can_be_activity_guest` in
# 00_society_rite_activity.txt, which the engine checks for every candidate however they got
# proposed. Belt and braces, and vanilla's witch ritual wears both for the same reason.
#
# ---- Why these iterators and not `every_living_character` ----
#
# Because you cannot invite someone you have never heard of. There is no world-wide roster in
# this prototype -- when the generator writes a real society it will keep one, as a global
# list maintained at game start, and this rule becomes a single walk over it. Until then the
# reachable sets below are both the honest answer and the cheap one: a ruler's court, his
# vassals, his family, his spouses and his liege are the people he could plausibly know the
# allegiance of.
#
# `bannable_serving_diarch_trigger = no` is copied from every vanilla invite rule. A serving
# diarch cannot leave, so offering one is offering a name that will always decline.

activity_invite_rule_society_members = {
	effect = {
		# ---- Instrumentation, and why it is worth carrying ----
		#
		# An empty guest tab has at least four possible causes and the UI reports none of them:
		# the rule never ran, the rule ran and matched nobody, the rule matched people and
		# `can_be_activity_guest` vetoed all of them, or the planner declined to draw a tab that
		# came back empty. Reading the list from the outside cannot tell those apart.
		#
		# These two lines can. The first says the effect executed at all; the second fires once
		# per name the rule is proposing, so `debug.log` carries the count the tab will not show.
		# No line at all means the rule is not being run and the fault is in the activity's
		# `guest_invite_rules` block, not here. Lines but an empty tab means the veto ate them.
		debug_log = "SOCIETY: invite rule running"

		# The household. In practice this is where the first few members come from, because
		# the induction interaction is easiest to use on somebody already standing there.
		every_courtier = {
			limit = {
				has_trait = society_member
				bannable_serving_diarch_trigger = no
			}
			debug_log = "SOCIETY: proposing a sworn courtier"
			add_to_list = characters
		}

		# Downward through the realm.
		every_vassal_or_below = {
			limit = {
				has_trait = society_member
				bannable_serving_diarch_trigger = no
			}
			add_to_list = characters
		}

		# Blood, which crosses realm borders where vassalage does not -- the nearest thing
		# this prototype has to the cross-realm reach a society is supposed to have.
		every_close_or_extended_family_member = {
			limit = {
				has_trait = society_member
				bannable_serving_diarch_trigger = no
			}
			add_to_list = characters
		}

		every_spouse = {
			limit = {
				has_trait = society_member
				bannable_serving_diarch_trigger = no
			}
			add_to_list = characters
		}

		# Upward. `?=` because a top-level ruler has no liege and the engine should not be
		# asked what trait nobody has.
		liege ?= {
			if = {
				limit = {
					has_trait = society_member
					bannable_serving_diarch_trigger = no
				}
				add_to_list = characters
			}
		}
	}
}

# TEMPORARY. Delete this rule and its line in 00_society_rite_activity.txt once the guest tab
# is known to work.
#
# It gathers the court with NO conditions whatsoever -- not the trait, not the diarch check,
# nothing -- so the planner draws two tabs side by side and the difference between them is the
# answer:
#
#   Both tabs empty                  the rule is not being run, or can_be_activity_guest is
#                                    vetoing every candidate. Check debug.log for the
#                                    "SOCIETY: invite rule running" line to tell those apart.
#   This tab full, the other empty   the rule runs and the veto passes, so the fault is the
#                                    trait limit -- nobody in court is actually sworn, or
#                                    has_trait is being asked of the wrong scope.
#   Both tabs full                   the guest list works and the bug was upstream of it.
#
# A control that shares as much as possible with the thing it is testing and differs in exactly
# one way is worth more than any amount of reading the rule again, which is what the last three
# rounds of this have been.
activity_invite_rule_society_debug = {
	effect = {
		debug_log = "SOCIETY: control rule running"

		every_courtier = {
			debug_log = "SOCIETY: control proposing a courtier"
			add_to_list = characters
		}
	}
}
