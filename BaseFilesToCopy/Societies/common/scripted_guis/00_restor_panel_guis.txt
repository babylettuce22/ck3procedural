# The society panel: everything the window asks the script layer.
#
# A .gui cannot run a trigger, compare numbers or read a global variable -- GetGlobalVariable
# type-checks, passes ck3-tiger and resolves to nothing in game (the prototype found this). So:
#
#   questions  are scripted_guis asked with IsShown (one per question, one per rung);
#   numbers    and people are COPIED onto the viewing player as character variables by
#              restor_panel_gather when the panel opens, and read with MakeScope.Var(...).
#
# society_panel_toggle and society_panel_window keep the prototype's names because
# Emit/GuiWriter.cs patches the HUD tab to call them by name.

# ---- Open state ------------------------------------------------------------------------------

society_panel_window = {
	scope = character
	is_shown = { has_variable = society_panel_open }
	effect = {
		remove_variable = society_panel_open
		# Closing lets the next opening rebuild "Within reach" at once rather than wait for its tick.
		remove_variable = restor_panel_reach_built
		remove_variable = cult_panel_reach_built
	}
}

# The tab. Shown only to a member of a society, so a non-member has no tab at all -- no society's
# existence is advertised to people outside it. One tab for every society: a character belongs to
# at most one, and each society's window shows only to its own members (restor_panel_viewer_member,
# cult_panel_viewer_member).
society_panel_toggle = {
	scope = character
	is_shown = {
		OR = {
			AND = {
				restor_active_trigger = yes
				has_trait = restor_member
			}
			AND = {
				cult_active_trigger = yes
				has_trait = cult_member
			}
		}
	}
	effect = {
		if = {
			limit = { has_variable = society_panel_open }
			remove_variable = society_panel_open
		}
		else = {
			set_variable = { name = society_panel_open value = yes }
		}
	}
}

# ---- Gather: copy what the window shows onto the viewer ----------------------------------------

restor_panel_gather = {
	scope = character
	is_shown = { restor_active_trigger = yes }
	effect = {
		# Run on opening (the window's _show) and then once a second by restor_panel_tick while it
		# stays open. The tick's widget may keep time while the window is hidden, so the whole body
		# is skipped unless the panel is actually open.
		if = {
			limit = { has_variable = society_panel_open }

			# The live value, read without recomputing the global: opening a window must not be what
			# crosses a milestone and sends every member an event.
			set_variable = { name = restor_panel_support value = restor_support_value }
			set_variable = { name = restor_panel_share value = restor_pretender_share_value }
			set_variable = { name = restor_panel_war_chest value = global_var:restor_war_chest }

			# People. Set when filled, removed when emptied -- never removed and re-set, which would
			# blink the portrait's datacontext every second.
			if = {
				limit = { exists = global_var:restor_pretender }
				set_variable = { name = restor_panel_pretender value = global_var:restor_pretender }
			}
			else = { remove_variable = restor_panel_pretender }
			if = {
				limit = { exists = global_var:restor_keeper }
				set_variable = { name = restor_panel_keeper value = global_var:restor_keeper }
			}
			else = { remove_variable = restor_panel_keeper }
			if = {
				limit = { exists = global_var:restor_usurper }
				set_variable = { name = restor_panel_usurper value = global_var:restor_usurper }
			}
			else = { remove_variable = restor_panel_usurper }
			if = {
				limit = {
					restor_restored_trigger = yes
					exists = global_var:restor_crown.holder
				}
				set_variable = { name = restor_panel_king value = global_var:restor_crown.holder }
			}
			else = { remove_variable = restor_panel_king }

			# The roster: the members this viewer would know (restor_panel_knows_trigger). Rebuilt
			# only when that set has changed -- someone to add, or someone listed who no longer
			# belongs -- because rebuilding the list every tick redraws every row under the mouse.
			save_scope_as = restor_viewer
			if = {
				limit = {
					OR = {
						any_in_global_list = {
							variable = restor_roster
							restor_panel_knows_trigger = yes
							NOT = {
								scope:restor_viewer = {
									is_target_in_variable_list = { name = restor_panel_roster target = prev }
								}
							}
						}
						any_in_list = {
							variable = restor_panel_roster
							NOT = { restor_panel_knows_trigger = yes }
						}
					}
				}
				clear_variable_list = restor_panel_roster
				every_in_global_list = {
					variable = restor_roster
					limit = { restor_panel_knows_trigger = yes }
					save_scope_as = restor_rostered
					scope:restor_viewer = {
						add_to_variable_list = {
							name = restor_panel_roster
							target = scope:restor_rostered
						}
					}
				}
			}

			# "Within reach" is the expensive part (it walks every ruler), so it is built once here on
			# opening and then only by restor_panel_reach_tick, not every second.
			if = {
				limit = {
					has_trait = restor_member
					NOT = { has_variable = restor_panel_reach_built }
				}
				restor_panel_gather_reach_effect = yes
			}
		}
	}
}

# The fifteen-second refresh of "Within reach" (restor_panel_reach_tick). Nothing unless open.
restor_panel_gather_reach = {
	scope = character
	is_shown = { restor_active_trigger = yes }
	effect = {
		if = {
			limit = {
				has_variable = society_panel_open
				has_trait = restor_member
			}
			restor_panel_gather_reach_effect = yes
		}
	}
}
