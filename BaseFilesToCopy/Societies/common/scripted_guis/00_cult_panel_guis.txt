# The cult's panel: everything its window asks the script layer. Same arrangement as the
# Restorationists' (00_restor_panel_guis.txt): questions are scripted_guis asked with IsShown, and
# numbers and people are copied onto the viewer by cult_panel_gather. The open/close pair
# (society_panel_toggle / society_panel_window) is shared and lives in that file.

cult_panel_viewer_member = {
	scope = character
	is_shown = {
		cult_active_trigger = yes
		has_trait = cult_member
	}
}

cult_panel_gather = {
	scope = character
	is_shown = { cult_active_trigger = yes }
	effect = {
		# On opening and then once a second while open (cult_panel_tick); skipped when closed. The
		# reasoning for each part is in restor_panel_gather.
		if = {
			limit = { has_variable = society_panel_open }

			# The live values, read without recomputing the globals: opening a window must not be
			# what crosses a milestone.
			set_variable = { name = cult_panel_rot value = cult_rot_value }
			set_variable = { name = cult_panel_sites value = cult_holy_site_share_value }

			if = {
				limit = { exists = global_var:cult_hierophant }
				set_variable = { name = cult_panel_hierophant value = global_var:cult_hierophant }
			}
			else = { remove_variable = cult_panel_hierophant }
			if = {
				limit = { exists = global_var:cult_host_faith.religious_head }
				set_variable = { name = cult_panel_head value = global_var:cult_host_faith.religious_head }
			}
			else = { remove_variable = cult_panel_head }
			if = {
				limit = {
					cult_secret_phase_trigger = yes
					exists = global_var:cult_hunter
				}
				set_variable = { name = cult_panel_hunter value = global_var:cult_hunter }
			}
			else = { remove_variable = cult_panel_hunter }

			# Rebuilt only when the set of members this viewer knows (cult_panel_knows_trigger) has
			# changed.
			save_scope_as = cult_viewer
			if = {
				limit = {
					OR = {
						any_in_global_list = {
							variable = cult_roster
							cult_panel_knows_trigger = yes
							NOT = {
								scope:cult_viewer = {
									is_target_in_variable_list = { name = cult_panel_roster target = prev }
								}
							}
						}
						any_in_list = {
							variable = cult_panel_roster
							NOT = { cult_panel_knows_trigger = yes }
						}
					}
				}
				clear_variable_list = cult_panel_roster
				every_in_global_list = {
					variable = cult_roster
					limit = { cult_panel_knows_trigger = yes }
					save_scope_as = cult_rostered
					scope:cult_viewer = {
						add_to_variable_list = {
							name = cult_panel_roster
							target = scope:cult_rostered
						}
					}
				}
			}

			# "Within reach" is the expensive part (it walks the faith's rulers), so it is built once
			# here on opening and then only by cult_panel_reach_tick, not every second.
			if = {
				limit = { NOT = { has_variable = cult_panel_reach_built } }
				cult_panel_gather_reach_effect = yes
			}
		}
	}
}

# The fifteen-second refresh of "Within reach" (cult_panel_reach_tick). Nothing unless open.
cult_panel_gather_reach = {
	scope = character
	is_shown = { cult_active_trigger = yes }
	effect = {
		if = {
			limit = {
				has_variable = society_panel_open
				has_trait = cult_member
			}
			cult_panel_gather_reach_effect = yes
		}
	}
}
