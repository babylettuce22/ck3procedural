# The inversion cult's core events: debug, introductions, the soul's cost, the rot's milestones, the
# capture of the church, the Unveiling, the Purification, and succession.
#
#   cult.0001-0004   debug (console)
#   cult.0010-0011   introductions: a sworn player; the hunter
#   cult.0050        what the cult took from you (soul corruption)
#   cult.0601-0603   the rot passes 25 / 50 / 75 (every member)
#   cult.0701        the church is hollow (every member)
#   cult.0710-0712   the Unveiling: the unveiler, the sworn, the faithful who were not
#   cult.0720-0722   the Purification: the head's choice, the sworn, the purifier
#   cult.0801        the church slips from the cult's hands (every member)
#   cult.0820        a new Hierophant

namespace = cult

# =====================================================================================
#  Debug
# =====================================================================================

cult.0001 = {
	type = character_event
	hidden = yes
	trigger = { cult_active_trigger = yes }
	immediate = {
		cult_swear_in_effect = { RANK = 50 }
		cult_power_gain_effect = { VALUE = 150 }
	}
}

# Taint every county of the host faith's holy sites and recompute: walks the rot up.
cult.0002 = {
	type = character_event
	hidden = yes
	trigger = { cult_active_trigger = yes }
	immediate = {
		global_var:cult_host_faith = {
			every_holy_site = {
				county = {
					add_county_modifier = {
						modifier = cult_county_tainted
						years = 10
					}
				}
			}
		}
		cult_rot_refresh_effect = yes
	}
}

	}
}

# =====================================================================================
#  The Purification
# =====================================================================================

# To a head of the host faith who is not sworn, yearly while the faith is unveiled.
cult.0720 = {
	type = character_event
	title = cult.0720.t
	desc = cult.0720.desc
	theme = faith
	left_portrait = {
		character = root
		animation = personality_zealous
	}
	trigger = {
		cult_unveiled_trigger = yes
		NOT = { has_trait = cult_member }
		# Still the head ten days after being asked.
		exists = global_var:cult_host_faith.religious_head
		this = global_var:cult_host_faith.religious_head
	}
	immediate = { cult_save_scopes_effect = yes }

	option = {
		name = cult.0720.a
		custom_tooltip = cult.0720.a.tt
		hidden_effect = { cult_purify_effect = yes }
		ai_chance = {
			base = 70
			modifier = { add = 30 has_trait = zealous }
			modifier = { add = -40 has_trait = cynical }
		}
	}
	option = {
		name = cult.0720.b
		add_piety = -100
		# Asked again in five years, not every year.
		add_character_flag = { flag = cult_declined_purify years = 5 }
		ai_chance = {
			base = 30
			modifier = { add = 40 has_trait = cynical }
		}
	}
}
