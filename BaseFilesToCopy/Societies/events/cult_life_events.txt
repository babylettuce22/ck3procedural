# The inversion cult: everyday life in the circle. cult.0200-0299.
#
# Fired by cult_life_pulse (common/on_action/00_cult_life_on_actions.txt), once a year at most and
# usually not at all. These are what it is like to worship the Adversary inside a pious court: the
# forms of the church kept every day by people who mean none of it, and what that costs them.
#
#   Hidden and hollow (secret)
#   0201   the holy day mass; the sacrament from a priest who may be sworn                   (random)
#   0202   your confessor asks questions you cannot answer                                    (random)
#   0203   a relic of the high god in your chapel -- AI too                                   (random)
#   0204   a pilgrim at the gate, bound for a shrine the circle holds                         (random)
#   0205   a sign in the night, or a dream                                                    (random)
#   0206   a member asks you to stand sponsor at a christening
#   0207   the hunter's preacher comes to your court                                          (random)
#   0208   an old member wants to leave the circle                                            (random)
#   0209   two of the Faithless quarrel over who serves the dark better                       (random)
#   0210   a heretic is burned in the square, and the circle watches                          (random)
#   0211   the bishop asks for money to rebuild the house of worship -- AI too                (random)
#   0212   a servant finds an inverted symbol among your things                               (random)
#   0213   the rot reaches a parish of yours -- AI too
#   0214   a quiet night: what you have become; stay, or walk away                            (random)
#   0215   hollow only: the sworn head of the faith preaches                                  (random)
#
#   Unveiled
#   0250   one of the old faithful at your court will not kneel
#   0251   zealots of the old faith riot in your capital                                      (random)
#   0252   was it worth it                                                                    (random)
#
# Every option with a roll ends each branch in a toast. Each event's immediate sets a twelve-year
# cult_life_seen_NNNN flag that its trigger reads, so a long life does not repeat itself. Anything
# that changes the map (a county tainted, a chaplain sworn) is followed by cult_rot_refresh_effect.

namespace = cult

# =====================================================================================
#  cult.0201 -- The Holy Day
# =====================================================================================

cult.0201 = {
	type = character_event
	title = cult.0201.t
	desc = {
		desc = cult.0201.desc
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:cult_life_chaplain }
				desc = cult.0201.desc_sworn_priest
			}
			desc = cult.0201.desc_priest
		}
	}
	theme = faith
	override_background = { reference = temple }
	left_portrait = {
		character = root
		animation = prayer
	}
	right_portrait = {
		trigger = { exists = scope:cult_life_chaplain }
		character = scope:cult_life_chaplain
		animation = chaplain
	}

	trigger = {
		cult_secret_phase_trigger = yes
		is_ai = no
		exists = global_var:cult_host_faith
		NOT = { has_character_flag = cult_life_seen_0201 }
	}

	immediate = {
		cult_save_scopes_effect = yes
		add_character_flag = { flag = cult_life_seen_0201 years = 12 }
		cp:councillor_court_chaplain ?= {
			if = {
				limit = { has_trait = cult_member }
				save_scope_as = cult_life_chaplain
			}
		}
	}

	# Take it, and mean none of it.
	option = {
		name = cult.0201.a
		add_piety = minor_piety_gain
		cult_exposure_loss_effect = { VALUE = 2 }
		stress_impact = {
			honest = minor_stress_impact_gain
		}
	}

	# Take it, and say the other words.
	option = {
		name = cult.0201.b
		cult_power_gain_effect = { VALUE = 10 }
		cult_corrupt_soul_effect = yes
		random_list = {
			75 = {
				modifier = { add = 20 exists = scope:cult_life_chaplain }
				modifier = { add = 10 intrigue >= 12 }
				send_interface_toast = {
					type = event_toast_effect_good
					title = cult.0201.b.unheard
					left_icon = root
					custom_tooltip = cult.0201.b.unheard.tt
				}
			}
			25 = {
				send_interface_toast = {
					type = event_toast_effect_bad
					title = cult.0201.b.heard
					left_icon = root
					cult_exposure_gain_effect = { VALUE = 4 }
				}
			}
		}
	}

	# A fever, and bed.
	option = {
		name = cult.0201.c
		add_piety = minor_piety_loss
		random_list = {
			60 = {
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = cult.0201.c.believed
					left_icon = root
					custom_tooltip = cult.0201.c.believed.tt
				}
			}
			40 = {
				modifier = { add = 20 has_trait = zealous }
				send_interface_toast = {
					type = event_toast_effect_bad
					title = cult.0201.c.noted
					left_icon = root
					cult_exposure_gain_effect = { VALUE = 2 }
				}
			}
