# The inversion cult: exposure, and the hunter's side. cult.0300-0399.
#
# The member's side
#   0300   exposure crossed 15: whispered about; a findable secret now exists (a crime or a disgrace,
#          by the host faith's own witchcraft doctrine)
#   0301   the member's secret was exposed: everyone knows
#   0302   exposure crossed 25: suspected heretic, the hunter is being told
#   0307   the hunter offers absolution for names
#   0308   the hunter demands a confession
#   0312   the church is searching the realm
#   0330-0333  yearly, exposure 10+: public penance, a showy donation, a scapegoat, the chaplain
#
# The hunter's side
#   0303   a name is suspected (a member crossed 25)
#   0305   the hunter's court found a member's secret
#   0306   a member's secret was exposed
#   0309   the trial for heresy (a member in the hunter's prison)
#   0310   Root Out Heresy names the most exposed member of the realm
#   0320-0324  yearly: a confessor's report, a desecrated shrine, a priest who prays wrongly, a witch
#              accused by villagers, black candles
#
# "Within the hunter's reach" is `any_liege_or_above = hunter`, or a guest at the hunter's own court.
# A hunter who is also the head of the faith may excommunicate anyone of the faith, anywhere.

namespace = cult

# =====================================================================================
#  cult.0300 -- whispered about
# =====================================================================================

cult.0300 = {
	type = character_event
	title = cult.0300.t
	desc = {
		desc = cult.0300.desc
		first_valid = {
			triggered_desc = {
				trigger = {
					global_var:cult_host_faith = {
						OR = {
							has_doctrine = doctrine_witchcraft_crime
							has_doctrine = doctrine_witchcraft_shunned
						}
					}
				}
				desc = cult.0300.desc_crime
			}
			desc = cult.0300.desc_shunned
		}
	}
	theme = witchcraft
	override_background = { reference = temple }
	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		trigger = {
			exists = scope:cult_hunter
			NOT = { scope:cult_hunter = root }
		}
		character = scope:cult_hunter
		animation = personality_zealous
	}
	trigger = {
		cult_secret_phase_trigger = yes
		has_trait = cult_member
	}
	immediate = { cult_save_scopes_effect = yes }

	# Keep up appearances.
	option = {
		name = cult.0300.a
		if = {
			limit = {
				exists = scope:cult_hunter
				NOT = { scope:cult_hunter = root }
			}
			custom_tooltip = cult.0300.a.tt
		}
		else = { custom_tooltip = cult.0300.a.tt_nohunter }
		ai_chance = { base = 70 }
	}

	# Clear the house of anything that could be found.
	option = {
		name = cult.0300.b
		cult_exposure_loss_effect = { VALUE = 4 }
		cult_power_spend_effect = { VALUE = 10 }
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = paranoid }
		}
	}
}

# =====================================================================================
#  cult.0301 -- the member's secret is exposed
# =====================================================================================

cult.0301 = {
	type = character_event
	title = cult.0301.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:cult_hunter
					NOT = { scope:cult_hunter = root }
					any_liege_or_above = { this = scope:cult_hunter }
				}
				desc = cult.0301.desc_inside
			}
			triggered_desc = {
				trigger = {
					exists = scope:cult_hunter
					NOT = { scope:cult_hunter = root }
				}
				desc = cult.0301.desc_outside
			}
			desc = cult.0301.desc_nohunter
		}
	}
	theme = secret
	override_background = { reference = temple }
	left_portrait = {
		character = root
		animation = shock
	}
	right_portrait = {
		trigger = {
			exists = scope:cult_hunter
			NOT = { scope:cult_hunter = root }
		}
		character = scope:cult_hunter
		animation = anger
	}
	lower_right_portrait = {
		trigger = { exists = scope:cult_refuge }
		character = scope:cult_refuge
	}
	trigger = {
		cult_secret_phase_trigger = yes
		has_trait = cult_member
	}
	immediate = {
		cult_save_scopes_effect = yes
		# A sworn lord outside the hunter's reach, for a courtier to run to.
		if = {
			limit = {
				is_ruler = no
				is_imprisoned = no
			}
			random_in_global_list = {
				variable = cult_roster
				limit = {
					is_alive = yes
					is_landed = yes
					is_imprisoned = no
					NOT = { this = root }
					OR = {
						NOT = { exists = global_var:cult_hunter }
						NOT = { any_liege_or_above = { this = global_var:cult_hunter } }
					}
				}
				save_scope_as = cult_refuge
			}
		}
	}

	# Deny it.
	option = {
		name = cult.0301.a
		duel = {
			skill = intrigue
			value = decent_skill_rating
			45 = {
				desc = cult.0301.a.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -40
				}
				modifier = { add = 10 has_trait = deceitful }
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = cult.0301.a.success
					left_icon = root
					add_prestige = minor_prestige_loss
				}
				cult_exposure_loss_effect = { VALUE = 5 }
			}
			55 = {
				desc = cult.0301.a.failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -50
				}
				modifier = { add = 10 has_trait = honest }
				send_interface_toast = {
					type = event_toast_effect_bad
					title = cult.0301.a.failure
					left_icon = root
					add_prestige = medium_prestige_loss
				}
				cult_exposure_gain_effect = { VALUE = 5 }
			}
		}
		stress_impact = { honest = minor_stress_impact_gain }
		ai_chance = {
			base = 30
			modifier = { add = 25 has_trait = deceitful }
			modifier = { add = -20 has_trait = honest }
		}
	}

	# Run to a sworn court.
	option = {
		name = cult.0301.b
		trigger = { exists = scope:cult_refuge }
		scope:cult_refuge = { add_courtier = root }
		cult_exposure_loss_effect = { VALUE = 5 }
		custom_tooltip = cult.0301.b.tt
		ai_chance = {
			base = 40
			modifier = { add = 30 has_trait = craven }
		}
	}

	# Confess to the hunter, and name names.
	option = {
		name = cult.0301.c
		trigger = {
			exists = scope:cult_hunter
			NOT = { scope:cult_hunter = root }
			NOT = { cult_is_hierophant_trigger = yes }
		}
		add_prestige = minor_prestige_loss
		cult_confess_and_name_effect = { TO = scope:cult_hunter }
		stress_impact = {
			honest = minor_stress_impact_gain
			brave = medium_stress_impact_gain
		}
		ai_chance = {
			base = 15
			modifier = { add = 30 has_trait = craven }
			modifier = { add = 15 has_trait = deceitful }
			modifier = { add = -15 cult_rank_at_least_trigger = { RANK = 50 } }
			modifier = { add = -10 has_trait = brave }
		}
	}

	# Own it.
	option = {
		name = cult.0301.d
		add_dread = medium_dread_gain
		cult_power_gain_effect = { VALUE = 20 }
		add_trait_xp = {
			trait = cult_member
			track = cult_rank
			value = 5
		}
		if = {
			limit = {
				exists = scope:cult_hunter
				NOT = { scope:cult_hunter = root }
				any_liege_or_above = { this = scope:cult_hunter }
			}
			custom_tooltip = cult.0301.d.tt_inside
		}
		stress_impact = { craven = medium_stress_impact_gain }
		ai_chance = {
			base = 20
			modifier = { add = 30 has_trait = brave }
			modifier = { add = 20 has_trait = arrogant }
			modifier = { add = 15 cult_rank_at_least_trigger = { RANK = 50 } }
		}
	}
}

# =====================================================================================
#  cult.0303 -- the hunter hears a name
# =====================================================================================

cult.0303 = {
	type = character_event
	title = cult.0303.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:cult_suspect = {
						OR = {
							any_liege_or_above = { this = root }
							is_foreign_court_guest_of = root
							is_pool_guest_of = root
						}
					}
				}
				desc = cult.0303.desc_inside
			}
			desc = cult.0303.desc_outside
		}
	}
	theme = faith
	override_background = { reference = council_chamber }
	left_portrait = {
		character = root
		animation = personality_zealous
	}
	right_portrait = {
		character = scope:cult_suspect
		animation = worry
	}
	trigger = {
		cult_secret_phase_trigger = yes
		exists = scope:cult_suspect
		scope:cult_suspect = {
			is_alive = yes
			has_trait = cult_member
		}
		NOT = { has_character_flag = cult_rooting_out }
		# Still the hunter, and not sworn since the report was sent five days ago.
		cult_is_hunter_trigger = yes
		NOT = { has_trait = cult_member }
	}
	immediate = { cult_save_scopes_effect = yes }

	# Watch them.
	option = {
		name = cult.0303.a
		cult_their_exposure_gain_effect = { WHO = scope:cult_suspect VALUE = 5 }
		ai_chance = { base = 35 }
	}

	# Examine them on the faith.
	option = {
		name = cult.0303.b
		trigger = {
			is_imprisoned = no
			scope:cult_suspect = {
				is_imprisoned = no
				OR = {
					any_liege_or_above = { this = root }
					is_foreign_court_guest_of = root
					is_pool_guest_of = root
				}
			}
		}
		cult_examine_suspect_effect = { SUSPECT = scope:cult_suspect }
		ai_chance = {
			base = 35
			modifier = { add = 20 learning >= 12 }
			modifier = { add = 20 has_trait = zealous }
		}
	}

	# Demand a confession.
	option = {
		name = cult.0303.c
		trigger = {
			is_imprisoned = no
			scope:cult_suspect = {
				is_imprisoned = no
				OR = {
					any_liege_or_above = { this = root }
					is_foreign_court_guest_of = root
					is_pool_guest_of = root
					AND = {
						exists = scope:cult_head
						scope:cult_head = root
						faith = root.faith
					}
				}
			}
		}
		save_scope_as = cult_inquisitor
		custom_tooltip = cult.0303.c.tt
		scope:cult_suspect = { trigger_event = { id = cult.0308 days = 3 } }
		ai_chance = {
			base = 25
			modifier = { add = 20 has_trait = zealous }
			modifier = { add = 10 has_trait = arrogant }
		}
	}

	# Ignore it.
	option = {
		name = cult.0303.d
		ai_chance = {
			base = 15
			modifier = { add = 20 has_trait = trusting }
			modifier = { add = 15 has_trait = cynical }
		}
	}
}

	}

		character = scope:cult_found_member
		animation = worry
	}
	lower_right_portrait = {
		trigger = { exists = scope:cult_spymaster }

	}

		cult_secret_phase_trigger = yes
		exists = scope:cult_found_member
		scope:cult_found_member = {
			is_alive = yes
			has_trait = cult_member
			any_secret = { secret_type = secret_cult_member }
		}
	}
	immediate = {
		cult_save_scopes_effect = yes
		cp:councillor_spymaster ?= { save_scope_as = cult_spymaster }
		cult_reveal_secret_to_root_effect = { WHO = scope:cult_found_member }
		scope:cult_found_member = {
			random_secret = {
				limit = { secret_type = secret_cult_member }
				save_scope_as = cult_found_secret
			}
		}
	}

	# Tell the faithful.
	option = {
		name = cult.0305.a
		trigger = { exists = scope:cult_found_secret }
		scope:cult_found_secret = { expose_secret = root }
		ai_chance = {
			base = 35
			modifier = { add = 20 has_trait = zealous }
			modifier = { add = 15 has_trait = wrathful }
		}
	}

	# Keep it as leverage.
	option = {
		name = cult.0305.b
		if = {
			limit = {
				exists = scope:cult_found_secret
				scope:cult_found_secret = { is_criminal_for = scope:cult_found_member }
				can_add_hook = {
					target = scope:cult_found_member
					type = strong_blackmail_hook
				}
			}
			add_hook = {
				type = strong_blackmail_hook
				target = scope:cult_found_member
				secret = scope:cult_found_secret
			}
		}
		else_if = {
			limit = {
				can_add_hook = {
					target = scope:cult_found_member
					type = suspicious_activity_hook
				}
			}
			add_hook = {
				type = suspicious_activity_hook
				target = scope:cult_found_member
			}
		}
		else = {
			custom_tooltip = cult.0305.b.nohook_tt
		}
		stress_impact = { zealous = minor_stress_impact_gain }
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = deceitful }
			modifier = { add = 15 has_trait = cynical }
		}
	}

	# Offer absolution for names.
	option = {
		name = cult.0305.c
		trigger = {
			scope:cult_found_member = {
				NOT = { cult_is_hierophant_trigger = yes }
			}
		}
		save_scope_as = cult_inquisitor
		custom_tooltip = cult.0305.c.tt
		scope:cult_found_member = { trigger_event = { id = cult.0307 days = 3 } }
		ai_chance = {
			base = 25
			modifier = { add = 20 learning >= 12 }
			modifier = { add = 10 has_trait = forgiving }
		}
	}

	# Watch them.
	option = {
		name = cult.0305.d
		cult_their_exposure_gain_effect = { WHO = scope:cult_found_member VALUE = 5 }
		ai_chance = { base = 10 }
