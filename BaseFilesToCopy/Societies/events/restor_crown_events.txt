# The line has ended. The Keeper chooses: take up another house, or let the oath end with the blood.
restor.0811 = {
	type = character_event
	title = restor.0811.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:restor_candidate }
				desc = restor.0811.desc
			}
			desc = restor.0811.desc_nobody
		}
	}
	theme = death
	left_portrait = {
		character = root
		animation = grief
	}
	right_portrait = {
		trigger = { exists = scope:restor_candidate }
		character = scope:restor_candidate
		animation = personality_bold
	}
	trigger = { restor_active_trigger = yes }

	immediate = {
		restor_save_scopes_effect = yes
		# The strongest sworn lord of the crown's own people, with a house of their own to give.
		ordered_in_global_list = {
			variable = restor_roster
			limit = {
				is_alive = yes
				is_adult = yes
				is_landed = yes
				exists = house
			}
			order_by = {
				value = prestige
				if = {
					limit = { restor_holds_crown_land_trigger = yes }
					add = 1000
				}
			}
			save_scope_as = restor_candidate
		}
	}

	# The cause goes on under a new house.
	option = {
		name = restor.0811.a
		trigger = { exists = scope:restor_candidate }
		scope:restor_candidate = {
			set_global_variable = {
				name = restor_house
				value = house
			}
			restor_make_pretender_effect = yes
		}
		# The new house may already wear the crown -- the strongest sworn lord often does. Then the
		# society has not lost its aim, it has just achieved it: no title changes hands, so
		# on_title_gain never fires, and the triumph has to be called here.
		if = {
			limit = {
				exists = global_var:restor_crown.holder
				global_var:restor_crown.holder = { restor_of_the_blood_trigger = yes }
			}
			global_var:restor_crown.holder = { restor_triumph_effect = yes }
		}
		else = {
			restor_recompute_usurper_effect = yes
			hidden_effect = {
				restor_notify_sworn_effect = { EVENT = restor.0810 DAYS = 3 }
			}
		}
		ai_chance = { base = 90 }
	}

	# The oath was to the blood, and the blood is gone.
	option = {
		name = restor.0811.b
		custom_tooltip = restor.0811.b.tt
		hidden_effect = {
			# Copied first: restor_leave_effect removes each member from the list being walked,
			# which would skip every other one and leave half the Sworn sworn to nothing.
			every_in_global_list = {
				variable = restor_roster
				add_to_temporary_list = restor_disbanded
			}
			every_in_list = {
				list = restor_disbanded
				restor_leave_effect = yes
			}
			clear_global_variable_list = restor_roster
			remove_global_variable = restor_crown
			if = {
				limit = { exists = global_var:restor_keeper }
				remove_global_variable = restor_keeper
			}
			if = {
				limit = { exists = global_var:restor_usurper }
				remove_global_variable = restor_usurper
			}
		}
		ai_chance = { base = 10 }
	}
}
