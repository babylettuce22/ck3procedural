namespace = cult

# The inversion cult: the Black Mass's random pool. cult.1000-1099.
#
# Drawn during the Mass (common/activities/activity_types/00_cult_mass_activity.txt) by the weekly
# pulse, through common/on_action/00_cult_mass_on_actions.txt: the host from the host pool, each
# player guest from the guest pool. Not every Mass gets every one.
#
#   HOST (root = host; AI hosts too)
#   1001   The Candle Gutters       a candle goes out at the wrong word
#   1002   The Initiate at the Door a new member panics and makes for the door
#   1003   The Chaplain             secret: a lord and the chaplain who hears their confession, both here
#   1004   A Knock at the Door      secret: someone knocks in the middle of the Mass
#   1005     Who Was at the Door    follow-up to 1004: a clerk, the village priest, or the hunter's rider
#   1006   The Wrong Hymn           the Faithless sing the true hymn from habit
#   1007   A Sign                   something happens that the circle reads as an answer
#   1008   Two Books                two senior Faithless, each with their own book of the rite
#   1009   The Offering             a guest lays something on the altar
#   1010   The Quiet Guest          secret: a guest who watches faces and says no responses
#   1011   The Preacher in Town     secret: the hunter's preacher, preaching every morning after
#   1012   The Founder's Tomb       the Mass said over the grave of the pious forebear who built the chapel
#   1020   The Doors Stand Open     unveiled: the town comes to watch
#   1021   The Old Priest           unveiled: a priest of the old way says the true Mass on the step
#   1022   A Late Convert           unveiled: a vassal who wept in church last year wants to be sworn
#
#   GUEST (root = a player guest)
#   1030   A Face Under a Hood      someone you know is here
#   1031   The Hierophant's Word    the Hierophant stops you on the stairs
#   1032   A Letter from Home       secret: your spouse writes, and asks where you are
#   1033   Cold Feet                the member beside you wants out
#   1034   The Lesson               the host has you read the lesson backwards
#
# Every host event marks the activity (cult_mass_pool_mark_effect) so it cannot repeat in one Mass,
# and the host for five years, which its weight_multiplier reads so the next Mass leans elsewhere.
# Every guest event sets a two-month character flag. Every roll ends in a toast.

# =====================================================================================
#  cult.1001 -- The Candle Gutters

# =====================================================================================
#  cult.1003 -- The Chaplain
# =====================================================================================
#
# A lord at the Mass whose own court chaplain is at the Mass too. The lord may be the host.

cult.1003 = {
	type = character_event
	title = cult.1003.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:cult_mass_chaplain_lord = root }
				desc = cult.1003.desc_yours
			}
			desc = cult.1003.desc_theirs
		}
		desc = cult.1003.desc_end
	}
	theme = faith
	left_portrait = {
		character = root
		animation = scheme
	}
	right_portrait = {
		character = scope:cult_mass_chaplain
		animation = chaplain
	}
	lower_right_portrait = {
		trigger = { NOT = { scope:cult_mass_chaplain_lord = root } }
		character = scope:cult_mass_chaplain_lord
	}
	weight_multiplier = {
		base = 1
		modifier = {
			has_character_flag = cult_mass_had_1003
			factor = 0.25
		}
	}
	trigger = {
		exists = scope:activity
		cult_secret_phase_trigger = yes
		NOT = { scope:activity = { has_variable = cult_mass_pool_1003 } }
		scope:activity = {
			any_attending_character = {
				exists = cp:councillor_court_chaplain
				cp:councillor_court_chaplain = {
					NOT = { this = root }
					involved_activity ?= scope:activity
				}
			}
		}
	}

	immediate = {
		cult_save_scopes_effect = yes
		cult_mass_pool_mark_effect = { EVENT = 1003 }
		scope:activity = {
			random_attending_character = {
				limit = {
					exists = cp:councillor_court_chaplain
					cp:councillor_court_chaplain = {
						NOT = { this = root }
						involved_activity ?= scope:activity
					}
				}
				save_scope_as = cult_mass_chaplain_lord
				cp:councillor_court_chaplain = { save_scope_as = cult_mass_chaplain }
			}
		}
	}

	# Say it with me at the altar.
	option = {
		name = cult.1003.a
		cult_power_gain_effect = { VALUE = 10 }
		cult_corrupt_soul_effect = yes
		scope:cult_mass_chaplain = {
			add_trait_xp = {
				trait = cult_member
				track = cult_rank
				value = 10
			}
		}
		cult_their_exposure_gain_effect = { WHO = scope:cult_mass_chaplain VALUE = 2 }
		ai_chance = {
			base = 40
			modifier = { add = 30 has_trait = arrogant }
		}
	}

	# Ask the chaplain what their lord confesses.
	option = {
		name = cult.1003.b
		trigger = { NOT = { scope:cult_mass_chaplain_lord = root } }
		random_list = {
			75 = {
				modifier = { add = 15 intrigue >= 12 }
				if = {
					limit = {
						scope:cult_mass_chaplain_lord = {
							any_secret = { NOT = { is_known_by = root } }
						}
					}
					send_interface_toast = {
						type = event_toast_effect_good
						title = cult.1003.b.tells
						left_icon = scope:cult_mass_chaplain
						right_icon = scope:cult_mass_chaplain_lord
						custom_tooltip = cult.1003.b.tells.tt
						hidden_effect = {
							scope:cult_mass_chaplain_lord = {
								random_secret = {
									limit = { NOT = { is_known_by = root } }
									reveal_to = root
								}
							}
						}
					}
				}
				else = {
					send_interface_toast = {
						type = event_toast_effect_neutral
						title = cult.1003.b.nothing
						left_icon = scope:cult_mass_chaplain
						right_icon = scope:cult_mass_chaplain_lord
						custom_tooltip = cult.1003.b.nothing.tt
					}
				}
			}
			25 = {
				send_interface_toast = {
					type = event_toast_effect_bad
					title = cult.1003.b.told
					left_icon = scope:cult_mass_chaplain_lord
					scope:cult_mass_chaplain_lord = {
						add_opinion = {
							target = root
							modifier = angry_opinion
							opinion = -20
						}
					}
				}
			}
		}
		stress_impact = {
			honest = minor_stress_impact_gain
			just = minor_stress_impact_gain
		}
		ai_chance = {
			base = 30
			modifier = { add = 30 has_trait = deceitful }
			modifier = { add = 20 has_trait = schemer }
			modifier = { add = -20 has_trait = honest }
		}
	}

	# Everyone here has a double life.
	option = {
		name = cult.1003.c
		scope:cult_mass_chaplain = {
			add_opinion = {
				target = root
				modifier = trust_opinion
				opinion = 10
			}
		}
		if = {
			limit = { NOT = { scope:cult_mass_chaplain_lord = root } }
			scope:cult_mass_chaplain_lord = {
				add_opinion = {
					target = root
					modifier = trust_opinion
					opinion = 10
				}
			}
		}
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = paranoid }
		}
	}
}
