# The Restorationists: the Gathering's random pool. restor.1000-1099.
#
# Drawn during the Gathering (common/activities/activity_types/00_restor_gathering_activity.txt) by
# the phase's on_weekly_pulse, through the on_actions in
# common/on_action/00_restor_gathering_pool_on_actions.txt -- vanilla's funeral_wake_pulse pattern.
# The spine (restor.0901-0908) is the same every time; these are what make one Gathering unlike the
# last.
#
#   Host pool -- once per Gathering each (var restor_pool_NNNN on the activity)
#   1001   Riders on the Road        a late arrival may have been followed
#   1002   Strike Now or Wait        two of the Sworn quarrel over the plan in front of you
#   1003   A Doubt at the Table      one guest says another stopped at the usurper's inn
#   1004   The Heir Among Us         the pretender has come in person, and asks to speak
#   1005   A Servant at the Door     one of your servants heard too much
#   1006   To the Last King          the toast to the dead king is yours to give
#   1007   A Letter from the Heir    the pretender, absent, has written to the hall
#   1008   An Uninvited Guest        a member brings a courtier nobody asked for
#   1009   News of the Usurper       at war, ailing, or watching the roads
#   1010   Counting Swords           the lords pledge men for the rising
#   1011   A Stranger in Town        someone is asking whose horses fill your stable
#   1012   Released from the Oath    a frightened member asks to be let go
#   1013   The Headman at the Gate   a village of the old land asks if the heir is coming
#   1014   The Keeper's Account      the Keeper wants to hear what you have done
#
#   Guest pool -- a 90-day flag each, so a guest does not see the same scene twice
#   1020   A Word in Private         the host asks you to carry letters
#   1021   Asked to Vouch            a suspected member wants you to speak up for them
#   1022   By the Fire               the heir or the Keeper, and nobody else awake
#   1023   A Face You Know           someone you have no love for is sworn too
#   1024   Too Much Wine             the old song, in the yard, at midnight
#
#   Follow-up
#   1030   The Watchers Report       from 1003: what the suspect did after the Gathering
#
# scope:activity and scope:host ride along from the phase's weekly pulse. Every pool event requires
# both in its trigger: they are scenes in a hall full of people and mean nothing outside one.
# Every random outcome ends in a toast (or, for 1003's watchers, a follow-up event that rolls in its
# immediate and says what it found).

namespace = restor

# =====================================================================================
#  HOST POOL
# =====================================================================================

# ---- restor.1001 -- Riders on the Road ---------------------------------------------------------
restor.1001 = {
	type = character_event
	title = restor.1001.t
	desc = restor.1001.desc
	theme = secret
	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		character = scope:restor_late
		animation = fear
	}
	trigger = {
		restor_underground_trigger = yes
		exists = scope:activity
		exists = scope:host
		this = scope:host
		scope:activity = {
			NOT = { has_variable = restor_pool_1001 }
			any_attending_character = {
				NOT = { this = scope:host }
				has_trait = restor_member
			}
		}
	}

	immediate = {
		restor_save_scopes_effect = yes
		scope:activity = {
			set_variable = { name = restor_pool_1001 }
			random_attending_character = {
				limit = {
					NOT = { this = scope:host }
					has_trait = restor_member
				}
				save_scope_as = restor_late
			}
		}
	}

	# Send men back down the road.
	option = {
		name = restor.1001.a
		random_list = {
			50 = {
				modifier = { add = 10 martial >= 12 }
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = restor.1001.a.empty
					left_icon = root
					custom_tooltip = restor.1001.a.empty.tt
				}
			}
			30 = {
				modifier = { add = 10 martial >= 14 }
				modifier = { add = 10 intrigue >= 14 }
				send_interface_toast = {
					type = event_toast_effect_good
					title = restor.1001.a.caught
					left_icon = root
					custom_tooltip = restor.1001.a.caught.tt
					restor_exposure_loss_effect = { VALUE = 2 }
				}
			}
			20 = {
				modifier = { add = 10 martial < 8 }
				send_interface_toast = {
					type = event_toast_effect_bad
					title = restor.1001.a.lost
					left_icon = root
					right_icon = scope:restor_late
					restor_exposure_gain_effect = { VALUE = 3 }
					restor_their_exposure_gain_effect = { WHO = scope:restor_late VALUE = 2 }
				}
			}
		}
		ai_chance = {
			base = 40
			modifier = { add = 20 has_trait = brave }
			modifier = { add = 20 has_trait = paranoid }
		}
	}

	# Bar the doors and say nothing.
	option = {
		name = restor.1001.b
		restor_exposure_gain_effect = { VALUE = 2 }
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = craven }
			modifier = { add = 10 has_trait = lazy }
		}
	}

	# Blame the one who led them here.
	option = {
		name = restor.1001.c
		restor_their_exposure_gain_effect = { WHO = scope:restor_late VALUE = 3 }
		scope:restor_late = {
			add_opinion = {
				target = root
				modifier = restor_life_scolded_me_opinion
			}
		}
		stress_impact = {
			compassionate = minor_stress_impact_gain
		}
		ai_chance = {
			base = 20
			modifier = { add = 25 has_trait = arrogant }
			modifier = { add = 20 has_trait = wrathful }
			modifier = { add = -15 has_trait = compassionate }
		}
	}
}

# ---- restor.1002 -- Strike Now or Wait -----------------------------------------------------------
#
# Two guests: one who wants the standard raised this year, one who wants to wait. Siding with either
# makes a friend and a grudge; handing it to the heir is a diplomacy roll that calms both or neither.
restor.1002 = {
	type = character_event
	title = restor.1002.t
	desc = restor.1002.desc
	theme = secret
	left_portrait = {
		character = scope:restor_hawk
		animation = anger
	}
	right_portrait = {
		character = scope:restor_dove
		animation = disapproval
	}
	lower_center_portrait = root
	trigger = {
		restor_underground_trigger = yes
		exists = scope:activity
		exists = scope:host
		this = scope:host
		scope:activity = {
			NOT = { has_variable = restor_pool_1002 }
			any_attending_character = {
				count >= 2
				NOT = { this = scope:host }
				has_trait = restor_member
			}
		}
	}

	immediate = {
		restor_save_scopes_effect = yes
		scope:activity = {
			set_variable = { name = restor_pool_1002 }
			random_attending_character = {
				limit = {
					NOT = { this = scope:host }
					has_trait = restor_member
				}
				weight = {
					base = 1
					modifier = { add = 4 has_trait = brave }
					modifier = { add = 4 has_trait = ambitious }
					modifier = { add = 3 has_trait = impatient }
					modifier = { add = 3 has_trait = wrathful }
				}
				save_scope_as = restor_hawk
			}
			random_attending_character = {
				limit = {
					NOT = { this = scope:host }
					NOT = { this = scope:restor_hawk }
					has_trait = restor_member
				}
				weight = {
					base = 1
					modifier = { add = 4 has_trait = craven }
					modifier = { add = 4 has_trait = patient }
					modifier = { add = 3 has_trait = paranoid }
					modifier = { add = 3 has_trait = calm }
				}
				save_scope_as = restor_dove
			}
		}
	}

	# The standard this year.
	option = {
		name = restor.1002.a
		scope:restor_hawk = {
			add_opinion = {
				target = root
				modifier = restor_life_took_my_side_opinion
			}
			restor_rank_gain_effect = { VALUE = 3 }
		}
		scope:restor_dove = {
			add_opinion = {
				target = root
				modifier = restor_life_took_their_side_opinion
			}
		}
		restor_exposure_gain_effect = { VALUE = 2 }
		stress_impact = {
			patient = minor_stress_impact_gain
			craven = minor_stress_impact_gain
		}
		ai_chance = {
			base = 35
			modifier = { add = 25 has_trait = brave }
			modifier = { add = 20 has_trait = ambitious }
			modifier = { add = 15 has_trait = impatient }
		}
	}

	# Wait.
	option = {
		name = restor.1002.b
		scope:restor_dove = {
			add_opinion = {
				target = root
				modifier = restor_life_took_my_side_opinion
			}
		}
		scope:restor_hawk = {
			add_opinion = {
				target = root
				modifier = restor_life_took_their_side_opinion
			}
		}
		restor_exposure_loss_effect = { VALUE = 1 }
		stress_impact = {
			impatient = minor_stress_impact_gain
			brave = minor_stress_impact_gain
		}
		ai_chance = {
			base = 35
			modifier = { add = 25 has_trait = craven }
			modifier = { add = 25 has_trait = patient }
		}
	}

	# The heir decides, not us.
	option = {
		name = restor.1002.c
		trigger = {
			exists = scope:restor_pretender
			NOT = { scope:restor_pretender = root }
		}
		random_list = {
			60 = {
				modifier = { add = 10 diplomacy >= 12 }
				modifier = { add = 10 diplomacy >= 16 }
				send_interface_toast = {
					type = event_toast_effect_good
					title = restor.1002.c.calm
					left_icon = scope:restor_hawk
					right_icon = scope:restor_dove
					scope:restor_hawk = {
						add_opinion = {
							target = root
							modifier = restor_life_kept_the_peace_opinion
						}
					}
					scope:restor_dove = {
						add_opinion = {
							target = root
							modifier = restor_life_kept_the_peace_opinion
						}
					}
				}
			}
			40 = {
				modifier = { add = 10 diplomacy < 8 }
				send_interface_toast = {
					type = event_toast_effect_bad
					title = restor.1002.c.sour
					left_icon = scope:restor_hawk
					right_icon = scope:restor_dove
					scope:restor_hawk = {
						add_opinion = {
							target = root
							modifier = annoyed_opinion
							opinion = -10
						}
					}
					scope:restor_dove = {
						add_opinion = {
							target = root
							modifier = annoyed_opinion
							opinion = -10
						}
					}
				}
			}
		}
		ai_chance = {
			base = 30
			modifier = { add = 25 has_trait = diplomat }
			modifier = { add = 15 has_trait = humble }
		}
	}

# ---- restor.1003 -- A Doubt at the Table ---------------------------------------------------------
#
# The accused is chosen with a lean toward the careless (high exposure), the deceitful and those who
# think little of the host, so the accusation is sometimes right. The public questioning is an
# intrigue roll; the quiet watch costs favour and reports later (restor.1030).
restor.1003 = {
	type = character_event
	title = restor.1003.t
	desc = {
		desc = restor.1003.desc
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:restor_usurper }
				desc = restor.1003.desc_usurper
			}
			desc = restor.1003.desc_nousurper
		}
		desc = restor.1003.desc_end
	}
	theme = secret
	left_portrait = {
		character = scope:restor_accuser
		animation = paranoia
	}
	right_portrait = {
		character = scope:restor_suspect
		animation = idle
	}
	lower_center_portrait = root
	trigger = {
		restor_underground_trigger = yes
		exists = scope:activity
		exists = scope:host
		this = scope:host
		scope:activity = {
			NOT = { has_variable = restor_pool_1003 }
			any_attending_character = {
				count >= 2
				NOT = { this = scope:host }
				has_trait = restor_member
			}
		}
	}

	immediate = {
		restor_save_scopes_effect = yes
		scope:activity = {
			set_variable = { name = restor_pool_1003 }
			random_attending_character = {
				limit = {
					NOT = { this = scope:host }
					has_trait = restor_member
				}
				weight = {
					base = 1
					modifier = {
						add = 4
						has_variable = restor_exposure
						var:restor_exposure >= 10
					}
					modifier = { add = 3 has_trait = deceitful }
					modifier = {
						add = 3
						opinion = {
							target = scope:host
							value < 0
						}
					}
				}
				save_scope_as = restor_suspect
			}
			random_attending_character = {
				limit = {
					NOT = { this = scope:host }
					NOT = { this = scope:restor_suspect }
					has_trait = restor_member
				}
				weight = {
					base = 1
					modifier = { add = 4 has_trait = paranoid }
					modifier = { add = 2 has_trait = honest }
				}
				save_scope_as = restor_accuser
			}
		}
	}

	# Put it to them in front of everyone.
	option = {
		name = restor.1003.a
		random_list = {
			55 = {
				modifier = { add = 10 has_trait = honest }
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = restor.1003.a.answer
					left_icon = scope:restor_suspect
					custom_tooltip = restor.1003.a.answer.tt
					scope:restor_suspect = {
						add_opinion = {
							target = root
							modifier = hurt_opinion
							opinion = -20
						}
						add_opinion = {
							target = scope:restor_accuser
							modifier = angry_opinion
							opinion = -20
						}
					}
				}
			}
			30 = {
				modifier = { add = 10 intrigue >= 12 }
				modifier = { add = 10 intrigue >= 16 }
				send_interface_toast = {
					type = event_toast_effect_good
					title = restor.1003.a.loose
					left_icon = scope:restor_suspect
					custom_tooltip = restor.1003.a.loose.tt
					restor_their_exposure_gain_effect = { WHO = scope:restor_suspect VALUE = 5 }
					scope:restor_suspect = {
						restor_rank_loss_effect = { VALUE = 10 }
					}
					restor_exposure_loss_effect = { VALUE = 2 }
				}
			}
			15 = {
				send_interface_toast = {
					type = event_toast_effect_bad
					title = restor.1003.a.walks
					left_icon = scope:restor_suspect
					scope:restor_suspect = {
						add_opinion = {
							target = root
							modifier = restor_named_before_the_sworn_opinion
						}
					}
					restor_their_exposure_gain_effect = { WHO = scope:restor_suspect VALUE = 3 }
				}
			}
		}
		ai_chance = {
			base = 30
			modifier = { add = 25 has_trait = paranoid }
			modifier = { add = 15 has_trait = wrathful }
			modifier = { add = -15 has_trait = trusting }
		}
	}

	# Have them watched afterwards.
	option = {
		name = restor.1003.b
		trigger = {
			has_variable = restor_favour
			var:restor_favour >= 10
		}
		restor_favour_spend_effect = { VALUE = 10 }
		custom_tooltip = restor.1003.b.tt
		hidden_effect = {
			trigger_event = {
				id = restor.1030
				days = { 30 60 }
			}
		}
		ai_chance = {
			base = 30
			modifier = { add = 25 has_trait = deceitful }
			modifier = { add = 15 has_trait = patient }
		}
	}

	# Drop it.
	option = {
		name = restor.1003.c
		scope:restor_accuser = {
			add_opinion = {
				target = root
				modifier = disappointed_opinion
				opinion = -10
			}
		}
		stress_impact = {
			paranoid = minor_stress_impact_gain
		}
		ai_chance = {
			base = 30
			modifier = { add = 30 has_trait = trusting }
			modifier = { add = 15 has_trait = forgiving }
		}
	}
}

# ---- restor.1004 -- The Heir Among Us ------------------------------------------------------------
restor.1004 = {
	type = character_event
	title = restor.1004.t
	desc = restor.1004.desc
	theme = secret
	left_portrait = {
		character = root
		animation = personality_honorable
	}
	right_portrait = {
		character = scope:restor_pretender
		animation = personality_bold
	}
	trigger = {
		restor_underground_trigger = yes
		exists = scope:activity
		exists = scope:host
		this = scope:host
		exists = global_var:restor_pretender
		NOT = { this = global_var:restor_pretender }
		scope:activity = {
			NOT = { has_variable = restor_pool_1004 }
			any_attending_character = { this = global_var:restor_pretender }
		}
	}

	immediate = {
		restor_save_scopes_effect = yes
		scope:activity = { set_variable = { name = restor_pool_1004 } }
	}

	# Let the heir speak.
	option = {
		name = restor.1004.a
		scope:activity = {
			every_attending_character = {
				limit = {
					has_trait = restor_member
					NOT = { this = scope:restor_pretender }
				}
				add_opinion = {
					target = scope:restor_pretender
					modifier = respect_opinion
					opinion = 10
				}
				restor_rank_gain_effect = { VALUE = 3 }
			}
		}
		scope:restor_pretender = { add_prestige = minor_prestige_gain }
		restor_exposure_gain_effect = { VALUE = 2 }
		ai_chance = {
			base = 45
			modifier = { add = 20 has_trait = zealous }
			modifier = { add = 20 has_trait = gregarious }
		}
	}

	# Better that fewer of them see the heir's face.
	option = {
		name = restor.1004.b
		scope:restor_pretender = {
			add_opinion = {
				target = root
				modifier = disappointed_opinion
				opinion = -10
			}
		}
		restor_exposure_loss_effect = { VALUE = 2 }
		ai_chance = {
			base = 30
			modifier = { add = 30 has_trait = paranoid }
			modifier = { add = 20 has_trait = craven }
		}
	}

	# Kneel first, so they all see it.
	option = {
		name = restor.1004.c
		scope:restor_pretender = {
			add_opinion = {
				target = root
				modifier = restor_life_served_the_house_opinion
			}
		}
		restor_rank_gain_effect = { VALUE = 5 }
		restor_exposure_gain_effect = { VALUE = 3 }
		stress_impact = {
			arrogant = medium_stress_impact_gain
		}
		ai_chance = {
			base = 25
			modifier = { add = 25 has_trait = humble }
			modifier = { add = 15 has_trait = honest }
			modifier = { add = -20 has_trait = arrogant }

		character = root
		animation = spymaster
	}
	right_portrait = {
		trigger = {
			exists = scope:restor_suspect
			scope:restor_suspect = { is_alive = yes }
		}
		character = scope:restor_suspect
		animation = idle
	}
	trigger = {
		restor_active_trigger = yes
		has_trait = restor_member
	}

	immediate = {
		restor_save_scopes_effect = yes
		if = {
			limit = {
				OR = {
					NOT = { exists = scope:restor_suspect }
					scope:restor_suspect = {
						OR = {
							is_alive = no
							NOT = { has_trait = restor_member }
						}
					}
				}
			}
			save_scope_value_as = { name = restor_watch_result value = flag:gone }
		}
		else = {
			random_list = {
				60 = {
					modifier = { add = 15 scope:restor_suspect = { has_trait = honest } }
					save_scope_value_as = { name = restor_watch_result value = flag:clean }
				}
				40 = {
					modifier = { add = 15 scope:restor_suspect = { has_trait = deceitful } }
					modifier = {
						add = 15
						scope:restor_suspect = {
							has_variable = restor_exposure
							var:restor_exposure >= 10
						}
					}
					save_scope_value_as = { name = restor_watch_result value = flag:loose }
				}
			}
		}
	}

	# Loose talk: stop it.
	option = {
		name = restor.1030.a
		trigger = { scope:restor_watch_result ?= flag:loose }
		restor_their_exposure_gain_effect = { WHO = scope:restor_suspect VALUE = 5 }
		scope:restor_suspect = {
			restor_rank_loss_effect = { VALUE = 10 }
		}
		restor_exposure_loss_effect = { VALUE = 3 }
		add_intrigue_lifestyle_xp = minor_lifestyle_xp
	}

	# Clean: a small apology owed, in private.
	option = {
		name = restor.1030.b
		trigger = { scope:restor_watch_result ?= flag:clean }
		restor_rank_gain_effect = { VALUE = 2 }
		stress_impact = {
			paranoid = minor_stress_impact_loss
		}
	}

	# Gone.
	option = {
		name = restor.1030.c
		trigger = { scope:restor_watch_result ?= flag:gone }
		restor_favour_gain_effect = { VALUE = 5 }
	}

	# Nothing matched (should not happen): close the window.
	option = {
		name = restor.1030.d
		fallback = yes
	}
}
