# The Restorationists: the Gathering and the Errand. restor.0900-0999.
#
# THE GATHERING (common/activities/activity_types/00_restor_gathering_activity.txt)
#   0901   host, day 1       The Sworn Gather: what the Gathering is for
#   0902   host              Plan the Rising: which counties to stir
#   0903   host              An Empty Seat: judge a member who was asked and stayed away
#   0904   host, day 17      Raise One of the Sworn: someone's standing, or your own
#   (the random pool drawn during the Gathering is restor.1000-1099, restor_gathering_pool_events.txt)
#   0905   each guest        Arrival: who is in the hall, and who is not
#   0906   each guest        what the host decided
#   0907   the absentee      named before the Sworn, or forgiven
#   0908   host, half the time   Talk in Town: someone outside noticed the riders
#
# THE ERRAND (common/activities/activity_types/00_restor_errand_activity.txt)
#   0910   host, day 1       Setting Out
#   0911   each guest        Riding with the host
#   0912   host, day 10      On the Road: careful or hasty
#   0913   host, day 18      The Witness: one of the usurper's people sees too much
#   0914   host, day 26      Home: the ONLY event of the Errand that pays
#   0915   host              An Old Soldier: swear the veteran in, or wait
#
# scope:activity and scope:host ride along from the activity's phase block. Every read of
# scope:activity is guarded (`?=`, or `exists`), so any of these can be fired from the DebugPanel
# outside an activity: each then degrades to the host alone instead of throwing "Undefined event
# target" once per render.

namespace = restor

# =====================================================================================
#  THE GATHERING
# =====================================================================================

# ---- restor.0901 -- The Sworn Gather ---------------------------------------------------------
#
# The host chooses the purpose after seeing who came, because who came decides what the Gathering
# can do: Plan the Rising needs crown land in or next to the hall, Judge the Absent needs somebody
# absent.
restor.0901 = {
	type = character_event
	title = restor.0901.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:restor_gathering_count > 0 }
				desc = restor.0901.desc
			}
			desc = restor.0901.desc_alone
		}
		triggered_desc = {
			trigger = { exists = scope:restor_absentee }
			desc = restor.0901.desc_absent
		}
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:restor_pretender
					NOT = { scope:restor_pretender = root }
					exists = scope:activity
					scope:activity = { any_attending_character = { this = scope:restor_pretender } }
				}
				desc = restor.0901.desc_heir
			}
			triggered_desc = {
				trigger = { scope:restor_usurper ?= { is_at_war = yes } }
				desc = restor.0901.desc_war
			}
		}
		desc = restor.0901.desc_end
	}
	theme = secret
	left_portrait = {
		character = root
		animation = personality_rational
	}
	right_portrait = {
		trigger = { exists = scope:restor_absentee }
		character = scope:restor_absentee
		animation = worry
	}
	trigger = { restor_active_trigger = yes }

	immediate = {
		restor_save_scopes_effect = yes

		save_scope_value_as = {
			name = restor_gathering_count
			value = {
				value = 0
				if = {
					limit = { exists = scope:activity }
					scope:activity = {
						every_attending_character = {
							limit = { NOT = { this = root } }
							add = 1
						}
					}
				}
			}
		}

		# Who is here.
		if = {
			limit = { exists = scope:activity }
			scope:activity = {
				every_attending_character = { add_to_list = restor_present }
			}
		}

		# Who was asked and is not. The activity's own record first (see on_start in the
		# activity file); failing that, a sworn member of the host's court or realm.
		if = {
			limit = {
				exists = scope:activity
				scope:activity = { has_variable_list = restor_gathering_invited }
			}
			scope:activity = {
				random_in_list = {
					variable = restor_gathering_invited
					limit = {
						is_alive = yes
						has_trait = restor_member
						NOT = { this = root }
						NOT = { is_in_list = restor_present }
					}
					save_scope_as = restor_absentee
				}
			}
		}
		if = {
			limit = { NOT = { exists = scope:restor_absentee } }
			random_courtier = {
				limit = {
					is_adult = yes
					has_trait = restor_member
					NOT = { is_in_list = restor_present }
				}
				save_scope_as = restor_absentee
			}
		}
		if = {
			limit = { NOT = { exists = scope:restor_absentee } }
			random_vassal_or_below = {
				limit = {
					is_adult = yes
					has_trait = restor_member
					NOT = { is_in_list = restor_present }
				}
				save_scope_as = restor_absentee
			}
		}

		# Whether Plan the Rising has anywhere to reach.
		restor_gathering_rising_lists_effect = yes
		restor_gathering_rising_counts_effect = yes
	}

	# Renew the oath.
	option = {
		name = restor.0901.a
		restor_gathering_reward_hall_effect = { XP = 5 FAVOUR = 15 }
		hidden_effect = {
			restor_gathering_tell_guests_effect = { PURPOSE = oath }
			trigger_event = { id = restor.0904 days = 16 }
		}
		ai_chance = {
			base = 40
			modifier = { add = 20 has_trait = gregarious }
			modifier = { add = 20 has_trait = honest }
		}
	}

	# Plan the rising.
	option = {
		name = restor.0901.b
		trigger = {
			OR = {
				scope:restor_rising_own_count > 0
				scope:restor_rising_near_count > 0
			}
		}
		custom_tooltip = restor.0901.b.tt
		hidden_effect = {
			trigger_event = { id = restor.0902 days = 2 }
			trigger_event = { id = restor.0904 days = 16 }
		}
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = ambitious }
			modifier = { add = 20 restor_is_pretender_trigger = yes }
			modifier = { add = -20 has_trait = craven }
		}
	}

	# Judge the absent.
	option = {
		name = restor.0901.c
		trigger = { exists = scope:restor_absentee }
		custom_tooltip = restor.0901.c.tt
		hidden_effect = {
			trigger_event = { id = restor.0903 days = 2 }
			trigger_event = { id = restor.0904 days = 16 }
		}
		ai_chance = {
			base = 15
			modifier = { add = 25 has_trait = vengeful }
			modifier = { add = 15 has_trait = arbitrary }
			modifier = { add = -10 has_trait = forgiving }
		}
	}
}

# ---- restor.0902 -- Plan the Rising -----------------------------------------------------------
#
# One county per landed lord in the hall (one at least, five at most), chosen from the crown's
# counties that are not yet stirred. Their own land is the safe choice and costs them their
# subjects' goodwill (the modifier lowers county opinion of whoever holds it); their neighbours'
# land costs someone else that goodwill, and the host a good deal more exposure.
restor.0902 = {
	type = character_event
	title = restor.0902.t
	desc = restor.0902.desc
	theme = secret
	left_portrait = {
		character = root
		animation = personality_bold
	}
	right_portrait = {
		trigger = { exists = scope:restor_pretender }
		character = scope:restor_pretender
	}
	trigger = { restor_active_trigger = yes }

	immediate = {
		restor_save_scopes_effect = yes
		restor_gathering_rising_lists_effect = yes
		restor_gathering_rising_counts_effect = yes
	}

	# Our own land.
	option = {
		name = restor.0902.a
		trigger = { scope:restor_rising_own_count > 0 }
		custom_tooltip = restor.0902.a.tt
		restor_exposure_gain_effect = { VALUE = 2 }
		hidden_effect = {
			restor_gathering_rising_lists_effect = yes
			restor_gathering_stir_list_effect = { LIST = restor_rising_own }
			send_interface_toast = {
				type = event_toast_effect_good
				title = restor.0902.toast
				left_icon = root
				custom_tooltip = restor.0902.toast.tt
			}
			restor_gathering_tell_guests_effect = { PURPOSE = rising }
		}
		ai_chance = {
			base = 50
			modifier = { add = 20 has_trait = craven }
			modifier = { add = 20 has_trait = paranoid }
		}
	}

	# Our neighbours' land.
	option = {
		name = restor.0902.b
		trigger = { scope:restor_rising_near_count > 0 }
		custom_tooltip = restor.0902.b.tt
		restor_exposure_gain_effect = { VALUE = 5 }
		hidden_effect = {
			restor_gathering_rising_lists_effect = yes
			restor_gathering_stir_list_effect = { LIST = restor_rising_near }
			send_interface_toast = {
				type = event_toast_effect_good
				title = restor.0902.toast
				left_icon = root
				custom_tooltip = restor.0902.toast.tt
			}
			restor_gathering_tell_guests_effect = { PURPOSE = rising }
		}
		ai_chance = {
			base = 40
			modifier = { add = 25 has_trait = brave }
			modifier = { add = 15 has_trait = ambitious }
		}
	}

	# Not yet.
	option = {
		name = restor.0902.c
		add_trait_xp = {
			trait = restor_member
			track = restor_rank
			value = 2
		}
		hidden_effect = {
			restor_gathering_tell_guests_effect = { PURPOSE = quiet }
		}
		ai_chance = {
			base = 10
			modifier = { add = 30 has_trait = craven }
		}
	}
}

# ---- restor.0903 -- An Empty Seat --------------------------------------------------------------
#
# scope:restor_absentee comes from restor.0901. Judging costs the absent member standing and makes
# an enemy of them; the host thinks the worse of them too. Letting it go costs nothing and earns
# some gratitude. Either way the absentee is told (restor.0907).
restor.0903 = {
	type = character_event
	title = restor.0903.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:restor_absentee }
				desc = restor.0903.desc
			}
			desc = restor.0903.desc_none
		}
	}
	theme = secret
	left_portrait = {
		character = root
		animation = disapproval
	}
	right_portrait = {
		trigger = { exists = scope:restor_absentee }
		character = scope:restor_absentee
		animation = shame
	}
	trigger = { restor_active_trigger = yes }

	immediate = {
		restor_save_scopes_effect = yes
	}

	# Name them.
	option = {
		name = restor.0903.a
		trigger = { exists = scope:restor_absentee }
		scope:restor_absentee = {
			add_trait_xp = {
				trait = restor_member
				track = restor_rank
				value = -15
			}
			add_opinion = {
				target = root
				modifier = restor_named_before_the_sworn_opinion
			}
		}
		add_opinion = {
			target = scope:restor_absentee
			modifier = restor_failed_charge_opinion
		}
		if = {
			limit = { exists = scope:activity }
			scope:activity = {
				every_attending_character = {
					limit = {
						NOT = { this = root }
						has_trait = restor_member
					}
					add_trait_xp = {
						trait = restor_member
						track = restor_rank
						value = 3
					}
				}
			}
		}
		hidden_effect = {
			save_scope_value_as = { name = restor_judged_harshly value = yes }
			scope:restor_absentee = { trigger_event = { id = restor.0907 days = 5 } }
			restor_gathering_tell_guests_effect = { PURPOSE = judged }
		}
		stress_impact = {
			forgiving = medium_stress_impact_gain
			compassionate = minor_stress_impact_gain
		}
		ai_chance = {
			base = 50
			modifier = { add = 30 has_trait = vengeful }
			modifier = { add = 15 has_trait = arbitrary }
			modifier = { add = -40 has_trait = forgiving }
		}
	}

	# Let it go.
	option = {
		name = restor.0903.b
		trigger = { exists = scope:restor_absentee }
		scope:restor_absentee = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 10
			}
		}
		hidden_effect = {
			scope:restor_absentee = { trigger_event = { id = restor.0907 days = 5 } }
			restor_gathering_tell_guests_effect = { PURPOSE = forgiven }
		}
		stress_impact = {
			vengeful = minor_stress_impact_gain
		}
		ai_chance = {
			base = 50
			modifier = { add = 40 has_trait = forgiving }
			modifier = { add = 20 has_trait = compassionate }
		}
	}

	# Nobody to judge (fired outside a Gathering, or the absentee has since died). `fallback`, so
	# the event can never render with no option to close it.
	option = {
		name = restor.0903.c
		fallback = yes
		restor_gathering_reward_hall_effect = { XP = 3 FAVOUR = 5 }
	}
}

		character = root
