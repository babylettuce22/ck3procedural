# The inversion cult: charges. cult.0500-0599.
#
#   0501        the Hierophant gives you a charge (accept / decline)
#   0510-0514   a charge kept: turn a priest, taint a county, defile a holy site, an offering, recruit
#   0520        a charge's three years ran out
#   0530-0534   a chance to do the charge in hand, one per kind
#
# The state and every transition are in 00_cult_charge_effects.txt; these events only show them.
# A charge is cleared before its ending event arrives, so the endings read the saved scopes
# (scope:cult_charge_target, scope:cult_charge_county, scope:cult_charge_kind), not the variables.

namespace = cult

# =====================================================================================
#  cult.0501 -- a charge from the Hierophant
# =====================================================================================

cult.0501 = {
	type = character_event
	title = cult.0501.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:cult_charge_kind ?= flag:turn_priest
					exists = scope:cult_charge_court
				}
				desc = cult.0501.desc_chaplain
			}
			triggered_desc = {
				trigger = { scope:cult_charge_kind ?= flag:turn_priest }
				desc = cult.0501.desc_theocrat
			}
			triggered_desc = {
				trigger = { scope:cult_charge_kind ?= flag:taint }
				desc = cult.0501.desc_taint
			}
			triggered_desc = {
				trigger = { scope:cult_charge_kind ?= flag:defile_site }
				desc = cult.0501.desc_defile_site
			}
			triggered_desc = {
				trigger = { scope:cult_charge_kind ?= flag:offering }
				desc = cult.0501.desc_offering
			}
			desc = cult.0501.desc_recruit
		}
	}
	theme = secret
	override_background = { reference = temple }
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		trigger = { exists = scope:cult_hierophant }
		character = scope:cult_hierophant
		animation = scheme
	}
	lower_right_portrait = {
		trigger = { exists = scope:cult_charge_target }
		character = scope:cult_charge_target
	}
	trigger = {
		cult_secret_phase_trigger = yes
		has_variable = cult_charge
		has_variable = cult_charge_given
		exists = global_var:cult_hierophant
	}
	immediate = {
		cult_save_scopes_effect = yes
		cult_charge_save_state_effect = yes
	}

	option = {
		name = cult.0501.a
		if = {
			limit = { scope:cult_charge_kind ?= flag:turn_priest }
			custom_tooltip = cult.0501.a.turn_priest_tt
		}
		else_if = {
			limit = { scope:cult_charge_kind ?= flag:taint }
			custom_tooltip = cult.0501.a.taint_tt
		}
		else_if = {
			limit = { scope:cult_charge_kind ?= flag:defile_site }
			custom_tooltip = cult.0501.a.defile_site_tt
		}
		else_if = {
			limit = { scope:cult_charge_kind ?= flag:offering }
			custom_tooltip = cult.0501.a.offering_tt
		}
		else = {
			custom_tooltip = cult.0501.a.recruit_tt
		}
		ai_chance = { base = 100 }
	}

	option = {
		name = cult.0501.b
		cult_charge_penalty_effect = { POWER = 10 }
		custom_tooltip = cult.0501.b.tt
		hidden_effect = {
			if = {
				limit = { has_variable = cult_charge }
				cult_charge_clear_effect = yes
			}
		}
		ai_chance = { base = 0 }
	}
}

# =====================================================================================
#  cult.0530 -- turn a priest: the questions after prayers
# =====================================================================================

cult.0530 = {
	type = character_event
	title = cult.0530.t
	desc = cult.0530.desc
	theme = faith
	override_background = { reference = temple }
	left_portrait = {
		character = root
		animation = scheme
	}
	right_portrait = {
		character = scope:cult_charge_target
		animation = thinking
	}
	trigger = {
		cult_secret_phase_trigger = yes
		has_variable = cult_charge
		var:cult_charge = flag:turn_priest
		exists = var:cult_charge_target
		var:cult_charge_target = {
			is_alive = yes
			is_imprisoned = no
			NOT = { has_trait = cult_member }
		}
	}
	immediate = {
		cult_save_scopes_effect = yes
		cult_charge_save_state_effect = yes
	}

	# Tell the truth.
	option = {
		name = cult.0530.a
		random_list = {
			30 = {
				modifier = { add = 15 intrigue >= 12 }
				modifier = { add = 15 intrigue >= 16 }
				modifier = {
					add = 25
					scope:cult_charge_target = { cult_sinner_trigger = yes }
				}
				modifier = {
					add = 15
					scope:cult_charge_target = { has_trait = cynical }
				}
				modifier = {
					add = 20
					scope:cult_charge_target = {
						opinion = {
							target = root
							value >= 30
						}
					}
				}
				send_interface_toast = {
					type = event_toast_effect_good
					title = cult.0530.success
					left_icon = root
					right_icon = scope:cult_charge_target
					cult_charge_swear_priest_effect = { PRIEST = scope:cult_charge_target }
				}
			}
			50 = {
				modifier = {
					add = 25
					scope:cult_charge_target = { has_trait = zealous }
				}
				modifier = {
					add = 15
					scope:cult_charge_target = {
						opinion = {
							target = root
							value < 0
						}
					}
				}
				send_interface_toast = {
					type = event_toast_effect_bad
					title = cult.0530.failure
					left_icon = root
					right_icon = scope:cult_charge_target
					scope:cult_charge_target = {
						add_opinion = {
							target = root
							modifier = cult_refused_the_offer_opinion
						}
					}
					cult_exposure_gain_effect = { VALUE = 8 }
				}
			}
		}
		hidden_effect = { cult_charge_check_effect = yes }
		ai_chance = { base = 50 }
	}

	# Let the dark make the argument.
	option = {
		name = cult.0530.b
		trigger = {
			has_variable = cult_power
			var:cult_power >= 30
		}
		cult_power_spend_effect = { VALUE = 30 }
		cult_corrupt_soul_effect = yes
		random_list = {
			60 = {
				modifier = {
					add = 25
					scope:cult_charge_target = { cult_sinner_trigger = yes }
				}
				modifier = {
					add = 15
					scope:cult_charge_target = { has_trait = cynical }
				}
				send_interface_toast = {
					type = event_toast_effect_good
					title = cult.0530.success
					left_icon = root
					right_icon = scope:cult_charge_target
					cult_charge_swear_priest_effect = { PRIEST = scope:cult_charge_target }
				}
			}
			40 = {
				modifier = {
					add = 25
					scope:cult_charge_target = { has_trait = zealous }
				}
				send_interface_toast = {
					type = event_toast_effect_bad
					title = cult.0530.failure
					left_icon = root
					right_icon = scope:cult_charge_target
					scope:cult_charge_target = {
						add_opinion = {
							target = root
							modifier = cult_refused_the_offer_opinion
						}
					}
					cult_exposure_gain_effect = { VALUE = 5 }
				}
			}
		}
		hidden_effect = { cult_charge_check_effect = yes }
		ai_chance = { base = 30 }
	}

	option = {
		name = cult.0530.c
		custom_tooltip = cult.0530.c.tt
		ai_chance = { base = 20 }
	}
}
