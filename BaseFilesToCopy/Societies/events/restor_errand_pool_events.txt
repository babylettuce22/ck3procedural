# THE ERRAND'S RANDOM POOL (restor.1100-1199)
#
# Drawn by common/on_action/00_restor_errand_pool_on_actions.txt, which the Errand's phase fires on
# day 5 (the road) and days 14 and 22 (the county). Host only. Every event pays for itself: the
# spine's own payment in restor.0914 is untouched by anything here.
#
#   THE ROAD (restor_errand_road_pool)
#     1101  The Bridge              the local lord's men check every traveller      (foreign county)
#     1102  Men in the Road         robbers who hold their spears like soldiers
#     1103  The Song at the Inn     the forbidden song of the last king
#     1104  Followed                a man in a brown cloak
#     1105  A Loose Tongue          a companion talks too much                        (companion)
#     1106  Fellow Travellers       pilgrims make good cover
#     1107  Letters in the Saddlebag  a patrol searches bags
#
#   THE COUNTY (restor_errand_county_pool)
#     1110  The Old King's Face     an old man with a coin asks if the line lives
#     1111  The Tax Men             the lord's collectors take a family's cow         (foreign county)
#     1112  The Garrison Sergeant   unpaid since spring                               (foreign county)
#     1113  The Soldier's Widow     a banner in a chest
#     1114  The Keeper of the Tombs the priest who swept the old kings' tombs
#     1115  The Empty Mill          a sworn contact who does not come
#     1116  A Second County         a neighbouring county asks for someone           (unstirred neighbour)
#     1117  The Proclamation        a reward for news of the house's friends          (foreign county)
#     1118  A Village Wedding       the old toast
#     1119  The Merchant's Price    coin only
#     1120  The Sergeant's Test     soldiers only
#     1121  The Headman's Doubt     stir only
#     1122  Names for Sale          the lord's informers, for twenty gold
#     1123  An Old Acquaintance     a companion is recognised                         (companion)
#
# Every event requires scope:activity (the on_action passes it), so none of these fires outside an
# Errand. Each sets restor_errand_seen_NNNN for 60 days: one Errand never draws the same event
# twice. Every random outcome ends in a toast, including the ones where nothing happens.

namespace = restor

#  THE ROAD
# =====================================================================================

# ---- restor.1101 -- The Bridge -------------------------------------------------------------------
restor.1101 = {
	type = character_event
	title = restor.1101.t
	desc = restor.1101.desc
	theme = secret
	override_background = { reference = terrain_travel }
	left_portrait = {
		character = root
		animation = worry
	}
	trigger = {
		exists = scope:activity
		restor_underground_trigger = yes
		restor_errand_foreign_county_trigger = yes
		NOT = { has_character_flag = restor_errand_seen_1101 }
	}
	immediate = {
		restor_save_scopes_effect = yes
		restor_errand_pool_scopes_effect = yes
		add_character_flag = { flag = restor_errand_seen_1101 days = 60 }
	}

	# The cover story.
	option = {
		name = restor.1101.a
		skill = intrigue
		duel = {
			skill = intrigue
			value = decent_skill_rating
			50 = {
				desc = restor.1101.a.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}
				send_interface_toast = {
					type = event_toast_effect_good
					title = restor.1101.a.success
					left_icon = root
					custom_tooltip = restor.1101.a.success.tt
				}
			}
			50 = {
				desc = restor.1101.a.failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -49
				}
				send_interface_toast = {
					type = event_toast_effect_bad
					title = restor.1101.a.failure
					left_icon = root
					restor_exposure_gain_effect = { VALUE = 2 }
				}
			}
		}
		ai_chance = {
			base = 40
			modifier = { add = 30 has_trait = deceitful }
		}
	}

	# Pay.
	option = {
		name = restor.1101.b
		trigger = { gold >= 15 }
		remove_short_term_gold = 15
		custom_tooltip = restor.1101.b.tt
		ai_chance = {
			base = 30
			modifier = { add = -20 has_trait = greedy }
		}
	}

	# The ford.
	option = {
		name = restor.1101.c
		random_list = {
			75 = {
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = restor.1101.c.dry
					left_icon = root
					custom_tooltip = restor.1101.c.dry.tt
				}
			}
			25 = {
				send_interface_toast = {
					type = event_toast_effect_bad
					title = restor.1101.c.wet
					left_icon = root
					add_stress = minor_stress_gain
				}
			}
		}
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = paranoid }
		}
	}
}

			skill = prowess

					value = scope:duel_value

					type = event_toast_effect_good
					title = restor.1102.a.success
					left_icon = root

				}
			}

				desc = restor.1102.a.failure

					value = scope:duel_value

				}

					type = event_toast_effect_bad
					title = restor.1102.a.failure
					left_icon = root

						limit = { gold >= 20 }

					}
				}
			}
		}
		ai_chance = {
			base = 30
			modifier = { add = 30 has_trait = brave }
			modifier = { add = -25 has_trait = craven }
		}

		name = restor.1102.b
		trigger = { gold >= 20 }
		remove_short_term_gold = 20
		custom_tooltip = restor.1102.b.tt
		ai_chance = {
			base = 40
			modifier = { add = 30 has_trait = craven }
		}
	}

	# Ask who taught them.
	option = {
		name = restor.1102.c
		random_list = {
			50 = {
				modifier = {
					add = 20
					restor_errand_for_soldiers_trigger = yes
				}
				modifier = {
					add = 15
					martial >= 12
				}
				send_interface_toast = {
					type = event_toast_effect_good
					title = restor.1102.c.success
					left_icon = root
					restor_favour_gain_effect = { VALUE = 5 }
					restor_exposure_gain_effect = { VALUE = 2 }
					if = {
						limit = { restor_errand_for_soldiers_trigger = yes }
						add_trait_xp = {
							trait = restor_member
							track = restor_rank
							value = 3
						}
					}
				}
			}
			50 = {
				send_interface_toast = {
					type = event_toast_effect_bad
					title = restor.1102.c.failure
					left_icon = root
					restor_exposure_gain_effect = { VALUE = 2 }
					if = {
						limit = { gold >= 20 }
						remove_short_term_gold = 20
					}
					else = {
						increase_wounds_no_death_effect = { REASON = beaten }
					}
				}
			}
		}
		ai_chance = {
			base = 20
			modifier = { add = 30 restor_errand_for_soldiers_trigger = yes }
		}
	}
