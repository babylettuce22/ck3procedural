# The Restorationists: everyday life among the Sworn. restor.0200-0299.
#
# Fired by restor_life_pulse (common/on_action/00_restor_life_on_actions.txt), once a year at most
# and usually not at all. These are what make being sworn feel like belonging to something: small
# scenes that can only happen to someone sworn to THIS crown and THIS house.
#
#   Underground
#   0201   the old king's coin turns up in your treasury
#   0202   an old soldier of the house at your gate knows the Sworn's greeting
#   0203   a letter from the heir, asking for help with the household
#   0204   the anniversary of the fall; the Sworn keep it in secret
#   0205   a singer sings of the house in your hall, with the usurper's knight at table     (random)
#   0206   your heir asks where you go at night                                               (random)
#   0207   two of the Sworn quarrel and the Keeper asks you to settle it                     (random)
#   0208   a young courtier who wants in and cannot keep quiet                               (random)
#   0209   the usurper's tax collector in a county of the crown -- AI too                    (random)
#   0210   the tombs of the old kings at the old seat are to be cleared                      (random)
#   0211   a member has died; the Sworn bury them quietly                                    (random)
#   0212   a sworn lord stays the night and tells old stories                                (random)
#   0213   the old royal banner for sale -- AI too                                           (random)
#   0214   a night of doubt; stay, lie low, or walk away
#
#   Restored
#   0250   the king: two of the old Sworn want the same reward
#   0251   an agent of the deposed house tries to buy you                                    (random)
#   0252   the old Sworn are named at a feast, and you are asked to speak                    (random)
#   0253   a latecomer who never swore wants you to sponsor them
#
# Every option with a roll ends each branch in a toast. Each event's immediate sets a twelve-year
# restor_life_seen_NNNN flag that its trigger reads, so a long life does not repeat itself.

namespace = restor

# =====================================================================================
#  restor.0201 -- The Old King's Coin
# =====================================================================================

restor.0201 = {
	type = character_event
	title = restor.0201.t
	desc = {
		desc = restor.0201.desc
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:restor_pretender NOT = { scope:restor_pretender = root } }
				desc = restor.0201.desc_heir
			}
			desc = restor.0201.desc_you
		}
	}
	theme = stewardship
	override_background = { reference = study }
	left_portrait = {
		character = root
		animation = interested
	}
	right_portrait = {
		trigger = { exists = scope:restor_life_steward }
		character = scope:restor_life_steward
		animation = steward
	}

	trigger = {
		restor_underground_trigger = yes
		is_ai = no
		is_landed = yes
		NOT = { has_character_flag = restor_life_seen_0201 }
	}

	immediate = {
		restor_save_scopes_effect = yes
		add_character_flag = { flag = restor_life_seen_0201 years = 12 }
		cp:councillor_steward ?= { save_scope_as = restor_life_steward }
	}

	# Carry it.
	option = {
		name = restor.0201.a
		add_character_modifier = {
			modifier = restor_life_kings_coin_modifier
			years = 10
		}
		restor_exposure_gain_effect = { VALUE = 2 }
	}

	# Send it to the heir.
	option = {
		name = restor.0201.b
		trigger = {
			exists = scope:restor_pretender
			NOT = { scope:restor_pretender = root }
		}
		scope:restor_pretender = {
			add_opinion = {
				target = root
				modifier = restor_life_served_the_house_opinion
			}
		}
		restor_favour_gain_effect = { VALUE = 10 }
	}

	# Melt it down.
	option = {
		name = restor.0201.c
		add_gold = 5
		stress_impact = {
			zealous = minor_stress_impact_gain
		}
	}
}

# =====================================================================================
#  restor.0202 -- The Old Soldier
# =====================================================================================

restor.0202 = {
	type = character_event
	title = restor.0202.t
	desc = restor.0202.desc
	theme = martial
	override_background = { reference = courtyard }
	left_portrait = {
		character = root
		animation = shock
	}
	right_portrait = {
		character = scope:restor_life_veteran
		animation = beg
	}

	trigger = {
		restor_underground_trigger = yes
		is_ai = no
		is_landed = yes
		exists = global_var:restor_seat
		NOT = { has_character_flag = restor_life_seen_0202 }
	}

	immediate = {
		restor_save_scopes_effect = yes
		add_character_flag = { flag = restor_life_seen_0202 years = 12 }
		create_character = {
			template = old_knight_teacher
			culture = global_var:restor_seat.culture
			faith = global_var:restor_seat.faith
			gender_female_chance = root_soldier_female_chance
			location = root.capital_province
			save_scope_as = restor_life_veteran
		}
	}

	# Take them in.
	option = {
		name = restor.0202.a
		add_courtier = scope:restor_life_veteran
		scope:restor_life_veteran = {
			add_opinion = {
				target = root
				modifier = restor_life_took_me_in_opinion
			}
		}
		restor_favour_gain_effect = { VALUE = 5 }
		random_list = {
			75 = {
				send_interface_toast = {
					type = event_toast_effect_good
					title = restor.0202.a.quiet
					left_icon = scope:restor_life_veteran
					custom_tooltip = restor.0202.a.quiet.tt
				}
			}
			25 = {
				send_interface_toast = {
					type = event_toast_effect_bad
					title = restor.0202.a.talks
					left_icon = scope:restor_life_veteran
					restor_exposure_gain_effect = { VALUE = 4 }
				}
			}
		}
		stress_impact = {
			greedy = minor_stress_impact_gain
		}
	}

	# Coin, and the road.
	option = {
		name = restor.0202.b
		trigger = { gold >= minor_gold_value }
		remove_short_term_gold = minor_gold_value
	}

	# Whip them out of the gate.
	option = {
		name = restor.0202.c
		restor_exposure_loss_effect = { VALUE = 2 }
		restor_favour_spend_effect = { VALUE = 5 }
		stress_impact = {
			compassionate = medium_stress_impact_gain
			just = minor_stress_impact_gain
		}
	}
}
