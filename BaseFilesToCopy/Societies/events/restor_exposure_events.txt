# The Restorationists: exposure, and the usurper's side. restor.0300-0399.
#
# The member's side
#   0300   exposure crossed 15: under suspicion, a findable secret now exists
#   0301   the member's secret was exposed: everyone knows
#   0302   exposure crossed 25: highly suspect, the usurper is being told
#   0307   the usurper offers a pardon for names
#   0312   the usurper is rooting out the Sworn in their realm
#   0330-0333  yearly, exposure 10+: a servant, a watched meeting place, a spouse, a moneylender
#
# The usurper's side
#   0303   a name is suspected (a member crossed 25)
#   0305   the spymaster found a member's secret
#   0306   a member's secret was exposed
#   0310   Root Out the Sworn names the most exposed member of the realm
#   0320-0324  yearly: an informer, the old songs, a silent priest, intercepted letters, a toast
#
# "Inside the usurper's realm" is `any_liege_or_above = usurper`, or a guest at the usurper's own
# court. That is also where secret_restor_member is criminal, so it is where arrest makes sense.

namespace = restor

# =====================================================================================
#  restor.0300 -- under suspicion
# =====================================================================================

restor.0300 = {
	type = character_event
	title = restor.0300.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:restor_usurper
					NOT = { scope:restor_usurper = root }
				}
				desc = restor.0300.desc
			}
			desc = restor.0300.desc_nousurper
		}
	}
	theme = intrigue
	override_background = { reference = corridor_night }
	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		trigger = {
			exists = scope:restor_usurper
			NOT = { scope:restor_usurper = root }
		}
		character = scope:restor_usurper
		animation = scheme
	}
	trigger = {
		restor_underground_trigger = yes
		has_trait = restor_member
	}
	immediate = { restor_save_scopes_effect = yes }

	# Lie low.
	option = {
		name = restor.0300.a
		if = {
			limit = {
				exists = scope:restor_usurper
				NOT = { scope:restor_usurper = root }
			}
			custom_tooltip = restor.0300.a.tt
		}
		else = { custom_tooltip = restor.0300.a.tt_nousurper }
		ai_chance = { base = 70 }
	}

	# Burn the letters.
	option = {
		name = restor.0300.b
		trigger = { restor_can_afford_favour_trigger = { VALUE = 10 } }
		show_as_unavailable = { always = yes }
		restor_exposure_loss_effect = { VALUE = 4 }
		restor_favour_spend_effect = { VALUE = 10 }
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = paranoid }
		}
	}
}

# =====================================================================================
#  restor.0301 -- the member's secret is exposed
# =====================================================================================

restor.0301 = {
	type = character_event
	title = restor.0301.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:restor_usurper
					NOT = { scope:restor_usurper = root }
					any_liege_or_above = { this = scope:restor_usurper }
				}
				desc = restor.0301.desc_inside
			}
			triggered_desc = {
				trigger = {
					exists = scope:restor_usurper
					NOT = { scope:restor_usurper = root }
				}
				desc = restor.0301.desc_outside
			}
			desc = restor.0301.desc_nousurper
		}
	}
	theme = secret
	override_background = { reference = throne_room }
	left_portrait = {
		character = root
		animation = shock
	}
	right_portrait = {
		trigger = {
			exists = scope:restor_usurper
			NOT = { scope:restor_usurper = root }
		}
		character = scope:restor_usurper
		animation = anger
	}
	lower_right_portrait = {
		trigger = { exists = scope:restor_refuge }
		character = scope:restor_refuge
	}
	trigger = {
		restor_underground_trigger = yes
		has_trait = restor_member
	}
	immediate = {
		restor_save_scopes_effect = yes
		# A sworn lord outside the usurper's reach, for a courtier to run to.
		if = {
			limit = {
				is_ruler = no
				is_imprisoned = no
			}
			random_in_global_list = {
				variable = restor_roster
				limit = {
					is_alive = yes
					is_landed = yes
					is_imprisoned = no
					NOT = { this = root }
					NOT = { restor_is_usurper_trigger = yes }
					OR = {
						NOT = { exists = global_var:restor_usurper }
						NOT = { any_liege_or_above = { this = global_var:restor_usurper } }
					}
				}
				save_scope_as = restor_refuge
			}
		}
	}

	# Deny it.
	option = {
		name = restor.0301.a
		duel = {
			skill = intrigue
			value = decent_skill_rating
			45 = {
				desc = restor.0301.a.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -40
				}
				modifier = { add = 10 has_trait = deceitful }
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = restor.0301.a.success
					left_icon = root
					add_prestige = minor_prestige_loss
				}
				restor_exposure_loss_effect = { VALUE = 5 }
			}
			55 = {
				desc = restor.0301.a.failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -50
				}
				modifier = { add = 10 has_trait = honest }
				send_interface_toast = {
					type = event_toast_effect_bad
					title = restor.0301.a.failure
					left_icon = root
					add_prestige = medium_prestige_loss
				}
				restor_exposure_gain_effect = { VALUE = 5 }
			}
		}
		stress_impact = { honest = minor_stress_impact_gain }
		ai_chance = {
			base = 30
			modifier = { add = 25 has_trait = deceitful }
			modifier = { add = -20 has_trait = honest }
		}
	}

	# Run to a sworn court.
	option = {
		name = restor.0301.b
		trigger = { exists = scope:restor_refuge }
		scope:restor_refuge = { add_courtier = root }
		restor_exposure_loss_effect = { VALUE = 5 }
		custom_tooltip = restor.0301.b.tt
		ai_chance = {
			base = 40
			modifier = { add = 30 has_trait = craven }
		}
	}

	# Confess, and name names.
	option = {
		name = restor.0301.c
		trigger = {
			exists = scope:restor_usurper
			NOT = { scope:restor_usurper = root }
			NOT = { restor_is_pretender_trigger = yes }
			NOT = { restor_is_keeper_trigger = yes }
		}
		restor_name_others_effect = yes
		custom_tooltip = restor.0301.c.tt
		hidden_effect = { restor_leave_effect = yes }
		scope:restor_usurper = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 20
			}
			hidden_effect = {
				if = {
					limit = { is_ai = no }
					send_interface_toast = {
						type = event_toast_effect_good
						title = restor_confessed_to_you_toast
						left_icon = root
					}
				}
			}
		}
		if = {
			limit = {
				exists = scope:restor_keeper
				NOT = { scope:restor_keeper = root }
			}
			scope:restor_keeper = {
				add_opinion = {
					target = root
					modifier = restor_betrayed_the_cause_opinion
				}
			}
		}
		stress_impact = {
			honest = minor_stress_impact_gain
			just = minor_stress_impact_gain
			brave = medium_stress_impact_gain
		}
		ai_chance = {
			base = 15
			modifier = { add = 30 has_trait = craven }
			modifier = { add = 15 has_trait = deceitful }
			modifier = { add = -15 restor_rank_at_least_trigger = { RANK = 50 } }
			modifier = { add = -10 has_trait = brave }
		}
	}

	# Own it.
	option = {
		name = restor.0301.d
		add_prestige = medium_prestige_gain
		restor_favour_gain_effect = { VALUE = 20 }
		add_trait_xp = {
			trait = restor_member
			track = restor_rank
			value = 5
		}
		if = {
			limit = {
				exists = scope:restor_usurper
				NOT = { scope:restor_usurper = root }
				any_liege_or_above = { this = scope:restor_usurper }
			}
			custom_tooltip = restor.0301.d.tt_inside
		}
		stress_impact = { craven = medium_stress_impact_gain }
		ai_chance = {
			base = 20
			modifier = { add = 30 has_trait = brave }
			modifier = { add = 20 has_trait = zealous }
			modifier = { add = 15 restor_rank_at_least_trigger = { RANK = 50 } }
		}
	}
}

# =====================================================================================
#  restor.0305 -- the spymaster finds a secret
# =====================================================================================

restor.0305 = {
	type = character_event
	title = restor.0305.t
	desc = restor.0305.desc
	theme = intrigue
	override_background = { reference = study }
	left_portrait = {
		character = root
		animation = schadenfreude
	}
	right_portrait = {
		character = scope:restor_found_member
		animation = worry
	}
	lower_right_portrait = {
		trigger = { exists = scope:restor_spymaster }
		character = scope:restor_spymaster
	}
	trigger = {
		restor_underground_trigger = yes
		exists = scope:restor_found_member
		scope:restor_found_member = {
			is_alive = yes
			has_trait = restor_member
			any_secret = { secret_type = secret_restor_member }
		}
	}
	immediate = {
		restor_save_scopes_effect = yes
		cp:councillor_spymaster ?= { save_scope_as = restor_spymaster }
		restor_reveal_secret_to_root_effect = { WHO = scope:restor_found_member }
		scope:restor_found_member = {
			random_secret = {
				limit = { secret_type = secret_restor_member }
				save_scope_as = restor_found_secret
			}
		}
	}

	# Tell everyone.
	option = {
		name = restor.0305.a
		trigger = { exists = scope:restor_found_secret }
		scope:restor_found_secret = { expose_secret = root }
		ai_chance = {
			base = 35
			modifier = { add = 20 has_trait = vengeful }
			modifier = { add = 15 has_trait = wrathful }
		}
	}

	# Keep it as leverage.
	option = {
		name = restor.0305.b
		if = {
			limit = {
				exists = scope:restor_found_secret
				scope:restor_found_secret = { is_criminal_for = scope:restor_found_member }
				can_add_hook = {
					target = scope:restor_found_member
					type = strong_blackmail_hook
				}
			}
			add_hook = {
				type = strong_blackmail_hook
				target = scope:restor_found_member
				secret = scope:restor_found_secret
			}
		}
		else_if = {
			limit = {
				can_add_hook = {
					target = scope:restor_found_member
					type = suspicious_activity_hook
				}
			}
			add_hook = {
				type = suspicious_activity_hook
				target = scope:restor_found_member
			}
		}
		else = {
			custom_tooltip = restor.0305.b.nohook_tt
		}
		ai_chance = {
			base = 35
			modifier = { add = 20 has_trait = deceitful }
			modifier = { add = 15 has_trait = patient }
		}
	}

	# Offer a pardon for names.
	option = {
		name = restor.0305.c
		trigger = {
			scope:restor_found_member = {
				NOT = { restor_is_pretender_trigger = yes }
				NOT = { restor_is_keeper_trigger = yes }
			}
		}
		save_scope_as = restor_turner
		custom_tooltip = restor.0305.c.tt
		scope:restor_found_member = { trigger_event = { id = restor.0307 days = 3 } }
		ai_chance = {
			base = 25
			modifier = { add = 20 diplomacy >= 12 }
		}
	}

	# Watch them.
	option = {
		name = restor.0305.d
		restor_their_exposure_gain_effect = { WHO = scope:restor_found_member VALUE = 5 }
		ai_chance = { base = 10 }
	}
}
