namespace = cult

# The inversion cult: the Black Mass and the Pilgrimage of Defilement. cult.0900-0999.
#
# THE BLACK MASS (common/activities/activity_types/00_cult_mass_activity.txt)
#   0901   host, day 1       The Black Mass: what the Mass is for (Invoke, Profane, Initiate)
#   0902   host              The Profanation: which counties to taint
#   0903   host              The Initiation: raise one of the circle, or swear in a courtier
#   0904   host, day 6       An Empty Place: judge a member who was asked and stayed away
#   0905   each guest        Arrival: who is in the chapel, and who is not
#   0906   each guest        what the host decided
#   0907   the absentee      named before the circle, or excused
#   0908   host, half the time, secret phases only   Talk in Town
#
# THE PILGRIMAGE OF DEFILEMENT (common/activities/activity_types/00_cult_defilement_activity.txt)
#   0910   host, day 1       Setting Out: the pilgrim's robe
#   0911   each guest        On the Pilgrim Road
#   0912   host, day 10      Among the Faithful: careful or hasty
#   0913   host, day 18      Seen at the Shrine: the site's priest, or the hunter's people
#   0914   host, day 26      Home: the ONLY event of the Pilgrimage that pays
#   0915   host              The Stolen Relic: burn it, or "find" it
#   0916   the site's holder (a player)   The Shrine Defiled / A Relic Stolen
#
# scope:activity and scope:host ride along from the activity's phase block. Every read of
# scope:activity is guarded (`?=`, or `exists`), so any of these can be fired from the DebugPanel
# outside an activity: each then degrades to the host alone.

# =====================================================================================
#  THE BLACK MASS
# =====================================================================================

# ---- cult.0901 -- The Black Mass ---------------------------------------------------------------
#
# The host chooses the purpose after seeing who came: Profane needs host-faith land in or next to
# the chapel's lords, Initiate needs somebody to raise or a courtier to bring. Invoke is always
# there. A member who stayed away is judged on day six (cult.0904) whatever is chosen.
cult.0901 = {
	type = character_event
	title = cult.0901.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:cult_mass_count > 0
					cult_unveiled_trigger = yes
				}
				desc = cult.0901.desc_open
			}
			triggered_desc = {
				trigger = { scope:cult_mass_count > 0 }
				desc = cult.0901.desc
			}
			desc = cult.0901.desc_alone
		}
		triggered_desc = {
			trigger = { exists = scope:cult_absentee }
			desc = cult.0901.desc_absent
		}
		# Loc in cult_mass_pool_l_english.yml.
		triggered_desc = {
			trigger = {
				cult_secret_phase_trigger = yes
				exists = scope:cult_hunter
				target_is_liege_or_above = scope:cult_hunter
			}
			desc = cult.0901.desc_hunter_liege
		}
		desc = cult.0901.desc_end
	}
	theme = witchcraft
	left_portrait = {
		character = root
		animation = scheme
	}
	right_portrait = {
		trigger = { exists = scope:cult_absentee }
		character = scope:cult_absentee
		animation = worry
	}
	trigger = { cult_active_trigger = yes }

	immediate = {
		cult_save_scopes_effect = yes

		save_scope_value_as = {
			name = cult_mass_count
			value = {
				value = 0
				if = {
					limit = { exists = scope:activity }
					scope:activity = {
						every_attending_character = {
							limit = { NOT = { this = root } }
							add = 1
						}
					}
				}
			}
		}

		# Who is here.
		if = {
			limit = { exists = scope:activity }
			scope:activity = {
				every_attending_character = { add_to_list = cult_mass_present }
			}
		}

		# Who was asked and is not. The activity's own record first; failing that, a sworn member
		# of the host's court or realm.
		if = {
			limit = {
				exists = scope:activity
				scope:activity = { has_variable_list = cult_mass_invited }
			}
			scope:activity = {
				random_in_list = {
					variable = cult_mass_invited
					limit = {
						is_alive = yes
						has_trait = cult_member
						NOT = { this = root }
						NOT = { is_in_list = cult_mass_present }
					}
					save_scope_as = cult_absentee
				}
			}
		}
		if = {
			limit = { NOT = { exists = scope:cult_absentee } }
			random_courtier = {
				limit = {
					is_adult = yes
					has_trait = cult_member
					NOT = { is_in_list = cult_mass_present }
				}
				save_scope_as = cult_absentee
			}
		}
		if = {
			limit = { NOT = { exists = scope:cult_absentee } }
			random_vassal_or_below = {
				limit = {
					is_adult = yes
					has_trait = cult_member
					NOT = { is_in_list = cult_mass_present }
				}
				save_scope_as = cult_absentee
			}
		}

		# Whether the Profanation has anywhere to reach, and the Initiation anyone to raise.
		cult_mass_profane_lists_effect = yes
		cult_mass_profane_counts_effect = yes
		cult_mass_find_initiates_effect = yes
	}

	# Invoke.
	option = {
		name = cult.0901.a
		custom_tooltip = cult.0901.a.tt
		cult_mass_invoke_effect = { POWER = 20 XP = 3 }
		hidden_effect = {
			cult_mass_tell_guests_effect = { PURPOSE = invoke }
			if = {
				limit = { exists = scope:cult_absentee }
				trigger_event = { id = cult.0904 days = 6 }
			}
		}
		stress_impact = {
			compassionate = minor_stress_impact_gain
		}
		ai_chance = {
			base = 40
			modifier = { add = 20 has_trait = ambitious }
			modifier = { add = 20 has_trait = arrogant }
		}
	}

	# Profane.
	option = {
		name = cult.0901.b
		trigger = {
			OR = {
				scope:cult_profane_own_count > 0
				scope:cult_profane_near_count > 0
			}
		}
		custom_tooltip = cult.0901.b.tt
		hidden_effect = {
			trigger_event = { id = cult.0902 days = 2 }
			if = {
				limit = { exists = scope:cult_absentee }
				trigger_event = { id = cult.0904 days = 6 }
			}
		}
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = callous }
			modifier = { add = 20 has_trait = sadistic }
			modifier = { add = -20 has_trait = craven }
		}
	}

	# Initiate.
	option = {
		name = cult.0901.c
		trigger = {
			OR = {
				exists = scope:cult_mass_candidate
				exists = scope:cult_mass_recruit
			}
		}
		custom_tooltip = cult.0901.c.tt
		hidden_effect = {
			trigger_event = { id = cult.0903 days = 2 }
			if = {
				limit = { exists = scope:cult_absentee }
				trigger_event = { id = cult.0904 days = 6 }
			}
		}
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = gregarious }
			modifier = { add = 20 cult_is_hierophant_trigger = yes }
		}
	}
}

# ---- cult.0903 -- The Initiation ---------------------------------------------------------------
#
# scope:cult_mass_candidate and scope:cult_mass_recruit come from cult.0901 (found again here if the
# event was fired from the console). The recruit is a courtier the host brought down to the chapel
# under some pretext; sworn in on the spot, they owe the host for it.

	type = character_event
	title = cult.0903.t
	desc = {
		desc = cult.0903.desc
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:cult_mass_candidate }
				desc = cult.0903.desc_candidate
			}
			desc = cult.0903.desc_nocandidate
		}
		triggered_desc = {
			trigger = { exists = scope:cult_mass_recruit }
			desc = cult.0903.desc_recruit
		}
		desc = cult.0903.desc_end
	}
	theme = witchcraft
	left_portrait = {
		character = root
		animation = personality_bold
	}
	right_portrait = {
		trigger = { exists = scope:cult_mass_candidate }
		character = scope:cult_mass_candidate
		animation = happiness
	}
	lower_right_portrait = {
		trigger = { exists = scope:cult_mass_recruit }
		character = scope:cult_mass_recruit
	}
	trigger = { cult_active_trigger = yes }

	immediate = {
		cult_save_scopes_effect = yes
		if = {
			limit = {
				NOT = { exists = scope:cult_mass_candidate }
				NOT = { exists = scope:cult_mass_recruit }
			}
			cult_mass_find_initiates_effect = yes
		}
	}

	# Raise them.
	option = {
		name = cult.0903.a
		trigger = {
			exists = scope:cult_mass_candidate
			scope:cult_mass_candidate = { is_alive = yes }
		}
		scope:cult_mass_candidate = {
			add_trait_xp = {
				trait = cult_member
				track = cult_rank
				value = 15
			}
			add_opinion = {
				target = root
				modifier = cult_raised_before_the_circle_opinion
			}
			hidden_effect = {
				send_interface_toast = {
					type = event_toast_effect_good
					title = cult.0903.a.toast
					left_icon = root
					right_icon = scope:cult_mass_candidate
				}
			}
		}
		add_trait_xp = {
			trait = cult_member
			track = cult_rank
			value = 3
		}
		hidden_effect = {
			cult_mass_tell_guests_effect = { PURPOSE = raised }
		}
		ai_chance = {
			base = 50
			modifier = { add = 30 has_trait = generous }
			modifier = { add = -30 has_trait = ambitious }
		}
	}

	# Swear in the courtier.
	option = {
		name = cult.0903.b
		trigger = {
			exists = scope:cult_mass_recruit
			scope:cult_mass_recruit = {
				is_alive = yes
				cult_recruitable_trigger = yes
			}
		}
		scope:cult_mass_recruit = {
			cult_swear_in_effect = { RANK = 0 }
			add_opinion = {
				target = root
				modifier = cult_brought_me_in_opinion
			}
		}
		add_trait_xp = {
			trait = cult_member
			track = cult_rank
			value = 5
		}
		cult_exposure_gain_effect = { VALUE = 2 }
		hidden_effect = {
			cult_mass_tell_guests_effect = { PURPOSE = sworn }
		}
		stress_impact = {
			honest = minor_stress_impact_gain
		}
		ai_chance = {
			base = 50
			modifier = { add = 20 has_trait = ambitious }
			modifier = { add = -30 has_trait = paranoid }
		}
	}

	# Raise yourself.
	option = {
		name = cult.0903.c
		add_trait_xp = {
			trait = cult_member
			track = cult_rank
			value = 15
		}
		if = {
			limit = { exists = scope:activity }
			scope:activity = {

					limit = { NOT = { this = root } }
					add_opinion = {
						target = root

# ---- cult.0904 -- An Empty Place ---------------------------------------------------------------
#
# scope:cult_absentee comes from cult.0901. Naming them costs the absent member standing and makes
# an enemy of them; letting it go earns some gratitude. Either way the absentee is told (cult.0907).
cult.0904 = {
	type = character_event
	title = cult.0904.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:cult_absentee
					scope:cult_absentee = { is_alive = yes }
				}
				desc = cult.0904.desc
			}
			desc = cult.0904.desc_none
		}
	}
	theme = witchcraft
	left_portrait = {
		character = root
		animation = disapproval
	}
	right_portrait = {
		trigger = { exists = scope:cult_absentee }
		character = scope:cult_absentee
		animation = shame
	}
	trigger = { cult_active_trigger = yes }

	immediate = {
		cult_save_scopes_effect = yes
	}

	# Name them.
	option = {
		name = cult.0904.a
		trigger = {
			exists = scope:cult_absentee
			scope:cult_absentee = { is_alive = yes }
		}
		scope:cult_absentee = {
			add_trait_xp = {
				trait = cult_member
				track = cult_rank
				value = -15
			}
			add_opinion = {
				target = root
				modifier = cult_shamed_before_the_circle_opinion
			}
		}
		add_opinion = {
			target = scope:cult_absentee
			modifier = cult_failed_charge_opinion
		}
		if = {
			limit = { exists = scope:activity }
			scope:activity = {
				every_attending_character = {
					limit = {
						NOT = { this = root }
						has_trait = cult_member
					}
					add_trait_xp = {
						trait = cult_member
						track = cult_rank
						value = 3
					}
				}
			}
		}
		hidden_effect = {
			save_scope_value_as = { name = cult_judged_harshly value = yes }
			scope:cult_absentee = { trigger_event = { id = cult.0907 days = 5 } }
		}
		stress_impact = {
			forgiving = medium_stress_impact_gain
		}
		ai_chance = {
			base = 50
			modifier = { add = 30 has_trait = vengeful }
			modifier = { add = 15 has_trait = arbitrary }
			modifier = { add = -40 has_trait = forgiving }
		}
	}

	# Let it go.
	option = {
		name = cult.0904.b
		trigger = {
			exists = scope:cult_absentee
			scope:cult_absentee = { is_alive = yes }
		}
		scope:cult_absentee = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 10
			}
		}
		hidden_effect = {
			scope:cult_absentee = { trigger_event = { id = cult.0907 days = 5 } }
		}
		stress_impact = {
			vengeful = minor_stress_impact_gain
		}
		ai_chance = {
			base = 50
			modifier = { add = 40 has_trait = forgiving }
			modifier = { add = 20 has_trait = paranoid }
		}
	}

	# Nobody to judge. `fallback`, so the event can never render with no way to close it.
	option = {
		name = cult.0904.c
		fallback = yes
	}
}

# activity's cult_defilement_home variable, which stops it.

# Pilgrimage can never be paid by an earlier one's flags.

	option = {

			}
			desc = cult.0916.desc_defile
		}
	}

	left_portrait = {
		character = root
		animation = rage
	}
	trigger = {
		cult_active_trigger = yes
		exists = scope:cult_defilement_county
		exists = scope:cult_defiler
		NOT = { has_trait = cult_member }
