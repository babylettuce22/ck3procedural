namespace = cult

# The inversion cult: powers and their outcomes. cult.0400-0499.
# The powers themselves are interactions: common/character_interactions/00_cult_powers.txt.
#
#   0401-0403   Turn a Priest: sworn / takes the gifts / reports it (to the member)
#   0404        Turn a Priest: the hunter hears of the approach
#   0405        Turn a Priest: to a player priest, who answers for themselves
#   0410-0412   Offer to the master: nobody asks / the circle hears / a jailer talks (to the member)
#   0420-0423   Curse: sickness / misfortune / nothing / traced back (to the member)
#   0424        Curse: a player victim learns who wished it on them
#   0430-0433   Defile a Relic: quietly / a scandal / interrupted / caught (to the member)
#   0434        Defile a Relic: a player lord learns who did it
#   0440-0442   Whisper to the See: sworn / wavers / appalled (to the member)
#   0443        Whisper to the See: the head, after refusing
#   0444        Whisper to the See: to a player head, who answers for themselves
#
# The outcome is decided where the power is used, so the interaction's tooltip shows the odds; these
# events report it. Where the other side is a player, they choose instead of a dice roll.

# To the hunter: a priest has reported being approached.
cult.0404 = {
	type = character_event
	title = cult.0404.t
	desc = cult.0404.desc
	theme = faith
	left_portrait = {
		character = root
		animation = anger
	}
	right_portrait = {
		character = scope:cult_priest
		animation = personality_zealous
	}
	lower_right_portrait = {
		character = scope:cult_priest_turner
	}
	trigger = { cult_secret_phase_trigger = yes }
	immediate = { cult_save_scopes_effect = yes }

	option = {
		name = cult.0404.a
		trigger = { gold >= minor_gold_value }
		pay_short_term_gold = {
			target = scope:cult_priest
			gold = minor_gold_value
		}
		scope:cult_priest = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 15
			}
		}
		ai_chance = { base = 50 }
	}
	option = {
		name = cult.0404.b
		add_opinion = {
			target = scope:cult_priest_turner
			modifier = suspicion_opinion
			opinion = -20
		}
		ai_chance = { base = 50 }
	}
}

# To the head of the faith, who refused.
cult.0443 = {
	type = character_event
	title = cult.0443.t
	desc = cult.0443.desc
	theme = faith
	left_portrait = {
		character = root
		animation = personality_zealous
	}
	right_portrait = {
		character = scope:cult_whisperer
		animation = scheme
	}
	trigger = { cult_secret_phase_trigger = yes }
	immediate = { cult_save_scopes_effect = yes }

	# Denounce them now.
	option = {
		name = cult.0443.a
		add_piety = minor_piety_gain
		scope:cult_whisperer = {
			cult_exposure_gain_effect = { VALUE = 5 }
		}
		ai_chance = {
			base = 50
			modifier = { add = 30 has_trait = zealous }
			modifier = { add = 10 has_trait = wrathful }
		}
	}
	# Watch them, and see who else comes.
	option = {
		name = cult.0443.b
		add_opinion = {
			target = scope:cult_whisperer
			modifier = suspicion_opinion
			opinion = -20
		}
		ai_chance = {
			base = 50
			modifier = { add = 30 has_trait = patient }
			modifier = { add = 20 has_trait = paranoid }
		}
	}
}
