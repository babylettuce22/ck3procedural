# The Restorationists: charges. restor.0500-0599.
#
#   0501        the Keeper gives you a charge (accept / decline)
#   0510-0514   a charge kept: win over, war chest, stir, shelter, recruit
#   0520        a charge's three years ran out
#   0530-0534   a chance to do the charge in hand, one per kind
#   0540        a charge from a player Keeper's own hand (hidden; restor_give_charge_interaction)
#
# The state and every transition are in 00_restor_charge_effects.txt; these events only show them.
# A charge is cleared before its ending event arrives, so the endings read the saved scopes
# (scope:restor_charge_target, scope:restor_charge_county, scope:restor_charge_kind), not the
# variables.

namespace = restor

# =====================================================================================
#  restor.0501 -- a charge from the Keeper
# =====================================================================================

restor.0501 = {
	type = character_event
	title = restor.0501.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:restor_charge_kind ?= flag:win_over }
				desc = restor.0501.desc_win_over
			}
			triggered_desc = {
				trigger = { scope:restor_charge_kind ?= flag:war_chest }
				desc = restor.0501.desc_war_chest
			}
			triggered_desc = {
				trigger = { scope:restor_charge_kind ?= flag:stir }
				desc = restor.0501.desc_stir
			}
			triggered_desc = {
				trigger = { scope:restor_charge_kind ?= flag:shelter }
				desc = restor.0501.desc_shelter
			}
			desc = restor.0501.desc_recruit
		}
	}
	theme = secret
	override_background = { reference = study }
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		trigger = { exists = scope:restor_keeper }
		character = scope:restor_keeper
		animation = scheme
	}
	lower_right_portrait = {
		trigger = { exists = scope:restor_charge_target }
		character = scope:restor_charge_target
	}
	trigger = {
		restor_underground_trigger = yes
		has_variable = restor_charge
		has_variable = restor_charge_given
		exists = global_var:restor_keeper
	}
	immediate = {
		restor_save_scopes_effect = yes
		restor_charge_save_state_effect = yes
	}

	option = {
		name = restor.0501.a
		if = {
			limit = { scope:restor_charge_kind ?= flag:win_over }
			custom_tooltip = restor.0501.a.win_over_tt
		}
		else_if = {
			limit = { scope:restor_charge_kind ?= flag:war_chest }
			custom_tooltip = restor.0501.a.war_chest_tt
		}
		else_if = {
			limit = { scope:restor_charge_kind ?= flag:stir }
			custom_tooltip = restor.0501.a.stir_tt
		}
		else_if = {
			limit = { scope:restor_charge_kind ?= flag:shelter }
			custom_tooltip = restor.0501.a.shelter_tt
		}
		else = {
			custom_tooltip = restor.0501.a.recruit_tt
		}
		ai_chance = { base = 100 }
	}

	option = {
		name = restor.0501.b
		restor_charge_penalty_effect = { FAVOUR = 10 }
		restor_rank_loss_effect = { VALUE = 5 }
		custom_tooltip = restor.0501.b.tt
		hidden_effect = {
			if = {
				limit = { has_variable = restor_charge }
				restor_charge_clear_effect = yes
			}
		}
		ai_chance = { base = 0 }
	}
}

# =====================================================================================
#  restor.0530 -- win over: supper with the lord
# =====================================================================================

restor.0530 = {
	type = character_event
	title = restor.0530.t
	desc = restor.0530.desc
	theme = diplomacy
	override_background = { reference = relaxing_room }
	left_portrait = {
		character = root
		animation = personality_rational
	}
	right_portrait = {
		character = scope:restor_charge_target
		animation = thinking
	}
	trigger = {
		restor_underground_trigger = yes
		exists = global_var:restor_pretender
		has_variable = restor_charge
		var:restor_charge = flag:win_over
		exists = var:restor_charge_target
		var:restor_charge_target = {
			is_alive = yes
			is_imprisoned = no
			NOT = { has_trait = restor_member }
		}
	}
	immediate = {
		restor_save_scopes_effect = yes
		restor_charge_save_state_effect = yes
	}

	# Say it out loud.
	option = {
		name = restor.0530.a
		random_list = {
			50 = {
				modifier = { add = 15 diplomacy >= 12 }
				modifier = { add = 15 diplomacy >= 16 }
				modifier = {
					add = 20
					scope:restor_charge_target = {
						opinion = {
							target = root
							value >= 30
						}
					}
				}
				send_interface_toast = {
					type = event_toast_effect_good
					title = restor.0530.a.success
					left_icon = root
					right_icon = scope:restor_charge_target
					scope:restor_charge_target = {
						add_opinion = {
							target = scope:restor_pretender
							modifier = friendliness_opinion
							opinion = 25
						}
					}
				}
			}
			50 = {
				modifier = { add = 15 has_trait = impatient }
				modifier = {
					add = 20
					scope:restor_charge_target = {
						opinion = {
							target = root
							value < 0
						}
					}
				}
				send_interface_toast = {
					type = event_toast_effect_bad
					title = restor.0530.a.failure
					left_icon = root
					right_icon = scope:restor_charge_target
					restor_exposure_gain_effect = { VALUE = 8 }
				}
			}
		}
		hidden_effect = { restor_charge_check_effect = yes }
		ai_chance = { base = 50 }
	}

	# A gift in the heir's name.
	option = {
		name = restor.0530.b
		trigger = { gold >= 50 }
		remove_short_term_gold = 50
		scope:restor_charge_target = {
			add_opinion = {
				target = scope:restor_pretender
				modifier = friendliness_opinion
				opinion = 15
			}
		}
		hidden_effect = { restor_charge_check_effect = yes }
		ai_chance = { base = 30 }
	}

	option = {
		name = restor.0530.c
		ai_chance = { base = 20 }
	}
}

# =====================================================================================
#  restor.0540 -- a charge from a player Keeper's hand (hidden)
# =====================================================================================
#
# Fired on the member by restor_give_charge_interaction, which leaves the kind asked for in
# var:restor_charge_order and the Keeper in var:restor_charge_giver. Runs as the member, which the
# charge effects need (ROOT is the member). The Keeper is told what was given; a player member gets
# the usual letter (restor.0501).

restor.0540 = {
	hidden = yes
	trigger = {
		has_variable = restor_charge_order
		has_variable = restor_charge_giver
	}
	immediate = {
		restor_save_scopes_effect = yes
		save_scope_as = restor_charge_member
		var:restor_charge_giver = { save_scope_as = restor_charge_giver }
		restor_charge_order_effect = yes
		# Did the kind asked for come up empty, so the usual draw gave something else? Compared as
		# variables: save_scope_value_as cannot hold a flag copied out of one.
		if = {
			limit = {
				NOT = { var:restor_charge_order = flag:any }
				NOT = { var:restor_charge_order = var:restor_charge }
			}
			save_scope_value_as = { name = restor_charge_fell_back value = yes }
		}
		remove_variable = restor_charge_order
		restor_charge_save_state_effect = yes

		scope:restor_charge_giver = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = restor_charge_given_toast
				left_icon = scope:restor_charge_member
				if = {
					limit = { scope:restor_charge_kind ?= flag:win_over }
					custom_tooltip = restor_charge_given_win_over_tt
				}
				else_if = {
					limit = { scope:restor_charge_kind ?= flag:stir }
					custom_tooltip = restor_charge_given_stir_tt
				}
				else_if = {
					limit = { scope:restor_charge_kind ?= flag:war_chest }
					custom_tooltip = restor_charge_given_war_chest_tt
				}
				else_if = {
					limit = { scope:restor_charge_kind ?= flag:shelter }
					custom_tooltip = restor_charge_given_shelter_tt
				}
				else = {
					custom_tooltip = restor_charge_given_recruit_tt
				}
				if = {
					limit = { exists = scope:restor_charge_fell_back }
					custom_tooltip = restor_charge_given_fallback_tt
				}
			}
		}
		restor_charge_notify_effect = yes
	}
}
