namespace = cult

# The inversion cult: recruitment. cult.0100-0199.
#
# A courtship in two steps (see common/scripted_effects/00_cult_recruitment_effects.txt), only for
# souls of the host faith -- the cult lives inside it:
#
#   0101   a player is sounded out by a member (sympathetic / cold / dangerous)
#   0102   the recruiter hears: sympathetic
#   0103   the recruiter hears: cold
#   0104   the recruiter hears: dangerous -- they will take it to their priest, or to the hunter
#   0105   a player member is pointed at a likely soul
#   0110   a player is offered the oath (accept / refuse / report)
#   0111   the recruiter hears: accepted
#   0112   the recruiter hears: refused -- buy the silence or not
#   0113   the recruiter hears: reported to a priest or the hunter
#   0115   the new member's first night in the circle, and what the cult wants now
#   0125   unveiled: the church invites a player lord to take the oath openly
#
# Events go only to players. An AI on either side is settled by the effects, so AI-to-AI recruiting
# makes no events at all.

# =====================================================================================
#  cult.0101 -- a player is sounded out
# =====================================================================================
#
# Often the first a player outside the cult hears of it, so the cult, what it serves and the faith it
# hides in are named plainly.

cult.0101 = {
	type = character_event
	title = cult.0101.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { cult_sinner_trigger = yes }
				desc = cult.0101.desc_intro_sinner
			}
			triggered_desc = {
				trigger = {
					OR = {
						has_trait = excommunicated
						AND = {
							exists = scope:cult_head
							opinion = { target = scope:cult_head value <= -20 }
						}
					}
				}
				desc = cult.0101.desc_intro_grudge
			}
			desc = cult.0101.desc_intro
		}
		desc = cult.0101.desc_society
	}
	theme = secret
	override_background = { reference = sitting_room }
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:cult_recruiter
		animation = scheme
	}

	trigger = {
		cult_secret_phase_trigger = yes
		cult_of_the_host_trigger = yes
		NOT = { has_trait = cult_member }
		NOT = { has_trait = restor_member }
		exists = scope:cult_recruiter
		scope:cult_recruiter = {
			is_alive = yes
			has_trait = cult_member
		}
	}

	immediate = {
		cult_save_scopes_effect = yes
		cult_recruit_find_confessor_effect = yes
	}

	# Sympathetic.
	option = {
		name = cult.0101.a
		custom_tooltip = cult.0101.a.tt
		if = {
			limit = { scope:cult_recruiter = { is_ai = yes } }
			custom_tooltip = cult.0101.a.offer_tt
		}
		hidden_effect = { cult_sound_out_sympathetic_effect = yes }
		ai_chance = {
			base = 10
			modifier = { add = cult_recruit_pull_value }
		}
	}

	# Cold.
	option = {
		name = cult.0101.b
		custom_tooltip = cult.0101.b.tt
		hidden_effect = { cult_sound_out_cold_effect = yes }
		ai_chance = { base = 35 }
	}

	# Dangerous.
	option = {
		name = cult.0101.c
		if = {
			limit = { exists = scope:cult_confessor }
			custom_tooltip = cult.0101.c.tt
		}
		else = {
			custom_tooltip = cult.0101.c.tt_nobody
		}
		hidden_effect = { cult_sound_out_dangerous_effect = yes }
		ai_chance = {
			base = 5
			modifier = { add = cult_recruit_push_value }
		}
	}
}

# =====================================================================================
#  cult.0105 -- a likely soul (a player member)
# =====================================================================================

# The yearly pulse's nudge, so a player finds the two interactions without reading a manual. The
# candidate is picked by the same weights the AI uses.

cult.0105 = {
	type = character_event
	title = cult.0105.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:cult_recruit = { has_character_flag = cult_sounded_sympathetic } }
				desc = cult.0105.desc_sympathetic
			}
			triggered_desc = {
				trigger = { scope:cult_recruit = { cult_recruit_is_chaplain_trigger = yes } }
				desc = cult.0105.desc_chaplain
			}
			triggered_desc = {
				trigger = { scope:cult_recruit = { cult_sinner_trigger = yes } }
				desc = cult.0105.desc_sinner
			}
			triggered_desc = {
				trigger = {
					exists = scope:cult_head
					scope:cult_recruit = { opinion = { target = scope:cult_head value <= -20 } }
				}
				desc = cult.0105.desc_grudge
			}
			desc = cult.0105.desc_doubter
		}
		first_valid = {
			triggered_desc = {
				trigger = { scope:cult_recruit = { has_trait = zealous } }
				desc = cult.0105.desc_zealous
			}
			triggered_desc = {
				trigger = { scope:cult_recruit = { cult_recruit_push_value >= 30 } }
				desc = cult.0105.desc_careful
			}
			desc = cult.0105.desc_unknown
		}
		desc = cult.0105.desc_how
	}
	theme = secret
	left_portrait = {
		character = root
		animation = scheme
	}
	right_portrait = {
		character = scope:cult_recruit
		animation = idle
	}

	trigger = {
		cult_secret_phase_trigger = yes
		has_trait = cult_member

		scope:cult_recruit = {
			is_alive = yes
			NOT = { has_trait = cult_member }
			cult_recruitable_trigger = yes
		}
	}

	immediate = {
		cult_save_scopes_effect = yes
		save_scope_as = cult_recruiter
	}

	option = {
		name = cult.0105.a

		custom_tooltip = cult.0105.b.tt
		hidden_effect = { cult_offer_oath_start_effect = yes }
	}

	}
