namespace = cult

# The Pilgrimage of Defilement's random pool. cult.1100-1199.
# The activity: common/activities/activity_types/00_cult_defilement_activity.txt; its spine is
# cult.0910-0916 in events/cult_rite_events.txt. The draw: common/on_action/
# 00_cult_defilement_pool_on_actions.txt (players only, one a week, three a pilgrimage, no repeats).
#
#   ON THE ROAD (cult_defilement_road_pool, from on_travel_state_pulse)
#   1101   A Companion on the Road       a genuine pilgrim walks with you (a new character)
#   1102   The Sick Pilgrim              an old man with a fever, a day short of the shrine
#   1103   A Face You Know               one of your vassals, or of your liege's court, on the road
#   1104   Robbers                       the pilgrims are stopped; a guest, if any, is ready to fight
#   1105   Talk at the Hostel            the hunter's people walk the pilgrim roads
#   1106   The Friar at the Crossroads   a friar asks every pilgrim to name a heretic
#
#   AT THE SHRINE (cult_defilement_shrine_pool, from the phase's on_weekly_pulse)
#   1110   The Keeper's Vigil            the shrine's priest keeps a vigil at night
#   1111   The Blind Man                 a miracle-seeker at the gate
#   1112   The Relic-Seller              a seller of false relics who has noticed you
#   1113   One of Us                     a sworn member serves at the shrine: cover (read by 0913/0914)
#   1114   The Sermon                    a priest preaches against the Adversary's servants
#   1115   On Your Knees                 you find yourself praying for real
#   1116   A Guest at the Altar          a guest caught praying in earnest
#   1117   Loose Talk                    a guest talked at the hostel
#   1118   The Hunter's Men              the hunter's people at the shrine gate
#   1119   The Offering Box              the pilgrims' coins
#   1120   The Blessed Well              ashes in the well: the county is tainted (map change)
#
#   HIDDEN, fired by the pulse actions (00_cult_defilement_actions.txt) so a meter moves in a
#   character's scope (the pulse action's root is the activity)
#   1190   exposure +1     1191   dark power +2     1192   exposure -1
#
# Every pool event: cult_save_scopes_effect, cult_defilement_save_county_effect, then
# cult_defilement_pool_open_effect { ID }. Every random branch ends in a toast. Reads of the site's
# holder and of scope:activity are guarded, so any of these can be fired from the console; the county
# name then reads blank, as the spine's does.

# =====================================================================================
#  ON THE ROAD
# =====================================================================================

# ---- cult.1101 -- A Companion on the Road --------------------------------------------------------
cult.1101 = {
	type = character_event
	title = cult.1101.t
	desc = cult.1101.desc
	theme = faith
	override_background = { reference = village }
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		trigger = { exists = scope:cult_defilement_pilgrim }
		character = scope:cult_defilement_pilgrim
		animation = happiness
	}
	trigger = {
		cult_secret_phase_trigger = yes
		has_trait = cult_member
		NOT = { has_character_flag = cult_defilement_seen_1101 }
	}
	immediate = {
		cult_save_scopes_effect = yes
		cult_defilement_save_county_effect = yes
		cult_defilement_pool_open_effect = { ID = 1101 }
		create_character = {
			template = location_pilgrim_template
			location = root.location
			culture = root.location.culture
			faith = global_var:cult_host_faith
			dynasty = none
			random_traits = no
			gender_female_chance = 50
			save_scope_as = cult_defilement_pilgrim
		}
	}

	# Walk with them.
	option = {
		name = cult.1101.a
		cult_exposure_loss_effect = { VALUE = 1 }
		scope:cult_defilement_pilgrim = {
			add_opinion = {
				target = root
				modifier = friendliness_opinion
				opinion = 20
			}
		}
		stress_impact = {
			shy = minor_stress_impact_gain
		}
		ai_chance = { base = 50 }
	}

	# Plant a doubt.
	option = {
		name = cult.1101.b
		random_list = {
			50 = {
				modifier = { add = 15 diplomacy >= 12 }
				modifier = { add = 15 learning >= 12 }
				send_interface_toast = {
					type = event_toast_effect_good
					title = cult.1101.b.doubts
					left_icon = root
					right_icon = scope:cult_defilement_pilgrim
					cult_power_gain_effect = { VALUE = 5 }
				}
			}
			50 = {
				send_interface_toast = {
					type = event_toast_effect_bad
					title = cult.1101.b.shocked
					left_icon = root
					right_icon = scope:cult_defilement_pilgrim
					cult_exposure_gain_effect = { VALUE = 2 }
				}
			}
		}
		ai_chance = { base = 25 }
	}

	# Walk alone.
	option = {
		name = cult.1101.c
		custom_tooltip = cult.1101.c.tt
		stress_impact = {
			gregarious = minor_stress_impact_gain
		}
		ai_chance = { base = 25 }
	}

# ---- cult.1103 -- A Face You Know ----------------------------------------------------------------
#
# One of the host's own vassals, else a courtier of the host's liege: somebody who knows the host's
# face and knows the host would not normally walk anywhere.
cult.1103 = {
	type = character_event
	title = cult.1103.t
	desc = cult.1103.desc
	theme = intrigue
	override_background = { reference = tavern }
	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		character = scope:cult_defilement_known
		animation = interested
	}
	trigger = {
		cult_secret_phase_trigger = yes
		has_trait = cult_member
		NOT = { has_character_flag = cult_defilement_seen_1103 }
		OR = {
			any_vassal = {
				is_adult = yes
				NOT = { has_trait = cult_member }
			}
			AND = {
				is_independent_ruler = no
				exists = liege
				liege = {
					any_courtier = {
						is_adult = yes
						NOT = { has_trait = cult_member }
						NOT = { this = root }
					}
				}
			}
		}
	}
	immediate = {
		cult_save_scopes_effect = yes
		cult_defilement_save_county_effect = yes
		cult_defilement_pool_open_effect = { ID = 1103 }
		random_vassal = {
			limit = {
				is_adult = yes
				NOT = { has_trait = cult_member }
			}
			save_scope_as = cult_defilement_known
		}
		if = {
			limit = {
				NOT = { exists = scope:cult_defilement_known }
				is_independent_ruler = no
				exists = liege
			}
			liege = {
				random_courtier = {
					limit = {
						is_adult = yes
						NOT = { has_trait = cult_member }
						NOT = { this = root }
					}
					save_scope_as = cult_defilement_known
				}
			}
		}
	}

	# I have sins to walk off.
	option = {
		name = cult.1103.a
		random_list = {
			60 = {
				modifier = { add = 15 intrigue >= 12 }
				modifier = { add = 10 has_trait = deceitful }
				send_interface_toast = {
					type = event_toast_effect_good
					title = cult.1103.a.believed
					left_icon = root
					right_icon = scope:cult_defilement_known
					scope:cult_defilement_known = {
						add_opinion = {
							target = root
							modifier = respect_opinion
							opinion = 10
						}
					}
				}
			}
			40 = {
				send_interface_toast = {
					type = event_toast_effect_bad
					title = cult.1103.a.doubted
					left_icon = root
					right_icon = scope:cult_defilement_known
					cult_exposure_gain_effect = { VALUE = 2 }
				}
			}
		}
		stress_impact = {
			honest = minor_stress_impact_gain
		}
		ai_chance = { base = 50 }
	}

	# Pay for silence.
	option = {
		name = cult.1103.b
		trigger = { gold >= 25 }
		pay_short_term_gold = {
			target = scope:cult_defilement_known
			gold = 25
		}
		custom_tooltip = cult.1103.b.tt
		ai_chance = { base = 25 }
	}

	# Walk with them, and pray loudly.
	option = {
		name = cult.1103.c
		add_piety = minor_piety_gain
		scope:cult_defilement_known = {
			add_opinion = {
