# A generated faith's Synod Seats. See common/scripted_effects/zz_gen_see_electors_effects.txt.

# Root: a character. May hold a Synod Seat of $FAITH$: of that faith, and a theocrat. Vanilla destroys
# an elector title held by any ruler who is not (pam_strip_illegal_cardinalate_effect, run on every
# government change, title gain and year), so handing a seat to anyone else loses it, and since
# zz_gen_pam_cardinal_title_effects.txt passes a stripped seat on instead, it would bounce between
# them daily. An unlanded character passes: the handover itself makes him a ruler, and our handovers
# to a new archbishop give him a theocrat's government.
gen_synod_rightful_holder_trigger = {
	faith = $FAITH$
	OR = {
		is_ruler = no
		government_has_flag = government_is_theocracy
		government_has_flag = government_is_holy_order
	}
}
