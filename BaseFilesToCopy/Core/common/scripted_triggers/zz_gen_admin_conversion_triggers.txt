# Loosens who may adopt All Under Heaven's three regional bureaucracies on a generated map.
#
# ---- What this is ----
#
# Single-object overrides of six vanilla scripted triggers from
# common/scripted_triggers/10_tgp_triggers.txt. Script is last-wins and this directory has no
# subfolders, so a zz_ file here replaces each trigger without touching the file it lives in.
#
# The first four loosen who may adopt a bureaucracy; the last two (minister eligibility, at the
# bottom of this file) tighten who may staff one, and fix a crash rather than a tooltip.
#
# The "Adopt <X> Government" items of convert_to_administrative_decision (06_ep3_admin_decisions.txt)
# gate on the first three. Vanilla ties them to Earth: the Hegemony of China, Han ancestry, the
# world_asia/world_tibet/world_burma/world_steppe_east regions, and a Japanese top liege. None of
# those exist on a generated map, so all three items showed as permanently unfulfillable — the
# tooltips in the screenshots that prompted this file.
#
# ---- The design ----
#
# * Celestial is the HEGEMON'S government. The generated hegemony (Titles.Crown) stands in for
#   h_china: hold it, or be a vassal of whoever does, and you may adopt Celestial. AI included —
#   that is vanilla's intent for the Chinese emperor, and there is one hegemony per world.
#
# * Meritocratic and Ritsuryō are opened to PLAYERS anywhere the structural rules allow
#   (independent, or a vassal of a non-administrative liege). The AI keeps vanilla's logic: the
#   decision's widget weights celestial/ritsuryō at 100 and meritocratic at 50 against
#   administrative at 10, so an AI allowed into all three would turn most converting empires
#   Ritsuryō — a government whose house blocs, shogunate and imperial-branch layers lean on
#   title:e_japan. Vanilla's "borders a merit realm" clause still lets AI neighbours of a
#   generated meritocratic hegemon follow it, as on Earth.
#
# * tgp_has_access_to_ministry_trigger is the one line the whole ministry layer (council
#   positions, minister titles, ministry projects) hangs off. Generalised from h_china to
#   "holds a hegemony-tier title" so a Celestial hegemon is not a name with nothing behind it.
#   The minister titles themselves are exempted from the vanilla-titular strip guard in
#   Emit/CompatibilityWriter.cs for the same reason. The dynastic-cycle situation and the imperial
#   examinations remain bound to title:h_china and are NOT reached by this file.
#
# Loc for the three decision tooltips is overridden alongside, in
# localization/english/zz_gen_admin_conversion_l_english.yml.

tgp_should_become_meritocratic_trigger = {
	has_tgp_dlc_trigger = yes
	OR = {
		# Vanilla: follow a neighbour or liege that already uses merit.
		top_liege = {
			OR = {
				has_government = meritocratic_government
				any_neighboring_and_across_water_top_liege_realm_owner = { government_allows = merit }
			}
		}
		# Vanilla's East Asia clause, with the region test replaced by a player test.
		AND = {
			is_ai = no
			OR = {
				is_independent_ruler = yes
				liege = {
					NOT = {
						government_has_flag = government_is_administrative
					}
				}
			}
		}
		# Becoming admin with creation of meritocratic great yuan
		exists = scope:great_yuan_ruler
	}
}

tgp_can_become_japan_administrative_trigger = {
	OR = {
		# Vanilla: inside a Ritsuryō/Sōryō realm, or restoring one.
		top_liege = {
			OR = {
				government_is_japanese_trigger = yes
				has_character_flag = tgp_japan_restore_japanese_government_flag
			}
		}
		government_is_japanese_trigger = yes
		has_character_flag = tgp_japan_restore_japanese_government_flag
		# Generated worlds: any player whose realm is not already another bureaucracy.
		gen_player_may_found_bureaucracy_trigger = yes
	}
}

tgp_can_become_celestial_trigger = {
	top_liege = {
		OR = {
			has_government = celestial_government
			highest_held_title_tier = tier_hegemony
		}
	}
}

tgp_has_access_to_ministry_trigger = {
	highest_held_title_tier = tier_hegemony
	government_has_flag = government_is_celestial
}

# The player-side loosening, shared by the Ritsuryō decision gate above and by
# japan_administrative_government.can_get_government (common/governments/zz_gen_japan_government_types.txt)
# so the engine's own validity check agrees with the decision that offers it.
gen_player_may_found_bureaucracy_trigger = {
	is_ai = no
	OR = {
		is_independent_ruler = yes
		liege = {
			NOT = {
				government_has_flag = government_is_administrative
			}
		}
	}
}

# ---- A landless ruler who still has vassals may not hold a ministry ----
#
# Verbatim copies of vanilla's two minister-eligibility triggers (10_tgp_triggers.txt 172 and 218)
# with one clause added to each. Both are single-object overrides, same as the four above.
#
# The bug they fix: vanilla asks only "is_ruler = no OR government_is_celestial", and nothing about
# vassals. A celestial ruler who has lost every county keeps his landless noble-family title AND
# whatever vassals were sworn to him -- crash ck3_20260907_175647's character 1453 held only
# x_nf_216 and still had three vassals, one of them his son on a real county. Appointing him closes
# a loop the engine cannot leave:
#
#     got_minister_position_effect grants the e_ tier ministry title, then (its own
#       `vassal_count > 0` block) moves every vassal he has up to the top liege
#       -> rank up fires on_rank_up 4023, which re-parents landless noble-family vassals again
#       -> the liege changes trip the engine's appointment-lease revocation
#       -> the ministry title is left holderless and destroyed
#       -> the council seat is vacant, so he is re-appointed
#
# Each turn nests one title-and-vassal change inside the last. The crash was a C00000FD stack
# overflow 7269 frames deep, ~170 turns a second for six seconds, all of it that one character.
#
# What the clause must NOT do is bar landless noble-family heads as a class, and the first version
# of this file did exactly that (`is_landed = no` plus `is_noble_family_title`). In a celestial
# realm the landless noble-family head IS the normal minister: claim_mandate turns every courtier
# with 3000 merit into one on purpose, and vanilla's own 10_tgp_triggers.txt 905 speaks of "the
# councilors of landless noble family heads" as an ordinary thing. With them barred, vanilla's
# fired-minister path -- fired_minister_position_effect hands the title to the title's
# appointment-succession heir and re-seats him through assign_councillor_type -- failed
# valid_character on every heir, left an e_ title in the hands of a non-councillor, and the engine
# resigned it (ResignInvalidTierNonDejure). Measured after nine years of that: eight of nine
# ministries vacant, e_minister_of_revenue created 22 times and destroyed 21, and every refill
# drawn from the rank-0 remainder, so the low-merit Dynastic Cycle catalysts fired on each one.
#
# `vassal_count > 0` is the discriminator because it is the one vanilla's own effect uses to decide
# a new minister needs cleaning up, and a fresh courtier-turned-noble-family-head has none. A
# landless ruler WITH vassals is by construction a former landed ruler, which is the crash shape
# and nothing else. Landed rulers are untouched: they are the governors, and can be ministers.
#
# Grand marshal carries the same hole and is overridden for the same reason -- left alone, the loop
# would move to e_minister_grand_marshal on the next appointment.

tgp_can_be_a_minister_trigger = {
	can_be_councillor_basics_trigger = yes
	# Has to be celestial or a non-ruler
	OR = {
		is_ruler = no
		government_has_flag = government_is_celestial
	}
	# Added: not a landless ruler who still has vassals. See the comment above.
	NOT = {
		AND = {
			is_landed = no
			vassal_count > 0
		}
	}
	# Gender is based on laws of the liege
	OR = {
		has_variable = can_be_councillor_override
		AND = {
			is_female = yes
			$COURT_OWNER$ = {
				NOR = {
					has_realm_law = male_only_law
					has_realm_law = male_preference_law
				}
			}
		}
		AND = {
			is_male = yes
			$COURT_OWNER$ = {
				NOR = {
					has_realm_law = female_only_law
					has_realm_law = female_preference_law
				}
			}
		}
	}
	# Players are only allowed if they have enough merit or is a councillor already, unless they are friends or disciples/elders
	# We do this to prevent the AI from "randomly" requesting the player to be on their council
	trigger_if = {
		limit = {
			is_ai = no
			$COURT_OWNER$ = { is_ai = yes }
			NOT = { is_councillor_of = $COURT_OWNER$ }
		}
		OR = {
			merit_level >= merit_level_expected_empire_value
			has_relation_friend = $COURT_OWNER$
			has_relation_disciple = $COURT_OWNER$
			has_relation_elder = $COURT_OWNER$
		}
	}
}

tgp_can_be_grand_marshal_trigger = {
	can_be_councillor_basics_trigger = yes
	# Has to be celestial or a non-ruler
	OR = {
		is_ruler = no
		government_has_flag = government_is_celestial
	}
	# Added: not a landless ruler who still has vassals. See the comment above.
	NOT = {
		AND = {
			is_landed = no
			vassal_count > 0
		}
	}
	# Has to be able to lead armies
	can_be_combatant_based_on_gender_trigger = { ARMY_OWNER = $COURT_OWNER$ }
}
