# Has this character's race already been decided?
#
# The guard for everything in 00_phenotype_effects.txt. All six traits share `group =
# phenotype_race` and list each other as `opposites`, so a character should never hold two.
#
# **The `gen_phenotype_human` character flag is deliberately in this list.** It exists only to say
# "this character was considered and came out human", which is a different state from "nobody has
# looked at this character yet". Without that distinction the yearly culture pulse cannot tell a
# human child of a mixed marriage from an engine-generated blank, and converts the former to the
# local race — overruling the parents, which is the one thing that must always win at birth.
#
# It is a FLAG rather than a seventh trait because CK3 traits cannot be hidden — there is no
# `hidden` field on a trait (the token in the binary belongs to events), so a marker trait would
# always render a chip in the character view. A character flag is invisible everywhere, persists in
# saves, and costs nothing, which is exactly the profile a bookkeeping marker wants.
#
# NOTE: the traits are `inherit_chance = 0`. They pass NOTHING genetically; every assignment is
# script, from either the birth hook or the culture pulse. An earlier version of this comment said
# they inherited at 100%, which is what the rest of this system used to be designed around — if the
# code and a comment here ever disagree again, suspect the comment.
gen_has_phenotype_trigger = {
	OR = {
		has_character_flag = gen_phenotype_human
		has_trait = phenotype_gracile
		has_trait = phenotype_stocky
		has_trait = phenotype_rough_hewn
		has_trait = phenotype_towering
		has_trait = phenotype_diminutive
		has_trait = phenotype_dusk_adapted
	}
}

# Is this character carrying more than one race at once?
#
# Should be impossible — `opposites` covers all six and the flag is only set by script — but the ruler designer and the console can
# both produce it, and a character wearing two races has two sets of forced genes fighting over the
# same portrait. Cheap enough to ask once a year alongside the assignment; repaired by
# gen_sanitize_phenotype_effect.
gen_has_conflicting_phenotype_trigger = {
	trigger_if = {
		limit = { has_character_flag = gen_phenotype_human }
		OR = {
			has_trait = phenotype_gracile
			has_trait = phenotype_stocky
			has_trait = phenotype_rough_hewn
			has_trait = phenotype_towering
			has_trait = phenotype_diminutive
			has_trait = phenotype_dusk_adapted
		}
	}
	trigger_else = {
		# Two or more of the six fantasy races at once.
		calc_true_if = {
			amount = 2
			has_trait = phenotype_gracile
			has_trait = phenotype_stocky
			has_trait = phenotype_rough_hewn
			has_trait = phenotype_towering
			has_trait = phenotype_diminutive
			has_trait = phenotype_dusk_adapted
		}
	}
}

# Checks if root/this and scope:target have different racial phenotypes
# Usage:
#   root = { gen_is_different_race_than = { TARGET = scope:other_char } }
gen_is_different_race_than = {
	$TARGET$ = { save_temporary_scope_as = gen_target_char }
	OR = {
		# One is human (flag or untagged), the other is fantasy
		AND = {
			OR = {
				has_character_flag = gen_phenotype_human
				NOT = { gen_has_phenotype_trigger = yes }
			}
			scope:gen_target_char = {
				gen_has_phenotype_trigger = yes
				NOT = { has_character_flag = gen_phenotype_human }
			}
		}
		AND = {
			gen_has_phenotype_trigger = yes
			NOT = { has_character_flag = gen_phenotype_human }
			scope:gen_target_char = {
				OR = {
					has_character_flag = gen_phenotype_human
					NOT = { gen_has_phenotype_trigger = yes }
				}
			}
		}
		# Both are fantasy races, but different traits
		AND = {
			has_trait = phenotype_gracile
			scope:gen_target_char = { NOT = { has_trait = phenotype_gracile } }
		}
		AND = {
			has_trait = phenotype_stocky
			scope:gen_target_char = { NOT = { has_trait = phenotype_stocky } }
		}
		AND = {
			has_trait = phenotype_rough_hewn
			scope:gen_target_char = { NOT = { has_trait = phenotype_rough_hewn } }
		}
		AND = {
			has_trait = phenotype_towering
			scope:gen_target_char = { NOT = { has_trait = phenotype_towering } }
		}
		AND = {
			has_trait = phenotype_diminutive
			scope:gen_target_char = { NOT = { has_trait = phenotype_diminutive } }
		}
		AND = {
			has_trait = phenotype_dusk_adapted
			scope:gen_target_char = { NOT = { has_trait = phenotype_dusk_adapted } }
		}
	}
}
