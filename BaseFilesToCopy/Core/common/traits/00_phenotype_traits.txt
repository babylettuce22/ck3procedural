#elf
phenotype_gracile = {
    physical = yes
    inherit_chance = 100
    both_parent_has_trait_inherit_chance = 100
    inherit_from_real_father = yes
    inherit_from_real_mother = yes
    shown_in_ruler_designer = yes
    ruler_designer_cost = 20

    group = phenotype_physique
    opposites = { phenotype_stocky phenotype_rough_hewn }

    icon = "gracile.dds"

    desc = {
        first_valid = {
            triggered_desc = {
                trigger = {
                    NOT = { exists = this }
                }
                desc = trait_phenotype_gracile_desc
            }
            desc = trait_phenotype_gracile_character_desc
        }
    }

	culture_modifier = {
		parameter = forest_trait_bonuses
		forest_attrition_mult = -0.1
		taiga_attrition_mult = -0.1
	}

    diplomacy = 1
    prowess = 1
    movement_speed = 0.05
    stress_loss_mult = 0.05
    same_opinion = 10

    # Elves outlive men, and both halves of that are needed.
    #
    # `life_expectancy` is literally years before natural death, and it also makes a character
    # visibly age slower -- which is the same idea gene_age's `no_aging` expresses on the portrait,
    # so the two agree instead of one contradicting the other. Vanilla's own range runs -10..+10
    # (fecund is +5), so +35 is far outside human scale on purpose: base expectancy sits around
    # sixty, and this is what puts an elf comfortably past a hundred.
    life_expectancy = 35

    # Health does the other half, and it is the part that actually stops the age-related ailments
    # -- faltering heart, clouded eyes, fragile bones, withering mind, infirm -- from arriving at a
    # human's schedule. Each of those events carries `modifier = { health >= good_health  age < 60
    # factor = 0 }`, so a character at or above good_health (5) simply cannot draw one before sixty.
    # Engine-generated characters spawn at health 4.0-5.0 (RANDOM_CHARACTER_MIN/MAX_HEALTH), so the
    # old +0.65 left roughly half of all elves under the bar and collecting ailments from 45. At
    # +1.25 the whole spawn range clears it.
    #
    # It keeps paying afterwards: the tracked ailments advance through a random_list in
    # health_on_actions.txt that subtracts 25 from the progression roll at `health >= fine_health`,
    # so an elf who does pick one up past sixty climbs toward the lethal top of the track far more
    # slowly than a man would.
    health = 1.25
}
