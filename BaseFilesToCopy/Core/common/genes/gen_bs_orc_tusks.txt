# Lower orc tusks. Emit/RaceHeadWriter.cs appends two tusks to the vanilla teeth mesh, collapsed onto
# their roots inside the mouth, and builds a teeth blendshape that grows them; this gene drives it.
# MapGen/OrcTusks.cs is the shape. The value is the blendshape weight, 0 none to 1 full-grown, so it
# tiers like the gene_bs_ genes (MapGen/Ethnicities.cs NeutralOf) and a half-orc gets half-grown tusks.
#
# The attribute lives on the TEETH entity; the head declares it too, mapped to its neutral shape, as
# vanilla does for its own teeth attributes.
#
# In Core beside gen_race_skin and gen_bs_elf_ears for the same reason: Emit/PortraitWriter.cs writes
# it into every persistent DNA record on every map. Index 0 must stay the empty template, and indices
# are only ever appended: a saved DNA names its template by index.
#
# Variants (MapGen/OrcTusks.cs Variants): lower is the standard tusk; stubby and great are inherited
# alternatives, mixed per orc culture in its ethnicity; broken_left and broken_right are scars, drawn
# from a character flag by the generated gen_race_tusks portrait group and never put in a DNA.

morph_genes = {
	gen_bs_orc_tusks = {
		inheritable = yes
		group = mouth

		gen_orc_tusks_none = {
			index = 0
			visible = no

		}
