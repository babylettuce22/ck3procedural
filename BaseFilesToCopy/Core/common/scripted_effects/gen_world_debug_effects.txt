# Manual world-state tools; never attach to pulses or AI decisions.
# Requires a character scope. War effects require debug mode.

gen_world_debug_collapse_effect = {
	# Snapshot held ordinary titles before rank changes mutate ruler lists.
	every_ruler = {
		limit = { gen_world_debug_ruler_trigger = yes }
		every_held_title = {
			limit = {
				tier > $TIER$
				is_landless_type_title = no
				is_holy_order = no
				is_mercenary_company = no
				is_titular = no
			}
			add_to_temporary_list = gen_world_debug_destroy_titles
		}
	}
	every_in_list = {
		list = gen_world_debug_destroy_titles
		if = {
			limit = { exists = holder }
			holder = { destroy_title = prev }
		}
	}
}

gen_world_debug_release_vassals_effect = {
	create_title_and_vassal_change = { type = independency save_scope_as = gen_world_debug_change }
	every_ruler = {
		limit = { gen_world_debug_ruler_trigger = yes exists = liege }
		becomes_independent = { change = scope:gen_world_debug_change }
	}
	resolve_title_and_vassal_change = scope:gen_world_debug_change
}

gen_world_debug_independent_counties_effect = {
	gen_world_debug_collapse_effect = { TIER = tier_county }
	gen_world_debug_release_vassals_effect = yes
}

gen_world_debug_white_peace_effect = {
	# Include landless participants and deduplicate wars before ending them.
	every_living_character = {
		every_character_war = { add_to_temporary_list = gen_world_debug_wars }
	}
	every_in_list = { list = gen_world_debug_wars end_war = white_peace }
}

gen_world_debug_neighbor_war_effect = {
	save_scope_as = gen_world_debug_attacker
	random_neighboring_top_liege_realm_owner = {
		limit = {
			gen_world_debug_ruler_trigger = yes
			$TARGET_FILTER$
			scope:gen_world_debug_attacker = {
				can_declare_war = {
					defender = prev
					casus_belli = debug_war
					target_titles = { prev.primary_title }
				}
			}
		}
		save_scope_as = gen_world_debug_defender
		scope:gen_world_debug_attacker = {
			start_war = {
				cb = debug_war
				target = scope:gen_world_debug_defender
				target_title = scope:gen_world_debug_defender.primary_title
			}
		}
	}
}

gen_world_debug_war_wave_effect = {
	every_independent_ruler = {
		limit = { gen_world_debug_ruler_trigger = yes is_ai = yes $ATTACKER_FILTER$ }
		save_scope_as = gen_world_debug_attacker
		random_neighboring_top_liege_realm_owner = {
			limit = {
				gen_world_debug_ruler_trigger = yes
				$TARGET_FILTER$
				scope:gen_world_debug_attacker = {
					can_declare_war = {
						defender = prev
						casus_belli = debug_war
						target_titles = { prev.primary_title }
					}
				}
			}
			save_scope_as = gen_world_debug_defender
			scope:gen_world_debug_attacker = {
				start_war = {
					cb = debug_war
					target = scope:gen_world_debug_defender
					target_title = scope:gen_world_debug_defender.primary_title
				}
			}
		}
	}
}

gen_world_debug_random_war_effect = {
	# Only choose attackers with a legal neighboring target.
	random_independent_ruler = {
		limit = {
			gen_world_debug_ruler_trigger = yes
			is_ai = yes
			any_neighboring_top_liege_realm_owner = {
				gen_world_debug_ruler_trigger = yes
				prev = { can_declare_war = { defender = prev casus_belli = debug_war target_titles = { prev.primary_title } } }
			}
		}
		gen_world_debug_neighbor_war_effect = { TARGET_FILTER = "always = yes" }
	}
}

gen_world_debug_consolidate_effect = {
	# Restore ordinary de jure duchies/kingdoms and attach lower-ranked counts.
	# Existing personal domains are preserved; cross-border holdings are not split.
	every_county = {
		limit = { exists = holder holder = { gen_world_debug_ruler_trigger = yes } }
		$TITLE_SCOPE$ = { add_to_temporary_list = gen_world_debug_consolidate_titles }
	}
	every_in_list = {
		list = gen_world_debug_consolidate_titles
		if = {
			limit = { exists = title_preferred_capital_county.holder title_preferred_capital_county.holder = { gen_world_debug_ruler_trigger = yes } }
			save_scope_as = gen_world_debug_title
			title_preferred_capital_county.holder = { save_scope_as = gen_world_debug_liege }
			create_title_and_vassal_change = { type = granted save_scope_as = gen_world_debug_change }
			change_title_holder = { holder = scope:gen_world_debug_liege change = scope:gen_world_debug_change }
			resolve_title_and_vassal_change = scope:gen_world_debug_change
			create_title_and_vassal_change = { type = granted save_scope_as = gen_world_debug_change }
			every_de_jure_county_holder = {
				limit = {
					gen_world_debug_ruler_trigger = yes
					NOT = { this = scope:gen_world_debug_liege }
					primary_title.tier < scope:gen_world_debug_title.tier
				}
				change_liege = { liege = scope:gen_world_debug_liege change = scope:gen_world_debug_change }
			}
			resolve_title_and_vassal_change = scope:gen_world_debug_change
		}
	}
}

gen_world_debug_gold_effect = {
	every_ruler = { limit = { gen_world_debug_ruler_trigger = yes } add_gold = 5000 }
}

gen_world_debug_refill_effect = {
	every_county = {
		limit = { exists = holder holder = { gen_world_debug_ruler_trigger = yes } }
		every_county_province = { refill_levy = yes refill_garrison = yes }
	}
}

	}

gen_world_debug_unrest_effect = {
	every_county = {
		limit = { exists = holder holder = { gen_world_debug_ruler_trigger = yes } }
		add_county_modifier = { modifier = gen_world_debug_unrest_modifier years = 5 }
	}
}

gen_world_debug_clear_unrest_effect = {
	every_county = { remove_county_modifier = gen_world_debug_unrest_modifier }
}

gen_world_debug_succession_effect = {
	# Snapshot rulers: successors must not enter the same death wave.
	every_ruler = {
		limit = { gen_world_debug_ruler_trigger = yes is_ai = yes $FILTER$ }
		add_to_temporary_list = gen_world_debug_deaths
	}
	every_in_list = { list = gen_world_debug_deaths death = natural }
}

gen_world_debug_random_succession_effect = {
	random_independent_ruler = { limit = { gen_world_debug_ruler_trigger = yes is_ai = yes } death = natural }
}

gen_world_debug_outbreak_effect = {
	random_county = {
		limit = { exists = holder holder = { gen_world_debug_ruler_trigger = yes } }
		title_province = { create_epidemic_outbreak = { type = smallpox intensity = $INTENSITY$ } }
	}
}

gen_world_debug_reduce_epidemics_effect = {
	every_epidemic = { set_epidemic_outbreak_intensity = minor }
}
