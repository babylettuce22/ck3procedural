# Global scope: one O(realms) census per year, not one per player or character.
gen_world_war_census_effect = {
	set_global_variable = { name = gen_world_realm_count value = 0 }
	set_global_variable = { name = gen_world_warring_realm_count value = 0 }
	set_global_variable = { name = gen_world_war_percent value = 0 }
	every_independent_ruler = {
		limit = { gen_world_census_realm_trigger = yes }
		change_global_variable = { name = gen_world_realm_count add = 1 }
		if = {
			limit = { is_at_war = yes }
			change_global_variable = { name = gen_world_warring_realm_count add = 1 }
		}
	}
	if = {
		limit = { global_var:gen_world_realm_count > 0 }
		set_global_variable = { name = gen_world_war_percent value = gen_world_war_percent_value }
	}
	# How many census years in a row have been quiet. Reset by any year that is not, so it
	# measures an unbroken stretch rather than a total.
	if = {
		limit = { NOT = { has_global_variable = gen_world_quiet_years } }
		set_global_variable = { name = gen_world_quiet_years value = 0 }
	}
	if = {
		limit = {
			global_var:gen_world_realm_count >= gen_world_war_minimum_realms_value
			global_var:gen_world_war_percent < gen_world_quiet_threshold_percent_value
		}
		change_global_variable = { name = gen_world_quiet_years add = 1 }
	}
	else = {
		set_global_variable = { name = gen_world_quiet_years value = 0 }
	}
	# Debug-mode telemetry lets observed campaigns establish whether 50% occurs.
	debug_log = "GEN_WORLD_WAR year=[GetCurrentDate.GetYear] realms=[GetGlobalVariable('gen_world_realm_count').GetValue] warring=[GetGlobalVariable('gen_world_warring_realm_count').GetValue] percent=[GetGlobalVariable('gen_world_war_percent').GetValue] quiet_years=[GetGlobalVariable('gen_world_quiet_years').GetValue]"
}
