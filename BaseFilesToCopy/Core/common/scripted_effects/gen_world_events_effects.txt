# Global scope: one O(realms) census per year, not one per player or character.
gen_world_war_census_effect = {
	set_global_variable = { name = gen_world_realm_count value = 0 }
	set_global_variable = { name = gen_world_warring_realm_count value = 0 }
	set_global_variable = { name = gen_world_war_percent value = 0 }
	every_independent_ruler = {
		limit = { gen_world_census_realm_trigger = yes }
		change_global_variable = { name = gen_world_realm_count add = 1 }
		if = {
			limit = { is_at_war = yes }
			change_global_variable = { name = gen_world_warring_realm_count add = 1 }
		}
	}
	if = {
		limit = { global_var:gen_world_realm_count > 0 }
		set_global_variable = { name = gen_world_war_percent value = gen_world_war_percent_value }
	}
	# Debug-mode telemetry lets observed campaigns establish whether 50% occurs.
	debug_log = "GEN_WORLD_WAR year=[GetCurrentDate.GetYear] realms=[GetGlobalVariable('gen_world_realm_count').GetValue] warring=[GetGlobalVariable('gen_world_warring_realm_count').GetValue] percent=[GetGlobalVariable('gen_world_war_percent').GetValue]"
}
