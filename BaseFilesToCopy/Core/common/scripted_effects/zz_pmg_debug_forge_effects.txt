# Shared body of the debug forge event's options. See BaseFilesToCopy/Core/events/
# zz_pmg_debug_forge_events.txt.
#
# One effect rather than six copies: every option differs only in which weapon-type flag it passes,
# and the interesting part -- create it, equip it, then report which look it actually got -- is
# identical. Six copies of that is six places to forget to update.

pmg_debug_forge_weapon_effect = {
	# hidden_effect, so the option's TOOLTIP does not evaluate any of this.
	#
	# Without it CK3 walks the whole body every time the tooltip is built - which is continuously,
	# while hovering - and each pass reports scope:newly_created_artifact as undefined, because the
	# artifact is only created when the option is actually taken. That produced 55,740 log lines from
	# this one event and an error.log of 23 MB, which buries everything genuinely worth reading.
	hidden_effect = {
		# Rarity is rolled HERE, before creation, and that placement is the whole point.
		#
		# The visual override in common/artifacts/visuals gates each forged look on the artifact's
		# rarity, and CK3 resolves a visual INSIDE create_artifact -- so set_artifact_rarity on the
		# finished artifact recolours the frame and leaves the model exactly as it was. Left to
		# itself the vanilla path lands on the same band nearly every time, which meant this event
		# could only ever show one look per weapon type however many times it was taken.
		#
		# Two vanilla effects make the band steerable, both in
		# common/scripted_effects/00_ep1_artifact_creation_effects.txt: get_artifact_quality_effect
		# and get_artifact_wealth_effect each open with `if = { limit = { NOT = { exists =
		# scope:quality } } }`, so a value saved before the call survives it untouched. Nothing on
		# the creation path moves either number afterwards -- the decoration effects only set
		# feature groups, and change_artifact_quality_effect belongs to reforging.
		#
		# Rarity is DERIVED from quality + wealth against ARTIFACT_THRESHOLD_MASTERWORK = 50,
		# _FAMED = 130 and _ILLUSTRIOUS = 180 in 00_defines.txt. The pairs below sum to 32, 90, 154
		# and 192, each sitting clear of the thresholds either side, and they are the midpoints of
		# the bands MapGen/Artifacts.cs rolls for generated artifacts.
		#
		# Even weights, deliberately not a world-realistic spread: a real world is roughly
		# 26/15/6/2 across the four bands, which would make illustrious one click in twenty-five.
		# This event exists to reach every band in a few clicks.
		random_list = {
			25 = {
				save_scope_value_as = { name = quality value = 16 }
				save_scope_value_as = { name = wealth value = 16 }
			}
			25 = {
				save_scope_value_as = { name = quality value = 45 }
				save_scope_value_as = { name = wealth value = 45 }
			}
			25 = {
				save_scope_value_as = { name = quality value = 77 }
				save_scope_value_as = { name = wealth value = 77 }
			}
			25 = {
				save_scope_value_as = { name = quality value = 96 }
				save_scope_value_as = { name = wealth value = 96 }
			}
		}

		# The game's own creation path, the same one inspirations and tournament prizes take. OWNER
		# and CREATOR both being root means the weapon reads as self-made, which is fine for a
		# diagnostic and avoids needing a courtier to exist.
		create_artifact_weapon_effect = {
			OWNER = root
			CREATOR = root
			SET_WEAPON_TYPE = $TYPE$
		}

		scope:newly_created_artifact = {
			# Creation does not equip. Without this the model is never drawn on the portrait, and
			# the weapon sits in the inventory looking like the render failed.
			equip_artifact_to_owner_replace = yes
		}

		# Which band the weapon actually came out in -- read back off the artifact rather than
		# echoed from the branch above, so a pair that ever drifts over a threshold reports the
		# band the player is really looking at. Tested highest-first, which is right whether the
		# trigger is an equality or a floor.
		if = {
			limit = { scope:newly_created_artifact = { rarity = illustrious } }
			send_interface_toast = {
				type = msg_artifact_gained
				title = pmg_debug_forge.0001.illustrious
				right_icon = scope:newly_created_artifact
			}
		}
		else_if = {
			limit = { scope:newly_created_artifact = { rarity = famed } }
			send_interface_toast = {
				type = msg_artifact_gained
				title = pmg_debug_forge.0001.famed
				right_icon = scope:newly_created_artifact
			}
		}
		else_if = {
			limit = { scope:newly_created_artifact = { rarity = masterwork } }
			send_interface_toast = {
				type = msg_artifact_gained
				title = pmg_debug_forge.0001.masterwork
				right_icon = scope:newly_created_artifact
			}
		}
		else = {
			send_interface_toast = {
				type = msg_artifact_gained
				title = pmg_debug_forge.0001.common
				right_icon = scope:newly_created_artifact
			}
		}

		# Reports whether the artifact carries a forged-look index. The answer is currently always
		# no, and the logs showed why: the visual is resolved INSIDE create_artifact, so nothing
		# that runs afterwards can influence which look it picks. Kept as a live check, because it
		# is the one line that will change the moment the selection is driven by something readable
		# at creation time.
		if = {
			limit = { scope:newly_created_artifact = { has_variable = gen_forge_index } }
			send_interface_toast = {
				type = msg_artifact_gained
				title = pmg_debug_forge.0001.indexed
				right_icon = scope:newly_created_artifact
			}
		}
		else = {
			send_interface_toast = {
				type = msg_artifact_gained
				title = pmg_debug_forge.0001.unindexed
				right_icon = scope:newly_created_artifact
			}
		}
	}
}

# Armour counterpart, for testing the portrait-modifier probe in gfx/portraits/portrait_modifiers/
# zz_pmg_armor_probe.txt. Same route the game uses - coronation gifts and tournament prizes both
# call this effect - so what appears is what a player would earn.
#
# TYPE is passed explicitly rather than left to the game. Every vanilla caller uses flag:no and
# lets the effect choose, which is useless for testing: the armour forge gives each of the six
# types its own palette, and telling plate from mail means minting one of each on purpose.
pmg_debug_forge_armor_effect = {
	hidden_effect = {
		create_artifact_armor_effect = {
			OWNER = root
			CREATOR = root
			SET_ARMOR_TYPE = $TYPE$
		}

		scope:newly_created_artifact = {
			equip_artifact_to_owner_replace = yes
		}

		send_interface_toast = {
			type = msg_artifact_gained
			title = pmg_debug_forge.0001.armor
			right_icon = scope:newly_created_artifact
		}
	}
}


# Mints an armour artifact of a fixed TYPE at a chosen RARITY, so the rarity axis can be tested
# on its own. Rarity is normally DERIVED from quality and wealth rather than chosen, but
# set_artifact_rarity is a real engine effect and vanilla uses it directly on an artifact scope
# (prelude_events.txt:5526, ep2_locale_events.txt:6831), so it can simply be overwritten after
# creation.
#
# Type is held constant on purpose. The forge varies the SURFACE by rarity and the PALETTE by type,
# so changing both at once would leave you unable to say which axis moved.
pmg_debug_forge_rarity_effect = {
	hidden_effect = {
		create_artifact_armor_effect = {
			OWNER = root
			CREATOR = root
			SET_ARMOR_TYPE = $TYPE$
		}

		scope:newly_created_artifact = {
			set_artifact_rarity = $RARITY$
			equip_artifact_to_owner_replace = yes
		}
	}

	send_interface_toast = {
		title = pmg_debug_forge.0003.toast
		left_icon = root
		right_icon = scope:newly_created_artifact
	}
}
