# Candidate scope. AUTHORITY is the current head. Both interactions use this atomic final guard.
# The success scope is cleared first; callers charge only after the title is actually held.
gen_synod_grant_personal_seat_effect = {
    clear_saved_scope = gen_personal_seat
    clear_saved_scope = gen_personal_primary_title
    clear_saved_scope = gen_personal_county
    if = {
        limit = { gen_synod_can_grant_personal_seat_trigger = { AUTHORITY = $AUTHORITY$ } }
        save_scope_as = gen_personal_recipient
        faith = { save_scope_as = gen_personal_faith }
        if = {
            limit = { is_ruler = yes }
            primary_title = { save_scope_as = gen_personal_primary_title }
            save_scope_as = gen_personal_government_base
        }
        else_if = {
            limit = { $AUTHORITY$ = { government_has_flag = government_is_theocracy } }
            $AUTHORITY$ = { save_scope_as = gen_personal_government_base }
        }
        else = { save_scope_as = gen_personal_government_base }
        if = { limit = { exists = location.county } location.county = { save_scope_as = gen_personal_county } }
        else_if = { limit = { exists = liege.capital_county } liege.capital_county = { save_scope_as = gen_personal_county } }
        create_dynamic_title = { tier = duchy name = gen_synod_personal_seat_name adj = gen_synod_personal_seat_adj }
        scope:new_title ?= {
            save_scope_as = gen_personal_seat
            set_variable = gen_synod_personal
            set_variable = { name = gen_synod_faith value = scope:gen_personal_faith }
            add_clerical_elector = scope:gen_personal_faith
            if = {
                limit = { exists = scope:gen_personal_county }
                set_capital_county = scope:gen_personal_county
                set_coa = scope:gen_personal_county
                set_color_from_title = scope:gen_personal_county
            }
            set_landless_title = yes
            set_destroy_on_succession = yes
            set_no_automatic_claims = yes
            set_can_be_named_after_dynasty = no
            set_can_use_nomadic_naming = no
        }
        if = {
            limit = { exists = scope:gen_personal_seat }
            create_title_and_vassal_change = { type = created save_scope_as = gen_personal_change add_claim_on_loss = no }
            scope:gen_personal_seat = {
                change_title_holder = {
                    holder = scope:gen_personal_recipient
                    change = scope:gen_personal_change
                    government_base = scope:gen_personal_government_base
                }
            }
            resolve_title_and_vassal_change = scope:gen_personal_change
            if = {
                limit = { exists = scope:gen_personal_primary_title }
                set_primary_title_to = scope:gen_personal_primary_title
            }
            else = { change_government = ecclesiastical_government }
            $AUTHORITY$ = { gen_see_electors_grant_law_effect = yes }
        }
    }
}
