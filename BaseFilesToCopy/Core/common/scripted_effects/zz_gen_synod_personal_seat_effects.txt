# Character scope: the chaplain. A personal Synod Seat for a generated faith, granted by the chaplain
# petition (zz_gen_synod_chaplain_interaction.txt) in place of vanilla's grant_cardinal_title_effect.
#
# It is a vote tied to the man, not to a see: destroy_on_succession, so it ends with him. It carries
# var:gen_synod_faith (for the faith-aware seat checks) and var:gen_synod_personal, which every
# permanent-seat path excludes: it yields to a see's seat (gen_synod_remove_personal_seats_effect), is
# cleared when its holder becomes Head of Faith, and goes through vanilla's destroy body when stripped.
# It never enters vanilla's shared roma_cardinalates pool and sets no cardinal_flag.
#
# Not called inside a title change: the interaction's on_accept is its own transaction.
gen_synod_grant_personal_seat_effect = {
	save_scope_as = gen_personal_recipient
	faith = { save_scope_as = gen_personal_faith }
	if = {
		limit = { exists = location.county }
		location.county = { save_scope_as = gen_personal_county }
	}
	else_if = {
		limit = { exists = liege.capital_county }
		liege.capital_county = { save_scope_as = gen_personal_county }
	}
	# As vanilla: a theocratic government_base keeps a lowborn from founding a dynasty
	if = {
		limit = { faith.religious_head ?= { government_has_flag = government_is_theocracy } }
		faith.religious_head = { save_scope_as = gen_personal_government_base }
	}
	else = {
		save_scope_as = gen_personal_government_base
	}
	create_dynamic_title = {
		tier = duchy
		name = gen_synod_personal_seat_name
		adj = gen_synod_personal_seat_adj
	}
	scope:new_title ?= {
		save_scope_as = gen_personal_seat
		set_variable = gen_synod_personal
		set_variable = { name = gen_synod_faith value = scope:gen_personal_faith }
		add_clerical_elector = scope:gen_personal_faith
		if = {
			limit = { exists = scope:gen_personal_county }
			set_capital_county = scope:gen_personal_county
			set_coa = scope:gen_personal_county
			set_color_from_title = scope:gen_personal_county
		}
		set_landless_title = yes
		set_destroy_on_succession = yes
		set_no_automatic_claims = yes
		set_can_be_named_after_dynasty = no
		set_can_use_nomadic_naming = no
	}
	if = {
		limit = { exists = scope:gen_personal_seat }
		create_title_and_vassal_change = {
			type = created
			save_scope_as = gen_personal_change
			add_claim_on_loss = no
		}
		scope:gen_personal_seat = {
			change_title_holder = {
				holder = scope:gen_personal_recipient
				change = scope:gen_personal_change
				government_base = scope:gen_personal_government_base
			}
		}
		resolve_title_and_vassal_change = scope:gen_personal_change
		if = {
			limit = { NOT = { government_has_flag = government_is_theocracy } }
			change_government = ecclesiastical_government
		}
		scope:gen_personal_faith.religious_head ?= { gen_see_electors_grant_law_effect = yes }
	}
}
