# Gnome Tinkering: the workbench menu, the test, and the outcome. See 00_gen_tinker_triggers.txt for the
# map of the system.
#
#   0010        The Workbench: pick a project and pay for it (fired by the Start a Contraption decision)
#   0300        hidden: the test is due (fired by gen_tk_next_stage_effect); picks one of three scenes
#   0310–0312   the test scenes. Every option that rolls ends in gen_tk_final_roll_effect, which
#               hands the result to 0350. "One last adjustment" changes the numbers and re-opens the same
#               scene, so the odds shown on the rolling options are always the real ones.
#   0350        the outcome: project x result text, then gen_tk_apply_outcome_effect and
#               gen_tk_finish_project_effect (once)
#
# All run on the tinkerer. Effects and dice: common/scripted_effects/00_gen_tinker_outcome_effects.txt.
# Prices and prestige stakes: common/script_values/00_gen_tinker_cost_values.txt.


# =====================================================================================
#  0010 The Workbench
# =====================================================================================

gen_tinker.0010 = {
	type = character_event
	title = gen_tinker.0010.t
	desc = gen_tinker.0010.desc
	theme = learning
	override_background = { reference = study }

	left_portrait = {
		character = root
		animation = thinking
	}

	# A clockwork curio
	option = {
		name = gen_tinker.0010.a
		# Every project option re-checks the bench: a commission may have started while this was open.
		trigger = {
			NOT = { has_character_flag = gen_tk_working }
			gold >= gen_tk_cost_curio_value
		}
		show_as_unavailable = { always = yes }
		remove_short_term_gold = gen_tk_cost_curio_value
		custom_tooltip = gen_tinker.0010.a.tt
		gen_tk_begin_project_effect = { PROJECT = flag:curio }

		ai_chance = {
			base = 10
			ai_value_modifier = { ai_rationality = 0.25 }
			modifier = {
				add = 20
				OR = {
					has_trait = scholar
					has_trait = curious
				}
			}
			modifier = {
				add = 10
				OR = {
					has_trait = arrogant
					has_trait = gregarious
				}
			}
		}
	}

	# Waterworks
	option = {
		name = gen_tinker.0010.b
		trigger = {
			NOT = { has_character_flag = gen_tk_working }
			is_landed = yes
			exists = capital_county
			gold >= gen_tk_cost_mill_value
		}
		show_as_unavailable = { is_landed = yes }
		remove_short_term_gold = gen_tk_cost_mill_value
		custom_tooltip = gen_tinker.0010.b.tt
		gen_tk_begin_project_effect = { PROJECT = flag:mill }

		ai_chance = {
			base = 10
			ai_value_modifier = { ai_greed = 0.5 }
			modifier = {
				add = 25
				has_trait = greedy
			}
			modifier = {
				add = 20
				has_trait = diligent
			}
		}
	}

	# A war engine
	option = {
		name = gen_tinker.0010.c
		trigger = {
			NOT = { has_character_flag = gen_tk_working }
			is_landed = yes
			gold >= gen_tk_cost_engine_value
		}
		show_as_unavailable = { is_landed = yes }
		remove_short_term_gold = gen_tk_cost_engine_value
		custom_tooltip = gen_tinker.0010.c.tt
		gen_tk_begin_project_effect = { PROJECT = flag:engine }

		ai_chance = {
			base = 5
			ai_value_modifier = {
				ai_boldness = 0.25
				ai_vengefulness = 0.25
			}
			modifier = {
				add = 20
				has_trait = wrathful
			}
			modifier = {
				add = 15
				has_trait = ambitious
			}
			modifier = {
				add = 25
				is_at_war = yes
			}
		}
	}

	# A clockwork limb, for a maker missing one
	option = {
		name = gen_tinker.0010.d
		trigger = {
			NOT = { has_character_flag = gen_tk_working }
			OR = {
				AND = {
					has_trait = maimed
					NOT = { has_character_modifier = gen_tk_clockwork_hand_modifier }
					NOT = { has_character_modifier = gen_tk_clockwork_hand_masterwork_modifier }
				}
				AND = {
					has_trait = one_legged
					NOT = { has_character_modifier = gen_tk_clockwork_leg_modifier }
					NOT = { has_character_modifier = gen_tk_clockwork_leg_masterwork_modifier }
				}
			}
			gold >= gen_tk_cost_limb_value
		}
		remove_short_term_gold = gen_tk_cost_limb_value
		custom_tooltip = gen_tinker.0010.d.tt
		gen_tk_begin_project_effect = { PROJECT = flag:limb }

		ai_chance = {
			base = 100
		}
	}

	# A grand invention
	option = {
		name = gen_tinker.0010.e
		trigger = {
			NOT = { has_character_flag = gen_tk_working }
			OR = {
				learning >= 18
				has_character_modifier = gen_tk_master_artificer_modifier
			}
			gold >= gen_tk_cost_invention_value
		}
		show_as_unavailable = { always = yes }
		remove_short_term_gold = gen_tk_cost_invention_value
		custom_tooltip = gen_tinker.0010.e.tt
		gen_tk_begin_project_effect = { PROJECT = flag:invention }

		ai_chance = {
			base = 10
			ai_value_modifier = { ai_rationality = 0.5 }
			modifier = {
				add = 25
				OR = {
					has_trait = scholar
					has_trait = curious
				}
			}
			modifier = {
				add = 10
				has_trait = ambitious
			}
		}
	}

	# Not now
	option = {
		name = gen_tinker.0010.f
		custom_tooltip = gen_tinker.0010.f.tt

		ai_chance = {
			base = 5
			modifier = {
				add = 20
				has_trait = lazy
			}
			modifier = {
				add = 15
				has_trait = content
			}
		}
	}
}


# =====================================================================================
#  0300 The test is due (hidden)
# =====================================================================================

gen_tinker.0300 = {
	hidden = yes

	# Ignore a test left over from an abandoned project (main's serial check).
	trigger = { gen_tk_event_is_current_trigger = yes }

	immediate = {
		if = {
			limit = {
				gen_tk_can_work_trigger = yes
				var:gen_tk_stage ?= 3
			}
			random_list = {
				50 = { trigger_event = gen_tinker.0310 }
				25 = { trigger_event = gen_tinker.0311 }
				25 = {
					modifier = {
						add = 25
						has_trait = gregarious
					}
					trigger_event = gen_tinker.0312
				}
			}
		}
		# A stale test from a project that already ended does nothing (as 0100/0200 do).
		else_if = {
			limit = {
				has_character_flag = gen_tk_working
				var:gen_tk_stage ?= 3
			}
			gen_tk_abandon_project_effect = yes
		}
	}
}
