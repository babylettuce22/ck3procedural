namespace = gen_blood_challenge

# The aftermath of a Blood Challenge: what the beaten fighter does on the ground, and what the victor
# does with them.
#
# The rest of the chain: MAIN's interaction (00_gen_blood_challenge_interactions.txt) and hidden events
# 0100/0101/0500 (gen_blood_challenge_events.txt); AGENT C's pre-fight scene 0010-0099
# (gen_blood_challenge_scene_events.txt). MAIN's 0100 sends 0200 to the loser (alive and able) or 0300
# to the victor (loser alive but unable). This file owns 0200-0399:
#
#   0200  hidden router: checks the loser can act, draws two risk options, sends 0201
#   0201  "On the Ground" - the loser: Yield, or two of five risks (Blood Rage, Fight Dirty,
#         Defiant Howl, Reach for the Blade, Last Insult)
#   0300  hidden router: checks both are still alive and the victor can act, sends 0301
#   0301  "The Verdict" - the victor: Spare, Finish It, Take a Tusk (or Make Them Crawl)
#
# Every path ends exactly once in 0300 or gen_bc_finish_chain_effect (MAIN -> 0500). 0200 is never sent
# twice: after a turnaround the NEW victor gets the verdict, never a second "On the Ground".
# Helpers (wounds, kills, the verdict hand-off) are in 00_gen_blood_challenge_aftermath_effects.txt.
#
# House rules: every roll ends in a toast to the roller and one to the other fighter, including the
# "nothing worse happened" branches; the odds are shown (random_list / duel chances).


# =====================================================================================
#  0200: router for the downed loser
# =====================================================================================

# Hidden so that a loser who cannot act still moves the chain on (a failed trigger on a visible
# event would strand it). The draw is done here, once, so 0201's two risk options never change while
# it is open.
gen_blood_challenge.0200 = {
	type = character_event
	hidden = yes

	immediate = {
		save_scope_as = gen_bc_ab_downed
		if = {
			limit = { exists = scope:gen_bc_victor }
			scope:gen_bc_victor = { save_scope_as = gen_bc_ab_foe }
		}
		if = {
			limit = {
				exists = scope:gen_bc_ab_foe
				scope:gen_bc_ab_foe = { is_alive = yes }
				is_alive = yes
				NOT = { has_trait = incapable }
				is_imprisoned = no
			}
			# First risk option. Characters who suit an option are likelier to be offered it.
			random_list = {
				10 = {
					modifier = {
						add = 10
						OR = {
							has_trait = wrathful
							has_trait = berserker
							has_character_flag = gen_bc_stance_fury
						}
					}
					save_scope_value_as = { name = gen_bc_ab_pick_a value = flag:rage }
				}
				10 = {
					modifier = {
						add = 10
						OR = {
							has_trait = deceitful
							has_character_flag = gen_bc_stance_cunning
						}
					}
					save_scope_value_as = { name = gen_bc_ab_pick_a value = flag:dirty }
				}
				10 = {
					modifier = {
						add = 10
						OR = {
							has_trait = zealous
							has_character_flag = gen_bc_stance_ancestors
						}
					}
					save_scope_value_as = { name = gen_bc_ab_pick_a value = flag:howl }
				}
				10 = {
					save_scope_value_as = { name = gen_bc_ab_pick_a value = flag:blade }
				}
				10 = {
					modifier = {
						add = 10
						has_trait = arrogant
					}
					save_scope_value_as = { name = gen_bc_ab_pick_a value = flag:insult }
				}
			}
			# Second risk option: same pool, never the first one again.
			random_list = {
				10 = {
					trigger = { NOT = { scope:gen_bc_ab_pick_a = flag:rage } }
					modifier = {
						add = 10
						OR = {
							has_trait = wrathful
							has_trait = berserker
							has_character_flag = gen_bc_stance_fury
						}
					}
					save_scope_value_as = { name = gen_bc_ab_pick_b value = flag:rage }
				}
				10 = {
					trigger = { NOT = { scope:gen_bc_ab_pick_a = flag:dirty } }
					modifier = {
						add = 10
						OR = {
							has_trait = deceitful
							has_character_flag = gen_bc_stance_cunning
						}
					}
					save_scope_value_as = { name = gen_bc_ab_pick_b value = flag:dirty }
				}
				10 = {
					trigger = { NOT = { scope:gen_bc_ab_pick_a = flag:howl } }
					modifier = {
						add = 10
						OR = {
							has_trait = zealous
							has_character_flag = gen_bc_stance_ancestors
						}
					}
					save_scope_value_as = { name = gen_bc_ab_pick_b value = flag:howl }
				}
				10 = {
					trigger = { NOT = { scope:gen_bc_ab_pick_a = flag:blade } }
					save_scope_value_as = { name = gen_bc_ab_pick_b value = flag:blade }
				}
				10 = {
					trigger = { NOT = { scope:gen_bc_ab_pick_a = flag:insult } }
					modifier = {
						add = 10
						has_trait = arrogant
					}
					save_scope_value_as = { name = gen_bc_ab_pick_b value = flag:insult }
				}
			}
			trigger_event = gen_blood_challenge.0201
		}
		# Alive but unable (MAIN normally catches this in 0100): straight to the verdict.
		else_if = {
			limit = {
				exists = scope:gen_bc_ab_foe
				scope:gen_bc_ab_foe = { is_alive = yes }
				is_alive = yes
			}
			scope:gen_bc_ab_foe = { trigger_event = gen_blood_challenge.0300 }
		}
		else = {
			gen_bc_finish_chain_effect = yes
		}
	}
}

# Sent to scope:gen_bc_victor by MAIN's 0100 (loser alive but unable), by 0200, or by 0201 (after a
# yield, a beating, or a turnaround, when then the NEW victor gets it). Hidden for the same reason as
# 0200: a verdict with no one left to judge, or a victor who cannot give one, must still end the chain.
gen_blood_challenge.0300 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				exists = scope:gen_bc_loser
				scope:gen_bc_loser = { is_alive = yes }
				is_alive = yes
				NOT = { has_trait = incapable }
				is_imprisoned = no
			}
			scope:gen_bc_loser = { save_scope_as = gen_bc_ab_fallen }
			save_scope_as = gen_bc_ab_judge	# names the victor in toasts sent to the fallen
			trigger_event = gen_blood_challenge.0301
		}
		# No one to judge, or no one fit to judge: the loser is simply left alive.
		else = {
			gen_bc_finish_chain_effect = yes
		}
	}
}
