namespace = gen_great_work

# The high elves' Great Work: the scenes of a maker's long labour. See 00_gen_great_work_triggers.txt for
# the map of the whole system; main owns the story, the oath decision and every effect called here.
#
# This file owns:
#   0010       The Oath — the kind of work is chosen (the decision has already paid for it)
#   0100       hidden beat dispatcher, fired by the story every few years on its owner
#   0101–0109  the beats of a life: the true material, a vision, the long doubt, a jealous kinsman, a
#              rival's work, a demand to see it, the work and the hearth, a pupil, fire or war
#   0510       The Work Is Finished — the maker is World-Weary and the work is far enough on
#   0600       hidden succession dispatcher, fired on the dead maker's heir
#   0601–0604  the maker's last wish, the heir's choice, the pupil's plea, a rival claim
#
# Every succession path ends in exactly one gen_gw_pass_work_effect or gen_gw_seal_unfinished_effect,
# and only characters passing gen_gw_can_take_up_trigger may be handed the work. The last option of each
# succession event is always available, so no path can stall.
#
# Beat modifiers: common/modifiers/00_gen_great_work_beat_modifiers.txt.

# =====================================================================================
#  0100: the beat dispatcher (hidden)
# =====================================================================================

# On the story's owner. Finds who is around to take part, then picks one beat, weighted by the kind of
# work, how far on it is, and what has happened lately (each beat leaves gen_gw_seen_01xx for 20 years).
gen_great_work.0100 = {
	type = character_event
	hidden = yes

	immediate = {
		gen_gw_save_story_effect = yes
		if = {
			limit = {
				exists = scope:gw_story
				is_alive = yes
			}

			# ---------------------------------------------------------------- who is around
			# A dwarf lord who might sell the true material.
			random_ruler = {
				limit = {
					has_trait = phenotype_stocky
					is_adult = yes
					highest_held_title_tier >= tier_county
					NOT = { this = root }
				}
				save_scope_as = gw_dwarf
			}
			# A kinsman with a grievance: the grasping sort first, anyone grown otherwise.
			random_close_family_member = {
				limit = {
					is_alive = yes
					is_adult = yes
					is_imprisoned = no
					NOT = { this = root }
					OR = {
						NOT = { exists = scope:gw_pupil }
						NOT = { this = scope:gw_pupil }
					}
					OR = {
						has_trait = ambitious
						has_trait = greedy
						has_trait = arrogant
					}
				}
				alternative_limit = {
					is_alive = yes
					is_adult = yes
					is_imprisoned = no
					NOT = { this = root }
					OR = {
						NOT = { exists = scope:gw_pupil }
						NOT = { this = scope:gw_pupil }
					}
				}
				save_scope_as = gw_kin
			}
			# A rival: another elf's Great Work if there is one, else an elf of standing and learning.
			random_ruler = {
				limit = {
					NOT = { this = root }
					gen_gw_has_work_trigger = yes
				}
				alternative_limit = {
					NOT = { this = root }
					NOT = { is_close_family_of = root }
					is_adult = yes
					gen_is_elf_phenotype_trigger = yes
					highest_held_title_tier >= tier_duchy
					learning >= 12
				}
				save_scope_as = gw_rival
			}
			# Someone who wants to see it: the liege, else a suitor for the unmarried, else a neighbour.
			if = {
				limit = {
					is_independent_ruler = no
					exists = liege
					liege = { NOT = { this = root } }
				}
				liege = { save_scope_as = gw_asker }
				save_scope_value_as = {
					name = gw_asker_kind
					value = flag:liege
				}
			}
			else_if = {
				limit = {
					is_adult = yes
					NOT = { exists = primary_spouse }
					any_neighboring_top_liege_realm_owner = {
						is_adult = yes
						NOT = { exists = primary_spouse }
					}
				}
				random_neighboring_top_liege_realm_owner = {
					limit = {
						is_adult = yes
						NOT = { exists = primary_spouse }
					}
					save_scope_as = gw_asker
				}
				save_scope_value_as = {
					name = gw_asker_kind
					value = flag:suitor
				}
			}
			else_if = {
				limit = {
					any_neighboring_top_liege_realm_owner = { is_adult = yes }
				}
				random_neighboring_top_liege_realm_owner = {
					limit = { is_adult = yes }
					save_scope_as = gw_asker
				}
				save_scope_value_as = {
					name = gw_asker_kind
					value = flag:neighbour
				}
			}
			# The family the work keeps the maker from.
			if = {
				limit = { primary_spouse ?= { is_alive = yes } }
				primary_spouse = { save_scope_as = gw_hearth }
				save_scope_value_as = {
					name = gw_hearth_kind
					value = flag:spouse
				}
			}
			else_if = {
				limit = {
					any_child = {
						is_alive = yes
						is_courtier_of = root
					}
				}
				random_child = {
					limit = {
						is_alive = yes
						is_courtier_of = root
					}
					save_scope_as = gw_hearth
				}
				save_scope_value_as = {
					name = gw_hearth_kind
					value = flag:child
				}
			}
			# A pupil, while there is none: the sharpest mind at court (elves first), and the maker's own child.
			if = {
				limit = { NOT = { exists = scope:gw_pupil } }
				ordered_courtier = {
					limit = {
						age >= 12
						learning >= 8
						is_imprisoned = no
						NOT = { has_trait = incapable }
					}
					order_by = {
						value = learning
						if = {
							limit = { gen_is_elf_phenotype_trigger = yes }
							add = 10
						}
					}
					save_scope_as = gw_candidate
				}
				if = {
					limit = { exists = scope:gw_candidate }
					ordered_child = {
						limit = {
							is_alive = yes
							age >= 12
							learning >= 6
							is_imprisoned = no
							NOT = { has_trait = incapable }
							NOT = { this = scope:gw_candidate }
						}
						order_by = learning
						save_scope_as = gw_child_candidate
					}
				}
			}

			# ---------------------------------------------------------------- which beat
			random_list = {
				# The true material: mostly early.
				10 = {
					modifier = {
						add = 20
						scope:gw_story.var:gw_progress < 25
					}
					modifier = {
						factor = 0.3
						scope:gw_story.var:gw_progress >= 60
					}
					modifier = {
						add = 5
						OR = {
							scope:gw_story.var:gw_kind = flag:jewel
							scope:gw_story.var:gw_kind = flag:regalia
							scope:gw_story.var:gw_kind = flag:lamp
						}
					}
					modifier = {
						add = 5
						exists = scope:gw_dwarf
					}
					modifier = {
						factor = 0.05
						has_character_flag = gen_gw_seen_0101
					}
					trigger_event = gen_great_work.0101
				}
				# A flash of vision: any time.
				15 = {
					modifier = {
						add = 5
						OR = {
							scope:gw_story.var:gw_kind = flag:harp
							scope:gw_story.var:gw_kind = flag:chronicle
						}
					}
					modifier = {
						factor = 0.1
						has_character_flag = gen_gw_seen_0102
					}
					trigger_event = gen_great_work.0102
				}
				# The long doubt: the middle years.
				5 = {
					modifier = {
						add = 20
						scope:gw_story.var:gw_progress >= 25
						scope:gw_story.var:gw_progress < 60
					}
					modifier = {
						add = 10
						scope:gw_story.var:gw_progress >= 60
					}
					modifier = {
						factor = 0.05
						has_character_flag = gen_gw_seen_0103
					}
					trigger_event = gen_great_work.0103
				}
				# A jealous kinsman.
				12 = {
					trigger = { exists = scope:gw_kin }
					modifier = {
						add = 10
						scope:gw_story.var:gw_kind = flag:regalia
					}
					modifier = {
						factor = 0.05
						has_character_flag = gen_gw_seen_0104
					}
					trigger_event = gen_great_work.0104
				}
				# A rival's work.
				8 = {
					trigger = { exists = scope:gw_rival }
					modifier = {
						add = 7
						scope:gw_rival = { gen_gw_has_work_trigger = yes }
					}
					modifier = {
						add = 5
						scope:gw_story.var:gw_kind = flag:jewel
					}
					modifier = {
						factor = 0.05
						has_character_flag = gen_gw_seen_0105
					}
					trigger_event = gen_great_work.0105
				}
				# A demand to see it: more as the work becomes something worth seeing.
				10 = {
					trigger = { exists = scope:gw_asker }
					modifier = {
						add = 8
						scope:gw_story.var:gw_progress >= 50
					}
					modifier = {
						factor = 0.05
						has_character_flag = gen_gw_seen_0106
					}
					trigger_event = gen_great_work.0106
				}
				# The work and the hearth: the middle years.
				12 = {
					trigger = { exists = scope:gw_hearth }
					modifier = {
						add = 8
						scope:gw_story.var:gw_progress >= 25
						scope:gw_story.var:gw_progress < 75
					}
					modifier = {
						factor = 0.05
						has_character_flag = gen_gw_seen_0107
					}
					trigger_event = gen_great_work.0107
				}
				# A pupil: more pressing as the years go on.
				5 = {
					trigger = {
						NOT = { exists = scope:gw_pupil }
						exists = scope:gw_candidate
					}
					modifier = {
						add = 20
						scope:gw_story.var:gw_progress >= 25
					}
					modifier = {
						add = 20
						scope:gw_story.var:gw_progress >= 60
					}
					modifier = {
						add = 10
						scope:gw_story.var:gw_kind = flag:chronicle
					}
					modifier = {
						factor = 0.1
						has_character_flag = gen_gw_seen_0108
					}
					trigger_event = gen_great_work.0108
				}
				# Fire or war at the workshop.
				6 = {
					modifier = {
						add = 20
						is_at_war = yes
					}
					modifier = {
						factor = 0.05
						has_character_flag = gen_gw_seen_0109
					}
					trigger_event = gen_great_work.0109
				}
			}
		}
	}
}

gen_great_work.0101 = {
	type = character_event
	title = gen_great_work.0101.t
	desc = {
		first_valid = {

				trigger = { scope:gw_story.var:gw_kind ?= flag:harp }
				desc = gen_great_work.0101.desc.harp
			}
			desc = gen_great_work.0101.desc.lamp
		}
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:gw_dwarf }
				desc = gen_great_work.0101.desc.dwarf
			}
			desc = gen_great_work.0101.desc.alone
		}
	}
	theme = learning
	override_background = { reference = wilderness_mountains }

	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		trigger = { exists = scope:gw_dwarf }
		character = scope:gw_dwarf
		animation = personality_rational
	}

	trigger = { gen_gw_has_work_trigger = yes }

	immediate = {
		gen_gw_save_story_effect = yes
		add_character_flag = {
			flag = gen_gw_seen_0101
			years = 20
		}
	}

	# Buy it from the dwarves.
	option = {
		name = gen_great_work.0101.a
		trigger = {
			exists = scope:gw_dwarf
			scope:gw_dwarf = { is_alive = yes }
			gold >= medium_gold_value
		}
		show_as_unavailable = { exists = scope:gw_dwarf }
		pay_short_term_gold = {
			target = scope:gw_dwarf
			gold = medium_gold_value
		}
		scope:gw_dwarf = {
			add_opinion = {
				target = root
				modifier = pleased_opinion
				opinion = 15
			}
		}
		add_character_modifier = {
			modifier = gen_gw_beat_true_material_modifier
			years = 5
		}
		gen_gw_progress_effect = {
			PROGRESS = 12
			QUALITY = 10
		}

		ai_chance = {
			base = 30
			ai_value_modifier = { ai_greed = -0.5 }
			modifier = {
				factor = 0.3
				has_trait = greedy
			}
			modifier = {
				factor = 0
				gold < medium_gold_value
			}
		}
	}

	# Go down into the deep places yourself.
	option = {
		name = gen_great_work.0101.b
		random_list = {
			60 = {
				desc = gen_great_work.0101.b.found
				modifier = {
					add = 20
					has_trait = brave
				}
				modifier = {
					add = 10
					prowess >= 12
				}
				send_interface_toast = {
					type = event_toast_effect_good
					title = gen_great_work.0101.b.found.toast
					left_icon = root
					custom_tooltip = gen_great_work.0101.b.found.tt
					add_prestige = minor_prestige_gain
					add_character_modifier = {
						modifier = gen_gw_beat_true_material_modifier
						years = 5
					}
				}
				gen_gw_progress_effect = {
					PROGRESS = 12
					QUALITY = 15
				}
			}
			40 = {
				desc = gen_great_work.0101.b.toll
				modifier = {
					add = 20
					has_trait = craven
				}
				send_interface_toast = {
					type = event_toast_effect_bad
					title = gen_great_work.0101.b.toll.toast
					left_icon = root
					custom_tooltip = gen_great_work.0101.b.toll.tt
					add_stress = medium_stress_gain
				}
				gen_gw_progress_effect = {
					PROGRESS = 6
					QUALITY = 5
				}
			}
		}
		stress_impact = {
			craven = minor_stress_impact_gain
		}

		ai_chance = {
			base = 20
			ai_value_modifier = { ai_boldness = 0.5 }
			modifier = {
				add = 20
				has_trait = brave
			}
			modifier = {
				factor = 0.2
				has_trait = craven
			}
		}
	}

		}

# =====================================================================================
#  0600: the succession dispatcher (hidden)
# =====================================================================================

# On the dead maker's heir, with scope:gw_story, scope:gw_dead_maker, scope:gw_pupil (if alive) and
# scope:gw_heir saved by the story. Looks for someone of the family who wants the work, then opens the
# last wish (if the maker named a pupil who could carry it) or the heir's choice.
gen_great_work.0600 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = { exists = scope:gw_story }
			# For the toasts others receive from this heir's choices.
			save_scope_as = gw_decider

			# The pupil, if the story did not pass one.
			if = {
				limit = {
					NOT = { exists = scope:gw_pupil }
					scope:gw_story.var:gw_pupil ?= { is_alive = yes }
				}
				scope:gw_story.var:gw_pupil = { save_scope_as = gw_pupil }
			}

			# Family who might want it: the maker's other children, and, when the heir is the maker's
			# child, the heir's other parent (as a rule the maker's spouse). The holder may be the pupil
			# rather than the heir (main's on_owner_death), whose parents are nothing to the work.
			scope:gw_dead_maker ?= {
				every_child = {
					limit = { is_alive = yes }
					add_to_temporary_list = gw_kin_list
				}
			}
			if = {
				limit = { is_child_of = scope:gw_dead_maker }
				mother ?= {
					if = {
						limit = {
							is_alive = yes
							NOT = { this = scope:gw_dead_maker }
						}
						add_to_temporary_list = gw_kin_list
					}
				}
				father ?= {
					if = {
						limit = {
							is_alive = yes
							NOT = { this = scope:gw_dead_maker }
						}
						add_to_temporary_list = gw_kin_list
					}
				}
			}
			ordered_in_list = {
				list = gw_kin_list
				limit = {
					gen_gw_can_take_up_trigger = yes
					NOT = { this = root }
					OR = {
						NOT = { exists = scope:gw_pupil }
						NOT = { this = scope:gw_pupil }
					}
				}
				order_by = {
					value = learning
					if = {
						limit = {
							OR = {
								has_trait = ambitious
								has_trait = diligent
							}
						}
						add = 10
					}
				}
				save_scope_as = gw_claimant
			}

			if = {
				limit = {
					exists = scope:gw_pupil
					scope:gw_pupil = {
						gen_gw_can_take_up_trigger = yes
						NOT = { this = root }
					}
				}
				trigger_event = gen_great_work.0601
			}
			else = {
				trigger_event = gen_great_work.0602
			}
		}
	}
}
