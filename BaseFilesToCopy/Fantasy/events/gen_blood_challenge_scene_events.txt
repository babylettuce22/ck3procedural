namespace = gen_blood_challenge

# Blood Challenge, part 2 of 4: the pre-fight scene (gen_blood_challenge.0010–0099).
#
#   interaction accepted (main: common/character_interactions/00_gen_blood_challenge_interactions.txt)
#     → 0010 here, a hidden dispatcher on scope:gen_bc_viewer
#       → one of six scenes, 0011–0016; every option picks a stance, then gen_bc_start_combat_effect
#         → vanilla single combat → 0100 (main) → aftermath 0200/0300 (events/gen_blood_challenge_aftermath_events.txt)
#           → 0500 (main), which calls gen_bc_clear_stance_effect on both fighters
#
# Scenes:
#   0011 The High Seat       upward only: the fight happens in the higher ruler's hall
#   0012 The Clan Circle     peer only: called out at the fire of a clan gathering
#   0013 An Example          downward only: a chief makes an example of a vassal
#   0014 Spit in the Dust    any kind: the opponent insults the viewer before the circle
#   0015 The Omens           any kind: a shaman draws the circle and offers war-paint and omens
#   0016 Your Blood Watches  any kind, needs a child or heir aged 8+: family watches the fight
#
# The viewer may be the challenger or the defender, so every scene has two openings (first_valid on
# root = scope:gen_bc_challenger) and a shared stakes paragraph keyed on scope:gen_bc_kind.
#
# Stances (common/scripted_effects/00_gen_blood_challenge_scene_effects.txt) set one of three flags
# the aftermath reads: gen_bc_stance_fury / _cunning / _ancestors. "Just fight" sets none.
# Every option calls gen_bc_start_combat_effect = yes exactly once, as its last line.
#
# Scene events have no trigger block on purpose: a scene that fails to fire would leave the
# accepted challenge hanging. Each scene has a plain option with a flat positive ai_chance, so an
# AI viewer always has something to pick.


# =====================================================================================
#  0010 Dispatcher
# =====================================================================================

# Hidden. Picks a scene. The three kind-specific scenes cover every kind, so at least one
# weighted entry is always valid. A 5-year "seen" flag set by each scene cuts its weight, so
# repeat challengers see different scenes.
gen_blood_challenge.0010 = {
	type = character_event
	hidden = yes

	immediate = {
		gen_bc_scene_setup_effect = yes
		random_list = {
			100 = {
				trigger = { scope:gen_bc_kind = flag:upward }
				modifier = {
					has_character_flag = gen_bc_seen_scene_0011
					factor = 0.3
				}
				trigger_event = gen_blood_challenge.0011
			}
			100 = {
				trigger = { scope:gen_bc_kind = flag:peer }
				modifier = {
					has_character_flag = gen_bc_seen_scene_0012
					factor = 0.3
				}
				trigger_event = gen_blood_challenge.0012
			}
			100 = {
				trigger = { scope:gen_bc_kind = flag:downward }
				modifier = {
					has_character_flag = gen_bc_seen_scene_0013
					factor = 0.3
				}
				trigger_event = gen_blood_challenge.0013
			}
			60 = {
				# More likely when the opponent is the kind to mouth off.
				modifier = {
					scope:gen_bc_scene_opponent = {
						OR = {
							has_trait = arrogant
							has_trait = wrathful
							has_trait = callous
						}
					}
					add = 30
				}
				modifier = {
					has_character_flag = gen_bc_seen_scene_0014
					factor = 0.3
				}
				trigger_event = gen_blood_challenge.0014
			}
			60 = {
				modifier = {
					has_trait = zealous
					add = 20
				}
				modifier = {
					has_character_flag = gen_bc_seen_scene_0015
					factor = 0.3
				}
				trigger_event = gen_blood_challenge.0015
			}
			50 = {
				# Same watcher test as 0016's immediate block.
				trigger = {
					OR = {
						primary_heir ?= {
							age >= 8
							is_imprisoned = no
							NOT = { this = scope:gen_bc_challenger }
							NOT = { this = scope:gen_bc_defender }
						}
						any_child = {
							age >= 8
							is_imprisoned = no
							NOT = { this = scope:gen_bc_challenger }
							NOT = { this = scope:gen_bc_defender }
						}
					}
				}
				modifier = {
					has_character_flag = gen_bc_seen_scene_0016
					factor = 0.3
				}
				trigger_event = gen_blood_challenge.0016
			}
		}
	}
}

				ai_rationality = -0.25
			}
			modifier = {
				has_trait = impatient
				add = 15
			}
		}
		gen_bc_start_combat_effect = yes
	}

	# Fury, berserker/wrathful
	option = {
		name = gen_blood_challenge.0011.a_blood
		trigger = {
			OR = {
				has_trait = berserker
				has_trait = wrathful
			}
		}
		gen_bc_apply_blood_fury_stance_effect = yes
		ai_chance = {
			base = 40
			ai_value_modifier = { ai_boldness = 0.5 }

			wrathful = minor_stress_impact_gain
