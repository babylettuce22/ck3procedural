namespace = gen_tinker

# Gnome Tinkering: the two middle stages of every project, the design and the setback.
# See 00_gen_tinker_triggers.txt for the map of the system.
#
# gen_tk_begin_project_effect (main) schedules 0100; gen_tk_next_stage_effect (main) takes 0100's scene to
# 0200 and 0200's scene to the test (0300, AGENT E). Every option of every scene here ends in exactly one
# gen_tk_next_stage_effect, except 0212's "find out who", whose success hands off to 0213 instead (which
# then ends in it).
#
#   0100        hidden: design dispatcher. Checks the maker can work (else abandons), saves the cast, draws one of:
#   0101          The Blueprint          careful / bigger / the dwarvish plan / simple / (scholar) do the sums
#   0102          Brass and Bolts        scrap / good brass / melt the silver / (steward) find it in the accounts
#   0103          An Apprentice          (a child or ward, and/or a clever courtier, and no apprentice yet)
#   0104          A Letter from a Rival  (the maker has a rival)
#   0105          The Patron's Letter    (a commission) bigger / faster / cheaper, rolled from the patron's traits
#   0106          An Idea in the Night   a gamble on a dream, shown honestly
#   0107          The Noise              (a spouse at court or a steward) complaint about the hammering
#   0200        hidden: setback dispatcher, same pattern. One of:
#   0201–0205     the project's own accident: the war engine's shed, the waterworks' flood, the curio's
#                 chiming, the limb's pinch, the invention's missing notes
#   0210          The Dwarf-Made Part
#   0211          Fire in the Workshop
#   0212          Tampered With          (a resentful courtier) -> 0213 Caught, if the maker finds them out
#   0215          The Apprentice         (an apprentice) a mistake or a breakthrough, rolled from their learning
#   0216          The Money Runs Out
#   0217          A Visitor in the Workshop (a small child at court)
#   0218          The Inspection         (a commission) the patron's envoy comes to look
#
# Weights: each scene is drawn by project, by the maker's traits, by who is at court, by the running
# thread (flags set by design options, read by the setback dispatcher) and at a quarter of its weight if it
# was seen in the last four years (gen_tk_seen_NNNN). The thread, the cast and the cleanup are documented in
# common/scripted_effects/00_gen_tinker_workshop_effects.txt. Quality/risk move only through main's
# gen_tk_adjust_effect.
#
# House rules: every roll ends in a toast or a follow-up event that says what happened; flags the tooltip
# cannot draw get a custom_tooltip; every option has an ai_chance.


# =====================================================================================
#  0100: design dispatcher
# =====================================================================================

gen_tinker.0100 = {
	type = character_event
	hidden = yes

	# Ignore a stage event left over from an abandoned project (main's serial check).
	trigger = { gen_tk_event_is_current_trigger = yes }

	immediate = {
		if = {
			limit = {
				gen_tk_can_work_trigger = yes
				var:gen_tk_stage ?= 1
			}
			gen_tk_ws_save_design_cast_effect = yes
			random_list = {
				# The Blueprint: always possible. Grand projects need the most planning.
				10 = {
					modifier = {
						add = 8
						OR = {
							gen_tk_project_is_trigger = { PROJECT = flag:engine }
							gen_tk_project_is_trigger = { PROJECT = flag:invention }
						}
					}
					modifier = {
						factor = 0.25
						has_character_flag = gen_tk_seen_0101
					}
					trigger_event = gen_tinker.0101
				}
				# Brass and Bolts: always possible. Fine work needs fine metal.
				10 = {
					modifier = {
						add = 8
						OR = {
							gen_tk_project_is_trigger = { PROJECT = flag:curio }
							gen_tk_project_is_trigger = { PROJECT = flag:limb }
						}
					}
					modifier = {
						add = 4
						exists = scope:gen_tk_d_steward
					}
					modifier = {
						factor = 0.25
						has_character_flag = gen_tk_seen_0102
					}
					trigger_event = gen_tinker.0102
				}
				# An Apprentice: someone to ask, and the stool is empty.
				12 = {
					trigger = {
						NOT = { exists = scope:gen_tk_d_appr }
						OR = {
							exists = scope:gen_tk_d_young
							exists = scope:gen_tk_d_clever
						}
					}
					modifier = {
						add = 6
						exists = scope:gen_tk_d_young
					}
					modifier = {
						add = 4
						has_trait = gregarious
					}
					modifier = {
						factor = 0.25
						has_character_flag = gen_tk_seen_0103
					}
					trigger_event = gen_tinker.0103
				}
				# A Letter from a Rival.
				8 = {
					trigger = { exists = scope:gen_tk_d_rival }
					modifier = {
						add = 8
						scope:gen_tk_d_rival = { gen_tk_has_tinkering_trigger = yes }
					}
					modifier = {
						factor = 0.25
						has_character_flag = gen_tk_seen_0104
					}
					trigger_event = gen_tinker.0104
				}
				# The Patron's Letter: a commission nearly always hears from its patron.
				25 = {
					trigger = { exists = scope:gen_tk_d_patron }
					modifier = {
						factor = 0.5
						has_character_flag = gen_tk_seen_0105
					}
					trigger_event = gen_tinker.0105
				}
				# An Idea in the Night: the curious and the learned dream of work.
				8 = {
					modifier = {
						add = 6
						OR = {
							has_trait = curious
							has_trait = erudite # 'scholar' until 1.20 renamed it
							has_trait = eccentric
						}
					}
					modifier = {
						add = 6
						gen_tk_project_is_trigger = { PROJECT = flag:invention }
					}
					modifier = {
						factor = 0.25
						has_character_flag = gen_tk_seen_0106
					}
					trigger_event = gen_tinker.0106
				}
				# The Noise: someone to complain. Engines and waterworks are the loudest.
				8 = {
					trigger = {
						OR = {
							exists = scope:gen_tk_d_spouse
							exists = scope:gen_tk_d_steward
						}
					}
					modifier = {
						add = 8
						OR = {
							gen_tk_project_is_trigger = { PROJECT = flag:engine }
							gen_tk_project_is_trigger = { PROJECT = flag:mill }
						}
					}
					modifier = {
						factor = 0.25
						has_character_flag = gen_tk_seen_0107
					}
					trigger_event = gen_tinker.0107
				}
			}
		}
		# The maker can no longer work at it (jailed, incapable). A stage mismatch is a stale event from an
		# earlier project and does nothing.
		else_if = {
			limit = {
				has_character_flag = gen_tk_working
				NOT = { gen_tk_can_work_trigger = yes }
			}
			gen_tk_abandon_project_effect = yes
		}
	}
}

			}
		}

		name = gen_tinker.0101.d

		stress_impact = {
			ambitious = minor_stress_impact_gain
		}

				ai_boldness = -0.5
			}

					has_trait = lazy

			}
		}
		gen_tk_next_stage_effect = yes

	# Do the sums properly: only a scholar or a genius can.

		name = gen_tinker.0101.e
		trigger = {
			OR = {
				has_trait = erudite
				has_trait = intellect_good

		}

				factor = 0.5

			}
		}
		gen_tk_next_stage_effect = yes
	}
}

gen_tinker.0102 = {
	type = character_event
	title = gen_tinker.0102.t
	desc = {
		desc = gen_tinker.0102.desc
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:gen_tk_d_steward }
				desc = gen_tinker.0102.desc.steward
			}
			desc = gen_tinker.0102.desc.alone
		}
	}
	theme = stewardship
	override_background = { reference = market }

	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		trigger = { exists = scope:gen_tk_d_steward }
		character = scope:gen_tk_d_steward
		animation = steward
	}

	immediate = {
		add_character_flag = {
			flag = gen_tk_seen_0102
			years = 4
		}
	}

	# Scrap.
	option = {
		name = gen_tinker.0102.a
		gen_tk_adjust_effect = { QUALITY = -5 RISK = 5 }
		add_character_flag = gen_tk_wf_cheap_parts
		custom_tooltip = gen_tk_ws_wf_cheap_parts.tt
		stress_impact = {
			greedy = minor_stress_impact_loss
		}
		ai_chance = {
			base = 15
			ai_value_modifier = {
				ai_greed = 0.5
			}
			modifier = {
				add = 20
				gold < minor_gold_value
			}
			min = 1	# the fallback: always some chance
		}
		gen_tk_next_stage_effect = yes
	}

	# Good brass.
	option = {
		name = gen_tinker.0102.b
		trigger = { gold >= minor_gold_value }
		show_as_unavailable = { always = yes }
		remove_short_term_gold = minor_gold_value
		gen_tk_adjust_effect = { QUALITY = 10 RISK = -5 }
		add_character_flag = gen_tk_wf_fine_brass
		custom_tooltip = gen_tk_ws_wf_fine_brass.tt
		stress_impact = {
			greedy = minor_stress_impact_gain
		}
		ai_chance = {
			base = 25
			ai_value_modifier = {
				ai_greed = -0.5
			}
			modifier = {
				add = 10
				has_trait = generous
			}
		}
		gen_tk_next_stage_effect = yes
	}

	# Melt the silver.
	option = {
		name = gen_tinker.0102.c
		add_prestige = minor_prestige_loss
		gen_tk_adjust_effect = { QUALITY = 10 RISK = 0 }
		if = {
			limit = { exists = scope:gen_tk_d_spouse }
			scope:gen_tk_d_spouse = {
				add_opinion = {
					target = root
					modifier = annoyed_opinion
					opinion = -10
				}
			}
		}
		stress_impact = {
			arrogant = minor_stress_impact_gain
		}
		ai_chance = {
			base = 10
			ai_value_modifier = {
				ai_boldness = 0.25
				ai_compassion = -0.25
			}
			modifier = {
				add = 10
				has_trait = eccentric
			}
		}
		gen_tk_next_stage_effect = yes
	}

	# Let the steward find it. The better the steward, the likelier the money turns up.
	option = {
		name = gen_tinker.0102.d
		trigger = { exists = scope:gen_tk_d_steward }
		random_list = {
			30 = {
				desc = gen_tinker.0102.d.found
				modifier = {
					add = {
						value = scope:gen_tk_d_steward.stewardship
						multiply = 3
					}
				}
				send_interface_toast = {
					type = event_toast_effect_good
					title = gen_tinker.0102.d.found.t
					left_icon = root
					right_icon = scope:gen_tk_d_steward
					gen_tk_adjust_effect = { QUALITY = 10 RISK = 0 }
				}
			}
			# Nothing found: it's the midden after all, and the steward resents the wasted month.
			50 = {
				desc = gen_tinker.0102.d.lost
				send_interface_toast = {
					type = event_toast_effect_bad
					title = gen_tinker.0102.d.lost.t
					left_icon = root
					right_icon = scope:gen_tk_d_steward
					gen_tk_adjust_effect = { QUALITY = -5 RISK = 5 }
					add_character_flag = gen_tk_wf_cheap_parts
					custom_tooltip = gen_tk_ws_wf_cheap_parts.tt
					scope:gen_tk_d_steward = {
						add_opinion = {
							target = root
							modifier = annoyed_opinion
							opinion = -10
						}
					}
				}
			}
		}
		ai_chance = {
			base = 15
			modifier = {
				add = 15
				scope:gen_tk_d_steward = { stewardship >= 14 }
			}
		}
		gen_tk_next_stage_effect = yes
	}
}

	option = {
		name = gen_tinker.0218.a

			limit = { has_character_flag = gen_tk_wf_patron_fooled }

			scope:gen_tk_s_patron = {
				add_opinion = {
					target = root
					modifier = disappointed_opinion

					target = root
					modifier = pleased_opinion
					opinion = 10
				}
			}
		}
		random_list = {
			30 = {
				desc = gen_tinker.0218.a.poked

					type = event_toast_effect_bad
					title = gen_tinker.0218.a.poked.t

					type = event_toast_effect_neutral

		if = {
			limit = { has_character_flag = gen_tk_wf_patron_fooled }
			custom_tooltip = gen_tinker.0218.secret_kept_tt
		}
		random_list = {
			60 = {
				desc = gen_tinker.0218.b.impressed
				modifier = {
					add = -20
					has_character_flag = gen_tk_wf_patron_fooled
				}
				send_interface_toast = {
					type = event_toast_effect_good
					title = gen_tinker.0218.b.impressed.t
					left_icon = root
					right_icon = scope:gen_tk_s_patron
					add_prestige = minor_prestige_gain
					scope:gen_tk_s_patron = {
						add_opinion = {
							target = root
							modifier = pleased_opinion
							opinion = 10
						}
					}
				}
				scope:gen_tk_s_patron = {
					send_interface_toast = {
						type = event_toast_effect_good
						title = gen_tinker.0218.b.impressed.patron
						left_icon = scope:gen_tk_maker
						custom_tooltip = gen_tinker.0218.b.impressed.patron.tt
					}
				}
			}
			40 = {
				desc = gen_tinker.0218.b.collapsed
				modifier = {
					add = 20
					has_character_flag = gen_tk_wf_patron_fooled
				}
				send_interface_toast = {
					type = event_toast_effect_bad
					title = gen_tinker.0218.b.collapsed.t
					left_icon = root
					right_icon = scope:gen_tk_s_patron
					gen_tk_adjust_effect = { QUALITY = -5 RISK = 0 }
					scope:gen_tk_s_patron = {
						add_opinion = {
							target = root
							modifier = disappointed_opinion
							opinion = -10
						}
					}
				}
				scope:gen_tk_s_patron = {
					send_interface_toast = {
						type = event_toast_effect_bad
						title = gen_tinker.0218.b.collapsed.patron
						left_icon = scope:gen_tk_maker
						custom_tooltip = gen_tinker.0218.b.collapsed.patron.tt
					}

		}
		ai_chance = {
			base = 15
			ai_value_modifier = {
				ai_boldness = 0.5
			}
		}
		gen_tk_next_stage_effect = yes
	}

	# Keep the envoy busy in the best room.
	option = {
		name = gen_tinker.0218.c
		trigger = { gold >= tiny_gold_value }
		show_as_unavailable = { always = yes }
		remove_short_term_gold = tiny_gold_value
		# Safe for the build; hides a change of plan until delivery.
		if = {
			limit = { has_character_flag = gen_tk_wf_patron_fooled }
			custom_tooltip = gen_tinker.0218.secret_kept_tt

				ai_sociability = 0.25

			modifier = {

				has_character_flag = gen_tk_wf_patron_fooled
			}
		}
		gen_tk_next_stage_effect = yes
	}
