namespace = gen_hornborn

# Horns at the Naming: a Hornborn child is born among a people who are not horned, and a parent
# decides how the child grows up. The system around it:
#   common/on_action/00_gen_hornborn_naming_on_actions.txt       — birth and 16th-birthday hooks
#   common/scripted_effects/00_gen_hornborn_naming_effects.txt   — the three paths, the cast
#   common/scripted_triggers/00_gen_hornborn_naming_triggers.txt
#   common/secret_types/00_gen_hornborn_secrets.txt              — hiding the child
#   common/modifiers/, common/opinion_modifiers/                 — 00_gen_hornborn_naming_*
#
#   0001        hidden, on the child a week after birth: picks the parent, sends 0003 or 0002
#   0002        Horns at the Naming — Hide / Embrace / Foster
#   0003        Whose Horns? — the real father is horned and the legal father is not; leads to 0002
#   0010        An Empty Cradle — the other parent learns the child was fostered
#   0020/0021   Embrace:  hidden dispatcher / The Chaplain's Omen (2–4 years on)
#   0030/0031   Foster:   hidden dispatcher (sets the ward) / A Letter from the Highlands (age ~6)
#   0040/0041   Hide:     hidden dispatcher / Why Must I Wear the Cap? (age ~8)
#   0045        Hide:     the secret is exposed
#   0050        hidden, on the child at 16: sends one of
#   0051–0053   The Cap Comes Off / Raised Unashamed / Two Homes — on the child
#
# Path flags on the child: gen_hb_path_hidden / gen_hb_path_embraced / gen_hb_path_fostered.
# The cast lives as variables on the child (gen_hb_store_cast_effect); every follow-up reloads it
# in a hidden event on the child and re-checks who is still alive before anything is shown.
#
# Twins each get their own naming: the parent may choose differently for each.
# A birth with no player parent and no landed parent gets nothing — the horns just show.

# =====================================================================================
#  0001 — who decides (hidden, on the child)
# =====================================================================================
gen_hornborn.0001 = {
	type = character_event
	hidden = yes

	trigger = { gen_hb_naming_eligible_trigger = yes }

	immediate = {
		add_character_flag = gen_hb_named
		save_scope_as = gen_hb_child

		# A player parent first, then the landed parent of the higher tier.
		if = {
			limit = { mother ?= { is_alive = yes is_ai = no } }
			mother = { save_scope_as = gen_hb_parent }
		}
		else_if = {
			limit = { father ?= { is_alive = yes is_ai = no } }
			father = { save_scope_as = gen_hb_parent }
		}
		else_if = {
			limit = {
				mother ?= {
					is_alive = yes
					is_landed = yes
					OR = {
						NOT = { exists = root.father }
						root.father = { is_alive = no }
						root.father = { is_landed = no }
						highest_held_title_tier >= root.father.highest_held_title_tier
					}
				}
			}
			mother = { save_scope_as = gen_hb_parent }
		}
		else_if = {
			limit = { father ?= { is_alive = yes is_landed = yes } }
			father = { save_scope_as = gen_hb_parent }
		}

		if = {
			limit = { exists = scope:gen_hb_parent }

			if = {
				limit = {
					exists = mother
					scope:gen_hb_parent = { this = root.father }
				}
				mother = { save_scope_as = gen_hb_other_parent }
			}
			else_if = {
				limit = {
					exists = father
					scope:gen_hb_parent = { this = root.mother }
				}
				father = { save_scope_as = gen_hb_other_parent }
			}

			gen_hb_find_foster_effect = yes
			gen_hb_store_cast_effect = yes

			if = {
				limit = { gen_hb_horns_not_his_trigger = yes }
				scope:gen_hb_parent = { trigger_event = gen_hornborn.0003 }
			}
			else = {
				scope:gen_hb_parent = { trigger_event = gen_hornborn.0002 }
			}
		}
	}
}

# =====================================================================================
#  0003 — Whose Horns? (the legal father)
# =====================================================================================
# Both options go on to the naming. "The truth" exposes vanilla's disputed-heritage secret when the
# child has one.
gen_hornborn.0003 = {
	type = character_event
	title = gen_hornborn.0003.t
	desc = gen_hornborn.0003.desc
	theme = secret
	override_background = { reference = bedchamber }

	left_portrait = {
		character = root
		animation = disbelief
	}
	right_portrait = {
		character = scope:gen_hb_other_parent
		animation = shame
	}

	trigger = {
		scope:gen_hb_child = { is_alive = yes }
	}

	option = {
		name = gen_hornborn.0003.a
		# Vanilla gives the disputed-heritage secret to the mother, with the child as its target
		# (00_pregnancy_effects.txt). Exposing it brings vanilla's own bastardy fallout with it.
		scope:gen_hb_other_parent ?= {
			random_secret = {
				type = secret_disputed_heritage
				limit = { secret_target = scope:gen_hb_child }
				expose_secret = root
			}
		}
		scope:gen_hb_other_parent ?= {
			save_scope_as = gen_hb_suspected
		}
		if = {
			limit = { exists = scope:gen_hb_suspected }
			add_opinion = {
				target = scope:gen_hb_suspected
				modifier = gen_hb_whose_horns_opinion
			}
		}
		trigger_event = gen_hornborn.0002
		ai_chance = {
			base = 40
			modifier = { add = 30 has_trait = wrathful }
			modifier = { add = 30 has_trait = paranoid }
			modifier = { add = 20 has_trait = vengeful }
			modifier = { factor = 0.3 has_trait = forgiving }
		}
	}

	option = {
		name = gen_hornborn.0003.b
		stress_impact = {
			base = medium_stress_impact_gain
			forgiving = medium_stress_impact_loss
		}
		trigger_event = gen_hornborn.0002
		ai_chance = {
			base = 40
			modifier = { add = 30 has_trait = forgiving }
			modifier = { add = 20 has_trait = content }
		}
	}
}

# =====================================================================================
#  0030/0031 — Foster: A Letter from the Highlands (age ~6)
# =====================================================================================
gen_hornborn.0030 = {
	type = character_event
	hidden = yes

	trigger = {
		is_alive = yes
		has_character_flag = gen_hb_path_fostered
	}

	immediate = {
		gen_hb_load_cast_effect = yes
		if = {
			limit = {
				scope:gen_hb_foster ?= { is_alive = yes }
				is_courtier_of = scope:gen_hb_foster
			}
			scope:gen_hb_foster = { set_relation_ward = scope:gen_hb_child }	# vanilla's direction

			# A match in the foster lord's own house, for the betrothal option. The foster lord is
			# often the horned parent's own kin, so their children can be the child's aunt, uncle or
			# cousin.
			scope:gen_hb_foster = {
				random_child = {
					limit = {
						is_alive = yes
						NOT = { this = scope:gen_hb_child }
						NOT = { is_close_or_extended_family_of = scope:gen_hb_child }
						age <= 12
						is_married = no
						is_betrothed = no
						sex_opposite_of = scope:gen_hb_child
					}
					save_scope_as = gen_hb_match
				}
			}

			if = {
				limit = { scope:gen_hb_parent ?= { is_alive = yes } }
				scope:gen_hb_parent = { trigger_event = gen_hornborn.0031 }
			}
		}
		# The fosterage fell through: the foster lord died, or the child was moved (a guardian assigned
		# elsewhere). From here the child grows up in the open wherever they are, so 0050 sends
		# 0052, not Two Homes.
		else = {
			gen_hb_switch_to_embraced_effect = yes
		}
	}
}

	}
