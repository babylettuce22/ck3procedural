# 00_phenotype_traits.txt

# No `group` on any of these: CK3 trait groups are for LEVELLED tracks (education tiers),
# and a grouped trait without a level is an error the engine "fixes" by unsetting the group at
# load — one error.log line per trait per boot. Mutual exclusivity is carried entirely by the
# `opposites` lists, which is also how vanilla does it for flat traits.

# Human — the baseline race, made visible. On a fantasy map every character of a human culture
# carries this (stamped by Emit/HistoryWriter.cs, spread by the culture pulse), which is what lets
# humans participate in the same_opinion/opposite_opinion web instead of being the one people with
# no stance. Deliberately light on modifiers: the fantasy races are the deviation, humans are what
# the rest of the game already balances for — their identity is fecundity and fast learning.
#
# Note this trait is NOT the same thing as the `gen_phenotype_human` character FLAG. The trait is
# broad, visible racial identity; the flag is a narrow render-time marker meaning "human born of a
# mixed line, snap any inherited skin shift off" (see gen_race_skin_human in the generated
# 99_gen_race_morphs.txt). Minority-race members hold this trait but never the flag — that is what
# lets them keep their fantasy looks while counting as members of a human culture.
phenotype_human = {
    physical = yes
    inherit_chance = 0
    shown_in_ruler_designer = yes
    ruler_designer_cost = 0

    opposites = {
        phenotype_gracile
        phenotype_stocky
        phenotype_rough_hewn
        phenotype_towering
        phenotype_diminutive
        phenotype_dusk_adapted
    }
    # Placeholder art, same policy as world_weary in 01_gen_aging_traits.txt: a vanilla icon that
    # reads well enough until a bespoke human.dds exists beside the race icons.
    icon = "human_phenotype.dds"

    fertility = 0.1
    monthly_lifestyle_xp_gain_mult = 0.1
    same_opinion = 5
    opposite_opinion = -5
}

# Elf
phenotype_gracile = {
    physical = yes
    inherit_chance = 0
    shown_in_ruler_designer = yes
    ruler_designer_cost = 20

    opposites = {
        phenotype_human
        phenotype_stocky
        phenotype_rough_hewn
        phenotype_towering
        phenotype_diminutive
        phenotype_dusk_adapted
    }
    icon = "gracile.dds"

    culture_modifier = {
        parameter = forest_trait_bonuses
        forest_attrition_mult = -0.1
        taiga_attrition_mult = -0.1
    }

    diplomacy = 1
    prowess = 1
    movement_speed = 0.05
    stress_loss_mult = 0.05
    same_opinion = 10
    life_expectancy = 50
    health = 1.25
    fertility = -0.5

	opposite_opinion = -2

    # Elves do not decay. `is_immortal` is a SCRIPT flag, not engine immortality (that is the
    # separate `immortal = yes` property, which stays off): it is what every vanilla elderly-health
    # gate checks — the five ailment events health.7000–7500 and
    # age_ranked_health_vulnerability_threshold_trigger, which gates the incapable-with-age chain
    # (harm.0001–0003). Natural death from life_expectancy still applies; the long decline is
    # carried by world_weary instead — see 01_gen_aging_traits.txt.
    flag = is_immortal
    no_prowess_loss_from_age = yes
}
