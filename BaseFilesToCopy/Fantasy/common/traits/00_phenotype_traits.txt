# 00_phenotype_traits.txt

# No `group` on any of these: CK3 trait groups are for LEVELLED tracks (education tiers),
# and a grouped trait without a level is an error the engine "fixes" by unsetting the group at
# load — one error.log line per trait per boot. Mutual exclusivity is carried entirely by the
# `opposites` lists, which is also how vanilla does it for flat traits.

# Human — the baseline race, made visible. On a fantasy map every character of a human culture
# carries this (stamped by Emit/HistoryWriter.cs, spread by the culture pulse), which is what lets
# humans participate in the same_opinion/opposite_opinion web instead of being the one people with
# no stance. Deliberately light on modifiers: the fantasy races are the deviation, humans are what
# the rest of the game already balances for — their identity is fecundity and fast learning.
#
# Note this trait is NOT the same thing as the `gen_phenotype_human` character FLAG. The trait is
# broad, visible racial identity; the flag is a narrow render-time marker meaning "human born of a
# mixed line, snap any inherited skin shift off" (see gen_race_skin_human in the generated
# 99_gen_race_morphs.txt). Minority-race members hold this trait but never the flag — that is what
# lets them keep their fantasy looks while counting as members of a human culture.
phenotype_human = {
    physical = yes
    inherit_chance = 0
    shown_in_ruler_designer = yes
    ruler_designer_cost = 0

    opposites = {
        phenotype_gracile
        phenotype_stocky
        phenotype_rough_hewn
        phenotype_towering
        phenotype_diminutive
        phenotype_dusk_adapted
        phenotype_sylvan
    }
    # Placeholder art, same policy as world_weary in 01_gen_aging_traits.txt: a vanilla icon that
    # reads well enough until a bespoke human.dds exists beside the race icons.
    icon = "human_phenotype.dds"

    fertility = 0.1
    monthly_lifestyle_xp_gain_mult = 0.1
    same_opinion = 5
    opposite_opinion = -5
}

# High elf (the wood elf is phenotype_sylvan below)
phenotype_gracile = {
    physical = yes
    inherit_chance = 0
    shown_in_ruler_designer = yes
    ruler_designer_cost = 20

    opposites = {
        phenotype_human
        phenotype_stocky
        phenotype_rough_hewn
        phenotype_towering
        phenotype_diminutive
        phenotype_dusk_adapted
        phenotype_sylvan
    }
    icon = "gracile.dds"

    # No forest block: that is the wood elf's (phenotype_sylvan), and it only fires for a culture
    # holding Forest Fighters, which a plains-dwelling high elf almost never does. A flat +1 learning
    # instead, matching the scholarly traditions Cultures.RaceTraditions gives them.
    diplomacy = 1
    learning = 1
    prowess = 1
    movement_speed = 0.05
    stress_loss_mult = 0.05
    same_opinion = 10
    life_expectancy = 50
    health = 1.25
    # Slow-breeding, not sterile. At -0.5 this matched vanilla's Infertile trait: base fertility is
    # 0.5, so a pure-elf couple sat on the engine's minimum conception chance, and the extra
    # life_expectancy bought nothing because female fertility is keyed to age, not lifespan.
    fertility = -0.35

	opposite_opinion = -2

    # Elves do not decay. `is_immortal` is a SCRIPT flag, not engine immortality (that is the
    # separate `immortal = yes` property, which stays off): it is what every vanilla elderly-health
    # gate checks — the five ailment events health.7000–7500 and
    # age_ranked_health_vulnerability_threshold_trigger, which gates the incapable-with-age chain
    # (harm.0001–0003). Natural death from life_expectancy still applies; the long decline is
    # carried by world_weary instead — see 01_gen_aging_traits.txt.
    flag = is_immortal
    no_prowess_loss_from_age = yes
}

# Wood elf — its own trait, split from phenotype_gracile (which is now the high elf alone).
#
# The two elves used to share one trait, and that cost both of them their looks at render time:
# the portrait modifier for a shared trait can only force the genes the two agree on, so it forced
# an AVERAGED height onto every elf (high elves lost their height) and could tell the skins apart
# only by culture key, which turned every wood elf outside a generation-time wood-elf culture —
# minorities, diverged and hybrid cultures, converts — high-elf skin. One trait per race lets
# 99_gen_race_morphs.txt force each elf's own table and skin with no culture test at all.
#
# Stats are gracile's as they were before the split, forest bonus included; the high elf has since
# traded that bonus for +1 learning.
phenotype_sylvan = {
    physical = yes
    inherit_chance = 0
    shown_in_ruler_designer = yes
    ruler_designer_cost = 20

    opposites = {
        phenotype_human
        phenotype_gracile
        phenotype_stocky
        phenotype_rough_hewn
        phenotype_towering
        phenotype_diminutive
        phenotype_dusk_adapted
    }
    # Placeholder art: gracile's icon until a bespoke sylvan.dds exists beside it.
    icon = "gracile.dds"

    culture_modifier = {
        parameter = forest_trait_bonuses
        forest_attrition_mult = -0.1
        taiga_attrition_mult = -0.1
    }

    diplomacy = 1
    prowess = 1
    movement_speed = 0.05
    stress_loss_mult = 0.05
    same_opinion = 10
    life_expectancy = 50
    health = 1.25
    fertility = -0.35 # see phenotype_gracile

	opposite_opinion = -2

    # Long-lived exactly as phenotype_gracile is — see the is_immortal note there.
    flag = is_immortal
    no_prowess_loss_from_age = yes
}

# Dwarf
phenotype_stocky = {
    physical = yes
    inherit_chance = 0
    shown_in_ruler_designer = yes
    ruler_designer_cost = 20

    opposites = {
        phenotype_human
        phenotype_gracile
        phenotype_rough_hewn
        phenotype_towering
        phenotype_diminutive
        phenotype_dusk_adapted
        phenotype_sylvan
    }
    icon = "stocky.dds"

    stewardship = 1
    prowess = 2
    health = 0.5
    defender_advantage = 3
    same_opinion = 15
	life_expectancy = 20

    # Dwarves age like humans, twenty years late. Vanilla hard-codes its ailment onset ages (45 for
    # infirm, 50 for the rest) into the five health.7x00 events and gates them on this script flag
    # only, so the flag switches vanilla onset OFF and gen_dwarf_ailment_pulse_effect
    # (00_gen_aging_effects.txt) switches it back on at 65, adding the same five vanilla traits.
    # Vanilla's own yearly progression then carries them exactly as it would a human's — the
    # extra twenty years are prime, not a longer decrepitude. Unlike the elves there is no
    # no_prowess_loss_from_age: a dwarf's arm still weakens with the years.
    flag = is_immortal

    culture_modifier = {
        parameter = mountain_trait_bonuses
        mountains_attrition_mult = -0.15
        desert_mountains_attrition_mult = -0.15
    }
	
	opposite_opinion = -10
}
