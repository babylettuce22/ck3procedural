# Hooks that decide a character's race. See common/scripted_effects/ for what the effects do; this
# file is only about WHEN to ask.
#
# **Every entry appends, none overrides.** Each vanilla on_action below is re-declared with nothing
# but an `on_actions` list. That is the sanctioned pattern, spelled out in
# common/on_action/_on_actions.info: data for an on-action may be declared in multiple files, and
# only a second `trigger` or `effect` block would conflict. Vanilla relies on it itself —
# `three_year_playable_pulse` is declared twice in the stock files. Writing a `trigger` or `effect`
# into one of these blocks instead would replace vanilla's own and silently delete whatever it did —
# the single commonest compatibility bug in CK3 modding.

# BIRTH — the authoritative moment.
#
# `root` is the newborn; vanilla documents the scopes on its own declaration in
# child_birth_on_actions.txt, and its birth.8001 confirms them with `is_grandparent_of = root`.
#
# This is where a child of two races is resolved to one of them, and it MUST run, because the traits
# are `inherit_chance = 0` and pass nothing genetically. (An earlier version of this file argued the
# exact opposite — that birth needed no hook because the traits inherited at 100% — and that block
# survived, contradicting the hook below, long after the traits were changed. The pulse effects are
# designed around births being handled here; without this hook they assign by CULTURE, which hands a
# human child born into a dwarven culture the local race.)
on_birth_child = {
	on_actions = { gen_phenotype_birth_on_action }
}

gen_phenotype_birth_on_action = {
	effect = {
		gen_resolve_child_phenotype_effect = yes
	}
}

# GAME START — one sweep over everyone alive on day one.
#
# The generator stamps every HISTORY character's trait at generation time, but the engine invents
# more people during setup — pool wanderers, filler courtiers, lowborn spouses — and before this
# hook they went a year or more looking human in a dwarven court until the yearly pulse reached
# them. One pass at the start closes that window. It also runs the sanitizer, so a save edited by
# console or ruler designer is repaired the moment it loads rather than at its first birthday.
#
# `on_game_start_after_lobby` rather than `on_game_start`, because vanilla documents it as the one
# that knows the game rules and the players — and it fires once, after the lobby, which is exactly
# the "before the player looks at anything" moment this wants. Vanilla declares its own `effect`
# block on it, which is precisely why this file must append via the on_actions list and never write
# an effect block of its own — see the header.
on_game_start_after_lobby = {
	on_actions = { gen_phenotype_game_start_on_action }
}

gen_phenotype_game_start_on_action = {
	effect = {
		every_living_character = {
			gen_sanitize_phenotype_effect = yes
			gen_reconcile_phenotype_with_genes_effect = yes
			gen_assign_phenotype_effect = yes
		}
	}
}

# THE NET — for characters the engine invents rather than births.
#
# Courtiers generated to fill a court, landless adventurers and their companions, pool guests, low
# nobles created for a barony, mercenary captains. Without this a dwarven kingdom slowly fills with
# human courtiers and the phenotype reads as a quirk of the starting cast rather than as what the
# culture is.
#
# Fires for every character once a year at a random point in their own year, so the cost spreads
# across the calendar rather than landing on January 1st. The obvious hook does not exist: there is
# no on_action for "a character was generated", and `on_join_court` is documented by vanilla as
# "Not fired for just generated characters" — precisely this population.
random_yearly_everyone_pulse = {
	on_actions = { gen_phenotype_assign_on_action }
}

# Pool characters specifically — not in anybody's court yet, so the yearly pulse may reach them
# before they have a liege or host to read a race from.
three_year_pool_pulse = {
	on_actions = { gen_phenotype_assign_on_action }
}

# Promptness, for the two cases where a newly generated character is immediately visible to the
# player. Both would be picked up by the yearly pulse anyway — but a guest arriving in court, or a
# low noble just handed a barony, is looked at straight away, and a dwarf realm's new baron being
# briefly human is exactly the seam this system exists to hide.
on_guest_arrived_from_pool = {
	on_actions = { gen_phenotype_assign_on_action }
}

on_baron_found_or_created_for_title = {
	on_actions = { gen_phenotype_assign_on_action }
}

gen_phenotype_assign_on_action = {
	effect = {
		# Repair first, assign second. A character wearing two races has two sets of forced genes
		# fighting over one portrait, and the repair is what makes the assignment meaningful —
		# gen_has_phenotype_trigger would otherwise read the broken character as already decided.
		gen_sanitize_phenotype_effect = yes

		# Between the two: a character the culture already called human, whose genome names a
		# fantasy race, is corrected before the assignment step looks at them. It has to run here
		# and not only at game start, because the pulse itself can produce the same disagreement —
		# an engine-generated courtier in a minority's host culture takes phenotype_human from the
		# culture head on some earlier pass, and the gene branch in gen_assign_phenotype_effect
		# never gets to see them again.
		gen_reconcile_phenotype_with_genes_effect = yes

		gen_assign_phenotype_effect = yes
	}
}
