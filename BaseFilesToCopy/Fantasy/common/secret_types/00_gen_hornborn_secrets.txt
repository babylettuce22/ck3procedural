# Hiding a Hornborn child (gen_hb_hide_child_effect). Owner: the parent; target: the child.
#
# Never a crime. A disgrace only for a hider who would feel it as one (gen_hb_unashamed_hider_trigger):
# shunned is what makes a secret blackmailable in vanilla (is_blackmailable_secret_trigger), so a
# paranoid, deceitful or callous hider cannot be squeezed over it, and anyone else can. Either way the
# lie costs prestige and vassal opinion when it comes out (on_expose), less for the shameless.
# Lapses by itself once the child comes of age or stops hiding (is_valid), so nothing has to remove it.

secret_gen_hidden_horns = {
	category = civil

	is_valid = {
		scope:secret_target ?= {
			is_alive = yes
			has_character_flag = gen_hb_path_hidden
			has_character_flag = gen_horns_filed
			NOT = { has_character_flag = gen_hb_came_of_age }
		}
	}

	is_criminal = {
		always = no
	}

	is_shunned = {
		scope:secret_owner = {
			NOT = { gen_hb_unashamed_hider_trigger = yes }
		}
	}

	on_expose = {
		scope:secret_target = {
			save_scope_as = gen_hb_child
			gen_hb_switch_to_embraced_effect = yes
			add_character_flag = gen_hb_exposed
		}
		scope:secret_owner = {
			save_scope_as = gen_hb_parent
			if = {
				limit = { gen_hb_unashamed_hider_trigger = yes }
				add_prestige = minor_prestige_loss
			}
			else = {
				add_prestige = medium_prestige_loss
			}
			every_vassal = {
				add_opinion = {
					target = scope:gen_hb_parent
					modifier = gen_hb_lied_about_child_opinion
				}
			}
			trigger_event = { id = gen_hornborn.0045 days = 1 }
		}
	}
}
