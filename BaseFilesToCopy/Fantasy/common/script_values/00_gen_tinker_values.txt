# Gnome Tinkering's numbers. See 00_gen_tinker_triggers.txt for the map of the system.
# All are read with the tinkerer as root/this.

# Where a project's quality starts: the maker's learning, their record at the bench, and their people's craft.
gen_tk_base_quality_value = {
	value = learning
	multiply = 3
	if = {
		limit = { has_character_modifier = gen_tk_master_artificer_modifier }
		add = 20
	}
	else_if = {
		limit = { has_character_modifier = gen_tk_tinkerer_modifier }
		add = 10
	}
	if = {
		limit = {
			culture ?= {
				OR = {
					has_cultural_tradition = tradition_artisans
					has_cultural_tradition = tradition_metal_craftsmanship
				}
			}
		}
		add = 5
	}
}

# Where a project's risk starts: every contraption can go wrong; weapons and grand ideas more so.
gen_tk_base_risk_value = {
	value = 10
	if = {
		limit = { var:gen_tk_project ?= flag:engine }
		add = 10
	}
	else_if = {
		limit = { var:gen_tk_project ?= flag:invention }
		add = 15
	}
}

# The four weights of the test (gen_tk_final_roll_effect). Quality ~0–100, risk ~0–60.
#
# A point of quality moves 1.7 weight (masterwork +0.8, works +0.4, dud -0.5); a point of risk moves 1
# (explodes). Tuned so +10 quality and -10 risk are worth about the same, and a "+10 quality, +10 risk"
# gamble keeps the success odds and trades duds for masterworks and explosions. At quality 60, risk 15:
# masterwork 23%, works 45%, dud 21%, explodes 11%.
gen_tk_masterwork_weight_value = {
	value = var:gen_tk_quality
	multiply = 0.8
	subtract = 15
	min = 0
}

gen_tk_works_weight_value = {
	value = var:gen_tk_quality
	multiply = 0.4
	add = 40
}

gen_tk_dud_weight_value = {
	value = var:gen_tk_quality
	multiply = -0.5
	add = 60
	min = 10
}

gen_tk_explodes_weight_value = {
	value = var:gen_tk_risk
	min = 5
}

gen_tk_total_weight_value = {
	value = gen_tk_masterwork_weight_value
	add = gen_tk_works_weight_value
	add = gen_tk_dud_weight_value
	add = gen_tk_explodes_weight_value
}

# The chance of an explosion as a whole percentage, for the AI's reading of the test's odds (the
# player sees all four chances on the roll itself, gen_tk_final_roll_effect).
gen_tk_explodes_pct_value = {
	value = gen_tk_explodes_weight_value
	multiply = 100
	divide = gen_tk_total_weight_value
	round = yes
}
