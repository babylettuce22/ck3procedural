# Gives a character the racial phenotype of the culture they belong to.
#
# Emit/HistoryWriter.cs stamps a phenotype trait onto every character it writes at generation time,
# and 00_phenotype_birth_effects.txt resolves every child born in-game to one parent's race. The
# traits are `inherit_chance = 0` and pass nothing genetically, so those two are the whole of it —
# this file exists only for the characters the ENGINE invents mid-game, who come from neither of
# those places: courtiers generated to fill a court, landless adventurers and
# the companions they recruit, pool guests, low nobles created for a barony, mercenary captains.
# Without this they are all humans, so a dwarven kingdom slowly fills up with human courtiers and
# the phenotype reads as a quirk of the 867 bookmark rather than as what the culture is.
#
# **Keyed on culture, not on who is standing nearby.** Phenotype is a property of a culture in this
# generator — Emit/HistoryWriter.cs derives the trait from the culture's ethnicity archetype — so the
# culture is the correct axis, and it is also the self-correcting one: culture passes from parent to
# child, so a human family living in a dwarven realm keeps its human culture and is never converted
# by anything here. Copying from the liege instead would turn every human servant in a dwarf's
# household into a dwarf.
#
# Culture is right for this population and wrong for a newborn, which is why births are handled
# somewhere else entirely: a child has parents to read, a courtier invented out of nothing does not,
# and its culture is the only honest signal available. gen_has_phenotype_trigger includes
# phenotype_human, so a child who came out human is already decided and this effect will not reach
# past its parents to convert it.
#
# **Why the culture route has to be indirect.** This file set ships identically for every seed and
# so may not name a generated key (see BaseFilesToCopy/README.txt), which rules out asking whether
# the character's culture is `culture:gen_culture_4` or their ethnicity `gen_ethnicity_7`. So the
# culture is identified by example instead — find a character the culture already agrees is typical,
# and copy their trait.
#
# `culture_head` is that character. It is a real per-culture pointer needing no generated key, and
# the head at game start is always a ruler out of history, which is to say one of the characters
# HistoryWriter already stamped. `liege` and `host` are fallbacks for a culture whose head has died
# or was never established, and both are gated on same-culture for the reason above.
#
# **The character's own ethnicity is asked FIRST.** That is what fixes the minority-race member, who
# is the one population where culture and body genuinely disagree: they roll their race's ethnicity
# out of a HUMAN culture's weighted ethnicity list (MapGen/Ethnicities.cs seats them at weight 15),
# so the culture head says human and their own body says gnome. Culture remains the fallback, and
# remains the right answer for the population this file was written for — a courtier the engine
# invented for a dwarven court rolled a dwarven ethnicity anyway. See
# gen_has_fantasy_race_ethnicity_trigger, and there for why this is not a `has_gene` read any more.

gen_assign_phenotype_effect = {
	if = {
		limit = {
			NOT = { gen_has_phenotype_trigger = yes }
			# Not the wilderness or ruins dummy. It is a placeholder, not a person, but it has a
			# culture (Unsettled) and so an ethnicity, and once the ethnicity probe started working
			# a map whose Unsettled culture drew a wood-elf ethnicity made both dummies sylvan — and
			# so mortal to the world_weary clock. Tested by government FLAG, not the `wilderness`
			# trait, because this set also ships on worlds without the Wilderness set (an unknown
			# flag is simply never held).
			NOT = { government_has_flag = government_is_wilderness }
		}

		# Saved rather than leaning on `root`, because this is called from several on_actions and
		# root is not the same character in all of them — in on_baron_found_or_created_for_title it
		# is the new baron, in the pulses it is whoever the pulse fired for.
		save_scope_as = gen_phenotype_target

		if = {
			limit = { gen_has_fantasy_race_ethnicity_trigger = yes }
			gen_assign_phenotype_from_genes_effect = yes
		}
		else_if = {
			limit = {
				culture.culture_head ?= {
					gen_has_phenotype_trigger = yes

					# A minority-race head is not a template for their own culture. The Rohand are
					# human with a gnome minority seated among them; if the head happens to be one
					# of the gnomes, copying from them would hand phenotype_diminutive to every
					# human Rohand courtier the engine invents — a 13% minority quietly taking over
					# the culture, and worse, the trait would then FORCE gnome genes onto human
					# bodies at render (see 99_gen_race_morphs.txt). The flag marks exactly the
					# characters whose race came from their own genome rather than their culture.
					NOT = { has_character_flag = gen_phenotype_from_gene }
				}
			}
			culture.culture_head = { save_scope_as = gen_phenotype_source }
		}
		else_if = {
			limit = {
				liege ?= {
					gen_has_phenotype_trigger = yes
					has_same_culture_as = scope:gen_phenotype_target
				}
			}
			liege = { save_scope_as = gen_phenotype_source }
		}
		else_if = {
			limit = {
				host ?= {
					gen_has_phenotype_trigger = yes
					has_same_culture_as = scope:gen_phenotype_target
				}
			}
			host = { save_scope_as = gen_phenotype_source }
		}
		# Any other member of the culture whose race came from the culture. Needed because the head
		# can be the one without a race: a peasant-revolt leader (vanilla create_character, back-dated
		# birth) who wins becomes the culture's most powerful member before their first pulse, then
		# has no liege or host, and every courtier invented for their court copies from them. A
		# headless 100-year run (ck3run soak, 2026-09-30) left three cultures locked like that for
		# 30+ years. This is a full scan, but it only runs for a character nothing above could
		# place, which is a handful per year.
		else_if = {
			limit = {
				any_living_character = {
					has_same_culture_as = scope:gen_phenotype_target
					gen_has_phenotype_trigger = yes
					NOT = { has_character_flag = gen_phenotype_from_gene }
				}
			}
			random_living_character = {
				limit = {
					has_same_culture_as = scope:gen_phenotype_target
					gen_has_phenotype_trigger = yes
					NOT = { has_character_flag = gen_phenotype_from_gene }
				}
				save_scope_as = gen_phenotype_source
			}
		}

		# No source is still possible: a culture (typically a new hybrid) with no raced member at all.
		# A character there without a fantasy ethnicity has a human body, since the gene branch above
		# already took every fantasy one, so it is human; see the last block. (This file set only
		# ships on fantasy maps — see Emit/StaticFileWriter.cs — and on a fantasy map nearly everyone
		# finds a source, humans included, which is what stamps phenotype_human onto engine-generated
		# human courtiers.)
		#
		# Re-asks gen_has_phenotype_trigger rather than trusting the branch structure above. The gene
		# branch assigns directly and saves no source, so it should fall through here harmlessly —
		# but `save_scope_as` inside an `every_living_character` loop is exactly the place not to bet
		# on a stale scope from the previous iteration being invisible. If the genome already
		# answered, nothing here may run.
		if = {
			limit = {
				NOT = { gen_has_phenotype_trigger = yes }
				exists = scope:gen_phenotype_source
			}
			gen_copy_phenotype_effect = yes
		}
		# Last resort, human by body. Without it such a character, and every courtier who later copies
		# from them, stays raceless for good.
		if = {
			limit = {
				NOT = { gen_has_phenotype_trigger = yes }
				NOT = { exists = scope:gen_phenotype_source }
				NOT = { gen_has_fantasy_race_ethnicity_trigger = yes }
			}
			add_trait = phenotype_human
		}
	}
}

# Gives a character the race their own ethnicity names.
#
# The ethnicity is the roll the character's body came from — the engine drew it out of their
# culture's weighted list and painted their genome with it. This is the only place in the system
# where a race is decided from the character rather than from the people around them, which is what
# makes it the right answer for a minority: a gnome seated among the Rohand rolls the gnome
# ethnicity no matter what the culture says. (The name still says "genes": it read the gen_race_skin
# gene with has_gene until that turned out to be interface-only and dead in script.)
#
# The two elves are two traits — phenotype_gracile for the high elf, phenotype_sylvan for the wood
# elf — so the ethnicity answers which elf this is directly, and Emit/RaceMorphWriter.cs forces each
# one's own shape and skin with no culture test.
#
# There is deliberately NO human branch: absent a fantasy ethnicity the character has said nothing,
# and the culture route in gen_assign_phenotype_effect decides.
#
# gen_phenotype_from_gene records that this character's race came from their body rather than their
# culture. Read by gen_assign_phenotype_effect to refuse a minority-race culture head as a source —
# see the comment there for why that matters.
gen_assign_phenotype_from_genes_effect = {
	if = {
		limit = { gen_is_high_elf_ethnicity_trigger = yes }
		add_trait = phenotype_gracile
		add_character_flag = gen_phenotype_from_gene
	}
	else_if = {
		limit = { gen_is_wood_elf_ethnicity_trigger = yes }
		add_trait = phenotype_sylvan
		add_character_flag = gen_phenotype_from_gene
	}
	else_if = {
		limit = { gen_is_dwarf_ethnicity_trigger = yes }
		add_trait = phenotype_stocky
		add_character_flag = gen_phenotype_from_gene
	}
	else_if = {
		limit = { gen_is_orc_ethnicity_trigger = yes }
		add_trait = phenotype_rough_hewn
		add_character_flag = gen_phenotype_from_gene
	}
	else_if = {
		limit = { gen_is_giantkin_ethnicity_trigger = yes }
		add_trait = phenotype_towering
		add_character_flag = gen_phenotype_from_gene
	}
	else_if = {
		limit = { gen_is_gnome_ethnicity_trigger = yes }
		add_trait = phenotype_diminutive
		add_character_flag = gen_phenotype_from_gene
	}
	else_if = {
		limit = { gen_is_deepkin_ethnicity_trigger = yes }
		add_trait = phenotype_dusk_adapted
		add_character_flag = gen_phenotype_from_gene
	}
	else_if = {
		limit = { gen_is_hornkin_ethnicity_trigger = yes }
		add_trait = phenotype_horned
		add_character_flag = gen_phenotype_from_gene
	}
}

# Repairs a character the CULTURE called human whose own genome says otherwise.
#
# gen_assign_phenotype_effect above only ever looks at characters who hold no phenotype yet, so it
# cannot reach this population: Emit/HistoryWriter.cs derives the trait from the culture's ethnicity
# archetype and stamps it on every history character at generation time, and a minority's host
# culture is human by definition. The minority members among them are therefore born already
# labelled phenotype_human and permanently invisible to the pulse — which is why a map could ship
# with gnome-bodied characters, no phenotype_diminutive anywhere in history, and no short people in
# the game at all.
#
# The generator cannot fix this at emit time. It does not know who the gnomes are: history
# characters carry no `dna` (only the bookmark cast does), so the ENGINE rolls each character's
# ethnicity out of the culture's weighted list when the save is created, long after the files are
# written. The genome only exists at runtime, so the correction has to live here.
#
# Narrow on purpose. It fires only on phenotype_human — never on a character already holding a
# fantasy race — and never on the mixed-line flag, so an inherited skin shift several generations
# down a half-elven line stays what the birth roll made it. Everywhere else trait and genome agree
# by construction and this is one trigger that answers no.
gen_reconcile_phenotype_with_genes_effect = {
	if = {
		limit = { gen_phenotype_contradicts_genes_trigger = yes }

		# Removed first: the race traits list each other as `opposites`, so add_trait would be
		# refused while phenotype_human is still on.
		remove_trait = phenotype_human
		gen_assign_phenotype_from_genes_effect = yes
	}
}

# Copies scope:gen_phenotype_source's phenotype onto the current character.
#
# Split out so the source-picking above stays readable, and written as one else_if chain rather than
# seven independent ifs so a source carrying two phenotypes — which the `opposites` lists are
# supposed to prevent, but which a ruler designer character or a console command can still produce —
# hands over exactly one instead of stacking them.
gen_copy_phenotype_effect = {
	if = {
		limit = { scope:gen_phenotype_source = { has_trait = phenotype_gracile } }
		add_trait = phenotype_gracile
	}
	else_if = {
		limit = { scope:gen_phenotype_source = { has_trait = phenotype_sylvan } }
		add_trait = phenotype_sylvan
	}
	else_if = {
		limit = { scope:gen_phenotype_source = { has_trait = phenotype_stocky } }
		add_trait = phenotype_stocky
	}
	else_if = {
		limit = { scope:gen_phenotype_source = { has_trait = phenotype_rough_hewn } }
		add_trait = phenotype_rough_hewn
	}
	else_if = {
		limit = { scope:gen_phenotype_source = { has_trait = phenotype_towering } }
		add_trait = phenotype_towering
	}
	else_if = {
		limit = { scope:gen_phenotype_source = { has_trait = phenotype_diminutive } }
		add_trait = phenotype_diminutive
	}
	else_if = {
		limit = { scope:gen_phenotype_source = { has_trait = phenotype_dusk_adapted } }
		add_trait = phenotype_dusk_adapted
	}
	else_if = {
		limit = { scope:gen_phenotype_source = { has_trait = phenotype_horned } }
		add_trait = phenotype_horned
	}
	else_if = {
		# The human race spreads through the pulse like any other: a courtier invented for a human
		# court takes phenotype_human from the culture head and is settled permanently instead of
		# being re-examined every year. Deliberately NOT copied: the gen_phenotype_human FLAG. The
		# flag means "mixed-line human, reset inherited skin at render" and is a fact about a
		# bloodline, not a culture — copying it from a mixed-line culture head would snap the skin
		# of every minority-race member of that culture back to human.
		limit = { scope:gen_phenotype_source = { has_trait = phenotype_human } }
		add_trait = phenotype_human
	}
}

# Strips a character back to one race.
#
# Should never fire: all nine traits list each other as `opposites`, and the mixed-line flag is
# only set by script. The ruler designer and the console can still produce a doubled character, and
# two races means two sets of forced genes fighting over one portrait, so the yearly pulse checks.
#
# Priority order is fixed rather than random so the same character repairs to the same race every
# time it is asked, instead of flickering between races year to year — fantasy traits outrank
# phenotype_human, so a designer character who stacked "human" on top of a race keeps the race.
# Every fantasy branch also clears the mixed-line flag: the flag means "human whose inherited skin
# must reset", which is nonsense beside a fantasy trait. Beside phenotype_human it is consistent
# and kept. remove_trait on a trait the character does not hold is a safe no-op, and so is removing
# an absent flag — what keeps this short.
gen_sanitize_phenotype_effect = {
	if = {
		limit = { gen_has_conflicting_phenotype_trigger = yes }

		if = {
			limit = { has_trait = phenotype_gracile }
			remove_trait = phenotype_sylvan
			remove_trait = phenotype_stocky
			remove_trait = phenotype_rough_hewn
			remove_trait = phenotype_towering
			remove_trait = phenotype_diminutive
			remove_trait = phenotype_dusk_adapted
			remove_trait = phenotype_horned
			remove_trait = phenotype_human
			remove_character_flag = gen_phenotype_human
		}
		else_if = {
			limit = { has_trait = phenotype_sylvan }
			remove_trait = phenotype_stocky
			remove_trait = phenotype_rough_hewn
			remove_trait = phenotype_towering
			remove_trait = phenotype_diminutive
			remove_trait = phenotype_dusk_adapted
			remove_trait = phenotype_horned
			remove_trait = phenotype_human
			remove_character_flag = gen_phenotype_human
		}
		else_if = {
			limit = { has_trait = phenotype_stocky }
			remove_trait = phenotype_rough_hewn
			remove_trait = phenotype_towering
			remove_trait = phenotype_diminutive
			remove_trait = phenotype_dusk_adapted
			remove_trait = phenotype_horned
			remove_trait = phenotype_human
			remove_character_flag = gen_phenotype_human
		}
		else_if = {
			limit = { has_trait = phenotype_rough_hewn }
			remove_trait = phenotype_towering
			remove_trait = phenotype_diminutive
			remove_trait = phenotype_dusk_adapted
			remove_trait = phenotype_horned
			remove_trait = phenotype_human
			remove_character_flag = gen_phenotype_human
		}
		else_if = {
			limit = { has_trait = phenotype_towering }
			remove_trait = phenotype_diminutive
			remove_trait = phenotype_dusk_adapted
			remove_trait = phenotype_horned
			remove_trait = phenotype_human
			remove_character_flag = gen_phenotype_human
		}
		else_if = {
			limit = { has_trait = phenotype_diminutive }
			remove_trait = phenotype_dusk_adapted
			remove_trait = phenotype_horned
			remove_trait = phenotype_human
			remove_character_flag = gen_phenotype_human
		}
		else_if = {
			limit = { has_trait = phenotype_dusk_adapted }
			remove_trait = phenotype_horned
			remove_trait = phenotype_human
			remove_character_flag = gen_phenotype_human
		}
		else_if = {
			limit = { has_trait = phenotype_horned }
			remove_trait = phenotype_human
			remove_character_flag = gen_phenotype_human
		}
	}
}
