# Gnome Tinkering: the project loop. See 00_gen_tinker_triggers.txt for the map of the system.
#
# A project spans months, so its state lives in variables on the TINKERER, not in saved scopes:
#   flag gen_tk_working        a project is on (one at a time)
#   var:gen_tk_project         flag:curio / mill / engine / limb / invention
#   var:gen_tk_quality         ~0–100; starts at gen_tk_base_quality_value
#   var:gen_tk_risk            ~0–60;  starts at gen_tk_base_risk_value
#   var:gen_tk_stage           1 design, 2 setback, 3 test
#   var:gen_tk_patron          who commissioned it, if anyone (they receive the result)
#   var:gen_tk_apprentice      who is at the bench with the maker, if anyone
#   var:gen_tk_successes       lifetime count of working results
#   var:gen_tk_serial          lifetime count of projects begun; never cleared (see below)
# plus a per-project character modifier, gen_tk_at_bench_<project>_modifier, so the sheet shows it.
#
# The clock, by project (gen_tk_schedule_stage_effect) — a quarter, a half, a quarter of the whole:
#   curio, limb      8–12 months:  60–90 d → design → 120–180 d → setback →  60–90 d → test
#   mill, engine    12–18 months:  90–135 d → design → 180–270 d → setback →  90–135 d → test
#   invention       18–24 months: 135–180 d → design → 270–360 d → setback → 135–180 d → test
# then final roll → outcome (0350) → finish. Every delayed event lands on the tinkerer; if they die
# first it never fires and nothing is left to clean up (the variables die with them).
#
# Stale events: a stage event scheduled by a project that was then abandoned can still arrive after a
# new project has begun. begin saves scope:gen_tk_serial_token = the new var:gen_tk_serial, and saved
# scopes ride along every delayed trigger_event of the chain, so each dispatcher first asks
# gen_tk_event_is_current_trigger and ignores anything from an older project.

# Start a project. On the tinkerer; PROJECT is one of the flags above. Costs are paid by the caller.
gen_tk_begin_project_effect = {
	add_character_flag = gen_tk_working
	if = {
		limit = { NOT = { has_variable = gen_tk_serial } }
		set_variable = {
			name = gen_tk_serial
			value = 0
		}
	}
	change_variable = {
		name = gen_tk_serial
		add = 1
	}
	save_scope_value_as = {
		name = gen_tk_serial_token
		value = var:gen_tk_serial
	}
	set_variable = {
		name = gen_tk_project
		value = $PROJECT$
	}
	gen_tk_add_bench_modifier_effect = yes
	set_variable = {
		name = gen_tk_quality
		value = gen_tk_base_quality_value
	}
	set_variable = {
		name = gen_tk_risk
		value = gen_tk_base_risk_value
	}
	set_variable = {
		name = gen_tk_stage
		value = 1
	}
	custom_tooltip = gen_tk_begin_project.tt
	gen_tk_schedule_edge_effect = { EVENT = gen_tinker.0100 }
}

# The next stage's event, after a delay set by the project's size (the table above). "Edge" is the
# wait before design and before the test (a quarter of the project each); "middle" the wait before
# the setback (half). On the tinkerer. A hurried project (gen_tk_hurry_project_effect) waits a quarter
# less at every stage scheduled after the hurry.
gen_tk_schedule_edge_effect = {
	if = {
		limit = {
			OR = {
				var:gen_tk_project ?= flag:mill
				var:gen_tk_project ?= flag:engine
			}
		}
		if = {
			limit = { has_character_flag = gen_tk_hurried }
			trigger_event = {
				id = $EVENT$
				days = { 68 101 }
			}
		}
		else = {
			trigger_event = {
				id = $EVENT$
				days = { 90 135 }
			}
		}
	}
	else_if = {
		limit = { var:gen_tk_project ?= flag:invention }
		if = {
			limit = { has_character_flag = gen_tk_hurried }
			trigger_event = {
				id = $EVENT$
				days = { 101 135 }
			}
		}
		else = {
			trigger_event = {
				id = $EVENT$
				days = { 135 180 }
			}
		}
	}
	else = {
		if = {
			limit = { has_character_flag = gen_tk_hurried }
			trigger_event = {
				id = $EVENT$
				days = { 45 68 }
			}
		}
		else = {
			trigger_event = {
				id = $EVENT$
				days = { 60 90 }
			}
		}
	}
}

gen_tk_schedule_middle_effect = {
	if = {
		limit = {
			OR = {
				var:gen_tk_project ?= flag:mill
				var:gen_tk_project ?= flag:engine
			}
		}
		if = {
			limit = { has_character_flag = gen_tk_hurried }
			trigger_event = {
				id = $EVENT$
				days = { 135 203 }
			}
		}
		else = {
			trigger_event = {
				id = $EVENT$
				days = { 180 270 }
			}
		}
	}
	else_if = {
		limit = { var:gen_tk_project ?= flag:invention }
		if = {
			limit = { has_character_flag = gen_tk_hurried }
			trigger_event = {
				id = $EVENT$
				days = { 203 270 }
			}
		}
		else = {
			trigger_event = {
				id = $EVENT$
				days = { 270 360 }
			}
		}
	}
	else = {
		if = {
			limit = { has_character_flag = gen_tk_hurried }
			trigger_event = {
				id = $EVENT$
				days = { 90 135 }
			}
		}
		else = {
			trigger_event = {
				id = $EVENT$
				days = { 120 180 }
			}
		}
	}
}

# The project is hurried: every wait scheduled from now on is a quarter shorter, and rushed hands slip
# (risk +5). On the tinkerer. Only while the test is still unscheduled (var:gen_tk_stage below 3) —
# a stage event already on its way cannot be moved, so after that there is nothing left to cut.
# Being asked is what stings, whatever the answer: gen_tk_hurried_me_opinion is given where the
# request is made (the hurry interaction, the patron's letter at the design stage), not here.
gen_tk_hurry_project_effect = {
	# Already hurried: no faster, and no second helping of risk.
	if = {
		limit = { has_character_flag = gen_tk_hurried }
		custom_tooltip = gen_tk_hurried_already.tt
	}
	else_if = {
		limit = {
			has_character_flag = gen_tk_working
			exists = var:gen_tk_stage
			var:gen_tk_stage < 3
		}
		add_character_flag = gen_tk_hurried
		custom_tooltip = gen_tk_hurried.tt
		gen_tk_adjust_effect = { QUALITY = 0 RISK = 5 }
	}
	else = {
		custom_tooltip = gen_tk_hurried_too_late.tt
	}
}

# The workbench shows on the character sheet while a project is on: which project, and a little calm
# from the work. Added at begin, removed with the rest of the state.
gen_tk_add_bench_modifier_effect = {
	if = {
		limit = { var:gen_tk_project ?= flag:curio }
		add_character_modifier = gen_tk_at_bench_curio_modifier
	}
	else_if = {
		limit = { var:gen_tk_project ?= flag:mill }
		add_character_modifier = gen_tk_at_bench_mill_modifier
	}
	else_if = {
		limit = { var:gen_tk_project ?= flag:engine }
		add_character_modifier = gen_tk_at_bench_engine_modifier
	}
	else_if = {
		limit = { var:gen_tk_project ?= flag:limb }
		add_character_modifier = gen_tk_at_bench_limb_modifier
	}
	else_if = {
		limit = { var:gen_tk_project ?= flag:invention }
		add_character_modifier = gen_tk_at_bench_invention_modifier
	}
}

gen_tk_remove_bench_modifier_effect = {
	remove_character_modifier = gen_tk_at_bench_curio_modifier
	remove_character_modifier = gen_tk_at_bench_mill_modifier
	remove_character_modifier = gen_tk_at_bench_engine_modifier
	remove_character_modifier = gen_tk_at_bench_limb_modifier
	remove_character_modifier = gen_tk_at_bench_invention_modifier
}

# A commission: who pays and who receives. On the tinkerer, before gen_tk_begin_project_effect.
gen_tk_set_patron_effect = {
	set_variable = {
		name = gen_tk_patron
		value = $PATRON$
	}
}

# Better or worse work, more or less danger. On the tinkerer. Either number may be 0 or negative.
gen_tk_adjust_effect = {
	change_variable = {
		name = gen_tk_quality
		add = $QUALITY$
	}
	change_variable = {
		name = gen_tk_risk
		add = $RISK$
	}
	clamp_variable = {
		name = gen_tk_quality
		min = 0
		max = 120
	}
	clamp_variable = {
		name = gen_tk_risk
		min = 0
		max = 80
	}
	# A literal cannot be compared in a trigger, so the two numbers are saved first.
	save_temporary_scope_value_as = {
		name = gen_tk_dq
		value = $QUALITY$
	}
	save_temporary_scope_value_as = {
		name = gen_tk_dr
		value = $RISK$
	}
	# Three sizes of change (5 / 10 / 15+), so options that differ only in how much read differently.
	if = {
		limit = { scope:gen_tk_dq >= 15 }
		custom_tooltip = gen_tk_quality_up_much.tt
	}
	else_if = {
		limit = { scope:gen_tk_dq >= 10 }
		custom_tooltip = gen_tk_quality_up.tt
	}
	else_if = {
		limit = { scope:gen_tk_dq > 0 }
		custom_tooltip = gen_tk_quality_up_little.tt
	}
	else_if = {
		limit = { scope:gen_tk_dq <= -15 }
		custom_tooltip = gen_tk_quality_down_much.tt
	}
	else_if = {
		limit = { scope:gen_tk_dq <= -10 }
		custom_tooltip = gen_tk_quality_down.tt
	}
	else_if = {
		limit = { scope:gen_tk_dq < 0 }
		custom_tooltip = gen_tk_quality_down_little.tt
	}
	if = {
		limit = { scope:gen_tk_dr >= 15 }
		custom_tooltip = gen_tk_risk_up_much.tt
	}
	else_if = {
		limit = { scope:gen_tk_dr >= 10 }
		custom_tooltip = gen_tk_risk_up.tt
	}
	else_if = {
		limit = { scope:gen_tk_dr > 0 }
		custom_tooltip = gen_tk_risk_up_little.tt
	}
	else_if = {
		limit = { scope:gen_tk_dr <= -15 }
		custom_tooltip = gen_tk_risk_down_much.tt
	}
	else_if = {
		limit = { scope:gen_tk_dr <= -10 }
		custom_tooltip = gen_tk_risk_down.tt
	}
	else_if = {
		limit = { scope:gen_tk_dr < 0 }
		custom_tooltip = gen_tk_risk_down_little.tt
	}
}

# Move on to the next stage. On the tinkerer; called once at the end of each stage's events.
gen_tk_next_stage_effect = {
	if = {
		limit = { var:gen_tk_stage = 1 }
		set_variable = {
			name = gen_tk_stage
			value = 2
		}
		gen_tk_schedule_middle_effect = { EVENT = gen_tinker.0200 }
	}
	else = {
		set_variable = {
			name = gen_tk_stage
			value = 3
		}
		gen_tk_schedule_edge_effect = { EVENT = gen_tinker.0300 }
	}
}

# Who receives what the bench makes: the patron of a commission while they live, else the maker.
gen_tk_save_beneficiary_effect = {
	if = {
		limit = {
			exists = var:gen_tk_patron
			var:gen_tk_patron = { is_alive = yes }
		}
		var:gen_tk_patron = { save_scope_as = gen_tk_beneficiary }
	}
	else = {
		save_scope_as = gen_tk_beneficiary
	}
}

# The test. On the tinkerer. Rolls the four results from quality and risk, saves scope:gen_tk_result,
# and hands the result to the outcome event.
gen_tk_final_roll_effect = {
	# The engine draws this list in the option's tooltip with its real chances; each branch's desc
	# names the result, so this IS the odds display (no separate odds line to fall out of step).
	random_list = {
		0 = {
			desc = gen_tk_result_masterwork.tt
			modifier = { add = gen_tk_masterwork_weight_value }
			save_scope_value_as = { name = gen_tk_result value = flag:masterwork }
		}
		0 = {
			desc = gen_tk_result_works.tt
			modifier = { add = gen_tk_works_weight_value }
			save_scope_value_as = { name = gen_tk_result value = flag:works }
		}
		0 = {
			desc = gen_tk_result_dud.tt
			modifier = { add = gen_tk_dud_weight_value }
			save_scope_value_as = { name = gen_tk_result value = flag:dud }
		}
		0 = {
			desc = gen_tk_result_explodes.tt
			modifier = { add = gen_tk_explodes_weight_value }
			save_scope_value_as = { name = gen_tk_result value = flag:explodes }
		}
	}
	gen_tk_save_beneficiary_effect = yes
	trigger_event = gen_tinker.0350
}

# The project is done. On the tinkerer; called once, at the end of the outcome event. Tallies a
# working result, tells a patron (0450), clears the bench, and starts the cooldown.
gen_tk_finish_project_effect = {
	if = {
		limit = {
			OR = {
				scope:gen_tk_result ?= flag:masterwork
				scope:gen_tk_result ?= flag:works
			}
		}
		gen_tk_tally_success_effect = yes
	}
	# A commission rests the bench for a year before the next commission; the maker's own project for
	# three before their next own one. Kept apart so a lord can't keep a gnome courtier on paid work
	# forever, and paid work doesn't lock the gnome out of their own bench.
	if = {
		limit = { exists = var:gen_tk_patron }
		add_character_flag = {
			flag = gen_tk_recently_commissioned
			years = 1
		}
		custom_tooltip = gen_tk_rest_commission.tt
	}
	else = {
		add_character_flag = {
			flag = gen_tk_recently_tinkered
			years = 3
		}
		custom_tooltip = gen_tk_rest_own.tt
	}
	if = {
		limit = {
			exists = var:gen_tk_patron
			var:gen_tk_patron = { is_alive = yes }
		}
		save_scope_as = gen_tk_tinkerer
		# A patron promised one thing and built another (workshop 0105) finds out on delivery (0450);
		# the flag itself is cleared with the bench below, so it rides along as a scope value.
		if = {
			limit = { has_character_flag = gen_tk_wf_patron_fooled }
			save_scope_value_as = {
				name = gen_tk_patron_fooled
				value = yes
			}
		}
		var:gen_tk_patron = { trigger_event = gen_tinker.0450 }
	}
	gen_tk_clear_project_effect = yes
}

# The project is given up: the maker was jailed or fell ill, or the bench was lost. On the tinkerer.
gen_tk_abandon_project_effect = {
	send_interface_toast = {
		type = event_toast_effect_bad
		title = gen_tk_project_abandoned.t
		left_icon = root
		if = {
			limit = { exists = var:gen_tk_patron }
			custom_tooltip = gen_tk_project_abandoned_commission.tt
		}
		else = {
			custom_tooltip = gen_tk_project_abandoned.tt
		}
	}
	# A paid commission refunds half and frees the patron to commission again (the same path as the
	# maker's death or jailing — 00_gen_tinker_commission_effects.txt). It clears var:gen_tk_patron,
	# so the plain notice below only reaches a patron with nothing on record to refund.
	gen_tk_commission_lost_effect = { REASON = abandoned }
	if = {
		limit = {
			exists = var:gen_tk_patron
			var:gen_tk_patron = { is_alive = yes }
		}
		save_scope_as = gen_tk_tinkerer
		var:gen_tk_patron = {
			send_interface_toast = {
				type = event_toast_effect_bad
				title = gen_tk_project_abandoned.t
				left_icon = scope:gen_tk_tinkerer
				right_icon = this
				custom_tooltip = gen_tk_commission_abandoned.tt
			}
		}
	}
	# The abandoned project's next stage event may still be on its way; it carries the old project's
	# serial and is ignored when it lands (gen_tk_event_is_current_trigger), so no lockout is needed.
	gen_tk_clear_project_effect = yes
}

# Everything a project leaves on the maker, gone. Shared by finish and abandon. Hidden: it is
# bookkeeping, and removing the five bench modifiers unconditionally drew a bare "You lose" line
# per modifier in the outcome option's tooltip.
gen_tk_clear_project_effect = {
	hidden_effect = {
		remove_character_flag = gen_tk_working
		remove_character_flag = gen_tk_hurried
		remove_variable = gen_tk_project
		remove_variable = gen_tk_quality
		remove_variable = gen_tk_risk
		remove_variable = gen_tk_stage
		remove_variable = gen_tk_patron
		remove_variable = gen_tk_apprentice
		gen_tk_remove_bench_modifier_effect = yes
		gen_tk_clear_workshop_flags_effect = yes
	}
}

# One more working contraption. Three make a Tinkerer; seven a Master Artificer.
gen_tk_tally_success_effect = {
	if = {
		limit = { NOT = { has_variable = gen_tk_successes } }
		set_variable = {
			name = gen_tk_successes
			value = 0
		}
	}
	change_variable = {
		name = gen_tk_successes
		add = 1
	}
	if = {
		limit = {
			var:gen_tk_successes >= 7
			NOT = { has_character_modifier = gen_tk_master_artificer_modifier }
		}
		if = {
			limit = { has_character_modifier = gen_tk_tinkerer_modifier }
			remove_character_modifier = gen_tk_tinkerer_modifier
		}
		add_character_modifier = gen_tk_master_artificer_modifier
	}
	else_if = {
		limit = {
			var:gen_tk_successes >= 3
			NOT = { has_character_modifier = gen_tk_tinkerer_modifier }
			NOT = { has_character_modifier = gen_tk_master_artificer_modifier }
		}
		add_character_modifier = gen_tk_tinkerer_modifier
	}
}
