# Gnome Tinkering: what a working contraption leaves behind (and what an exploding one does).
# Applied by 00_gen_tinker_outcome_effects.txt from the outcome event gen_tinker.0350. Each project has a
# "works" and a stronger "masterwork" version. See 00_gen_tinker_triggers.txt for the map of the system.

# ---------------------------------------------------------------- Curio (artifact modifiers)
# Ten contraptions, each with a "works" and a stronger masterwork version (gen_tk_deliver_curio_effect
# picks one). Trinkets go in a personal slot and anyone can have them; court pieces need a royal court.
# Each leans on one thing, so two curios rarely do the same job.

# --- Trinkets

	learning = 1
	monthly_prestige = 0.1

	attraction_opinion = 5
}
gen_tk_curio_songbird_masterwork_modifier = {
	learning = 2
	monthly_prestige = 0.25
	stress_loss_mult = 0.15
	attraction_opinion = 10
}
