# Who fades instead of decaying. The two long-lived races — see 00_phenotype_traits.txt for the
# is_immortal flag that switches vanilla elderly health OFF for exactly this pair, and
# 01_gen_aging_traits.txt for the world_weary trait that replaces it.
gen_long_lived_race_trigger = {
	OR = {
		has_trait = phenotype_gracile
		has_trait = phenotype_sylvan
		has_trait = phenotype_dusk_adapted
	}
}

# May the weight of centuries settle on this character this year?
#
# Age 80: with life_expectancy = 50 on both race traits the engine's natural-death window opens
# around 105, so onset at 80 gives a two-to-three-decade decline — the same proportional runway
# vanilla gives a human between infirmity at ~50 and death at ~70.
#
# Respects off_elderly_health for the same reason vanilla's ailment events do: a player who turned
# elderly decay off asked for no decay clock, and this is one — the life_expectancy backstop (and
# the flavored death reason) still applies either way.
gen_world_weary_eligible_trigger = {
	gen_long_lived_race_trigger = yes
	NOT = { has_trait = world_weary }
	age >= 80
	NOT = { has_game_rule = off_elderly_health }
	# A truly immortal character (ruler designer / console) never fades. Checked by TRAIT, not by
	# the is_immortal flag — the flag is deliberately on every elf and deepkin now, the engine
	# `immortal = yes` property lives only on this trait.
	NOT = { has_trait = immortal }
}

# May age catch up with this dwarf this year?
#
# Vanilla's onset is 45 for infirm and 50 for the other four ailments; phenotype_stocky's
# life_expectancy = 20 moves the whole curve twenty years, so onset here is 65. The race trait
# carries is_immortal purely to switch vanilla's own onset off (see 00_phenotype_traits.txt).
#
# One ailment at a time is not the rule — vanilla lets them stack, and so does this — but a
# character must lack at least one of the five, or there is nothing left to add.
gen_dwarf_ailment_eligible_trigger = {
	has_trait = phenotype_stocky
	age >= 65
	is_incapable = no
	NOT = { has_game_rule = off_elderly_health }
	NOT = { has_trait = immortal }
	# Mirrors the per-option triggers of the random_list in gen_dwarf_ailment_pulse_effect, so the
	# list can never be reached with every option ruled out.
	OR = {
		NOT = { has_trait = infirm }
		AND = {
			NOT = { has_game_rule = infirm_only_elderly_health }
			OR = {
				NOT = { has_trait = withering_mind }
				NOT = { has_trait = clouded_eyes }
				NOT = { has_trait = faltering_heart }
				NOT = { has_trait = fragile_bones }
			}
		}
	}
}

# May this woman be offered a late child this year? Scope: the would-be mother.
#
# Vanilla stops female conception at MAX_FEMALE_REPRODUCTION_AGE = 45 for every race, so a
# long-lived woman's extra decades bought her no extra children. gen_late_child_pulse_effect puts
# a rare, chosen pregnancy back in the years past 45 (the offer is gen_aging.0010).
#
# Windows follow each race's aging, not a flat number:
#   - Elves and deepkin: from 45 until world_weary reaches its first level (25 XP, usually a few
#     years after onset at ~85). The fading is the end of the fertile life too.
#   - Dwarves: from 45 to 61. life_expectancy = 20 moves their ailments to 65, and the window
#     closes just ahead of that.
# Explicit trait exclusions rather than a `fertility` threshold: that trigger's relation to the
# engine's age multiplier is unmeasured, and one that folds age in would read zero past 45 and
# silently switch the whole feature off.
gen_late_child_mother_trigger = {
	is_female = yes
	age >= 45
	OR = {
		AND = {
			gen_long_lived_race_trigger = yes
			gen_late_child_not_fading_trigger = yes
		}
		AND = {
			has_trait = phenotype_stocky
			age < 62
		}
	}
	is_pregnant = no
	is_imprisoned = no
	is_incapable = no
	NOT = { has_trait = infertile }
	NOT = { has_trait = celibate }
	NOT = { has_character_flag = gen_late_child_cooldown }
	is_married = yes
	primary_spouse = { gen_late_child_father_trigger = yes }
}

# The primary spouse, checked from the mother's primary_spouse scope (so `prev` is the mother).
# Any race: a mixed couple's child is settled by the on_birth_child race roll like any other.
# Must share her court or her location, so a wife ruling one realm and a husband ruling another
# are not offered a child they could not have.
gen_late_child_father_trigger = {
	is_alive = yes
	is_male = yes
	is_imprisoned = no
	is_incapable = no
	NOT = { has_trait = infertile }
	NOT = { has_trait = eunuch }
	NOT = { has_trait = celibate }
	# A fading elf father is past it too, the same rule as the mother's.
	gen_late_child_not_fading_trigger = yes
	OR = {
		is_in_the_same_court_as = prev
		location = prev.location
	}
}

# Not yet at world_weary's first level. The trait test comes first so the XP is only ever read on
# someone who has the track.
gen_late_child_not_fading_trigger = {
	OR = {
		NOT = { has_trait = world_weary }
		NOT = { has_trait_xp = { trait = world_weary track = world_weary value >= 25 } }
	}
}
