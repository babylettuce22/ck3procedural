# The eight fantasy races, as one question. Not the same as gen_has_phenotype_trigger below —
# phenotype_human is a race too, just not a fantastical one.
gen_has_fantasy_phenotype_trigger = {
	OR = {
		has_trait = phenotype_gracile
		has_trait = phenotype_sylvan
		has_trait = phenotype_stocky
		has_trait = phenotype_rough_hewn
		has_trait = phenotype_towering
		has_trait = phenotype_diminutive
		has_trait = phenotype_dusk_adapted
		has_trait = phenotype_horned
	}
}

# Does this character have horns at all — the horned people, or a Hornborn of any other race?
# What gen_file_horns_decision and the horn-style effects ask. (The portrait side asks the same
# question in the generated gen_race_horns group.)
gen_has_horns_trigger = {
	OR = {
		has_trait = phenotype_horned
		has_trait = hornborn
	}
}

# A Hornborn whose horns are on show — not filed down. What the AI's marriage reluctance asks
# (AI_HORNS_SHOWING_MARRIAGE_PENALTY, Emit/InteractionWriter.cs): the stigma is in what people see.
# The horned people are not here — among their own horns are ordinary, and their marriages outside
# the race already carry the cross-race penalty.
gen_hornborn_horns_showing_trigger = {
	has_trait = hornborn
	NOT = { has_trait = phenotype_horned }
	NOT = { has_character_flag = gen_horns_filed }
}

# Does this character have tusks at all — an orc, or a half-orc at half size? (00_tusk_effects.txt)
gen_has_tusks_trigger = {
	OR = {
		has_trait = phenotype_rough_hewn
		has_character_flag = gen_half_orc
	}
}

# Has one of those tusks been snapped? (00_tusk_effects.txt; drawn by the generated gen_race_tusks group)
gen_has_broken_tusk_trigger = {

		has_character_flag = gen_tusk_broken_left
		has_character_flag = gen_tusk_broken_right
	}
}

# Does this character already have a horn-style flag (00_horn_effects.txt)?
gen_has_horn_style_trigger = {

		has_character_flag = gen_horn_style_ram
		has_character_flag = gen_horn_style_forward
		has_character_flag = gen_horn_style_nubs

# The three elves — high, wood and dusk — who share the pointed ear. What a human child needs one
# parent to be to come out half-elf (gen_resolve_half_race_effect).
gen_is_elf_phenotype_trigger = {
	OR = {
		has_trait = phenotype_gracile
		has_trait = phenotype_sylvan
		has_trait = phenotype_dusk_adapted
	}
}

# Elf or orc by trait OR by genome — for reading the PARENTS of history characters at game start
# (gen_resolve_history_half_race_effect). A minority-race parent who died before the bookmark still
# wears the phenotype_human the generator stamped by culture: the game-start reconcile only reaches
# the living. Their ethnicity was rolled by the engine and says what they really were. The
# ethnicity triggers are generated per map (99_gen_race_ethnicity_triggers.txt, RaceMorphWriter),
# always all of them, `always = no` for a race the map lacks.
gen_is_elf_blooded_trigger = {
	OR = {
		gen_is_elf_phenotype_trigger = yes
		gen_is_high_elf_ethnicity_trigger = yes
		gen_is_wood_elf_ethnicity_trigger = yes
		gen_is_deepkin_ethnicity_trigger = yes
	}
}

gen_is_orc_blooded_trigger = {
	OR = {
		has_trait = phenotype_rough_hewn
		gen_is_orc_ethnicity_trigger = yes
	}
}

# Has this character's race already been decided?
#
# The guard for everything in 00_phenotype_effects.txt. All nine traits (eight fantasy races plus
# human) list each other as `opposites`, so a character should never hold two.
#
# Humans used to be marked with the `gen_phenotype_human` character FLAG instead of a trait,
# because a marker that exists only for bookkeeping should not render a chip in the character view.
# Now that humans carry a visible identity of their own, phenotype_human is one of the traits here
# and answers "was this character considered and found human" — which is a different state from
# "nobody has looked at this character yet", and is what stops the yearly culture pulse converting
# a human child of a mixed marriage to the local race, overruling the parents.
#
# The flag still exists, with a NARROWER meaning: "human born of a mixed line — snap any inherited
# skin shift, ears and tusks off at render time" (consumed by gen_race_skin_human and
# gen_race_morph_mixed_human in the generated 99_gen_race_morphs.txt, which since 2026-09-29 also
# fire on the trait alone, for history characters who are never flagged). It always travels WITH the
# trait, so it no longer needs to be in this OR; a minority-race member holds the trait but never the
# flag, which is what keeps their looks.
#
# NOTE: the traits are `inherit_chance = 0`. They pass NOTHING genetically; every assignment is
# script: the birth hook, the culture pulse, or the ethnicity reconcile. An earlier version of this
# comment said they inherited at 100%, which is what the rest of this system used to be designed
# around — if the code and a comment here ever disagree again, suspect the comment.
gen_has_phenotype_trigger = {
	OR = {
		has_trait = phenotype_human
		gen_has_fantasy_phenotype_trigger = yes
	}
}

# Is this character carrying more than one race at once?
#
# Should be impossible — `opposites` covers all nine and the flag is only set by script — but the
# ruler designer and the console can both produce it, and a character wearing two races has two
# sets of forced genes fighting over the same portrait. Cheap enough to ask once a year alongside
# the assignment; repaired by gen_sanitize_phenotype_effect.
#
# The mixed-line flag conflicts only with a FANTASY trait (it means "human whose inherited skin
# shift must be reset" — nonsense on an orc). Flag beside phenotype_human is the normal, consistent
# state of a mixed-line human.
gen_has_conflicting_phenotype_trigger = {
	OR = {
		# Two or more of the nine race traits at once.
		calc_true_if = {
			amount = 2
			has_trait = phenotype_human
			has_trait = phenotype_gracile
			has_trait = phenotype_sylvan
			has_trait = phenotype_stocky
			has_trait = phenotype_rough_hewn
			has_trait = phenotype_towering
			has_trait = phenotype_diminutive
			has_trait = phenotype_dusk_adapted
			has_trait = phenotype_horned
		}
		AND = {
			has_character_flag = gen_phenotype_human
			gen_has_fantasy_phenotype_trigger = yes
		}
	}
}

# Does this character's own ETHNICITY name a fantasy race?
#
# The per-character race probe the culture pulse never had. The pulse asks what a character's
# CULTURE is, which is right for a courtier invented out of nothing and wrong for a minority-race
# member: they roll their race's ethnicity out of a human culture's weighted list, so the culture
# says human while their body says gnome. The ethnicity IS that roll.
#
# This used to read the gen_race_skin gene with `has_gene`, and never worked: has_gene is an
# INTERFACE trigger, which ck3-tiger accepts and the game refuses at load ("Reading an interface
# trigger 'has_gene' in forbidden area" in error.log), so it answered no for every character. Every
# minority member stayed phenotype_human wearing their race's skin — a lowborn human with dusk elf
# violet skin was the bug that surfaced it.
#
# The seven gen_is_<race>_ethnicity_trigger live in the GENERATED
# common/scripted_triggers/99_gen_race_ethnicity_triggers.txt (Emit/RaceMorphWriter.cs), because
# ethnicity keys are generated and this file may not name them.
gen_has_fantasy_race_ethnicity_trigger = {
	OR = {
		gen_is_high_elf_ethnicity_trigger = yes
		gen_is_wood_elf_ethnicity_trigger = yes
		gen_is_dwarf_ethnicity_trigger = yes
		gen_is_orc_ethnicity_trigger = yes
		gen_is_gnome_ethnicity_trigger = yes
		gen_is_giantkin_ethnicity_trigger = yes
		gen_is_deepkin_ethnicity_trigger = yes
		gen_is_hornkin_ethnicity_trigger = yes
	}
}

# A character whose trait says human and whose ethnicity says otherwise — the minority-race case,
# and the ONLY case this fires on, because everywhere else the two agree by construction.
#
# The `gen_phenotype_human` flag is the exclusion that makes this safe. It means "human of a mixed
# line": the birth roll deliberately made this child human although a parent was not, and a child
# can carry that parent's ethnicity. Reading the ethnicity without this guard would convert every
# half-breed back into the fantasy race the birth roll dropped.
gen_phenotype_contradicts_genes_trigger = {
	has_trait = phenotype_human
	NOT = { has_character_flag = gen_phenotype_human }
	gen_has_fantasy_race_ethnicity_trigger = yes
}

# Checks if root/this and scope:target have different racial phenotypes
# Usage:
#   root = { gen_is_different_race_than = { TARGET = scope:other_char } }
gen_is_different_race_than = {
	$TARGET$ = { save_temporary_scope_as = gen_target_char }
	OR = {
		# One is human (trait, or still untagged), the other is a fantasy race. Anyone without a
		# fantasy trait counts as human here, so a pre-pulse engine courtier compares the same as
		# their traited culture-mates.
		AND = {
			NOT = { gen_has_fantasy_phenotype_trigger = yes }
			scope:gen_target_char = { gen_has_fantasy_phenotype_trigger = yes }
		}
		AND = {
			gen_has_fantasy_phenotype_trigger = yes
			scope:gen_target_char = { NOT = { gen_has_fantasy_phenotype_trigger = yes } }
		}
		# Both are fantasy races, but different traits
		# The two elves are kin for this question: high and wood elves are separate races (each is
		# drawn with its own body) but they married freely while they shared one trait, and the only
		# consumer of this trigger is the AI's marriage reluctance. Their opinion of each other still
		# follows the opposites web.
		AND = {
			OR = {
				has_trait = phenotype_gracile
				has_trait = phenotype_sylvan
			}
			scope:gen_target_char = {
				NOT = { has_trait = phenotype_gracile }
				NOT = { has_trait = phenotype_sylvan }
			}
		}
		AND = {
			has_trait = phenotype_stocky
			scope:gen_target_char = { NOT = { has_trait = phenotype_stocky } }
		}
		AND = {
			has_trait = phenotype_rough_hewn
			scope:gen_target_char = { NOT = { has_trait = phenotype_rough_hewn } }
		}
		AND = {
			has_trait = phenotype_towering
			scope:gen_target_char = { NOT = { has_trait = phenotype_towering } }
		}
		AND = {
			has_trait = phenotype_diminutive
			scope:gen_target_char = { NOT = { has_trait = phenotype_diminutive } }
		}
		AND = {
			has_trait = phenotype_dusk_adapted
			scope:gen_target_char = { NOT = { has_trait = phenotype_dusk_adapted } }
		}
		AND = {
			has_trait = phenotype_horned
			scope:gen_target_char = { NOT = { has_trait = phenotype_horned } }
		}
	}
}
