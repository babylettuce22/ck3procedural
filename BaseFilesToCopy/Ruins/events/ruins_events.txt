namespace = gen_ruins

# Ruination, from the holder's side.
#
#   0001  The warning. Sent once, well before anything is lost.
#   0005  The last chance. Fires at the threshold; pay and the county gets a few more years.
#   0010  The fall. This is the event that actually ruins the county.
#   9000  The debug console for all of it.
#   9010  The mass-ruin debug event: whole domains and whole realms at once.
#
# ---- Why the collapse is an event and not just an effect ----
#
# gen_ruin_county_effect could be called straight from the yearly tick and the county would fall
# exactly the same way. It is fired through an event instead so that a window is on screen when it
# happens. A county disappearing off a player's realm list with no notification is indistinguishable
# from a bug, and the first thing anyone would do is go looking for one.
#
# ---- The scope every event here needs ----
#
#   scope:gen_ruin_target   the county. Saved by whatever fired the event, never rediscovered.
#
# The one exception is the debug hub, which falls back to the player's capital county when it was
# opened from the Events tab rather than from a county. That is the same fallback shape
# wilderness_colonization.0062 uses when it is fired without a colony.

# =====================================================================================
#  0001 — The warning
# =====================================================================================
#
# Fires once per slide toward the threshold, gated by the `ruin_warned` variable in
# gen_ruin_tick_county_effect. The county is not lost at this point and the event says so: there is
# a year or more of margin left, and raising control is what spends it.
#
# No option that fixes anything. A button here that restored control would make the whole mechanic
# a formality — the fix is the ordinary game, played in the ordinary way.

gen_ruins.0001 = {
	type = character_event
	title = gen_ruins.0001.t
	desc = gen_ruins.0001.desc
	theme = realm
	override_background = { reference = wilderness }

	left_portrait = {
		character = root
		animation = worry
	}

	# The debug panel's Events tab bakes a button for every shipped event and fires it at the player
	# with nothing attached. Without this that button would open a window whose title read
	# "[gen_ruin_target.GetName] Is Failing" with the name resolving to nothing — a half-rendered
	# event is worse than no event, and "the button did nothing" is already documented behaviour for
	# that tab. Use gen_ruins.9000 to reach this deliberately.
	trigger = {
		exists = scope:gen_ruin_target
	}

	option = {
		name = gen_ruins.0001.a
		add_stress = minor_stress_impact_gain
	}
}


# =====================================================================================
#  0005 — The last chance
# =====================================================================================
#
# Fires at the threshold INSTEAD of the collapse. The county is one decision away from being lost
# and the holder is offered three ways to buy it back: gold out of the treasury, prestige spent
# calling in what the family name is worth, or piety spent asking the faith for relief. Any of them
# applies the `ruin_reprieve` county modifier for gen_ruin_reprieve_years and knocks the counter back
# to the warning line. The fourth option lets it go, and only that option fires gen_ruins.0010.
#
# ---- Why this exists ----
#
# Ruination is supposed to be rare, and the counter alone cannot make it rarer without also making
# it unreachable: every knob that slows the slide slows it for the neglected county and the tended
# one alike. This separates the two questions. A holder who is paying attention now has something to
# spend, and a county nobody has looked at in fifteen years still falls on exactly the old schedule —
# because nobody is there to press the button.
#
# ---- It is not a free rescue, in three separate ways ----
#
#   the price scales    `major_gold_value` is about a year of the payer's own income, so this is a
#                       real decision at every tier rather than a formality above some wealth line
#   it escalates        each save on the same county makes the next dearer, to three times the base
#   it does not cure    the counter goes back to the WARNING line, not to zero, and the county is
#                       still on the watchlist and still failing. Six more unbroken bad years and it
#                       is here again, at a higher price
#
# What actually saves a county for good is the ordinary tick: one year that is not suffering sets
# the count to zero. The modifier's control, opinion and development bonuses exist to make that year
# reachable — see common/modifiers/00_ruins_modifiers.txt.
#
# ---- The AI plays it too ----
#
# Same event, same prices, same effects; the ai_chance weights on each option decide. That is
# deliberate rather than incidental: a rescue only the player can perform would mean the AI's map
# keeps ruining at the old rate while the player's does not, and the mechanic would read as a tax on
# playing rather than as something happening in the world.
#
# One thing it does NOT do: let a liege save a vassal's county. The event goes to the holder, and
# the holder is whoever is failing. A liege who wants that county tended has the ordinary tools.

gen_ruins.0005 = {
	type = character_event
	title = gen_ruins.0005.t
	desc = gen_ruins.0005.desc
	theme = realm
	override_background = { reference = wilderness }

	left_portrait = {
		character = root
		animation = disapproval
	}

	# Same guard as its neighbours: the debug panel's Events tab fires every shipped event at the
	# player with nothing attached, and this one would otherwise open with no county, no prices and
	# a decline option that calls the collapse. Use gen_ruins.9000 to reach it deliberately.
	trigger = {
		exists = scope:gen_ruin_target
	}

	immediate = {
		# The three prices, lifted into scope values so the DESCRIPTION can state them.
		#
		# Stating them there rather than only on the options is the point: an option whose trigger
		# fails is not drawn at all, so a holder who cannot afford any of the three would otherwise
		# be shown a window with one button on it and no explanation of what the other three would
		# have cost. Printed in the text, the price is visible whether or not it can be paid.
		#
		# Scope values and not script values, for the reason documented on gen_ruins.9000: a .yml
		# cannot read a script value, and the bracket renders as nothing with no error anywhere.
		save_scope_value_as = { name = gen_ruin_gold_cost value = gen_ruin_save_gold_cost }
		save_scope_value_as = { name = gen_ruin_prestige_cost value = gen_ruin_save_prestige_cost }
		save_scope_value_as = { name = gen_ruin_piety_cost value = gen_ruin_save_piety_cost }
		save_scope_value_as = { name = gen_ruin_reprieve_n value = gen_ruin_reprieve_years }
	}

	# ---- Pay for it ----
	#
	# The plainest of the three and the one most rulers will take: money out of the treasury, into
	# the mills and the roads and the wages of people who would otherwise have left.
	option = {
		name = gen_ruins.0005.a_gold

		trigger = {
			gold >= gen_ruin_save_gold_cost
		}

		remove_short_term_gold = gen_ruin_save_gold_cost
		gen_ruin_reprieve_effect = { COUNTY = scope:gen_ruin_target }

		ai_chance = {
			base = 60

			# Affordable is not the same as comfortable. An AI at exactly the price would be
			# spending its last coin on its worst county, which is not what a careful ruler does.
			modifier = {
				add = 60
				gold >= gen_ruin_save_gold_cost_comfortable
			}

			# Worth the money. A developed county is a real asset and losing it is a real loss; a
			# county at development 2 is worth about what the reprieve costs.
			modifier = {
				add = 40
				scope:gen_ruin_target = { development_level >= 10 }
			}

			modifier = {
				add = -40
				has_trait = greedy
			}

			modifier = {
				add = 25
				has_trait = diligent
			}
		}
	}

	# ---- Go there ----
	#
	# Prestige rather than gold: the lord spends what the name is worth — calls in favours, leans on
	# neighbours, puts his own household on the road — instead of opening the treasury. Costs the
	# same in kind and lands in the same place.
	option = {
		name = gen_ruins.0005.b_prestige

		trigger = {
			prestige >= gen_ruin_save_prestige_cost
		}

		add_prestige = gen_ruin_save_prestige_cost_neg
		gen_ruin_reprieve_effect = { COUNTY = scope:gen_ruin_target }

		# A little stress, because this is the option where the ruler personally goes and deals with
		# it. The gold option is a decision; this one is a season of somebody's life.
		add_stress = minor_stress_impact_gain

		ai_chance = {
			base = 30

			modifier = {
				add = 50
				prestige_level >= 3
			}

			modifier = {
				add = 30
				has_trait = ambitious
			}

			modifier = {
				add = -30
				has_trait = lazy
			}
		}
	}

	# ---- Ask the faith ----
	#
	# Piety, and the relief arrives as grain and hands rather than coin. The cheapest of the three
	# for anybody who has been pious and the most expensive for anybody who has not, which is what
	# having three currencies is for.
	option = {
		name = gen_ruins.0005.c_piety

		trigger = {
			piety >= gen_ruin_save_piety_cost
		}

		add_piety = gen_ruin_save_piety_cost_neg
		gen_ruin_reprieve_effect = { COUNTY = scope:gen_ruin_target }

		ai_chance = {
			base = 30

			modifier = {
				add = 50
				piety_level >= 3
			}

			modifier = {
				add = 40
				has_trait = zealous
			}

			modifier = {
				add = 30
				is_theocratic_lessee = yes
			}
		}
	}

	# ---- Let it go ----
	#
	# The only option with no trigger on it, so this window can never be a dead end: a holder who can
	# afford nothing sees one button, and it is this one. It fires the collapse rather than doing it,
	# so that gen_ruins.0010 stays the single place a county is lost and its letter is still read.
	option = {
		name = gen_ruins.0005.d_decline

		trigger_event = gen_ruins.0010

		ai_chance = {
			base = 25

			# An AI that cannot pay for any of the three will be choosing this whatever the weight
			# says — the others are not drawn. What this modifier is for is the AI that CAN pay and
			# should not: one county among forty, poor, and not worth a year's income.
			modifier = {
				add = 60
				scope:gen_ruin_target = { development_level <= 4 }
			}

			modifier = {
				add = 40
				has_trait = greedy
			}
		}
	}
}


# =====================================================================================
#  0010 — The fall
# =====================================================================================

gen_ruins.0010 = {
	type = character_event
	title = gen_ruins.0010.t
	desc = gen_ruins.0010.desc
	theme = realm
	override_background = { reference = wilderness }

	left_portrait = {
		character = root
		animation = grief
	}

	# Reached from gen_ruins.0005's decline option, never from the yearly tick directly — the tick
	# fires the last chance and this is what the last chance costs when nobody takes it. The debug
	# console reaches the collapse by its effect instead, past every veto.
	#
	# Same guard as 0001, and it matters more here: this event RUINS A COUNTY in its immediate. Fired
	# from the Events tab with no target it would either half-render or, with a fallback, quietly
	# destroy the player's capital past every veto. Declining is the only right answer. The debug
	# console's "Collapse it now" is the deliberate way in.
	trigger = {
		exists = scope:gen_ruin_target
	}

	immediate = {
		# Before the effect, not after. gen_ruin_county_effect revokes the title, so anything asking
		# a question about the county's holder has to ask it here.
		# Before the collapse for the same reason, and it matters more now than it did when this was
		# one literal string: gen_ruin_county_effect calls gen_ruin_forget_county_effect on its way
		# through, which wipes the tally. Everything this line reports stops existing four lines
		# below it.
		if = {
			limit = { debug_only = yes }
			scope:gen_ruin_target = {
				gen_ruin_debug_payload_effect = yes
				debug_log = gen_ruin_log_fall
			}
		}

		gen_ruin_county_effect = { COUNTY = scope:gen_ruin_target }
	}

	option = {
		name = gen_ruins.0010.a
		add_prestige = medium_prestige_loss
	}
}


# =====================================================================================
#  9000 — Debug console
# =====================================================================================
#
# One event with many options rather than many events, and that is the point: the Events tab of the
# generated debug panel bakes one button per event, so a dozen separate debug events would be a
# dozen buttons in a list of sixty. This is one button that opens everything.
#
# Every rung of the mechanic is reachable from here without waiting for a war:
#
#   read      what the counter currently says, and whether the vetoes would allow a fall
#   warn      jump to the warning point — the next tick sends the letter
#   doom      jump to the threshold — the next tick collapses it
#   tick      run a year's evaluation right now, instead of waiting for one
#   fall      collapse it immediately, BYPASSING the vetoes
#   stones    place ruins at either rung without collapsing, to test the discovery path on its own
#   clear     take the stones back off, so the same county can be used again
#   reset     forget everything and take the county off the watchlist
#   aim       point the console at a different county and reopen it
#
# `fall` bypassing the vetoes is deliberate and is the single most useful thing here. The vetoes
# exist so the yearly pulse cannot break a de jure title or unland a ruler, and testing what the
# collapse LOOKS like should not require first finding a county that satisfies all of them.
#
# ---- How it picks a county ----
#
# Reached from the debug panel's Events tab, which — like every button on that tab — fires the event
# at the player with nothing else attached. So it opens on the CAPITAL county and the `aim` options
# at the bottom move it.
#
# It cannot open on whatever county you happen to be looking at, and that is an engine limit rather
# than a decision. There is no global "selected province" datafunction: the county window's province
# lives in `HoldingView`, which appears in exactly one of vanilla's 373 .gui files — its own — and is
# supplied by the engine to that window alone. Any button that could aim at the open county has to
# BE on that window, which is why one briefly was. The `aim` options are what replaced it.

gen_ruins.9000 = {
	type = character_event
	title = gen_ruins.9000.t
	desc = {
		desc = gen_ruins.9000.desc_open

		first_valid = {
			triggered_desc = {
				trigger = { NOT = { exists = scope:gen_ruin_target } }
				desc = gen_ruins.9000.desc_no_county
			}
			triggered_desc = {
				trigger = {
					scope:gen_ruin_target = { gen_county_has_ruins_trigger = yes }
				}
				desc = gen_ruins.9000.desc_already_ruined
			}
			triggered_desc = {
				trigger = {
					scope:gen_ruin_target = { gen_ruin_can_fall_trigger = no }
				}
				desc = gen_ruins.9000.desc_vetoed
			}
			# Somebody has paid. Its own line rather than folding into the one below, because a
			# reprieve is a thing a PLAYER did and the tester's next question is when it runs out,
			# not what is standing on the county.
			triggered_desc = {
				trigger = {
					scope:gen_ruin_target = { has_county_modifier = ruin_reprieve }
				}
				desc = gen_ruins.9000.desc_reprieved
			}
			# Structurally allowed, but not this year: occupied, or the fighting is too recent. A
			# separate line from the one above because the two mean opposite things to a tester —
			# that county will never fall, this one is merely waiting.
			triggered_desc = {
				trigger = {
					scope:gen_ruin_target = { gen_ruin_timing_allows_fall_trigger = no }
				}
				desc = gen_ruins.9000.desc_waiting
			}
			desc = gen_ruins.9000.desc_eligible
		}

		desc = gen_ruins.9000.desc_counter

		# ---- Which veto, by name ----
		#
		# `desc_vetoed` above can only say that SOMETHING refuses this county, because the trigger it
		# asks is an AND of six reasons that answers one boolean. This chain asks the six separately
		# (see the readout triggers at the foot of 00_ruins_triggers.txt) and names the first that
		# refuses, which is the difference between "this will never fall" and knowing whether moving
		# the capital would fix it.
		#
		# First-match rather than a list of every failing clause: a first_valid stops at one, and one
		# is what a tester acts on. The order is the order the real trigger states them in.
		first_valid = {
			# First, and it is not redundant with the chain above. Every clause below reaches through
			# `scope:gen_ruin_target ?= { ... }`, which answers FALSE when there is no target — so a
			# landless player, whose case the chain above already handled with desc_no_county, would
			# fall past all seven clauses and land on desc_veto_unknown, telling them a county they
			# do not have is being refused for an unknown reason.
			triggered_desc = {
				trigger = { NOT = { exists = scope:gen_ruin_target } }
				desc = gen_ruins.9000.desc_blank
			}
			triggered_desc = {
				trigger = { scope:gen_ruin_target ?= { gen_ruin_can_fall_trigger = yes } }
				desc = gen_ruins.9000.desc_veto_none
			}
			triggered_desc = {
				trigger = { scope:gen_ruin_target ?= { gen_ruin_veto_settled_trigger = no } }
				desc = gen_ruins.9000.desc_veto_settled
			}
			triggered_desc = {
				trigger = { scope:gen_ruin_target ?= { gen_ruin_veto_capital_trigger = no } }
				desc = gen_ruins.9000.desc_veto_capital
			}
			triggered_desc = {
				trigger = { scope:gen_ruin_target ?= { gen_ruin_veto_only_county_trigger = no } }
				desc = gen_ruins.9000.desc_veto_only_county
			}
			triggered_desc = {
				trigger = { scope:gen_ruin_target ?= { gen_ruin_veto_dejure_trigger = no } }
				desc = gen_ruins.9000.desc_veto_dejure
			}
			triggered_desc = {
				trigger = { scope:gen_ruin_target ?= { gen_ruin_veto_special_trigger = no } }
				desc = gen_ruins.9000.desc_veto_special
			}
			triggered_desc = {
				trigger = { scope:gen_ruin_target ?= { gen_ruin_veto_holysite_trigger = no } }
				desc = gen_ruins.9000.desc_veto_holysite
			}
			# The six above restate gen_ruin_can_fall_trigger rather than composing it, so they can
			# in principle disagree with it — a clause added there and not here lands exactly here.
			# Saying so is better than rendering nothing and letting the console look broken.
			desc = gen_ruins.9000.desc_veto_unknown
		}

		# ---- Why it is failing ----
		#
		# The per-cause tally, which only exists in debug mode: everything that writes it is behind
		# `debug_only`, so outside -debug_mode there is genuinely nothing to show and saying so is
		# more useful than fifteen zeroes that look like data.
		first_valid = {
			# No target means the readout effect never ran, so every field in desc_telemetry would
			# render blank rather than zero — worse than saying nothing.
			triggered_desc = {
				trigger = { NOT = { exists = scope:gen_ruin_target } }
				desc = gen_ruins.9000.desc_blank
			}
			triggered_desc = {
				trigger = { NOT = { debug_only = yes } }
				desc = gen_ruins.9000.desc_no_telemetry
			}
			desc = gen_ruins.9000.desc_telemetry
		}
	}

	theme = realm
	override_background = { reference = council_chamber }

	left_portrait = {
		character = root
		animation = disbelief
	}

	immediate = {
		# The Events-tab route arrives with no county at all. Falling back to the capital means the
		# button does something rather than opening a window with nothing in it — and the capital is
		# a county every player certainly has.
		if = {
			limit = { NOT = { exists = scope:gen_ruin_target } }
			capital_county ?= { save_scope_as = gen_ruin_target }
		}

		# Read by the description. A county that has never been touched by the counter has no
		# variable at all, and a description that said "0" and one that said "never measured" would
		# be telling the player two different things.
		if = {
			limit = { scope:gen_ruin_target ?= { has_variable = ruin_decay } }
			scope:gen_ruin_target = {
				save_scope_value_as = { name = gen_ruin_decay_now value = var:ruin_decay }
			}
		}
		else = {
			save_scope_value_as = { name = gen_ruin_decay_now value = 0 }
		}

		# The two lines of the ladder, lifted into scope values so the description can print them.
		#
		# A .yml cannot read a script value. `[gen_ruin_decay_threshold|0]` looks like it should
		# work and does not — localisation resolves datafunctions and scopes, and a script value is
		# neither, so the whole bracket renders as nothing. ck3-tiger calls this out as an unknown
		# datafunction, which is the only reason it was caught; the game says nothing at all.
		#
		# Printing them rather than hardcoding "14" and "24" in the string is the point: the numbers
		# live in 00_ruins_values.txt and this readout has to follow them when they are tuned.
		save_scope_value_as = { name = gen_ruin_warn_at value = gen_ruin_decay_warning }
		save_scope_value_as = { name = gen_ruin_fall_at value = gen_ruin_decay_threshold }

		# ---- The tally, for desc_telemetry ----
		#
		# Behind `debug_only` like every other telemetry call, because the readout effect initialises
		# the variables it reads, and initialising them outside debug mode would put telemetry state
		# on a county in an ordinary game. The description has its own branch for that case.
		#
		# `gen_ruin_debug_readout_effect` and not `gen_ruin_debug_payload_effect`, which saves the
		# same numbers under different names. The difference is lifetime and it is not cosmetic: the
		# payload saves TEMPORARY scopes, which end with the effect that saved them, and an event
		# description is evaluated after the immediate has returned — so a description reading them
		# would render blanks. The three scope values above are permanent for exactly this reason.
		if = {
			limit = { debug_only = yes }
			scope:gen_ruin_target ?= { gen_ruin_debug_readout_effect = yes }
		}
	}

	# ---- read ----
	#
	# No effect at all. The description above is the readout; this is the way out of the window
	# without changing anything, and it is first because looking should be the default.
	option = {
		name = gen_ruins.9000.a_read
	}

	# ---- warn ----
	option = {
		name = gen_ruins.9000.b_warn
		trigger = { exists = scope:gen_ruin_target }

		gen_ruin_debug_set_effect = {
			COUNTY = scope:gen_ruin_target
			POINTS = gen_ruin_decay_warning
		}

		custom_tooltip = gen_ruins.9000.b_warn_tt
	}

	# ---- doom ----
	option = {
		name = gen_ruins.9000.c_doom
		trigger = { exists = scope:gen_ruin_target }

		gen_ruin_debug_set_effect = {
			COUNTY = scope:gen_ruin_target
			POINTS = gen_ruin_decay_threshold
		}

		custom_tooltip = gen_ruins.9000.c_doom_tt
	}

	# ---- tick ----
	#
	# Runs the same evaluation the yearly pulse runs, on this county only. Pressing `doom` and then
	# `tick` is the whole collapse path end to end, through the real code, in two clicks.
	option = {
		name = gen_ruins.9000.d_tick
		trigger = { exists = scope:gen_ruin_target }

		scope:gen_ruin_target = { gen_ruin_tick_county_effect = yes }

		custom_tooltip = gen_ruins.9000.d_tick_tt
	}

	# ---- the last chance ----
	#
	# Opens gen_ruins.0005 on this county without waiting for the counter to reach the threshold, so
	# the offer, its three prices and the escalation can be read at any time. Declining it there
	# collapses the county for real, exactly as it would in a game — this is not a preview.
	#
	# Fired rather than simulated: the prices are computed in that event's immediate from the
	# county's own `ruin_saves`, and a debug copy of them would be a second thing to keep in step.
	option = {
		name = gen_ruins.9000.m_offer
		trigger = {
			exists = scope:gen_ruin_target
			scope:gen_ruin_target = { gen_county_has_ruins_trigger = no }
		}

		trigger_event = gen_ruins.0005

		custom_tooltip = gen_ruins.9000.m_offer_tt
	}

	# ---- fall ----
	#
	# Straight to the effect, past gen_ruin_can_fall_trigger. Shown even on a county the vetoes
	# refuse, because that is exactly the case worth being able to look at.
	option = {
		name = gen_ruins.9000.e_fall
		trigger = {
			exists = scope:gen_ruin_target
			scope:gen_ruin_target = { gen_county_has_ruins_trigger = no }
		}

		gen_ruin_county_effect = { COUNTY = scope:gen_ruin_target }

		custom_tooltip = gen_ruins.9000.e_fall_tt
	}

	# ---- stones, large ----
	#
	# The discovery path without the collapse: a county that keeps its holder and its development but
	# has ruins standing in it. That is not a state the mechanic ever produces on its own, and it is
	# the fastest way to reach wilderness_ruins.0001.
	#
	# Two options rather than one because the RUNG is the thing being tested. From large the ladder
	# is two construction jobs; from small it is one, and only the second of them fires the event.
	option = {
		name = gen_ruins.9000.f_large
		trigger = {
			exists = scope:gen_ruin_target
			scope:gen_ruin_target = { gen_county_has_ruins_trigger = no }
		}

		# `add_building` directly rather than through gen_ruins_place_effect, because that effect is
		# a 50/50 coin and the whole point of these two options is to force a KNOWN rung. Guarded
		# the same way it is, so pressing either twice cannot stack a second building.
		scope:gen_ruin_target = {
			title_province ?= {
				if = {
					limit = {
						NOR = {
							has_building = ruins_large
							has_building = ruins_small
							has_building = ruins_cleared
						}
					}
					add_building = ruins_large
				}
			}
		}

		custom_tooltip = gen_ruins.9000.f_large_tt
	}

	# ---- stones, small ----
	option = {
		name = gen_ruins.9000.f_small
		trigger = {
			exists = scope:gen_ruin_target
			scope:gen_ruin_target = { gen_county_has_ruins_trigger = no }
		}

		scope:gen_ruin_target = {
			title_province ?= {
				if = {
					limit = {
						NOR = {
							has_building = ruins_large
							has_building = ruins_small
							has_building = ruins_cleared
						}
					}
					add_building = ruins_small
				}
			}
		}

		custom_tooltip = gen_ruins.9000.f_small_tt
	}

	# ---- clear ----
	#
	# Removes whichever rung is standing, directly. This is the ONE thing nothing else in the mod
	# does — the whole design says a ruin may only leave through ruins_cleared's on_complete. Doing
	# it here does not weaken that: it is what lets the same county be used for a second run without
	# reloading, and it deliberately does NOT fire the discovery event, so the two paths stay
	# distinguishable in the log and on screen.
	option = {
		name = gen_ruins.9000.g_clear
		trigger = {
			exists = scope:gen_ruin_target
			scope:gen_ruin_target = { gen_county_has_ruins_trigger = yes }
		}

		scope:gen_ruin_target = {
			title_province ?= {
				# All three, unconditionally guarded. Only one can be standing, but which one is the
				# question this option exists to stop having to ask.
				if = { limit = { has_building = ruins_large }   remove_building = ruins_large }
				if = { limit = { has_building = ruins_small }   remove_building = ruins_small }
				if = { limit = { has_building = ruins_cleared } remove_building = ruins_cleared }
			}

			if = { limit = { has_variable = ruin_former_culture } remove_variable = ruin_former_culture }
			if = { limit = { has_variable = ruin_former_faith }   remove_variable = ruin_former_faith }
			if = { limit = { has_variable = ruin_fell_year }      remove_variable = ruin_fell_year }
		}

		custom_tooltip = gen_ruins.9000.g_clear_tt
	}

	# ---- reset ----
	option = {
		name = gen_ruins.9000.h_reset
		trigger = { exists = scope:gen_ruin_target }

		gen_ruin_forget_county_effect = { COUNTY = scope:gen_ruin_target }

		# The paid-for reprieve as well, which that effect deliberately leaves alone — see the note
		# on it. A tester wants the county back in its virgin state; a player whose county recovered
		# keeps what they bought.
		scope:gen_ruin_target = {
			if = {
				limit = { has_county_modifier = ruin_reprieve }
				remove_county_modifier = ruin_reprieve
			}
		}

		custom_tooltip = gen_ruins.9000.h_reset_tt
	}

	# =================================================================================
	#  Retargeting
	# =================================================================================
	#
	# Each of these picks a different county and reopens the console on it. `save_scope_as` overwrites
	# a scope of the same name, and an event inherits the scopes saved where it was fired from, so
	# re-firing this same event is all the retarget needs to be.
	#
	# This is the ONLY way the console reaches a county other than the capital — see the note at the
	# top of this event for why it cannot simply use whatever county is open. A button on the county
	# window could, and one briefly existed, but it put a debug control on a vanilla window every
	# player sees when the debug decision is the front door for everything else here.

	# The county already furthest along toward falling. The single most useful aim: it finds whatever
	# the real mechanic has been quietly working on while you were not looking.
	option = {
		name = gen_ruins.9000.i_worst
		trigger = {
			any_held_title = {
				tier = tier_county
				has_variable = ruin_decay
			}
		}

		# Descending is the default, and highest decay is what "worst" means here, so no negation.
		ordered_held_title = {
			limit = {
				tier = tier_county
				has_variable = ruin_decay
			}
			order_by = var:ruin_decay
			max = 1

			save_scope_as = gen_ruin_target
		}

		trigger_event = gen_ruins.9000
		custom_tooltip = gen_ruins.9000.i_worst_tt
	}

	# The county with least control, whether or not the counter has ever noticed it. What the mechanic
	# WOULD start on, as against what it already has.
	option = {
		name = gen_ruins.9000.j_lowest
		trigger = {
			any_held_title = { tier = tier_county }
		}

		# `multiply = -1`, because an `ordered_` list sorts DESCENDING and there is no flag to turn
		# that around. Written plainly as `order_by = county_control` this option silently picked the
		# BEST-held county in the realm and called it the worst. Vanilla negates the same way to find
		# its `least_developed_county` in 00_accolades_scripted_effects.txt.
		ordered_held_title = {
			limit = { tier = tier_county }
			order_by = {
				value = county_control
				multiply = -1
			}
			max = 1

			save_scope_as = gen_ruin_target
		}

		trigger_event = gen_ruins.9000
		custom_tooltip = gen_ruins.9000.j_lowest_tt
	}

	# Somebody else's ruin, anywhere on the map — the seeded ones, which nobody holds. The only way
	# to look at a county the generator ruined rather than one the game did.
	option = {
		name = gen_ruins.9000.k_seeded
		trigger = {
			exists = title:k_gen_ruins.holder
			title:k_gen_ruins.holder = {
				any_held_title = { tier = tier_county }
			}
		}

		title:k_gen_ruins.holder = {
			random_held_title = {
				limit = { tier = tier_county }
				save_scope_as = gen_ruin_target
			}
		}

		trigger_event = gen_ruins.9000
		custom_tooltip = gen_ruins.9000.k_seeded_tt
	}

	# Back to where it opens by default, so a retarget is not a one-way door.
	option = {
		name = gen_ruins.9000.l_capital
		trigger = {
			exists = capital_county
			NOT = { scope:gen_ruin_target ?= capital_county }
		}

		capital_county = { save_scope_as = gen_ruin_target }

		trigger_event = gen_ruins.9000
		custom_tooltip = gen_ruins.9000.l_capital_tt
	}
}

# =====================================================================================
#  9010 — Mass ruin
# =====================================================================================
#
# Collapses whole realms at once, for looking at what a map full of holes does to everything
# downstream: the ruins realm's size, the discovery path at scale, the Wilds situation's counters,
# the AI's appetite for reclaiming. The console above does one county at a time and that is the
# right tool for testing the mechanic; this is the tool for testing what the mechanic produces.
#
# Four reaches, in two pairs. `Domain` is what root holds directly; `realm` adds every vassal, and
# their vassals, all the way down. Each comes in a version that leaves every ruler's CAPITAL standing
# and one that does not. Sparing capitals is the one to reach for first: a ruler with no county at
# all is landless, a vassal with no county drops out of the realm, and a player with no county and no
# titular title above it gets the game-over screen. The all-in versions are there for exactly that
# case — seeing what the collapse looks like with nothing held back — and their tooltips say so.
#
# Everything goes through gen_ruin_county_effect, and so through abandon_county_effect under it, so a
# county ruined here is indistinguishable from one the yearly pulse took: same stones, same
# obstacles, same memory of who lived there, same hand-off to the ruins dummy. The vetoes are
# bypassed the way the console's `fall` bypasses them.
#
# Fired from the debug panel's Events tab at the player with nothing attached, so root is the
# player and that is the only scope it needs.

gen_ruins.9010 = {
	type = character_event
	title = gen_ruins.9010.t
	desc = {
		desc = gen_ruins.9010.desc_open

		first_valid = {
			triggered_desc = {
				trigger = { NOT = { exists = title:k_gen_ruins.holder } }
				desc = gen_ruins.9010.desc_no_dummy
			}
			triggered_desc = {
				trigger = { gen_ruin_debug_realm_count <= 0 }
				desc = gen_ruins.9010.desc_nothing
			}
			desc = gen_ruins.9010.desc_counts
		}
	}

	theme = realm
	override_background = { reference = council_chamber }

	left_portrait = {
		character = root
		animation = worry
	}

	immediate = {
		# Lifted into scope values for the description; a .yml cannot read a script value directly.
		save_scope_value_as = { name = gen_ruin_domain_n value = gen_ruin_debug_domain_count }
		save_scope_value_as = { name = gen_ruin_realm_n value = gen_ruin_debug_realm_count }
	}

	# ---- nothing ----
	option = {
		name = gen_ruins.9010.a_cancel
	}

	# ---- domain, sparing the capital ----
	option = {
		name = gen_ruins.9010.b_domain_spare
		trigger = {
			exists = title:k_gen_ruins.holder
			gen_ruin_debug_domain_count > 0
		}

		gen_ruin_debug_collect_domain_effect = { SPARE_CAPITAL = yes }
		gen_ruin_debug_ruin_batch_effect = yes

		custom_tooltip = gen_ruins.9010.b_domain_spare_tt
	}

	# ---- domain, everything ----
	option = {
		name = gen_ruins.9010.c_domain_all
		trigger = {
			exists = title:k_gen_ruins.holder
			gen_ruin_debug_domain_count > 0
		}

		gen_ruin_debug_collect_domain_effect = { SPARE_CAPITAL = no }
		gen_ruin_debug_ruin_batch_effect = yes

		custom_tooltip = gen_ruins.9010.c_domain_all_tt
	}

	# ---- realm, sparing every capital ----
	#
	# The whole list is gathered before the first county falls, across root and every vassal below.
	# That matters more here than in the domain options: collapsing a vassal's last county unlands
	# them and drops them out of `every_vassal_or_below`, and a walk that was still in progress
	# would then be walking a tree that is changing under it.
	option = {
		name = gen_ruins.9010.d_realm_spare
		trigger = {
			exists = title:k_gen_ruins.holder
			gen_ruin_debug_realm_count > 0
			any_vassal_or_below = { always = yes }
		}

		gen_ruin_debug_collect_domain_effect = { SPARE_CAPITAL = yes }
		every_vassal_or_below = {
			gen_ruin_debug_collect_domain_effect = { SPARE_CAPITAL = yes }
		}
		gen_ruin_debug_ruin_batch_effect = yes

		custom_tooltip = gen_ruins.9010.d_realm_spare_tt
	}

	# ---- realm, everything ----
	option = {
		name = gen_ruins.9010.e_realm_all
		trigger = {
			exists = title:k_gen_ruins.holder
			gen_ruin_debug_realm_count > 0
			any_vassal_or_below = { always = yes }
		}

		gen_ruin_debug_collect_domain_effect = { SPARE_CAPITAL = no }
		every_vassal_or_below = {
			gen_ruin_debug_collect_domain_effect = { SPARE_CAPITAL = no }
		}
		gen_ruin_debug_ruin_batch_effect = yes

		custom_tooltip = gen_ruins.9010.e_realm_all_tt
	}
}
