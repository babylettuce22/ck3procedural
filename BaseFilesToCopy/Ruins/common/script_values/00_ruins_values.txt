# The numbers county ruination is tuned with. All in one file so a change is a change to one line
# and not a hunt through effects.
#
# ---- Why a counter and not a threshold ----
#
# The obvious gate is "development below N", and it is wrong. Every tribal and nomadic realm starts
# near zero development, so an absolute floor either condemns a third of the map on day one or is
# set so low it never fires. The number is not the problem: the QUESTION is. A level says how rich a
# county is; ruination is about whether it is dying, and those come apart hardest exactly where this
# has to be right — a tribal county sitting at 2 forever is healthy, and one that fell from 9 to 2 is
# not, and no threshold can tell them apart.
#
# So nothing here is a level. Every point is scored against a county's own recent past or against
# conditions that mean the same thing at any development: control lost, ground occupied, plague in
# the province, a population that hates whoever claims to hold it. A county has to hold one of those
# every year without a break before anything happens, and a single good year wipes the count.

# ---- The ladder ------------------------------------------------------------------------------
#
# At GEN_RUIN_DECAY_WARNING the holder is told, in as many words, that the county is failing. At
# GEN_RUIN_DECAY_THRESHOLD it falls.
#
# The gap between them is the point of having two numbers. It is not a courtesy: a county that
# vanishes with no warning is indistinguishable from a bug, and the player has to have had years
# in which the answer was still "raise the control and it stops".
#
# ---- These count CONSECUTIVE years. They are not a running total ----
#
# They used to be. The counter accumulated and bled off slowly, and the result was that ruination
# measured whether a county had changed hands rather than whether anyone was governing it. Measured
# over 57 years of one 819-county world: 119 counties ruined, 2.1 a year, 14.5% of the map, and the
# rate was not falling. 59 of them had been conquered exactly once, and 48 had never been conquered
# at all. One war was worth twelve points against a threshold of twenty-four while a peaceful year
# paid back two, so a single war took six good years to undo and two wars inside six years was a
# ruin — which is most border counties, in most games.
#
# So a good year no longer pays points back at a couple of points a year. What these numbers measure
# is a sustained run of bad years, and a county that is genuinely put back together never falls at
# all. That is the whole intent: ruination should mean nobody has held this place together for a
# generation, and no running total can say that — it can only say that something bad once happened.
#
# ---- Forgiveness is two-tier, and that is the 2026-09-09 change ----
#
# It used to be one tier: ANY year that was not suffering set the count to zero. Measured over 50
# years of one 1405-county world with the telemetry running, that turned out to be a hair trigger
# firing on noise rather than on recovery. Eighteen counties accumulated past the old threshold of
# fifteen and were wiped anyway — one of them (Vārdia) had THIRTY consecutive bad years erased in a
# single year in which control moved from 3 to 8 and the county was still at development 3 with
# opinion -15. Nothing about that county recovered. `county_control <= var:ruin_last_ctl + 2` simply
# failed for one year because control wobbled up by three instead of two.
#
# The bar for total forgiveness is now what "put back together" actually looks like —
# `gen_ruin_is_recovering_trigger`, control climbing hard or a healthy level of it or development
# rising — and a year that is merely NOT BAD pays back `gen_ruin_decay_bleed` instead of everything.
# So the counter is a leaky integrator with a hard reset on real governance, not a coin toss against
# an annual signal that is noisy at the bottom.
#
# What the numbers buy:
#
#   neglect alone, unbroken        +1 a year         warned in 13 years, falls in 22
#   besieged and looted as well    up to +5 a year   count is full in 5 — but see the shock veto
#   drifting, neither bad nor good -3 a year         three years of neglect undone per quiet year
#   actually being governed        reset to 0        never falls, whatever came before
#
# The last row is the one to check after any retune, and it is guaranteed rather than tuned: a
# recovering year zeroes the count outright, so no amount of past violence can carry a county to the
# threshold on its own so long as somebody eventually governs it.
#
# ---- Where 22 and 3 came from ----
#
# Replayed against the observed RUINLOG of that same 50-year world, using the technique in the
# telemetry notes: reconstruct every county's year-by-year state from the log, apply a candidate
# rule, count threshold crossings, and read the result as a RATIO against the same replay of the
# shipped rule — never as an absolute, because the replay cannot see gen_ruin_can_fall_trigger, the
# timing veto or the reprieve and overcounts by 3.14x on this world.
#
# Two-tier at the old threshold of 15 came out at 1.18x the shipped fall rate, so the threshold had
# to rise to pay for it. 22 lands at 0.59x — about four falls per fifty years in a 1405-county world,
# one ruin every twelve years or so, against the seven per fifty the shipped rule produced.
#
# The bleed is a weak lever and was chosen for meaning rather than for its effect on the rate: 2, 3,
# 4 and 6 all land within a rounding error of each other, because a streak long enough to reach the
# threshold rarely contains a drift year at all. Three is legible — a quiet year undoes three years
# of neglect — and that is the whole reason it is three.
gen_ruin_decay_threshold = 22
gen_ruin_decay_warning = 13

# What one unbroken year of being in a bad way is worth. One, because what the counter counts is
# years — the shocks below are the only thing that can make a year worth more than a year.
gen_ruin_decay_yearly = 1

# What one year of drifting is worth back. A year that is not suffering but is not recovering either
# — nobody is grinding the county down and nobody is putting it right.
#
# Three rather than one because the asymmetry is the mechanic: neglect has to be sustained to count,
# so a year of quiet has to be worth more than a year of rot or the counter would creep upward on any
# county that is merely mediocre forever. Three rather than the whole count because that was the old
# behaviour and it is what this change exists to undo — see the two-tier note above.
#
# This is NOT the old `gen_ruin_decay_recovery`. That one bled a running total that never reset, so
# it was the only brake and had to carry the whole design. This one is the middle rung of three, and
# real recovery still zeroes the count outright.
gen_ruin_decay_bleed = 3

# ---- What counts as being put back together ----------------------------------------------------
#
# Read only by gen_ruin_is_recovering_trigger, which is asked only on a year that is already not
# suffering. Neither of these is a rate lever — see the note on that trigger — so retuning the fall
# rate means moving gen_ruin_decay_threshold, not these.
#
# A level of control at which the county is simply well held. Above the +2 slack in the suffering
# arm by a wide margin, so the two can never both be true.
gen_ruin_recovering_control_level = 20

# Or a one-year gain in control this large. Five rather than three because the suffering arm already
# forgives a gain of two as "treading water": the gap between them is the drift band, and collapsing
# it to nothing would put the hair trigger straight back.
gen_ruin_recovering_control_gain = 5

# ---- What the world does to a county, outside the yearly tick -----------------------------------
#
# Stamped by the on_action hooks the moment the thing happens, rather than inferred a year later
# from its aftermath. A siege that is lifted in March would be invisible to a December sweep, and it
# should not be: somebody still burned the place.
#
# A quarter of what they were. Their job is to make an already-failing county fail faster, never to
# condemn a healthy one: a county that is sacked and then governed properly is forgiven entirely by
# the next good year, and these numbers have to be small enough that the sacking alone cannot
# outrun that. At the old values one war was half a ruin.
gen_ruin_shock_occupied = 1
gen_ruin_shock_siege = 1
gen_ruin_shock_looted = 2

# How long after the last violence a county is safe from falling, whatever the count says.
#
# A ruin is the aftermath, not the battle. Without this a county collapses at the moment its count
# peaks, which is while an army is still standing on it and the war has not yet decided who owns it
# — the behaviour that made ruination read as a casualty of war rather than as neglect. Every shock
# restarts the clock, so a county fought over for a decade cannot fall until the fighting stops, and
# then only if it is still failing five years later.
#
# Read by gen_ruin_can_fall_trigger, so it is a veto like the others: the debug console's "collapse
# it now" still goes straight past it, which is the point of that button.
gen_ruin_shock_immunity_years = 5

# ---- Debug readouts ----------------------------------------------------------------------------
#
# Character scope. How many counties the mass-ruin debug event (gen_ruins.9010) would take: the
# character's own domain, and the domain of everyone below them. Counties that already carry ruins
# are not counted, because the event will not touch them either.
gen_ruin_debug_domain_count = {
	value = 0
	every_held_title = {
		limit = {
			tier = tier_county
			gen_county_has_ruins_trigger = no
		}
		add = 1
	}
}

gen_ruin_debug_realm_count = {
	value = gen_ruin_debug_domain_count
	every_vassal_or_below = {
		add = gen_ruin_debug_domain_count
	}
}

# ---- The reprieve: what it costs to refuse to let a county go ---------------------------------
#
# At the threshold the tick no longer collapses the county. It fires gen_ruins.0005, which offers
# the holder — player or AI, the same event and the same numbers — three ways to buy the place
# another few years. Paying applies the `ruin_reprieve` county modifier and knocks the counter back
# down the ladder; declining fires gen_ruins.0010 and the county falls exactly as it used to.
#
# This exists to make ruination rarer than the counter alone makes it, and it does that in two
# different ways at once. A holder who pays gets a county that is measurably easier to govern for a
# few years — see the effects on the modifier — and one good year in that window sets the counter to
# zero for good. A holder who cannot or will not pay loses the county on the same schedule as before.
# So the frequency falls exactly where somebody is paying attention, which is where it should.

# How long the paid-for reprieve holds. Read by gen_ruin_reprieve_effect as the modifier's duration,
# and the modifier is a veto in gen_ruin_timing_allows_fall_trigger — so this is also the shortest
# possible gap between two payments on the same county.
#
# Three rather than one. One year is not enough time for anything the money bought to show up in the
# county's numbers: control growth is monthly, development growth is monthly, and a single year of
# either moves a failing county barely at all. At three the holder has a real chance of a good year,
# which is the only thing that ends the count permanently — and still has to keep paying if they do
# not get one.
gen_ruin_reprieve_years = 3

# Where the counter is put back to when somebody pays.
#
# The warning line, deliberately, and not zero. Zero would mean the payment did the work the holder
# was supposed to do — the county would leave the watchlist entirely and the whole slide would have
# to happen again from nothing, at which point money simply repeals the mechanic.
#
# Put back to the warning line, the county is still visibly failing, the warning flag is still set
# so no second letter is sent, and the earliest it can be back at the threshold is
# (threshold - warning) unbroken bad years. With the shipped ladder that is six.
gen_ruin_reprieve_setback = {
	value = gen_ruin_decay_warning
}

# ---- The price -------------------------------------------------------------------------------
#
# One multiplier and three currencies. All three prices are vanilla's own "major" band, which is the
# band vanilla reserves for a decision a ruler makes a handful of times in a life — and `major_gold_value`
# is not a flat number: it is roughly a year of the payer's income, so a count pays what a count has
# and an emperor pays what an emperor has. A flat price would be ruinous at one end of the map and
# invisible at the other, and ruination happens at both ends.
#
# The escalation is what stops a rich realm from propping a dead county up forever at a fixed price.
# Each save on the same county makes the next one dearer, up to three times the base; a county that
# has been bought back three times is one the holder should either fix or let go.
gen_ruin_save_escalation = {
	value = 1

	# `ruin_saves` is stamped on the county by gen_ruin_reprieve_effect and cleared with the rest of
	# the counter's memory when the county recovers — so a county that gets a good year, drops off
	# the watchlist and starts failing again years later is priced as a first save. That is right:
	# what escalates is repeatedly buying off the SAME collapse, not having had a bad century.
	if = {
		limit = {
			exists = scope:gen_ruin_target
			scope:gen_ruin_target = { has_variable = ruin_saves }
		}
		scope:gen_ruin_target = { add = var:ruin_saves }
	}

	max = 3
}

gen_ruin_save_gold_cost = {
	value = major_gold_value
	multiply = gen_ruin_save_escalation
}

gen_ruin_save_prestige_cost = {
	value = major_prestige_value
	multiply = gen_ruin_save_escalation
}

gen_ruin_save_piety_cost = {
	value = major_piety_value
	multiply = gen_ruin_save_escalation
}

# The same three, negated, because `add_prestige` and `add_piety` are the only way to spend either
# and they take the amount as a signed number. Gold has `remove_short_term_gold` and so needs no
# negative twin.
gen_ruin_save_prestige_cost_neg = {
	value = gen_ruin_save_prestige_cost
	multiply = -1
}

gen_ruin_save_piety_cost_neg = {
	value = gen_ruin_save_piety_cost
	multiply = -1
}

# Three times the gold price, read only by the AI weights on gen_ruins.0005.
#
# "Can afford" and "should spend" are different questions and the AI has to be able to ask both. An
# AI holding exactly the price would be emptying its treasury on its worst county, which is not a
# thing a careful ruler does — so affordability opens the option and this decides how eagerly it is
# taken.
gen_ruin_save_gold_cost_comfortable = {
	value = gen_ruin_save_gold_cost
	multiply = 3
}
