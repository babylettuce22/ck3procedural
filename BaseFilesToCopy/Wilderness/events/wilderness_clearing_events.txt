namespace = wilderness_clearing

# What happens when the build queue finishes clearing an obstacle.
#
# Fired by wilderness_clearing_works_done_effect (common/scripted_effects/00_clearing_effects.txt)
# from each `x_cleared` building's on_complete, for human holders only, about half the time. The
# other half keeps the plain toast. Three events per obstacle:
#
#   01xx beast_dens    02xx bandit_camps    03xx dense_growth
#   04xx flooded_lands 05xx scree_slopes    06xx killing_cold
#
# Scopes arriving from the building: scope:obstacle_province (the capital province, where the
# building was) and scope:obstacle_county. wilderness_clearing_setup_effect adds scope:foreman, who
# always exists, and scope:champion, who may not.
#
# House rules these follow:
#   * Every roll ends in a toast, on the failure branch too, and a toast does not repeat what the
#     event window already said.
#   * A gamble has a sure option beside it. Where the risk is losing the clearance itself, it goes
#     through wilderness_clearing_obstacle_returns_effect, whose tooltip names the blocker.
#   * The immediate block sets a 15-year "seen" flag, which the dispatcher reads to make repeats less
#     likely. The trigger asks only for the scopes, so a fired event cannot fail silently.

#  Beasts

# The Pack Leader: one animal the hunters could not kill.
wilderness_clearing.0101 = {
	type = character_event
	title = wilderness_clearing.0101.t
	desc = wilderness_clearing.0101.desc
	theme = hunting
	override_background = { reference = wilderness_forest }

	left_portrait = {
		character = root
		animation = personality_bold
	}
	right_portrait = {
		character = scope:foreman
		animation = worry
	}
	lower_right_portrait = {
		trigger = { exists = scope:champion }
		character = scope:champion
	}

	trigger = {
		exists = scope:obstacle_province
		exists = scope:obstacle_county
	}

	immediate = {
		add_character_flag = { flag = wilderness_clearing_seen_0101 years = 15 }
		wilderness_clearing_setup_effect = yes
	}

	# Go yourself
	option = {
		name = wilderness_clearing.0101.a
		skill = prowess

		duel = {
			skill = prowess
			value = decent_skill_rating
			50 = {
				desc = wilderness_clearing.0101.a.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}
				send_interface_toast = {
					type = event_toast_effect_good
					title = wilderness_clearing.0101.a.success
					left_icon = root
					right_icon = scope:obstacle_county
					add_prestige = medium_prestige_gain
					if = {
						limit = { has_royal_court = yes }
						custom_tooltip = wilderness_clearing.0101.a.pelt_tt
						create_artifact_animal_hide_effect = {
							OWNER = root
							HUNTER = root
							LEGENDARY = no
							ANIMAL = flag:wolf
						}
					}
				}
			}
			50 = {
				desc = wilderness_clearing.0101.a.failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -49
				}
				send_interface_toast = {
					type = event_toast_effect_bad
					title = wilderness_clearing.0101.a.failure
					left_icon = root
					right_icon = scope:obstacle_county
					increase_wounds_no_death_effect = { REASON = wild_animal }
					add_prestige = minor_prestige_loss
					wilderness_clearing_obstacle_returns_effect = { OBSTACLE = beast_dens }
				}
			}
		}
		stress_impact = {
			craven = medium_stress_impact_gain
		}
	}

	# Send the champion
	option = {
		name = wilderness_clearing.0101.b
		trigger = { exists = scope:champion }
		highlight_portrait = scope:champion

		scope:champion = {
			duel = {
				skill = prowess
				value = decent_skill_rating
				60 = {
					desc = wilderness_clearing.0101.b.success
					compare_modifier = {
						value = scope:duel_value
						multiplier = 3.5
						min = -59
					}
					root = {
						send_interface_toast = {
							type = event_toast_effect_good
							title = wilderness_clearing.0101.b.success
							left_icon = scope:champion
							right_icon = scope:obstacle_county
							add_prestige = minor_prestige_gain
							scope:champion = {
								add_prestige = medium_prestige_gain
								add_opinion = {
									target = root
									modifier = grateful_opinion
									opinion = 15
								}
							}
						}
					}
				}
				40 = {
					desc = wilderness_clearing.0101.b.failure
					compare_modifier = {
						value = scope:duel_value
						multiplier = -3.5
						min = -39
					}
					root = {
						send_interface_toast = {
							type = event_toast_effect_bad
							title = wilderness_clearing.0101.b.failure
							left_icon = scope:champion
							right_icon = scope:obstacle_county
							scope:champion = { increase_wounds_no_death_effect = { REASON = wild_animal } }
							wilderness_clearing_obstacle_returns_effect = { OBSTACLE = beast_dens }
						}
					}
				}
			}
		}
	}

	# Hire trappers
	option = {
		name = wilderness_clearing.0101.c
		trigger = { gold >= minor_gold_value }
		show_as_unavailable = { always = yes }

		remove_short_term_gold = minor_gold_value
		custom_tooltip = wilderness_clearing.0101.c.tt
	}
}

# The Chief: the last bandit leader, taken alive.
wilderness_clearing.0201 = {
	type = character_event
	title = wilderness_clearing.0201.t
	desc = wilderness_clearing.0201.desc
	theme = martial
	override_background = { reference = army_camp }

	left_portrait = {
		character = root
		animation = personality_honorable
	}
	right_portrait = {
		character = scope:bandit_chief
		animation = prisonhouse
	}

	trigger = {
		exists = scope:obstacle_province
		exists = scope:obstacle_county
	}

	immediate = {
		add_character_flag = { flag = wilderness_clearing_seen_0201 years = 15 }
		wilderness_clearing_setup_effect = yes
		create_character = {
			template = bandit_character
			location = scope:obstacle_province
			culture = scope:obstacle_county.culture
			faith = scope:obstacle_county.faith
			dynasty = none
			save_scope_as = bandit_chief
		}
	}

	# Hang them
	option = {
		name = wilderness_clearing.0201.a

		scope:bandit_chief = {
			death = {
				death_reason = death_execution
				killer = root
			}
		}
		scope:obstacle_county = {
			add_county_modifier = {
				modifier = clearing_gallows
				years = 10
			}
		}
		add_dread = minor_dread_gain
		stress_impact = {
			compassionate = minor_stress_impact_gain
			forgiving = minor_stress_impact_gain
			vengeful = minor_stress_impact_loss
		}
	}

	# Take their service
	option = {
		name = wilderness_clearing.0201.b

		add_courtier = scope:bandit_chief
		scope:bandit_chief = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 20
			}
		}
		stress_impact = {
			vengeful = minor_stress_impact_gain
		}
	}

	# Let them go
	option = {
		name = wilderness_clearing.0201.c

		random_list = {
			70 = {
				desc = wilderness_clearing.0201.c.success
				send_interface_toast = {
					type = event_toast_effect_good
					title = wilderness_clearing.0201.c.success
					left_icon = root
					right_icon = scope:obstacle_county
					add_piety = minor_piety_gain
				}
			}
			30 = {
				desc = wilderness_clearing.0201.c.failure
				send_interface_toast = {
					type = event_toast_effect_bad
					title = wilderness_clearing.0201.c.failure
					left_icon = scope:bandit_chief
					right_icon = scope:obstacle_county
					wilderness_clearing_obstacle_returns_effect = { OBSTACLE = bandit_camps }
				}
			}
		}
		stress_impact = {
			forgiving = minor_stress_impact_loss
			compassionate = minor_stress_impact_loss
			vengeful = medium_stress_impact_gain
		}
	}
}
