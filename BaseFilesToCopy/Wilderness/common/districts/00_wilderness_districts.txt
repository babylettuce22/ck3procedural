# The holding districts of the two colonisation holdings (CK3 1.21).
#
# Since 1.21 every building stands in a district, and a holding is its `primary_district`
# (common/holdings/00_wilderness_holdings.txt). These two carry what the holdings themselves
# used to: the primary building (the holding's own level chain) and the buildings it may hold.
#
# The blocker buildings appear in BOTH lists on purpose. A blocker is a fact about the ground, not
# about who owns it, so claiming a county does not clear its wolves — it inherits them. That single
# decision is what makes colonisation a project rather than a purchase.
#
# ---- Slots ----
#
# A district has COUNTY_BUILDING_SLOTS / BARONY_BUILDING_SLOTS = 1 slot of its own, plus whatever
# `building_slots_for_host_district` its primary building grants (vanilla's castle_01 grants 1).
# The primary buildings in 00_wilderness_buildings.txt and 00_settlement_buildings.txt set that so
# the counts are what 1.20 measured: an unsettled capital holds two obstacles (see
# wilderness_seed_obstacle_effect), a colony one slot on its first two rungs and two on the third.

wilderness_holding_district = {
	primary_building = wilderness_01

	background = {
		reference = "gfx/interface/illustrations/event_scenes/ep3_campfire.dds"
	}

	type = holding

	buildings = {
		beast_dens
		bandit_camps
		dense_growth
		flooded_lands
		scree_slopes
		killing_cold

		# The two spawn rungs of the ruins ladder — see the note in 00_wilderness_buildings.txt.
		# Listed here so a ruin can stand on a county nobody has claimed yet, which is the normal
		# state of one: a county that falls is unclaimed by definition.
		#
		# `ruins_cleared` is deliberately absent. The wilderness dummy cannot build (its government
		# has `buildings = no`), so the bottom rung would be unreachable here anyway, and a ruin is
		# only ever cleared by somebody who has claimed the ground — at which point the holding is a
		# settlement_holding and the list below applies.
		ruins_large
		ruins_small
	}
}

settlement_holding_district = {
	primary_building = settlement_01

	background = {
		reference = "gfx/interface/illustrations/event_scenes/ep3_campfire.dds"
	}

	type = holding

	buildings = {
		beast_dens
		beast_dens_cleared
		bandit_camps
		bandit_camps_cleared
		dense_growth
		dense_growth_cleared
		flooded_lands
		flooded_lands_cleared
		scree_slopes
		scree_slopes_cleared
		killing_cold
		killing_cold_cleared

		# All three rungs. A colony inherits whatever rung the ground was on when it was claimed and
		# has to work down the rest of it.
		ruins_large
		ruins_small
		ruins_cleared
		colony_timber_works_01
		colony_timber_works_02
		colony_pallisade_01
		colony_pallisade_02
		colony_office_01
		colony_office_02
		colony_office_03

		# The tribal chains. Which set a colony sees is decided in the buildings' own
		# can_construct_potential (colony_is_tribal_trigger), not here — a district's list cannot
		# ask who holds it.
		colony_hall_01
		colony_hall_02
		colony_hall_03
		colony_hunting_camps_01
		colony_hunting_camps_02
		colony_kin_muster_01
		colony_kin_muster_02
	}
}
