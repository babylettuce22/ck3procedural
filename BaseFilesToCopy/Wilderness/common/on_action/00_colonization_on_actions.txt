# Where the system hooks into the running game.
#
# Every hook below APPENDS a custom on_action to a vanilla one rather than redefining the vanilla
# one's effect. That is the single most important compatibility rule in CK3 modding: redefining
# `on_game_start_after_lobby = { effect = ... }` would silently delete every other mod's — and
# several of vanilla's own — game-start work.

# --- Setup ---------------------------------------------------------------------------------------

on_game_start_after_lobby = {
	on_actions = {
		wilderness_on_game_start
	}
}

# Seeds obstacles across every wilderness county.
#
# After the lobby, not during history load. Adding buildings to a holding that history is still
# assembling errors; AGOT hit this and worked around it by commenting the call out of its game-start
# block, leaving the dead code in place with a note. Running it here is the fix rather than the
# workaround.
#
# The generator has already done the expensive half — deciding which counties are wild, and writing
# them into history — so this is one pass over one character's titles, not a world scan.
wilderness_on_game_start = {
	effect = {
		every_ruler = {
			limit = { has_trait = wilderness }
			wilderness_setup_on_start_effect = yes
		}
	}
}


# --- The AI's yearly look at the frontier ---------------------------------------------------------

yearly_playable_pulse = {
	on_actions = {
		wilderness_yearly_pulse
	}
}

wilderness_yearly_pulse = {
	# Gated hard at the top. `trigger` here means the effect below never runs for the great majority
	# of rulers, which is what keeps a yearly pulse off the performance budget — the list iteration
	# in wilderness_pick_adjacent_effect is only reached by someone who actually borders wilderness.
	trigger = {
		is_ai = yes
		can_colonize_at_all_trigger = yes
		NOT = { has_variable = colonization_cooldown }
	}

	effect = {
		# An AI over the cap sheds its worst colony before considering another.
		release_excess_colony_effect = yes

		# --- Finish what it started ------------------------------------------------------------
		#
		# Promotion runs here rather than through the promote_colony_* interactions, and that is not
		# a shortcut — the AI genuinely cannot reach those. CK3 only drives a hidden interaction for
		# the AI when it carries a `special_interaction` tag the engine knows, which is why vanilla's
		# own feudalize reaches the AI and any mod's conversion interaction does not. Ours have
		# ai_will_do weights that nothing ever reads.
		#
		# Without this an AI colonises, clears its blockers, builds up to settlement_03 — and then
		# sits on a permanent colony forever, paying upkeep and taking the wrong-holding penalty,
		# because the last step was the one step it had no way to take.
		#
		# Same effect the player's buttons call, and the same price, so the AI is not quietly on a
		# cheaper path than the player.
		if = {
			limit = {
				gold > promote_colony_cost
				any_held_title = {
					tier = tier_county
					can_promote_colony_trigger = yes
				}
			}

			random_held_title = {
				limit = {
					tier = tier_county
					can_promote_colony_trigger = yes
				}
				promote_colony_effect = { COUNTY = this }
			}

			remove_short_term_gold = promote_colony_cost
		}

		# --- And consider starting another -----------------------------------------------------
		#
		# 6%, down from 40%. At 40 every AI ruler bordering wilderness claimed everything within
		# reach inside two or three years and the frontier was gone before the first succession —
		# the map spent one reign becoming a normal map. Colonisation is supposed to be a thing a
		# realm does once in a while and remembers, not a land rush.
		#
		# The modifiers are what stop 6% being merely slow-and-uniform. A greedy or bold ruler with
		# money to spare is several times likelier to try than a cautious one, so which realms push
		# into the wilderness says something about who is running them — and two neighbours in the
		# same position behave differently, which is most of what makes the frontier feel alive
		# rather than scheduled.
		random = {
			chance = 6

			modifier = { ai_greed > 50     add = 5 }
			modifier = { ai_boldness > 50  add = 4 }
			modifier = { has_trait = ambitious  add = 5 }

			# Genuinely rich, not merely solvent. A ruler sitting on ten times the price is one for
			# whom this is pocket money, and those are the ones who historically did it.
			modifier = {
				gold > { value = colonize_cost  multiply = 10 }
				add = 6
			}

			# Nothing better to do with the men.
			modifier = { is_at_war = no  add = 2 }

			trigger_event = wilderness_colonization.0001
		}
	}
}


# --- Keeping the frontier honest -------------------------------------------------------------------

# When a title changes hands there is nothing to reconcile, because nothing caches a colony count —
# the upkeep rides on the settlement buildings and moves with the county by itself. This hook exists
# only to clear the per-character claim cooldown, which is meaningless on a new holder.
#
# (That absence is the design. See the note at the top of 00_settlement_buildings.txt for what the
# alternative costs: AGOT reconciles a counter from eight call sites, two of them debug buttons.)
on_title_gain = {
	on_actions = {
		wilderness_on_title_gain
	}
}

wilderness_on_title_gain = {
	effect = {
		if = {
			limit = {
				has_variable = colonization_cooldown
				scope:title ?= { tier = tier_county }
				scope:title ?= { is_colony_county_trigger = yes }
			}
			remove_variable = colonization_cooldown
		}
	}
}
