# Everything that MOVES a county between wilderness, colony and settled.
#
# There are exactly three such transitions and each has exactly one effect below. Every entry point
# — the AI pulse, the player decision, the promote interactions — calls one of them and adds
# nothing of its own. AGOT's equivalent logic is inlined at four call sites with the culture rules
# copy-pasted verbatim into each, which is why its adventurer decision and its AI event drifted
# apart; the rule here is that if a caller needs to do something extra, the extra goes in the
# effect, behind a parameter.


# =====================================================================================
#  Seeding: what is in the way
# =====================================================================================

# Province scope. Puts one obstacle on a wilderness province, chosen by the ground it sits on.
#
# Terrain-keyed rather than a flat random draw. AGOT rolls a uniform list with a couple of triggers
# bolted on, so its tundra grows dense forest as readily as its jungle does; the point of a blocker
# is to say something true about the place, and CK3 already knows what the place is.
#
# The weights within each terrain are the flavour; the terrain is the constraint. Every branch is
# reachable for some terrain, and the fallback at the bottom means no province comes out empty.
wilderness_seed_obstacle_effect = {
	random_list = {
		# Cold kills before anything else does.
		10 = {
			trigger = {
				NOT = { has_building = killing_cold }
				OR = { terrain = taiga  terrain = mountains  terrain = desert_mountains }
			}
			modifier = { terrain = taiga  add = 20 }
			add_building = killing_cold
		}

		# Standing water: marsh, delta, river bottom, and any low coast.
		10 = {
			trigger = {
				NOT = { has_building = flooded_lands }
				OR = {
					terrain = wetlands
					terrain = floodplains
					AND = { is_riverside_province = yes  NOT = { terrain = mountains } }
				}
			}
			modifier = { terrain = wetlands  add = 25 }
			add_building = flooded_lands
		}

		# Ground you cannot get a cart across.
		10 = {
			trigger = {
				NOT = { has_building = scree_slopes }
				OR = { terrain = mountains  terrain = desert_mountains  terrain = hills }
			}
			modifier = { terrain = mountains  add = 20 }
			modifier = { terrain = desert_mountains  add = 20 }
			add_building = scree_slopes
		}

		# Trees, and what lives in them.
		10 = {
			trigger = {
				NOT = { has_building = dense_growth }
				OR = { terrain = forest  terrain = jungle  terrain = taiga }
			}
			modifier = { terrain = jungle  add = 25 }
			add_building = dense_growth
		}
		10 = {
			trigger = {
				NOT = { has_building = beast_dens }
				OR = { terrain = forest  terrain = taiga  terrain = hills  terrain = jungle }
			}
			add_building = beast_dens
		}

		# People. Anywhere at all, but likeliest on open ground somebody could live off.
		10 = {
			trigger = { NOT = { has_building = bandit_camps } }
			modifier = {
				OR = { terrain = plains  terrain = steppe  terrain = drylands  terrain = farmlands }
				add = 15
			}
			add_building = bandit_camps
		}
	}
}

# Province scope. The full seeding for one wilderness province: one obstacle usually, two sometimes.
#
# Never zero, unlike AGOT — where a 1-in-6 weight leaves the county with nothing in the way at all,
# which makes a wilderness county indistinguishable from a merely poor one. If a county is worth
# marking as wild, something is stopping people living there.
wilderness_seed_province_effect = {
	# Guaranteed two, chance for 3rd

	wilderness_seed_obstacle_effect = yes
	wilderness_seed_obstacle_effect = yes

	if = {
		limit = { always = yes }
		random = {
			chance = 40 # 40% chance of a third blocker
			wilderness_seed_obstacle_effect = yes
		}
	}
}

# Character scope, called on the wilderness dummy at game start. Seeds every county it holds.
#
# Runs from on_game_start_after_lobby rather than during history load: adding buildings to a holding
# that history is still assembling is the error AGOT worked around by commenting the call out of its
# game-start block and moving it to a post-lobby effect. Do it after the lobby and it just works.
wilderness_setup_on_start_effect = {
	# --- Stop the holder being a person -------------------------------------------------------
	#
	# History can declare the dummy but it cannot stop the engine treating it as a character
	# afterwards, and the engine is thorough: it rolls personality traits, generates a court, finds
	# an heir, and offers the whole thing to the player as a neighbouring ruler with an opinion of
	# them. `disallow_random_traits` in history covers only the first of those.
	#
	# So the cleanup runs here, once, after history has finished loading. Everything below is
	# undoing something the engine did on its own — none of it is flavour.
	clear_traits = yes
	add_trait = wilderness

	# The single most important line. `incapable` takes the character out of the diplomacy,
	# scheme and marriage systems wholesale, which is most of what made it show up as somebody the
	# player could interact with.
	add_trait = incapable

	# A court is a hundred generated characters per dummy, each with an opinion, a portrait and a
	# succession claim on land nobody should inherit.
	every_courtier_or_guest = {
		death = { death_reason = death_vanished }
	}

	# --- Then seed what is in the way ---------------------------------------------------------
	every_held_title = {
		limit = { tier = tier_county }

		title_province ?= {
			if = {
				limit = { has_holding_type = wilderness_holding }
				wilderness_seed_province_effect = yes
			}
		}
	}
}


# =====================================================================================
#  Choosing where to go
# =====================================================================================

# Character scope. Sets scope:wilderness to the best wilderness county this ruler can reach, or
# leaves it unset if there is none.
#
# Shared by the player decision and the AI pulse. That sharing is the point: AGOT scores its
# adjacency separately in the AI event and in the conqueror story cycle, and the two lists have
# drifted — the conqueror version orders on a `scope:curr_county` it sets inside the same block it
# reads it from, which cannot mean what it looks like it means. One list, one scoring, both callers.
#
# The scoring prefers, in order: counties inside a duchy the ruler already holds land in, then a
# kingdom, then an empire, then merely bordering. Filling in the holes in your own duchy before
# striking off across the map is both what a real lord would do and what keeps AI realms compact.
wilderness_pick_adjacent_effect = {
	every_sub_realm_county = {
		every_title_to_title_neighboring_and_across_water_county = {
			limit = {
				is_wilderness_county_trigger = yes
				NOT = { has_county_modifier = block_settlement_ability }
			}
			add_to_list = adjacent_wilderness
		}
	}

	ordered_in_list = {
		list = adjacent_wilderness
		order_by = {
			value = 0

			# Bordering anything of the ruler's at all.
			every_neighboring_county = {
				limit = { holder = root }
				add = 2
			}

			# And progressively better the closer it sits to land they already govern. The three
			# blocks are scored against THIS county's own de jure parents, which is why each one
			# compares `prev` — inside every_neighboring_county, prev is the wilderness county
			# being scored.
			every_neighboring_county = {
				limit = { holder = root  duchy ?= prev.duchy }
				add = 16
			}
			every_neighboring_county = {
				limit = { holder = root  kingdom ?= prev.kingdom }
				add = 8
			}
			every_neighboring_county = {
				limit = { holder = root  empire ?= prev.empire }
				add = 4
			}
		}
		position = 0
		save_scope_as = wilderness
	}
}


# =====================================================================================
#  Transition 1: wilderness -> colony
# =====================================================================================

# Claims a wilderness county for a ruler.
#
#   $COUNTY$     the wilderness county title
#   $COLONIZER$  the character claiming it
#
# The county keeps its obstacles. Claiming ground is not the same as clearing it, and the whole
# shape of the mechanic depends on that: what the player buys here is a liability with potential,
# not a finished county.
colonize_county_effect = {
	$COUNTY$ = { save_scope_as = colony }
	$COLONIZER$ = { save_scope_as = colonizer }

	scope:colony = {
		create_title_and_vassal_change = {
			type = returned
			save_scope_as = change
		}
		change_title_holder = {
			holder = scope:colonizer
			change = scope:change
			take_baronies = yes
		}
		resolve_title_and_vassal_change = scope:change

		# Settlers bring their own. A later pass should derive these from a founder character rather
		# than from the liege — that is what produces frontier heresies and cultural drift — but
		# nothing here may name a specific culture or faith key: these files ship identically for
		# every seed, and the generator's culture keys differ on every one.
		set_county_faith = scope:colonizer.faith
		set_county_culture = scope:colonizer.culture

		generate_coa = yes

		# The capital becomes the colony; anything else in the county stays empty until the colony
		# is promoted and can support real vassals.
		every_county_province = {
			limit = {
				has_holding = yes
				has_holding_type = wilderness_holding
				is_county_capital = yes
			}
			set_holding_type = settlement_holding
		}
		every_county_province = {
			limit = {
				has_holding = yes
				has_holding_type = wilderness_holding
				is_county_capital = no
			}
			remove_holding = yes
		}
		every_county_province = {
			barony ?= { generate_coa = yes }
		}

		# A colony starts at whatever the ground was worth, which is nothing.
		#
		# It used to be handed 2-4 development on founding, and that was too much: development is
		# the one county statistic a ruler can already move on their own, through a steward and
		# through time, so a free two-to-four levels was worth more than the county itself and made
		# claiming wilderness the cheapest development in the game. Settling somewhere empty should
		# buy you the *right* to develop it, not the result.
		#
		# What replaces it is a growth multiplier on settlement_01 — a colony develops faster than
		# ordinary land, because everything built there is new and there is nothing to work around.
		# That rewards the same investment over years instead of at a stroke, and it disappears when
		# the colony is promoted, which is the point at which it has stopped being new.
	}

	scope:colonizer = {
		# The one player-facing moment of colonisation. Fired here rather than from each entry
		# point so the button, the decision and the interaction all reach it; the event's own
		# trigger keeps it away from the AI.
		trigger_event = wilderness_colonization.0010

		# NOTE: this effect does NOT charge for the claim. Every caller pays for itself.
		#
		# That looks like the duplication this file exists to avoid, and it is the one place the rule
		# has to live twice, because `cost` on a decision or an interaction is not a payment — it is
		# the UI affordance. The engine reads it to print the price, grey the button out when the
		# player cannot afford it, and only then deduct. Charging in here as well billed the player
		# twice for the same county while the AI, which has no cost block, paid once.
		#
		# So: the two player entry points declare `cost = { gold = ... }` and the engine handles it;
		# wilderness_colonization.0001 deducts explicitly, one line, right where it fires.

		# One claim at a time, and not soon after the last. Two years was not nearly enough: with
		# the old 40% pulse and a cheap claim, a wealthy AI could take a county every other year
		# and clear a region inside a decade. Five means a ruler founds a couple of colonies in a
		# reign rather than a province of them, and it applies to the player for the same reason —
		# the pause is where a colony either takes hold or does not.
		set_variable = {
			name = colonization_cooldown
			years = 5
		}
	}
}


# =====================================================================================
#  Transition 2: colony -> settled
# =====================================================================================

# Turns a finished colony into a real holding of whatever kind its holder can actually seat.
#
#   $COUNTY$  the colony county title
#
# Government-driven, not a choice, for the same reason the generator's Development.Holding is: each
# government names exactly one primary_holding, and a ruler seated on anything else cannot hold his
# own capital. The promote interactions offer city and church as alternatives where the holder's
# innovations allow it; this is the default path.
promote_colony_effect = {
	$COUNTY$ = {
		# Read from the county, not from a scope the caller was supposed to have set.
		#
		# This used to test `scope:promoting_holder`, which the promote interactions save and the AI
		# pulse does not — and because the test was written `scope:promoting_holder ?= { ... }`, the
		# missing scope failed silently and fell through to castle. Every AI tribal ruler in the game
		# has been raising castles on its colonies since, with nothing in the log to say so.
		#
		# A county always knows its holder. Nothing has to be passed and no caller can forget.
		holder ?= { save_scope_as = colony_promoter }

		# Saved rather than passed on to colony_completion_effect as a parameter, and that is not
		# tidiness. The AI pulse calls this effect as `promote_colony_effect = { COUNTY = this }`, so
		# `$COUNTY$` is the literal text `this` — and handing that down into a block that runs in the
		# colonist's scope would substitute to the COLONIST. A saved scope means the same county
		# whoever is reading it.
		save_scope_as = finished_colony

		# The presence modifier belongs to a colony, and this county is about to stop being one.
		# Cleared here rather than left to colony_refresh_presence_effect because that effect can
		# only reach counties through a character, and a promoted county is no longer one the refresh
		# would look at.
		if = {
			limit = { has_county_modifier = settlement_overseen }
			remove_county_modifier = settlement_overseen
		}

		title_province ?= {
			# Tear the colony down before raising anything on top of it.
			#
			# set_holding_type brings the new holding's own primary building with it, but the
			# settlement chain belongs to no holding type once this county stops being a colony, so
			# nothing removes it. Left in place it holds the slot the castle needs, and the map
			# shows a promoted county as a banner standing on nothing — a holding with no model.
			#
			# Guarded individually rather than assumed: an upgrade replaces the tier below it, so
			# only one of the three is ever present, but which one depends on how far the colony got
			# and it costs nothing to be sure.
			if = { limit = { has_building = settlement_01 }  remove_building = settlement_01 }
			if = { limit = { has_building = settlement_02 }  remove_building = settlement_02 }
			if = { limit = { has_building = settlement_03 }  remove_building = settlement_03 }

			if = {
				limit = { colony_promotes_to_tribe_trigger = yes }
				set_holding_type = tribal_holding
			}
			else = {
				set_holding_type = castle_holding
			}
		}
		set_color_from_title = de_jure_liege
	}

	# --- The lord who came out to live here can go home now ------------------------------------
	#
	# This is what actually ends activity_oversee_colony. Its own phase runs for fifty years, which
	# is a backstop rather than a duration — the real ending is the work being finished, and this is
	# the line that says so.
	#
	# Checked against the activity's LOCATION as well as its type, because a lord may hold several
	# colonies and be living in a different one. Promoting the colony he is not standing in must not
	# call him home from the one he is.
	#
	# `invalidate_activity` rather than `complete_activity`: completion runs an activity's conclusion
	# and its rewards, and there are none here. The activity simply stops mattering, and the engine
	# sends him home the way it sends anybody home from anywhere — see the note at the top of
	# 00_oversee_colony_activity.txt on why that is now the correct behaviour rather than the bug.
	scope:colony_promoter ?= {
		if = {
			limit = {
				exists = involved_activity
				involved_activity = {
					has_activity_type = activity_oversee_colony
					activity_host = prev
					activity_location.county ?= scope:finished_colony
				}
			}
			invalidate_activity = involved_activity
		}
	}

	# A colonist's colony finishing is not a building upgrade, it is the end of an arc: the man stops
	# being a colonist, the frontier stops being a frontier, and there is exactly one moment at which
	# he might decide he no longer needs the lord who sent him out.
	#
	# Tested BEFORE anything changes his government, because the whole question is what he was.
	if = {
		limit = {
			exists = scope:colony_promoter
			scope:colony_promoter = {
				is_colonist_trigger = yes
				exists = liege
			}
		}
		scope:colony_promoter = {
			colony_completion_effect = yes
		}
	}
}


# =====================================================================================
#  Transition 2b: colonist -> ordinary lord, or colonist -> nobody's man
# =====================================================================================

# Character scope: the COLONIST whose colony has just been promoted.
#
# Takes no parameters. The county arrives as `scope:finished_colony`, saved by promote_colony_effect
# — see the note there for why a parameter would have resolved to the wrong thing entirely.
#
# ---- Why the roll lives here ----
#
# Because this is the only moment it means anything. A colonist spends a generation holding a county
# that produces nothing, under a contract with one line in it, at the far end of somebody else's
# map. Seceding at any point in that generation wins him a palisade. Seceding on the day the seat is
# raised wins him a county — and on that same day he stops needing the man who paid for it.
#
# It is a small chance, and it has to be able to not happen: see colony_secession_chance for why a
# frontier that always breaks away is a tax on granting rather than a risk worth taking.
#
# ---- Why the mechanical work happens before the event, not inside it ----
#
# Events can be dropped. A liege who dies in the same tick, a save loaded mid-chain, an on_action
# ordering nobody predicted — any of them leaves a state where the roll happened and the world does
# not reflect it, and that is the class of bug that shows up as a county with two lords. So the
# world is changed here, unconditionally and in one place, and the events below are narration of
# something that has already happened. The player's choice in 0021 is about what to do NEXT, which
# is a choice an event can safely own.
colony_completion_effect = {
	save_scope_as = colonist
	liege = { save_scope_as = colony_liege }

	# He stops being a colonist because there is now a seat here, and a seat is the one thing
	# colony_government cannot hold. Done before the secession roll: after it there is no liege to
	# read, and which government he lands in depends on the realm that sent him.
	if = {
		limit = { colony_promotes_to_tribe_trigger = yes }
		change_government = tribal_government
	}
	else = {
		change_government = feudal_government
	}

	# The roll. Written as a variable rather than as a `random` wrapping both outcomes because
	# `random` has no else branch, and duplicating the whole consequence block inside two random_list
	# entries is how the two halves drift apart.
	random = {
		chance = colony_secession_chance
		set_variable = { name = colony_secedes }
	}

	if = {
		limit = { has_variable = colony_secedes }
		remove_variable = colony_secedes

		create_title_and_vassal_change = {
			type = independency
			save_scope_as = change
		}
		change_liege_or_become_independent = {
			CHANGE = scope:change
			VASSAL = this
		}
		resolve_title_and_vassal_change = scope:change

		# A truce, for the same reason vanilla grants one after every independence war: without it
		# the liege reconquers on the next tick and the whole moment was theatre.
		add_truce_both_ways = {
			character = scope:colony_liege
			years = 5
			name = TRUCE_COLONY_INDEPENDENCE
		}

		add_prestige = major_prestige_gain
		scope:colony_liege = {
			add_prestige = major_prestige_loss

			# The rest of the court noticed. Stacking, because each frontier lost is its own failure.
			every_vassal = {
				limit = { NOT = { this = scope:colonist } }
				add_opinion = {
					target = scope:colony_liege
					modifier = lost_a_colony_opinion
					opinion = -10
				}
			}

			trigger_event = wilderness_colonization.0021
		}
	}
	else = {
		scope:colony_liege = {
			trigger_event = wilderness_colonization.0020
		}
	}
}


# =====================================================================================
#  Granting a colony away
# =====================================================================================

# Character scope: the character who has just GAINED a colony county. Called from on_title_gain.
#
#   $COUNTY$  the county just gained
#
# ---- Why this hangs off on_title_gain rather than an interaction of its own ----
#
# Because vanilla's Grant Title already does the granting, and the AI already uses it. A dedicated
# "settle a colonist here" interaction would be a second way to do a thing the game can do, reachable
# only by the player, and the AI would go on granting colonies through the vanilla path with none of
# this applying. Hooking the gain instead means every route in — the player's grant, the AI's
# surplus-holding logic, inheritance, a war — comes out the same, and none of them had to know.
#
# ---- The asymmetry, which is the content ----
#
# A landless courtier handed a county has been raised up. He becomes a colonist, and he is grateful.
#
# A vassal who already holds land has been sent away. He does NOT become a colonist — he is a count
# with a castle who now also holds a colony, and it is a burden he did not ask for. The penalty
# scales with his rank, which is what makes granting a colony to a troublesome duke read, correctly,
# as exile.
# ---- Split into two, because a grant and an inheritance are not the same event ----
#
# Becoming a colonist follows from the facts on the ground: a man whose only county is a colony is a
# colonist, however he came by it. That half runs for an heir as readily as for a grantee.
#
# Being grateful or resentful about it does not. An heir who inherits his father's colony was not
# raised up by anybody and was not sent anywhere — the opinion half is a reaction to a decision
# somebody made about him, and there was no decision. Running both halves off one hook would give
# every colonist's son a grudge, or a debt, against a liege who did nothing.

# Character scope: whoever now holds a colony county. Makes them a colonist if that is what they are.
colonization_make_colonist_effect = {
	if = {
		# Landless until a moment ago, or landless-but-for-colonies now: this is the whole of what he
		# holds. Testing what he holds NOW rather than what he held before is the reliable form —
		# on_title_gain fires after the transfer, so "held nothing before" is not a question the game
		# can still answer.
		limit = {
			is_colonist_trigger = no
			NOT = {
				any_held_title = {
					tier = tier_county
					is_settled_county_trigger = yes
				}
			}
		}

		change_government = colony_government
	}
}

# Character scope: a character who has just been GRANTED a colony county.
#
#   $COUNTY$  the county just granted
#
# The asymmetry, which is the content.
#
# A landless courtier handed a county has been raised up, and he is grateful. A vassal who already
# holds land has been sent away — he is a count with a castle who now also holds somebody's
# frontier, and the penalty scales with his rank, which is what makes granting a colony to a
# troublesome duke read, correctly, as exile.
colonization_grant_reaction_effect = {
	# An independent ruler cannot be granted anything, but title transfers reach odd states during
	# wars and successions, and a grudge against a liege who does not exist is an error in the log.
	if = {
		limit = { exists = liege }

		$COUNTY$ = { save_scope_as = granted_colony }
		liege = { save_scope_as = granting_liege }

		if = {
			limit = { is_colonist_trigger = yes }

			add_opinion = {
				target = scope:granting_liege
				modifier = colonist_raised_up_opinion
				opinion = 40
			}

			save_scope_as = colonist
			scope:granting_liege = {
				if = { limit = { is_ai = no }  trigger_event = wilderness_colonization.0015 }
			}
		}
		else_if = {
			limit = { highest_held_title_tier >= tier_kingdom }
			add_opinion = {
				target = scope:granting_liege
				modifier = colonist_sent_to_the_frontier_opinion
				opinion = -40
			}
		}
		else_if = {
			limit = { highest_held_title_tier = tier_duchy }
			add_opinion = {
				target = scope:granting_liege
				modifier = colonist_sent_to_the_frontier_opinion
				opinion = -25
			}
		}
		else = {
			add_opinion = {
				target = scope:granting_liege
				modifier = colonist_sent_to_the_frontier_opinion
				opinion = -12
			}
		}
	}
}


# =====================================================================================
#  Presence: the lord is here, or he is not
# =====================================================================================

# Character scope. Brings this character's presence modifiers into line with where they are standing.
#
# ---- Why residence is a variable and not just a location ----
#
# Because CK3 will not leave a landed ruler parked anywhere. His location is derived from where he
# belongs — his capital, or an activity, or a travel plan in progress — and once a one-way plan
# completes the engine puts him back. Every one-way `start_travel_plan` in vanilla belongs to
# somebody whose HOME moved: a ward sent to a new court, a hostage, an adventurer whose domicile
# travels with him. There is no supported "a lord lives here for a while".
#
# So the intent is stored. `var:overseeing_colony` is the county the lord has decided to live in; it
# is set on arrival, survives whatever the engine does about his position, and is cleared only when
# he decides to go home, loses the county, or the colony stops being one.
#
# And because the intent is stored, the location can be ASSERTED. The refresh re-pins him with
# `set_location` whenever the engine has drifted him back to his capital. That is not fighting the
# engine for its own sake: the point of presence over a capital move was that the cost — schemes
# maturing, the council drifting, an heir raised elsewhere — is already in the base game and keys
# off where the man actually is. A residence the engine does not believe in would have the benefits
# and none of the cost, which is the one version of this mechanic not worth shipping.
#
# A colonist needs none of it. His capital IS his colony, so the plain location branch below covers
# him and he never has a variable set.
#
# ---- Why it is a refresh rather than a grant and a revoke ----
#
# Because there are a dozen ways to stop being somewhere and only one way to be there. A lord can
# leave a colony by travelling, by being imprisoned, by losing the county, by the colony being
# promoted out from under him, by dying and passing it on. Writing a matching removal for each of
# those is how a modifier ends up stuck on somebody who is three provinces away.
#
# So nothing removes the modifier in particular. Every hook calls this, it strips whatever is there
# and reapplies only what the world currently justifies, and the answer is correct whatever happened
# in between — including things nobody thought of when it was written.
#
# ---- Who this runs for ----
#
# Everybody who holds a colony, which importantly includes colonists. A colonist's capital IS his
# colony, so he passes the test permanently and holds the modifier permanently — that is not a
# special case, it is the same rule ("a colony produces for whoever is in it") read straight, and it
# is what makes a granted colony worth anything to the man who was given it.
colony_refresh_presence_effect = {
	if = {
		limit = { has_character_modifier = overseeing_the_settlement }
		remove_character_modifier = overseeing_the_settlement
	}

	every_held_title = {
		limit = {
			tier = tier_county
			has_county_modifier = settlement_overseen
		}
		remove_county_modifier = settlement_overseen
	}

	# --- Is he standing in a colony of his own? ------------------------------------------------
	#
	# One branch, one question. This used to be two: a stored `overseeing_colony` variable naming a
	# county he had DECIDED to live in, asserted back onto him with `set_location` whenever the engine
	# had walked him home, and this plain location test underneath it as the fallback.
	#
	# All of that existed to work around a landed ruler not staying where a one-way travel plan put
	# him. activity_oversee_colony makes the engine keep him there itself, which means his location is
	# now simply true and there is nothing left to store or to assert. The variable, the assertion and
	# its `is_travelling` guard all came out together — see the note at the top of
	# 00_oversee_colony_activity.txt.
	#
	# Colonists are excluded deliberately. The modifier says a LIEGE is here overseeing the work; a
	# colonist living in his own colony is not overseeing anything, he is the one being overseen.
	if = {
		limit = {
			character_is_at_own_colony_trigger = yes
			is_colonist_trigger = no
		}

		add_character_modifier = overseeing_the_settlement

		location.county ?= {
			add_county_modifier = settlement_overseen
		}
	}
}


# =====================================================================================
#  Transition 3: colony -> wilderness
# =====================================================================================

# Hands a county back to the wilderness.
#
#   $COUNTY$  the county to abandon
#
# Currently reached only when a ruler is over the cap. It is written as a general effect rather than
# folded into that one caller because reversion is the interesting half of the mechanic and is
# meant to grow teeth later: a colony at low control, with its obstacles uncleared, through a bad
# winter, ought to be able to fail. The machinery for that is this effect and a trigger.
abandon_county_effect = {
	$COUNTY$ = {
		save_scope_as = abandoned

		# The dummy is found by trait, never by character id — the generator names it whatever it
		# likes and this file cannot know what that was.
		random_ruler = {
			limit = { has_trait = wilderness }
			save_scope_as = wilderness_holder
		}

		# Same reason as in promote_colony_effect: nobody is overseeing a county that has gone back to
		# being empty country, and no character-side refresh will ever look at it again.
		if = {
			limit = { has_county_modifier = settlement_overseen }
			remove_county_modifier = settlement_overseen
		}

		if = {
			limit = { exists = scope:wilderness_holder }

			hidden_effect = {
				create_title_and_vassal_change = {
					type = revoked
					save_scope_as = change
				}
				change_title_holder = {
					holder = scope:wilderness_holder
					change = scope:change
					take_baronies = yes
				}
				resolve_title_and_vassal_change = scope:change

				# Culture and faith are left alone. AGOT wipes them to a dedicated unknown pair;
				# we cannot, since those keys are generated per seed — and leaving the last
				# inhabitants' culture on abandoned ground is the better story anyway.
				change_development_level = -100

				every_county_province = {
					limit = { has_holding = yes  is_county_capital = no }
					remove_holding = yes
				}
				every_county_province = {
					limit = { has_holding = yes  is_county_capital = yes }
					if = {
						limit = { NOT = { has_holding_type = wilderness_holding } }
						set_holding_type = wilderness_holding
					}
					# The land goes back to what it was, obstacles and all.
					wilderness_seed_province_effect = yes
				}
			}
		}
	}
}

# Province scope, from every `*_cleared` building's on_complete.
#
# One effect for all six pairs, so a seventh blocker needs no new hook. Deliberately quiet for now:
# clearing an obstacle is its own reward, and the event content that should hang here — the wolves
# that come back, the bandits who ask to be hired, the ruin found under the trees — is the next
# piece of work rather than part of the framework.
wilderness_obstacle_cleared_effect = {
	save_scope_as = cleared_province

	county ?= {
		holder ?= {
			send_interface_toast = {
				title = colonize_obstacle_cleared_toast
				left_icon = scope:cleared_province.county
			}
		}
	}
}

# Character scope. Called from the yearly pulse: an AI over the cap gives one back.
#
# AI only. A player over the cap keeps their colonies and pays for them — the cap exists to stop the
# AI hoarding wilderness it will never develop, and taking a county off a player without asking is
# the kind of thing that reads as a bug however well it is explained.
release_excess_colony_effect = {
	if = {
		limit = {
			is_ai = yes
			wilderness_settlement_headroom < 0
		}

		# The worst one: lowest development, so the AI abandons its failures rather than its
		# investments.
		#
		# `multiply = -1` is load-bearing and is NOT a sign convention quibble. Every `ordered_`
		# list in CK3 sorts DESCENDING, so position 0 is the highest-scoring member; negating the
		# key is how vanilla asks for the lowest, and it does so in 54 places (see
		# 00_accolades_scripted_effects.txt, "Now, find our least developed county"). Without it
		# this effect reads as "abandon the worst" and does the exact opposite.
		ordered_held_title = {
			limit = {
				tier = tier_county
				is_colony_county_trigger = yes
				NOT = { this = root.capital_county }
			}
			order_by = {
				value = development_level
				multiply = -1
			}
			position = 0

			abandon_county_effect = { COUNTY = this }
		}
	}
}
