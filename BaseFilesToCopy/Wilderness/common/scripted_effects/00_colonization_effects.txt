# Everything that MOVES a county between wilderness, colony and settled.
#
# There are exactly three such transitions and each has exactly one effect below. Every entry point
# — the AI pulse, the player decision, the promote interactions — calls one of them and adds
# nothing of its own.

# =====================================================================================
#  Seeding: what is in the way
# =====================================================================================

wilderness_seed_obstacle_effect = {
	random_list = {
		# Cold kills before anything else does.
		10 = {
			trigger = {
				NOT = { has_building = killing_cold }
				OR = { terrain = taiga  terrain = mountains  terrain = desert_mountains }
			}
			modifier = { terrain = taiga  add = 20 }
			add_building = killing_cold
		}

		# Standing water: marsh, delta, river bottom, and any low coast.
		10 = {
			trigger = {
				NOT = { has_building = flooded_lands }
				OR = {
					terrain = wetlands
					terrain = floodplains
					AND = { is_riverside_province = yes  NOT = { terrain = mountains } }
				}
			}
			modifier = { terrain = wetlands  add = 25 }
			add_building = flooded_lands
		}

		# Ground you cannot get a cart across.
		10 = {
			trigger = {
				NOT = { has_building = scree_slopes }
				OR = { terrain = mountains  terrain = desert_mountains  terrain = hills }
			}
			modifier = { terrain = mountains  add = 20 }
			modifier = { terrain = desert_mountains  add = 20 }
			add_building = scree_slopes
		}

		# Trees, and what lives in them.
		10 = {
			trigger = {
				NOT = { has_building = dense_growth }
				OR = { terrain = forest  terrain = jungle  terrain = taiga }
			}
			modifier = { terrain = jungle  add = 25 }
			add_building = dense_growth
		}
		10 = {
			trigger = {
				NOT = { has_building = beast_dens }
				OR = { terrain = forest  terrain = taiga  terrain = hills  terrain = jungle }
			}
			add_building = beast_dens
		}

		# People. Anywhere at all, but likeliest on open ground somebody could live off.
		10 = {
			trigger = { NOT = { has_building = bandit_camps } }
			modifier = {
				OR = { terrain = plains  terrain = steppe  terrain = drylands  terrain = farmlands }
				add = 15
			}
			add_building = bandit_camps
		}
	}
}

wilderness_seed_province_effect = {
	wilderness_seed_obstacle_effect = yes
	wilderness_seed_obstacle_effect = yes

	if = {
		limit = { always = yes }
		random = {
			chance = 40 # 40% chance of a third blocker
			wilderness_seed_obstacle_effect = yes
		}
	}
}

wilderness_setup_on_start_effect = {
	clear_traits = yes
	add_trait = wilderness
	add_trait = incapable

	every_courtier_or_guest = {
		death = { death_reason = death_vanished }
	}

	every_held_title = {
		limit = { tier = tier_county }

		title_province ?= {
			if = {
				limit = { has_holding_type = wilderness_holding }
				wilderness_seed_province_effect = yes
			}
		}
	}
}


# =====================================================================================
#  Choosing where to go
# =====================================================================================

wilderness_pick_adjacent_effect = {
	every_sub_realm_county = {
		every_title_to_title_neighboring_and_across_water_county = {
			limit = { county_can_be_claimed_trigger = yes }
			add_to_list = adjacent_wilderness
		}
	}

	ordered_in_list = {
		list = adjacent_wilderness
		order_by = {
			value = 0

			every_neighboring_county = {
				limit = { holder = root }
				add = 2
			}
			every_neighboring_county = {
				limit = { holder = root  duchy ?= prev.duchy }
				add = 16
			}
			every_neighboring_county = {
				limit = { holder = root  kingdom ?= prev.kingdom }
				add = 8
			}
			every_neighboring_county = {
				limit = { holder = root  empire ?= prev.empire }
				add = 4
			}
		}
		position = 0
		save_scope_as = wilderness
	}
}


# =====================================================================================
#  Before any of it: sending the expedition
# =====================================================================================

colony_launch_expedition_effect = {
	$COUNTY$ = { save_scope_as = colony }
	$COLONIZER$ = { save_scope_as = colonizer }

	scope:colony = {
		add_county_modifier = settling_expedition_underway
		set_variable = { name = expedition_fortune  value = 0 }
	}

	scope:colonizer = {
		set_variable = {
			name = expedition_underway
			value = scope:colony
		}

		colony_pick_expedition_leader_effect = yes

		trigger_event = wilderness_colonization.0010

		trigger_event = {
			id = wilderness_colonization.0014
			months = 3
		}

		trigger_event = {
			id = wilderness_colonization.0011
			months = 7
		}
	}
}

colony_road_roll_effect = {
	if = {
		limit = { has_variable = road_roll_took }
		remove_variable = road_roll_took
	}

	random = {
		chance = $CHANCE$
		set_variable = { name = road_roll_took }
	}
}

# Takes the temporary expedition marks off both the character and the county.
# NOTE: settlers_were_promised is deliberately preserved here so it survives to mature promotion.
colony_clear_expedition_effect = {
	if = {
		limit = { has_variable = expedition_underway }
		remove_variable = expedition_underway
	}

	if = {
		limit = { has_variable = road_roll_took }
		remove_variable = road_roll_took
	}

	$COUNTY$ = {
		if = {
			limit = { has_county_modifier = settling_expedition_underway }
			remove_county_modifier = settling_expedition_underway
		}
		if = { limit = { has_variable = colony_leader }         remove_variable = colony_leader }
		if = { limit = { has_variable = colony_led_by_soldier } remove_variable = colony_led_by_soldier }
		if = { limit = { has_variable = expedition_fortune }    remove_variable = expedition_fortune }
	}
}


# =====================================================================================
#  Transition 1: wilderness -> colony
# =====================================================================================

colonize_county_effect = {
	$COUNTY$ = { save_scope_as = colony }
	$COLONIZER$ = { save_scope_as = colonizer }

	scope:colony = {
		create_title_and_vassal_change = {
			type = returned
			save_scope_as = change
		}
		change_title_holder = {
			holder = scope:colonizer
			change = scope:change
			take_baronies = yes
		}
		resolve_title_and_vassal_change = scope:change

		set_county_faith = scope:colonizer.faith
		set_county_culture = scope:colonizer.culture

		generate_coa = yes

		every_county_province = {
			limit = {
				has_holding = yes
				has_holding_type = wilderness_holding
				is_county_capital = yes
			}
			set_holding_type = settlement_holding
		}
		every_county_province = {
			limit = {
				has_holding = yes
				has_holding_type = wilderness_holding
				is_county_capital = no
			}
			remove_holding = yes
		}
		every_county_province = {
			barony ?= { generate_coa = yes }
		}

		# --- Set Baseline Authority & Rebound Protection ---
		if = {
			limit = { county_control < 50 }
			change_county_control  = 50
		}

		set_variable = {
			name = colony_just_founded
			years = 3
		}
	}

	# The expedition marks come off
	colony_clear_expedition_effect = { COUNTY = $COUNTY$ }
}


# =====================================================================================
#  Who goes out with them
# =====================================================================================

colony_pick_expedition_leader_effect = {
	ordered_courtier_or_guest = {
		limit = {
			is_adult = yes
			is_available_ai_adult = yes
			NOT = { this = root }
		}
		order_by = colony_settling_chance
		position = 0
		save_scope_as = expedition_leader
	}

	if = {
		limit = { exists = scope:expedition_leader }
		scope:colony = {
			set_variable = {
				name = colony_leader
				value = scope:expedition_leader
			}
		}
	}

	ordered_courtier_or_guest = {
		limit = {
			is_adult = yes
			is_available_ai_adult = yes
			NOT = { this = root }
		}
		order_by = {
			value = prowess
			add = martial
		}
		position = 0
		save_scope_as = expedition_soldier
	}
}


# =====================================================================================
#  What the expedition came to
# =====================================================================================

colony_settling_success_effect = {
	# THE claim
	colonize_county_effect = {
		COUNTY = scope:colony
		COLONIZER = root
	}

	# NOTE: If settlers_were_promised is on scope:colony, we leave it untouched.
	# It will silently sit on the county until promotion triggers Event 0025.

	if = {
		limit = { exists = scope:expedition_leader }

		scope:expedition_leader = {
			add_prestige = medium_prestige_gain
			add_character_flag = founded_a_colony
		}
	}

	# The soldier's payoff
	if = {
		limit = { scope:colony ?= { has_variable = colony_led_by_soldier } }

		scope:colony = {
			remove_variable = colony_led_by_soldier
			change_county_control = 15

			title_province ?= {
				if = { limit = { has_building = killing_cold }  remove_building = killing_cold }
				else_if = { limit = { has_building = flooded_lands } remove_building = flooded_lands }
				else_if = { limit = { has_building = bandit_camps }  remove_building = bandit_camps }
				else_if = { limit = { has_building = beast_dens }    remove_building = beast_dens }
				else_if = { limit = { has_building = dense_growth }  remove_building = dense_growth }
				else_if = { limit = { has_building = scree_slopes }  remove_building = scree_slopes }
			}
		}
	}

	trigger_event = wilderness_colonization.0012
}

colony_settling_failure_effect = {
	if = {
		limit = { exists = scope:expedition_leader }

		scope:expedition_leader = {
			random = {
				chance = 35
				death = { death_reason = death_starved }
			}

			if = {
				limit = { is_alive = yes }
				add_stress = major_stress_gain
			}
		}
	}

	# Clean up promise if the expedition never took root
	scope:colony = {
		if = { limit = { has_variable = settlers_were_promised } remove_variable = settlers_were_promised }
	}

	colony_clear_expedition_effect = { COUNTY = scope:colony }

	trigger_event = wilderness_colonization.0013
}


# =====================================================================================
#  Transition 2: colony -> settled
# =====================================================================================

promote_colony_effect = {
	$COUNTY$ = {
		holder ?= { save_scope_as = colony_promoter }
		save_scope_as = finished_colony

		if = {
			limit = { has_county_modifier = settlement_overseen }
			remove_county_modifier = settlement_overseen
		}

		title_province ?= {
			if = { limit = { has_building = settlement_01 }  remove_building = settlement_01 }
			if = { limit = { has_building = settlement_02 }  remove_building = settlement_02 }
			if = { limit = { has_building = settlement_03 }  remove_building = settlement_03 }

			if = {
				limit = { colony_promotes_to_tribe_trigger = yes }
				set_holding_type = tribal_holding
			}
			else = {
				set_holding_type = castle_holding
			}
		}
		set_color_from_title = de_jure_liege
	}

	# End lord in residence activity if present
	scope:colony_promoter ?= {
		if = {
			limit = {
				exists = involved_activity
				involved_activity = {
					has_activity_type = activity_oversee_colony
					activity_host = prev
					activity_location.county ?= scope:finished_colony
				}
			}
			invalidate_activity = involved_activity
		}
	}

	# Transition government for colonists — vassal AND independent alike. This used to also ask
	# `exists = liege`, which looked like a vassals-only gate and was not: the liege scope falls
	# back to self for independent rulers, so the test always passed. That accident was load-
	# bearing — an independent colonist (the pioneer path's normal product) must still convert out
	# of colony_government here or he is a colonist forever — so the condition is simply gone
	# rather than tightened. colony_completion_effect itself now splits on is_independent_ruler
	# and keeps the secession theatre for actual vassals.
	if = {
		limit = {
			exists = scope:colony_promoter
			scope:finished_colony = { is_colony_county_trigger = no }
			scope:colony_promoter = { is_colonist_trigger = yes }
		}
		scope:colony_promoter = {
			colony_completion_effect = yes
		}
	}

	# --- Settlers' Promised Land Reckoning (Payoff Event 0025) ---
	if = {
		limit = {
			scope:finished_colony = { has_variable = settlers_were_promised }
		}

		# is_independent_ruler, not `exists = liege` — the liege scope is self for an independent
		# ruler, so the exists form sent every personal-domain promotion down the "liege" branch.
		# Same recipient either way in that case, but only by luck; say what is meant.
		if = {
			limit = {
				exists = scope:colony_promoter
				scope:colony_promoter = { is_independent_ruler = no }
			}
			scope:colony_promoter.liege = {
				trigger_event = wilderness_colonization.0025
			}
		}
		# If completed directly in personal domain
		else_if = {
			limit = { exists = scope:colony_promoter }
			scope:colony_promoter = {
				trigger_event = wilderness_colonization.0025
			}
		}
	}

	# --- Founder honors ------------------------------------------------------------------------
	#
	# The trait goes to the DIRECT holder at the moment of promotion and to nobody else —
	# scope:colony_promoter is saved from the county's own holder at the top of this effect, so
	# there is no way to collect it from a liege's chair.
	#
	# The tenure gate is the colony_recent_holder clock, set by wilderness_on_title_gain whenever a
	# colony county changes hands (see the note there for why any transfer type counts). No clock
	# means the promoter either claimed the wilderness themselves or has held the colony for ten
	# years since acquiring it; a clock still running means they took a place somebody else raised,
	# and finishing another man's work is not founding.
	#
	# Applied here rather than in the ceremony event, for the reason 0020/0021 give at their top:
	# the effect is the funnel every promotion goes through — player buttons and AI pulse alike —
	# so an AI colonist earns the trait silently, and the trait shows in the promote interaction's
	# own tooltip where a player can see what they are about to earn.
	if = {
		limit = {
			exists = scope:colony_promoter
			NOT = { scope:finished_colony = { has_variable = colony_recent_holder } }
		}
		scope:colony_promoter = { add_trait = colony_founder }
	}

	# The ceremony fires for founder and latecomer both — a promotion is news either way, and the
	# desc split inside 0022 is what tells them apart. Player-only, gated in the event itself.
	scope:colony_promoter ?= {
		trigger_event = wilderness_colonization.0022
	}
}


# =====================================================================================
#  Transition 2b: colonist -> ordinary lord, or colonist -> nobody's man
# =====================================================================================

colony_completion_effect = {
	save_scope_as = colonist

	if = {
		limit = { colony_promotes_to_tribe_trigger = yes }
		change_government = tribal_government
	}
	else = {
		change_government = feudal_government
	}

	# ---- Everything below is about the LIEGE, so an independent colonist skips all of it ----
	#
	# And "independent" has to be asked with is_independent_ruler, not `exists = liege`: for an
	# independent ruler the engine's liege scope falls back to THEMSELF, so the exists test passes
	# and every liege-facing line here fires reflexively — a lord seceding from himself, taking a
	# five-year truce with himself, and being handed 0020 to decide whether to revoke his own
	# county. The pioneer path mints independent colonists as a matter of course, so this is not a
	# corner: it is half the ways a colony gets finished.
	if = {
		limit = { is_independent_ruler = no }

		liege = { save_scope_as = colony_liege }

		# Roll or check for forced test variable
		if = {
			limit = { has_variable = force_colony_secession }
			remove_variable = force_colony_secession
			set_variable = { name = colony_secedes }
		}
		else = {
			random = {
				chance = colony_secession_chance
				set_variable = { name = colony_secedes }
			}
		}

		if = {
			limit = { has_variable = colony_secedes }
			remove_variable = colony_secedes

			# --- Clean Engine Independence Execution ---
			create_title_and_vassal_change = {
				type = independency
				save_scope_as = change
			}
			becomes_independent = {
				change = scope:change
			}
			resolve_title_and_vassal_change = scope:change

			add_truce_both_ways = {
				character = scope:colony_liege
				years = 5
				name = TRUCE_COLONY_INDEPENDENCE
			}

			add_prestige = major_prestige_gain
			scope:colony_liege = {
				add_prestige = major_prestige_loss

				every_vassal = {
					limit = { NOT = { this = scope:colonist } }
					add_opinion = {
						target = scope:colony_liege
						modifier = lost_a_colony_opinion
						opinion = -10
					}
				}

				trigger_event = wilderness_colonization.0021
			}
		}
		else = {
			scope:colony_liege = {
				trigger_event = wilderness_colonization.0020
			}
		}
	}
}


# =====================================================================================
#  Granting a colony away
# =====================================================================================

colonization_make_colonist_effect = {
	if = {
		limit = {
			is_colonist_trigger = no
			NOT = {
				any_held_title = {
					tier = tier_county
					is_settled_county_trigger = yes
				}
			}
		}

		change_government = colony_government

		# ---- Pitch the camp ----
		#
		# colony_government declares `domicile_type = camp`, and the line above is the whole of what
		# brings a camp into being: the engine reads the field and the domicile exists. There is no
		# create_domicile effect in CK3 to call instead — vanilla's own nomad conversion in
		# 00_religious_war.txt goes straight from `change_government = nomad_government` to
		# `domicile = { set_domicile_culture = ... }` with nothing between them, and does not even
		# bother with the `?=` guard.
		#
		# Everything below has to run AFTER that line, because none of it has anything to write into
		# until the government that declares the domicile is the one the character has.
		#
		# ---- Why the move is not optional ----
		#
		# A domicile created this way has no location, and a camp standing nowhere is a camp that
		# does not appear: no map pin, and the HUD's own `ShouldShowDomicileInHUD` has nothing to
		# show. This was the bug — the field alone was shipped, and colonists got no camp.
		#
		# Vanilla never leaves this to chance either. Every path that hands a character a domicile
		# places it in the same breath: game_start.txt moves them to `capital_province`, the Japan
		# house-fief decision does it for the AI after changing government, title_on_actions moves an
		# estate when its owner's realm changes under it. The yurt gets away with `move_with_realm_
		# capital = yes` in its type; the camp does not carry that flag, so a colonist's camp is
		# placed once, here, and thereafter by the player through the move-domicile planner —
		# `can_move_manually = yes` is part of why reusing vanilla's camp type was the right call.
		#
		# The ground is the COLONY, not simply `capital_province`, and the difference is real on the
		# pioneer path: pioneer_success_effect does not call set_realm_capital until after this
		# effect has returned, so a newly-landed pioneer's capital is whatever the engine picked
		# while he was being unmade as an adventurer. Asking for a colony he holds gets the stake
		# either way.
		save_scope_as = new_colonist

		if = {
			limit = { capital_county ?= { is_colony_county_trigger = yes } }
			capital_province ?= { save_scope_as = colony_camp_ground }
		}
		else_if = {
			limit = {
				any_held_title = {
					tier = tier_county
					is_colony_county_trigger = yes
				}
			}
			ordered_held_title = {
				limit = {
					tier = tier_county
					is_colony_county_trigger = yes
				}
				order_by = development_level
				position = 0

				title_province ?= { save_scope_as = colony_camp_ground }
			}
		}

		domicile ?= {
			if = {
				limit = { exists = scope:colony_camp_ground }
				move_domicile = scope:colony_camp_ground
			}

			# ---- The main tent ----
			#
			# An empty domicile window is worse than none: it reads as a broken feature rather than
			# as a camp waiting to be built. camp_main_01 is the right thing to hand over rather than
			# a courtesy — vanilla calls it "always built" in 00_camp_buildings.txt, gives it
			# construction_time 1 and an ai_value of 10000, gates it behind no can_construct at all,
			# and hangs the camp's external building slots off its character_modifier, so a camp
			# without it has nowhere to put anything else.
			#
			# Guarded on absence for the pioneer path in particular. A pioneer already had a camp and
			# vanilla's destroy_laamp_effect struck it on the way through — but that effect is
			# vanilla's and its teardown list is not ours to predict, so this asks rather than
			# assumes. A colonist made by a feudal realm's expedition never had a camp at all.
			if = {
				limit = { NOT = { has_domicile_building = camp_main_01 } }
				add_domicile_building = camp_main_01
			}
		}
	}
}

colonization_grant_reaction_effect = {
	if = {
		limit = { exists = liege }

		$COUNTY$ = { save_scope_as = granted_colony }
		liege = { save_scope_as = granting_liege }

		if = {
			limit = { is_colonist_trigger = yes }

			add_opinion = {
				target = scope:granting_liege
				modifier = colonist_raised_up_opinion
				opinion = 40
			}

			save_scope_as = colonist
			scope:granting_liege = {
				if = { limit = { is_ai = no }  trigger_event = wilderness_colonization.0015 }
			}
		}
		else_if = {
			limit = { highest_held_title_tier >= tier_kingdom }
			add_opinion = {
				target = scope:granting_liege
				modifier = colonist_sent_to_the_frontier_opinion
				opinion = -40
			}
		}
		else_if = {
			limit = { highest_held_title_tier = tier_duchy }
			add_opinion = {
				target = scope:granting_liege
				modifier = colonist_sent_to_the_frontier_opinion
				opinion = -25
			}
		}
		else = {
			add_opinion = {
				target = scope:granting_liege
				modifier = colonist_sent_to_the_frontier_opinion
				opinion = -12
			}
		}
	}
}


# =====================================================================================
#  Presence: the lord is here, or he is not
# =====================================================================================

colony_refresh_presence_effect = {
	if = {
		limit = { has_character_modifier = overseeing_the_settlement }
		remove_character_modifier = overseeing_the_settlement
	}

	every_held_title = {
		limit = {
			tier = tier_county
			has_county_modifier = settlement_overseen
		}
		remove_county_modifier = settlement_overseen
	}

	if = {
		limit = {
			character_is_at_own_colony_trigger = yes
			is_colonist_trigger = no
		}

		add_character_modifier = overseeing_the_settlement

		location.county ?= {
			add_county_modifier = settlement_overseen
		}
	}
}


# =====================================================================================
#  Transition 3: colony -> wilderness
# =====================================================================================

abandon_county_effect = {
	$COUNTY$ = {
		save_scope_as = abandoned

		random_ruler = {
			limit = { has_trait = wilderness }
			save_scope_as = wilderness_holder
		}

		if = {
			limit = { has_county_modifier = settlement_overseen }
			remove_county_modifier = settlement_overseen
		}

		# Clean up any persistent colony variables
		if = { limit = { has_variable = settlers_were_promised } remove_variable = settlers_were_promised }
		if = { limit = { has_variable = colony_just_founded }    remove_variable = colony_just_founded }

		if = {
			limit = { exists = scope:wilderness_holder }

			hidden_effect = {
				create_title_and_vassal_change = {
					type = revoked
					save_scope_as = change
				}
				change_title_holder = {
					holder = scope:wilderness_holder
					change = scope:change
					take_baronies = yes
				}
				resolve_title_and_vassal_change = scope:change

				change_development_level = -100

				every_county_province = {
					limit = { has_holding = yes  is_county_capital = no }
					remove_holding = yes
				}
				every_county_province = {
					limit = { has_holding = yes  is_county_capital = yes }
					if = {
						limit = { NOT = { has_holding_type = wilderness_holding } }
						set_holding_type = wilderness_holding
					}
					wilderness_seed_province_effect = yes
				}
			}
		}
	}
}

# Province scope. Takes the hardest blocker standing on this province off it, and does nothing at
# all on a province that has none.
#
# The ladder itself, in one place. It lived in pioneer_clear_one_obstacle_effect, whose own comment
# says "nothing else should ever grow its own copy of this ladder" — and then the Warden's
# task_clear_the_wilds arrived needing exactly it, from a scope that has a province rather than a
# scope:stake. So the ladder moved down here into the shared file and the pioneer effect became the
# one line that reaches it. Two callers now, no second copy.
#
# Hardest first, which is the rule that makes a single clearance worth a fixed amount: the killing
# cold is the difference between a winnable county and an unwinnable one, and the scree is a rounding
# error. Priced the same way in colony_settling_chance, and the two orders must not drift apart.
#
# No toast and no bookkeeping — deliberately. Both callers report in their own voice (the pioneer
# through pioneer_outcome_toast_effect, the council task through wilderness_obstacle_cleared_effect),
# and an effect that announced itself could not be used by either without doubling up.
wilderness_clear_hardest_obstacle_effect = {
	if = { limit = { has_building = killing_cold }        remove_building = killing_cold }
	else_if = { limit = { has_building = flooded_lands }  remove_building = flooded_lands }
	else_if = { limit = { has_building = bandit_camps }   remove_building = bandit_camps }
	else_if = { limit = { has_building = beast_dens }     remove_building = beast_dens }
	else_if = { limit = { has_building = dense_growth }   remove_building = dense_growth }
	else_if = { limit = { has_building = scree_slopes }   remove_building = scree_slopes }
}


wilderness_obstacle_cleared_effect = {
	save_scope_as = cleared_province

	county ?= {
		holder ?= {
			send_interface_toast = {
				title = colonize_obstacle_cleared_toast
				left_icon = scope:cleared_province.county
			}
		}
	}
}
