# The colony. Three levels between claiming a county and it becoming a real holding.
#
# ---- Where the cost of an empire of colonies lives ----
#
# On `settlement_01.character_modifier`, and nowhere else.
#
# The obvious design — and the one AGOT ships — is a counter variable on the holder, incremented on
# colonisation and decremented on promotion, driving an escalating malus. That counter is a cache of
# something the game already knows, and it goes stale on every path that moves a county without
# passing through colonisation script: inheritance, revocation, conquest, title destruction. AGOT
# carries a 40-line reconciliation effect for it, called from eight places including two debug
# interactions.
#
# Putting the malus on the building instead means the engine maintains it. Ten colonies is ten
# copies of this modifier because it is ten buildings; lose five in a war and the cost halves the
# same tick, with no script involved and nothing to reconcile. `wilderness_settlement_count` in
# script_values still reports the number for anything that wants to *read* it, but nothing has to
# write it.
#
# The trade is that the cost is linear where AGOT's ladder accelerates. That is a fair price for
# deleting an entire class of bug, and a colony that stays merely expensive rather than ruinous is
# also the better game.

settlement_01 = {
	# Placed by script when a county is colonised, never built directly.
	can_construct = { always = no }
	can_construct_potential = { always = no }

	next_building = settlement_02

	# One generic asset — see the note in 00_wilderness_buildings.txt on why there is no
	# graphical_regions filter on generated maps.
	asset = {
		type = pdxmesh
		names = { "building_western_tribal_01_a_mesh" }
		illustration = "gfx/interface/illustrations/holding_types/tribe_grass.dds"
		soundeffect = {
			soundeffect = "event:/SFX/Ambience/3DMapEmitters/Holdings/City/tribal_city"
			soundparameter = { "Tier" = 0 }
		}
	}

	levy = 25
	max_garrison = 100
	garrison_reinforcement_factor = 0.01

	province_modifier = {
		# A colony occupies a slot a real holding would have filled.
		building_slot_add = -1
	}

	county_modifier = {
		# Still a rough place to travel through, though far better than the wilderness it replaced.
		travel_danger = 20

		# ---- Why a colony develops SLOWLY, and why this number is so large ----
		#
		# This was +0.25 — a bonus — on the reasoning that settling empty ground should buy the
		# right to develop it rather than the result, so a compounding factor beat the flat 2-4
		# development colonies used to be handed at founding. That reasoning still stands and the
		# lump sum is still gone. The sign was wrong because it left out where a colony's
		# development actually comes from.
		#
		# CK3 spreads development between adjacent counties: every month a county gains
		# `(highest neighbouring development - its own) * DEVELOPMENT_SPREAD_NEIGHBOUR_FACTOR`,
		# which is 0.1 in 00_defines.txt. A colony is the maximum case of that formula in both
		# terms — it starts at zero development, and it is by definition adjacent to the realm that
		# colonised it, because colonisation requires a bordering county. Claim wilderness next to
		# a 30-development capital and the new colony draws 3.0 a month from the neighbour alone,
		# against the 0.20-0.30 the buildings below contribute. The neighbour is worth ten to
		# fifteen times everything else in this file put together.
		#
		# So the colony's development was never really being earned. It was the capital's
		# development, arriving through a border, and the +0.25 was multiplying it.
		#
		# ---- Why not just switch the spread off ----
		#
		# `DEVELOPMENT_SPREAD_NEIGHBOUR_FACTOR = 0` in a defines override is the one-line version,
		# and it is what to reach for if this ever needs to be harsher. It is not the default here
		# because the spread is a good mechanic that is only wrong in this one case: land near a
		# rich neighbour genuinely should develop faster, and turning that off changes every county
		# in the world to fix a problem that exists on colonies. Doing it as a modifier also means
		# it expires on its own — the building goes when the colony is promoted, and the malus goes
		# with it, with no script and nothing to reconcile. Same reason the upkeep lives on
		# `character_modifier` below rather than on a counter.
		#
		# ---- The floor this must not cross ----
		#
		# can_promote_colony_trigger requires `development_level >= x`. At -1.0 a colony's
		# development freezes, it can never clear that bar, and the whole system deadlocks with no
		# error — the promote button simply never enables. Anything tuned here has to leave real
		# growth behind, which is also why the flat `development_growth` on the two tiers below is
		# kept rather than removed: those are the floor that lets a colony founded out in deep
		# wilderness, with no developed neighbour to draw on at all, still reach promotion.
		#
		# The malus is steepest here and eases at each tier (-0.75, -0.50, -0.25). A colony with
		# nobody in it yet cannot absorb its neighbour's prosperity; one that is nearly a real
		# holding increasingly can. That is the same ramp the levies and the tax malus already use.
		development_growth_factor = -0.95
	}

	character_modifier = {
		# THE upkeep. One copy per colony held, maintained by the engine. See the note above.
		domain_tax_mult = -0.06
	}

	type_icon = "icon_building_outposts.dds"
	ai_value = { base = 100 }
}

settlement_02 = {
	construction_time = slow_construction_time
	cost = { gold = settlement_upgrade_cost }

	can_construct = { wilderness_can_grow_settlement_trigger = yes }
	can_construct_potential = { wilderness_can_grow_settlement_trigger = yes }

	next_building = settlement_03

	asset = {
		type = pdxmesh
		names = { "building_western_tribal_01_a_mesh" }
		illustration = "gfx/interface/illustrations/holding_types/tribe_grass.dds"
		soundeffect = {
			soundeffect = "event:/SFX/Ambience/3DMapEmitters/Holdings/City/tribal_city"
			soundparameter = { "Tier" = 1 }
		}
	}

	levy = 75
	max_garrison = 200
	garrison_reinforcement_factor = 0.02

	province_modifier = {
		building_slot_add = -1
	}

	county_modifier = {
		travel_danger = 10

		# factor is how much of the neighbour's
		# development it can absorb. They pull in opposite directions on purpose — see the long
		# note on settlement_01. Note the factor scales the flat add too, so this is really 0.10 a
		# month of guaranteed local growth, which is the point: it does not depend on there being
		# a rich neighbour.
		development_growth_factor = -0.95
	}

	character_modifier = {
		# The colony is paying for more of itself now.
		domain_tax_mult = -0.03
	}

	type_icon = "icon_building_outposts.dds"
	ai_value = { base = 120 }
}

settlement_03 = {
	construction_time = very_slow_construction_time
	cost = { gold = settlement_upgrade_cost }

	can_construct = { wilderness_can_grow_settlement_trigger = yes }
	can_construct_potential = { wilderness_can_grow_settlement_trigger = yes }

	# No next_building: the step past this one is not a building at all but a change of holding
	# type, which is what the promote interactions in 00_colonization_interactions.txt do.

	asset = {
		type = pdxmesh
		names = { "building_western_tribal_01_a_mesh" }
		illustration = "gfx/interface/illustrations/holding_types/tribe_grass.dds"
		soundeffect = {
			soundeffect = "event:/SFX/Ambience/3DMapEmitters/Holdings/City/tribal_city"
			soundparameter = { "Tier" = 2 }
		}
	}

	levy = 150
	max_garrison = 300
	garrison_reinforcement_factor = 0.03

	county_modifier = {
		# The last tier, and the shallowest malus. A colony this far along has people in it, and
		# letting the neighbour's prosperity reach them is no longer a windfall for empty ground —
		# it is what happens to a real place next to a rich one. This is also the tier promotion
		# is gated on (`has_building = settlement_03`), so it is the tier that has to be able to
		# push a county to development 3 without help.
		development_growth_factor = -0.95
	}

	character_modifier = {
	}

	type_icon = "icon_building_outposts.dds"
	ai_value = { base = 140 }
}
