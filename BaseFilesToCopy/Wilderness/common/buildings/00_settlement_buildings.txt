colony_office_03 = {
	construction_time = standard_construction_time
	cost = { gold = settlement_cost_tier_4 }

	can_construct_potential = {
		has_holding_type = settlement_holding
	}
	can_construct = {
		has_holding_type = settlement_holding
	}

	province_modifier = {
		monthly_income = 1.5
	}

	# ---- Both numbers were outside anything vanilla ships, and this is the one building the
	#      whole promotion clock runs on ----
	#
	# `development_growth` is the ONLY flat growth anywhere in this file set, so the office chain
	# is not one contributor among many — it is the entire numerator. Everything else in the
	# system, this mod's modifiers and the Wilds situation alike, is a factor multiplying it.
	# That makes these two lines the colony's speed, and they were set as if they were flavour.
	#
	# The flat was 1.5. Vanilla's largest building `development_growth` is 0.5, on one special
	# building, and its named band tops out at `very_large_development_growth_gain = 0.5`
	# (00_basic_values.txt); an ordinary building's base is 0.02-0.04. 1.5 was three times the
	# ceiling and better than half a stewardship-15 steward running Increase Development
	# (0.1 + 15 x 0.175 = 2.725) — permanently, on one county, from one building.
	#
	# The factor was 0.80. Vanilla's largest building `development_growth_factor` is 0.4, on the
	# top tier of the city chain.
	#
	# 0.5 and 0.40 are those two ceilings exactly. The colony still finishes — the gate is
	# development_level >= 4 in can_promote_colony_trigger, which is 1000 progress against
	# LEVELS_DEVELOPMENT — it finishes on a clock a player can watch rather than one that
	# resolves while they are looking at something else.
	county_modifier = {
		development_growth_factor = 0.40
		development_growth = 0.5
	}

	type_icon = "icon_structure_ghana_palace.dds"

	ai_value = {
		base = 1800
		modifier = {
			add = 700
			scope:holder ?= { is_colonist_trigger = yes }
		}
	}
}

# =====================================================================================
#  Tribal colonies: Chieftain's Hall, Hunting Camps, Kin Muster
# =====================================================================================
#
# A colony held for a tribe (colony_is_tribal_trigger — a tribal holder, or a colonist whose liege
# is tribal) builds these instead of the office and the timber works. Two reasons, and the first is
# the one that mattered in play:
#
#   * Price. The office chain is 1150 gold and is also the colony's only flat development_growth,
#     so it IS the promotion clock (see the note on colony_office_03). A chief cannot pay that, so a
#     tribal colony simply stalled. These are priced like vanilla's own tribal buildings — mostly
#     prestige, a token of gold — see colony_hall_* in 00_colonization_values.txt.
#   * Fit. A tribe does not run its frontier through scribes and a chancery. The same growth comes
#     from a chief's hall, the hunters who feed the first winters, and the kin who follow.
#
# ---- The hall carries the office's development numbers exactly ----
#
# 0.1/0.3/0.5 flat and 0.15/0.30/0.40 factor, identical to the office. The flat figure is the
# numerator of the whole promotion clock, so matching it keeps one clock for every colony; what the
# hall gives up is gold (poor-tax band, not the office's 1.5) and what it gets back is prestige and
# control. The extra speed a tribe needs — its neighbours are thinly developed — comes from the kin
# muster's FACTOR, never from a second flat source.
#
# ---- The split lives on the entry tier only ----
#
# The government test and the mutual exclusion sit on each chain's _01. Upgrades ask only for the
# holding: a chain a ruler has started is one his successor can finish, whatever government the
# successor keeps. Gating the upgrades too would strand a colony at hall_01 the day its chief
# adopted feudalism — with the office also shut to it, because the hall already fills that role.
# The has_building_or_higher lines are what stop a colony that changed hands from stacking both
# development chains.
