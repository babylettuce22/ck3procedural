# The wilderness holding's primary building, and the six obstacles that sit on top of it.
#
# ---- The blocker pattern ----
#
# Each obstacle is a PAIR: `x` is the obstacle itself, `x_cleared` is the work of removing it.
# `next_building` makes the second an upgrade of the first, so clearing a wolf den is, mechanically,
# constructing over one — which means it costs time and gold, shows in the build queue, and can be
# interrupted by a siege, all without a line of script.
#
# ONLY the `x` half carries `flag = obstacle`. That flag is the whole interface: nothing downstream
# knows the names below, it only ever asks `has_building_with_flag = { flag = obstacle count = 0 }`.
# Adding a seventh blocker therefore needs no change anywhere else in this system.
#
# ---- Two flags, because "in the way" and "a pioneer can chop it down" are different ----
#
# The six natural blockers carry `clearable_obstacle` as well. `obstacle` still means what it always
# meant — this is in the way, promotion waits — and every promotion gate still asks only for that.
# `clearable_obstacle` means something narrower: a pioneer or the Warden's council task can be sent
# at this and will remove it.
#
# They were one flag until the ruins ladder became a blocker, and that is what forced them apart.
# Ruins are in the way, so they must count for `obstacle`; but
# `wilderness_clear_hardest_obstacle_effect` removes things BY NAME from a fixed ladder, so a pioneer
# sent at a county whose only blocker was a ruin would have found nothing to do and reported nothing
# — a silent no-op, chosen by a trigger that said there was work there. And adding the ruins rungs to
# that ladder is worse, not better: it would strip the building without going through
# `ruins_cleared`'s `on_complete`, so the discovery event that hangs off it would be skipped in
# exactly the case somebody went looking.
#
# So the pioneer's targeting asks `clearable_obstacle` and the ladder stays as it is. The invariant
# that buys is worth stating: **every removal of a ruin goes through `ruins_cleared.on_complete`**,
# which is the only reason an event can be hung there and trusted to fire.
#
# A further NATURAL blocker wants both flags. Something that blocks but cannot be chopped — a
# flooded ford, a curse — wants `obstacle` alone, like the ruins rungs.
#
# The `x_cleared` half must NOT carry it, and this was originally wrong. When both halves had the
# flag the count could never reach zero — upgrading a blocker replaced an obstacle with another
# obstacle — so `county_obstacles_cleared_trigger` was false forever and no colony could be
# promoted. The flag means "this is in the way"; ground somebody has cleared is not in the way.
#
# The cleared building stays rather than deleting itself, which is the other way to fix this (AGOT
# removes it in on_complete). Keeping it means the county carries a visible record of what was done
# to it, and gives each cleared blocker somewhere to hang a reward later — drained marsh is good
# farmland, felled forest is timber — which a building that erases itself cannot do.
#
# The `x` half is never constructible (`can_construct = always no`) — it is placed by script when
# the county is seeded. The `x_cleared` half is constructible by anyone EXCEPT the wilderness dummy,
# whose AI is off anyway; the guard is there so the build menu of an unclaimed county is empty
# rather than merely unusable.
#
# ---- Why the maluses land where they do ----
#
# `county_modifier` for anything about the place being hard to live in, because those stack across
# every province in the county and a county is what colonisation claims. `province_modifier` for
# anything about the specific ground. `character_modifier` only where the obstacle should follow the
# holder off the county — disease does, wolves do not.


# =====================================================================================
#  The wilderness itself
# =====================================================================================

wilderness_01 = {
	# Placed by script when the generator seeds a county, never built.
	can_construct = { always = no }
	can_construct_potential = { always = no }

	# A lone tree where a holding would be. Not a building model: an empty county should read as
	# empty at map zoom, and the terrain and foliage the generator paints are the real visual.
	#
	# One generic asset and no `graphical_regions` filter, deliberately. CompatibilityWriter
	# re-declares vanilla's graphical regions against generated provinces by splitting them evenly
	# (`i % graphical.Count`), so region membership on our maps has no relation to climate — a
	# per-region asset would scatter desert huts through tundra. Filter on nothing, or on terrain.
	asset = {
		type = pdxmesh
		names = {
			"tree_pine_single_01_a_mesh"
			"tree_pine_single_01_b_mesh"
			"tree_pine_single_01_c_mesh"
		}
		illustration = "gfx/interface/illustrations/holding_types/tribe_grass.dds"
	}

	county_modifier = {
		# -1.0 is total: no tax, no levy, no garrison, and development that cannot creep upward on
		# its own. A wilderness county must be worth exactly nothing until somebody settles it,
		# or the AI will hoard them as free income.
		tax_mult = -1.0
		levy_size = -1.0
		garrison_size = -1.0
		development_growth_factor = -1.0

		# The one number that makes the map feel different rather than just look emptier. Crossing
		# the wilderness is dangerous, so an army or a travelling character pays for the shortcut.
		travel_danger = 30
	}

	province_modifier = {
		fort_level = -1
	}

	levy = 0
	max_garrison = 0

	type_icon = "icon_building_royal_forest.dds"

	# Placed by script; the AI must never consider it.
	ai_value = { base = -100 }
}

# =====================================================================================
#  Ruins — a three-rung ladder, and the only obstacle nobody can simply chop down
# =====================================================================================
#
# Somebody built here before, and whatever happened to them is the point.
#
#     ruins_large  ->  ruins_small  ->  ruins_cleared  ->  (nothing)
#
# Three rungs rather than the two-step pair the natural blockers use, because the middle rung is
# worth something on its own. Clearing a wolf den is one job with one outcome; taking a ruined town
# apart is years of work in which the place is steadily less of a problem well before it is gone.
# `ruins_small` is that halfway state: the same obstacle, most of the malus lifted, one job left.
#
# ---- Which rung a ruin starts on ----
#
# Both `ruins_large` and `ruins_small` are spawn states, and neither is constructible from nothing.
# The Ruins file set is what chooses, in gen_ruins_place_effect, and it is an even coin: a county
# seeded at generation and a county that falls in play are equally likely to come out on either rung.
#
# So the rung reads as what was STANDING here rather than as how long ago it emptied — a walled town
# leaves a big ruin whenever it fell, a handful of steadings leaves foundations. The ladder below
# then turns that into how much work is left: two construction jobs from large, one from small.
#
# ---- The flags ----
#
# `ruins_large` and `ruins_small` both carry `flag = obstacle` and NOT `flag = clearable_obstacle`,
# and that pair of facts is the whole of what makes ruins different from a wolf den:
#
#   * `obstacle` means a county with ANY rung standing cannot be promoted out of a colony. That is
#     the point of the flag here — the discovery event talks about silence and no signs of life, and
#     none of that survives a bustling town having been raised around the stones first. BOTH rungs
#     carry it, because a half-cleared ruin is still a ruin.
#   * NOT `clearable_obstacle` means a pioneer cannot be sent at either, and the Warden's council
#     task will not touch them, because `wilderness_clear_hardest_obstacle_effect` clears by NAME
#     from a fixed ladder of six and no rung of this one is in it.
#
# That leaves exactly one route down the ladder — somebody building the next rung and waiting — and
# it is the invariant the whole system rests on: **`ruins_cleared` is the only way for a ruin to
# leave a province, and completing it fires the event.** Put any rung into the clearing ladder and a
# pioneer strips it silently, taking the discovery with it.
#
# `ruins_cleared` removes itself on completion like the other cleared halves, so a finished county
# carries no permanent building slot for having once been a ruin.

ruins_large = {
	# Placed by script, never built. Nothing upgrades INTO this rung.
	can_construct = { always = no }
	can_construct_potential = { always = no }

	county_modifier = {
		# Peasants will not settle next to it and will not go near it after dark.
		county_opinion_add = -10
		development_growth_factor = -0.20
		travel_danger = 40
	}

	province_modifier = {
		# Dressed stone already cut and lying where it fell. Robbing it out is cheaper than
		# quarrying — which is exactly why nobody has ever bothered to look underneath.
		build_gold_cost = -0.15
	}

	next_building = ruins_small
	type_icon = "icon_megalith.dds"

	# In the way, but not something a work party can be pointed at — see the two-flag note at the
	# top of this file. `clearable_obstacle` is deliberately absent.
	flag = obstacle
	ai_value = { base = -100 }
}

# The middle rung. Reached two ways: built as the upgrade out of ruins_large, or placed directly by
# script on a ruin meant to start old.
#
# CONSTRUCTIBLE, unlike ruins_large, and that is what makes the ladder work at all: `next_building`
# still honours the target's own can_construct, so a rung nothing may build is a rung nothing may
# climb to. `add_building` ignores can_construct entirely, so the script-placed route is unaffected
# by the guard below.
#
# `has_building = ruins_large` is what stops it being offered as a fresh build on a province with no
# ruin on it. During the upgrade the old building is still standing, so the test is true exactly
# when it should be and false everywhere else.
ruins_small = {
	construction_time = slow_construction_time
	cost = { gold = wilderness_clear_cost_dear }

	can_construct = {
		wilderness_can_clear_obstacle_trigger = yes
		has_building = ruins_large
	}
	can_construct_potential = {
		wilderness_can_clear_obstacle_trigger = yes
		has_building = ruins_large
	}

	# Most of the malus lifted, which is the reason this rung exists rather than being folded into
	# one long job to nothing. The walls are down and carted off; what is left is foundations, a
	# cellar or two, and a reputation.
	county_modifier = {
		county_opinion_add = -3
		development_growth_factor = -0.05
		travel_danger = 10
	}

	province_modifier = {
		build_gold_cost = -0.05
	}

	next_building = ruins_cleared
	type_icon = "icon_megalith.dds"

	# Still in the way. Promotion still waits.
	flag = obstacle

	ai_value = {
		base = 150
		modifier = {
			add = 1000
			scope:holder ?= { is_colonist_trigger = yes }
		}
	}
}

ruins_cleared = {
	construction_time = standard_construction_time
	cost = { gold = wilderness_clear_cost_standard }

	# No `has_building` guard here, unlike ruins_small above, and the asymmetry is deliberate: this
	# rung is only ever reachable as ruins_small's next_building — no other building's ladder ends
	# at it — so the upgrade path is already the only path and a second guard would say nothing.
	can_construct = { wilderness_can_clear_obstacle_trigger = yes }
	can_construct_potential = { wilderness_can_clear_obstacle_trigger = yes }

	type_icon = "icon_megalith.dds"
	ai_value = {
			base = 200
			modifier = {
				add = 1000
				scope:holder ?= { is_colonist_trigger = yes }
			}
		}
	on_complete = {
		# The discovery. Modelled on the wonder capstone in Core/events/gen_wonder_events.txt, and
		# for the same reason: what the player did is the same in every case, so one static event
		# serves every ruin on every map.
		#
		# Fires HERE and only here, at the bottom of the ladder, however the county reached it —
		# whether it started large and was worked down over two construction jobs, or started small
		# and took one. What was under the stones does not depend on how big the pile was.
		#
		# on_complete runs in PROVINCE scope. Both scopes are saved HERE rather than rediscovered in
		# the event, because the event is a static file that knows nothing about which ruin fired it
		# — being able to name the county is the whole reason it can say more than "it".
		#
		# No toast beside it. The obstacle toast that used to be here was announcing the same act a
		# second time, and an event window plus a toast for one thing reads as a double-fire.
		#
		# `?=` on the holder, not `=`. A wilderness county is held by a dummy that cannot build, so
		# in practice somebody real is always here — but a scope that resolves to nothing would take
		# the whole on_complete down with it, including the remove_building below.
		save_scope_as = gen_ruins_site
		county ?= { save_scope_as = gen_ruins_county }
		county.holder ?= { trigger_event = wilderness_ruins.0001 }

		remove_building = ruins_cleared
	}
}
