# The wilderness holding's primary building, and the six obstacles that sit on top of it.
#
# ---- The blocker pattern ----
#
# Each obstacle is a PAIR: `x` is the obstacle itself, `x_cleared` is the work of removing it.
# `next_building` makes the second an upgrade of the first, so clearing a wolf den is, mechanically,
# constructing over one — which means it costs time and gold, shows in the build queue, and can be
# interrupted by a siege, all without a line of script.
#
# ONLY the `x` half carries `flag = obstacle`. That flag is the whole interface: nothing downstream
# knows the names below, it only ever asks `has_building_with_flag = { flag = obstacle count = 0 }`.
# Adding a seventh blocker therefore needs no change anywhere else in this system.
#
# The `x_cleared` half must NOT carry it, and this was originally wrong. When both halves had the
# flag the count could never reach zero — upgrading a blocker replaced an obstacle with another
# obstacle — so `county_obstacles_cleared_trigger` was false forever and no colony could be
# promoted. The flag means "this is in the way"; ground somebody has cleared is not in the way.
#
# The cleared building stays rather than deleting itself, which is the other way to fix this (AGOT
# removes it in on_complete). Keeping it means the county carries a visible record of what was done
# to it, and gives each cleared blocker somewhere to hang a reward later — drained marsh is good
# farmland, felled forest is timber — which a building that erases itself cannot do.
#
# The `x` half is never constructible (`can_construct = always no`) — it is placed by script when
# the county is seeded. The `x_cleared` half is constructible by anyone EXCEPT the wilderness dummy,
# whose AI is off anyway; the guard is there so the build menu of an unclaimed county is empty
# rather than merely unusable.
#
# ---- Why the maluses land where they do ----
#
# `county_modifier` for anything about the place being hard to live in, because those stack across
# every province in the county and a county is what colonisation claims. `province_modifier` for
# anything about the specific ground. `character_modifier` only where the obstacle should follow the
# holder off the county — disease does, wolves do not.


# =====================================================================================
#  The wilderness itself
# =====================================================================================

wilderness_01 = {
	# Placed by script when the generator seeds a county, never built.
	can_construct = { always = no }
	can_construct_potential = { always = no }

	# A lone tree where a holding would be. Not a building model: an empty county should read as
	# empty at map zoom, and the terrain and foliage the generator paints are the real visual.
	#
	# One generic asset and no `graphical_regions` filter, deliberately. CompatibilityWriter
	# re-declares vanilla's graphical regions against generated provinces by splitting them evenly
	# (`i % graphical.Count`), so region membership on our maps has no relation to climate — a
	# per-region asset would scatter desert huts through tundra. Filter on nothing, or on terrain.
	asset = {
		type = pdxmesh
		names = {
			"tree_pine_single_01_a_mesh"
			"tree_pine_single_01_b_mesh"
			"tree_pine_single_01_c_mesh"
		}
		illustration = "gfx/interface/illustrations/holding_types/tribe_grass.dds"
	}

	county_modifier = {
		# -1.0 is total: no tax, no levy, no garrison, and development that cannot creep upward on
		# its own. A wilderness county must be worth exactly nothing until somebody settles it,
		# or the AI will hoard them as free income.
		tax_mult = -1.0
		levy_size = -1.0
		garrison_size = -1.0
		development_growth_factor = -1.0

		# The one number that makes the map feel different rather than just look emptier. Crossing
		# the wilderness is dangerous, so an army or a travelling character pays for the shortcut.
		travel_danger = 30
	}

	province_modifier = {
		fort_level = -1
	}

	levy = 0
	max_garrison = 0

	type_icon = "icon_building_royal_forest.dds"

	# Placed by script; the AI must never consider it.
	ai_value = { base = -100 }
}


# =====================================================================================
#  Obstacles
# =====================================================================================

# --- Beasts: wolves, bears, big cats. Forest, taiga and hill country. ---

beast_dens = {
	can_construct = { always = no }
	can_construct_potential = { always = no }

	county_modifier = {
		development_growth_factor = -0.20
		supply_limit_mult = -0.30
		tax_mult = -0.20
		garrison_size = -0.15
		levy_size = -0.10
		travel_danger = 50
	}

	next_building = beast_dens_cleared
	type_icon = "icon_building_hunting_grounds.dds"
	flag = obstacle
	ai_value = { base = -100 }
}

beast_dens_cleared = {
	construction_time = quick_construction_time
	cost = { gold = wilderness_clear_cost_cheap }

	can_construct = { wilderness_can_clear_obstacle_trigger = yes }
	can_construct_potential = { wilderness_can_clear_obstacle_trigger = yes }

	type_icon = "icon_building_hunting_grounds.dds"
	ai_value = {
			base = 200
			modifier = {
				add = 1000
				scope:holder ?= { is_colonist_trigger = yes }
			}
		}
	on_complete = {
		wilderness_obstacle_cleared_effect = yes
		remove_building = beast_dens_cleared  # <-- Deletes the building and frees the slot
	}
}
