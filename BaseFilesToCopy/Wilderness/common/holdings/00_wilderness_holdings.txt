# The two holdings colonisation moves a county between.
#
#     wilderness_holding  ->  settlement_holding  ->  castle / city / church / tribal
#
# ---- CK3 1.21: a holding is its primary district ----
#
# Since 1.21 (Silk and Silver free update) a holding no longer lists a primary building and its
# buildings itself. It names a `primary_district` (type = holding), and that district carries the
# primary building and the building list — see common/districts/00_wilderness_districts.txt. The
# lists that used to be here, and why the blockers appear in both, now live there.
#
# Neither holding lists any other `districts`: nothing but the holding district can be built on
# unsettled ground or in a colony, which is what the old single building list meant.

wilderness_holding = {
	primary_district = wilderness_holding_district

	# The dummy holder is immortal and blocks inheritance anyway, but saying it here means a
	# wilderness barony cannot leak into a real succession even if the dummy is somehow removed.
	can_be_inherited = no
}

settlement_holding = {
	primary_district = settlement_holding_district

	# ---- There is no `can_be_built` here, and there cannot be ----
	#
	# CK3 holdings accept a fixed set of fields (since 1.21: primary_district, districts,
	# can_be_inherited, counts_toward_domain_limit_if_disabled, required_heir_government_types,
	# parameters). `can_be_built` is not one of them — it appears nowhere in the game files — so a
	# block by that name is read, ignored, and reported by nothing except ck3-tiger.
	#
	# What it was reaching for is already true by a different route. No government lists
	# settlement_holding in `required_county_holdings`, which is the engine's precondition for
	# raising anything new in a county — so a county whose only holding is a colony never satisfies
	# it and the empty baronies stay shut until the colony is promoted. See the long note in
	# 00_wilderness_government.txt, which is where that rule is explained and where it belongs.
}
