# The government the dummy holder rules under. It exists to make every wilderness county
# uninteractable rather than to be played: no council, no court, no AI, no income.
#
# `government_is_wilderness` is the flag the rest of this system tests. Everything asks the flag
# rather than `has_government = wilderness_government`, so a second uninteractable government can be
# added later without touching any trigger.

wilderness_government = {
	government_rules = {
		court_generate_spouses = no
		council = no
		create_cadet_branches = no
		rulers_should_have_dynasty = no
		dynasty_named_realms = no
		allow_accolades = no
		legitimacy = no
	}

	primary_holding = wilderness_holding
	valid_holdings = { wilderness_holding }

	# `wilderness_holding` is deliberately ABSENT from this list.
	#
	# This field is the engine's precondition for building anything new in a county: per
	# _governments.info, "which holdings must be present in a county before more of an existing type
	# or others can be built". Listing wilderness_holding here made a wilderness county satisfy its
	# own precondition, so the empty baronies around the capital offered "build new holding" —
	# letting a county be developed from the outside without ever being settled, which skips the
	# entire mechanic.
	#
	# Leaving it out means a county whose only holding is wilderness_holding never meets the
	# requirement, and the button is gone until something real stands there. The same is true of a
	# colony: settlement_holding appears in no government's list, vanilla's included, so the empty
	# baronies stay shut until the colony is promoted to a castle, city or temple.
	#
	# Castle, city and church stay listed so a county the player has *finished* colonising still
	# satisfies the requirement while the title is mid-transfer.
	required_county_holdings = { castle_holding city_holding church_holding }

	character_modifier = {
		# The wilderness never raises anyone and never pays anyone.
		knight_limit = -100
		monthly_income_mult = -10

		# Its counties never revolt: a peasant rising in an empty county is a war the player cannot
		# reach and the AI cannot resolve.
		county_opinion_add = 100
	}

	ai = {
		use_lifestyle = no
		arrange_marriage = no
		use_goals = no
		use_decisions = no
		use_scripted_guis = no
		use_legends = no
		perform_religious_reformation = no
	}

	flags = {
		government_is_wilderness
		government_is_uninteractable
		government_is_settled
	}

	# Near-black, so realm-coloured map modes read wilderness as absence rather than as a power.
	color = hsv{ 1 0.0 0.1 }
}
