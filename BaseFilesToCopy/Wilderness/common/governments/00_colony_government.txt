# The government of a ruler whose only land is a colony.

colony_government = {
	government_rules = {
		# A colonist keeps a name and passes it on. Everything else that makes a court is stripped:
		create_cadet_branches = no
		rulers_should_have_dynasty = yes
		dynasty_named_realms = no
		legitimacy = no
		allow_accolades = no
		court_generate_spouses = no

		# ---- Why this is `yes` when everything around it is `no` ----
		#
		# It was `no`, and `no` did not mean what it looks like it means. It does not hide the
		# council; it leaves the ruler with no positions and no tasks while window_council.gui goes
		# on drawing its six hardcoded frames — so a colonist got a Council button that opened onto
		# six blank panels, each reading "Click to appoint a" with no office named after it and
		# nothing behind the portrait to click.
		#
		# A frontier post does have a handful of people the colonist actually listens to, so the
		# answer was to give them offices rather than to keep pretending the screen is not there.
		# See common/council_positions/00_colony_council_positions.txt for the five, and
		# Emit/GuiWriter.cs for the window slots they appear in — vanilla's own five are hidden for
		# a colonist in the same patch. Vanilla's spouse seat is deliberately left standing.
		council = yes
	}

	court_generate_commanders = no

	# ---- A domicile for the colony: CURRENTLY A CONTROL, NOT THE ANSWER ----
	#
	# `domicile_type` is a TOP-LEVEL field, not a government_rules entry — same trap as
	# court_generate_commanders above and royal_court in 00_wilderness_government.txt. Inside the
	# rules block it is silently ignored and the colonist simply has no domicile, with nothing logged.
	#
	# ---- What has been measured ----
	#
	# `domicile_type = camp` was shipped first, on the reasoning that nothing in vanilla's camp type
	# objects to a landed owner: it carries no `allowed_for_character` at all (only `estate` and
	# `east_asian_estate` do, and theirs want a noble family title), and `camp_main_01` has no
	# can_construct. Colonists got no camp. Placing it explicitly after the government change did not
	# help either, and the debug readout on wilderness_colonization.9998 then answered why:
	# `has_domicile = no`. The engine never made one. It is creation that fails, not placement and
	# not the HUD.
	#
	# ---- Why `yurt`, of all things ----
	#
	# It is the one type that isolates the variable, and it is temporary. Line up what vanilla ships:
	#
	#   estate, east_asian_estate   landed owner, no resource, GATED on a noble family title  — works
	#   japanese_manor              landed owner, no resource                                 — works
	#   yurt                        landed owner (nomads hold counties), no gate, `herd`       — works
	#   camp                        landless owner, no gate, `provisions`                      — ours failed
	#
	# `yurt` differs from `camp` in exactly two things that could matter: it manages `herd` where the
	# camp manages `provisions`, and it carries `move_with_realm_capital = yes` so the engine places
	# it without being asked. If a colonist gets a yurt, then a gateless domicile on a landed
	# government is fine and `provisions` — an adventurer resource — is what the engine is refusing.
	# If a colonist gets no yurt either, the government field cannot grant a landed character a
	# domicile at all here, and a type of our own would fail the same way.
	#
	# Do not ship this. A colonist with a yurt gets a herd meter, which is worse than the provisions
	# meter this was meant to avoid.
	#
	# ---- What replaces it once the control has answered ----
	#
	# If provisions is the culprit: a `colony_camp` type of our own with `provisions = no` and
	# `move_with_realm_capital = yes`, which also retires the explicit placement in
	# colonization_make_colonist_effect. That owes one thing — domicile buildings are bound to a type
	# BY KEY, and every entry in vanilla's 00_camp_buildings.txt carries
	# `allowed_domicile_types = { camp }` — so it needs a small building set of its own rather than
	# an override of twenty-odd vanilla files. That is the reason `camp` was reached for first.
	#
	# Promotion needs no cleanup whichever way this lands: promote_colony_effect moves the ruler off
	# colony_government, the new government declares no domicile_type, and the domicile stops
	# existing on its own.
	domicile_type = yurt

	primary_holding = settlement_holding
	valid_holdings = { settlement_holding }

	# Empty baronies around a colony stay shut until it is promoted.
	required_county_holdings = { castle_holding city_holding church_holding }

	vassal_contract_group = colony_vassal

	can_get_government = {
		any_held_title = {
			tier = tier_county
			is_colony_county_trigger = yes
		}
		NOT = {
			any_held_title = {
				tier = tier_county
				is_settled_county_trigger = yes
			}
		}
	}

	character_modifier = {
		# Nobody rides out to a frontier post looking for service.
		knight_limit = -2

		# Hard living, honestly priced: it kills slowly and it wears on a man.
		health = -0.1
		stress_gain_mult = 0.15

		# Quick control recovery on virgin ground
		monthly_county_control_growth_add = 0.05

		# --- Frontier Palisade Resilience ---
		garrison_size = 0.35                 # Every settler grabs a spear when attacked
		defender_holding_advantage = 5       # Deep knowledge of the local trails and woods
		hostile_county_attrition  = 0.25 # Invaders suffer in the untamed brush
	}

	# Respects high-prestige lieges; resents weak/absentee lords
	opinion_of_liege = {
		value = -5
		scope:liege = {
			add = {
				value = prestige_level
				multiply = 2
			}
		}
	}
	opinion_of_liege_desc = colony_opinion_of_liege_desc

	ai = {
		use_lifestyle = yes
		arrange_marriage = yes
		use_goals = yes
		use_decisions = yes
		use_scripted_guis = yes
		use_legends = no
		perform_religious_reformation = no

	}

	flags = {
		government_is_colony
		government_is_settled
	}

	# Pale earth. Reads as a place on the map rather than as a power on it.
	color = hsv{ 0.09 0.35 0.55 }
}
