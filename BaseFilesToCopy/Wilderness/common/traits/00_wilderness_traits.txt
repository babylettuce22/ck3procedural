# The marker the generated dummy holder carries.
#
# Nothing in this system references a character id: the generator names its dummy whatever it likes,
# and every effect and trigger here finds it by `has_trait = wilderness` instead. That also means a
# second dummy — ruins, an off-limits region — can be added later without any of this knowing.
#
# The values below are not flavour. Each one stops the engine from treating the holder as a person:
# it cannot die, breed, marry, be schemed against, be inherited from, or accumulate anything. A
# dummy that can be murdered leaves every wilderness county on the map ownerless.

wilderness = {
	icon = lifestyle_seasoned_pastor.dds
	shown_in_ruler_designer = no

	flag = can_not_marry

	# Never dies of anything. An heirless dummy dying would scatter its counties to the succession
	# code, which would hand a continent of wilderness to whichever real ruler had the nearest claim.
	immortal = yes
	health = 100
	fertility = -1
	inheritance_blocker = all
	claim_inheritance_blocker = all

	# Holds an unbounded domain, because it holds every unsettled county on the map at once.
	domain_limit = 10000

	# Earns nothing and cannot be preyed on. Without the scheme lines a player can simply murder the
	# wilderness, and without the income lines it quietly becomes the richest character in the world.
	monthly_income_mult = -1
	monthly_prestige_gain_mult = -1
	monthly_piety_gain_mult = -1
	enemy_hostile_scheme_phase_duration_add = 100
	enemy_hostile_scheme_success_chance_max_add = -100
	enemy_hostile_scheme_success_chance_growth_add = -100

	# The AI must never take an interest in anything.
	ai_sociability = -1000
	ai_energy = -1000
	ai_rationality = -1000
	ai_boldness = -1000
	ai_zeal = -1000
	ai_greed = 1000
}
