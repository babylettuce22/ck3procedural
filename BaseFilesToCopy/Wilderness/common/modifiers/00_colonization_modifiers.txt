# =====================================================================================
#  Presence: what changes when the lord is actually there
# =====================================================================================
#
# A colony produces nothing for an absentee lord. That is the rule the whole mechanic rests on, and
# these two modifiers are the one honest way out of it that does not repeal it: you go and live in
# the place.
#
# ---- Why two modifiers rather than one ----
#
# Because the two halves of "the lord is here" land in different places. What the lord gets is
# income, and income is a character modifier. What the COUNTY gets is growth, control and a
# willingness to be governed, and those are county modifiers. One object cannot be both, and
# splitting them is also what lets the county half stay put for a moment while the character half
# comes off — which is what happens on the day he rides home.
#
# They are applied and removed together by colony_refresh_presence_effect and by nothing else. If a
# third thing ever needs to know whether a lord is present, it asks that effect's modifier, not its
# own copy of the location test.
#
# ---- Why the numbers are not simply "cancel the penalty" ----
#
# They cannot be, and the reason is stronger than "the penalty is large".
#
# A government declares `primary_holding` and `valid_holdings`. settlement_holding is neither of
# those for feudal_government — it is the primary_holding of colony_government and of nothing else —
# so a feudal lord holding a colony directly trips the engine's wrong-holding-type rule. That rule
# is not a reduction. Vanilla states it in gui/holding_view_l_english.yml:
#
#     GOVERNMENT_WRONG_HOLDING_TYPE: "...the holding produces no levies or taxes"
#
# No levies and no taxes: the holding's output is ZERO before any modifier in this file is consulted.
# It is engine-side, it is keyed off the government rather than off anything scriptable, and there is
# no escape hatch — holding `parameters` are inert (they exist to be read back by
# has_holding_parameter and nothing else), and no define governs it.
#
# So a multiplier is useless here in both directions, and only ONE of the two halves can be
# substituted:
#
#   Tax  — substitutable. `monthly_income` is a flat add on the CHARACTER, so it never passes through
#          the holding and the zero never touches it. That is why the income below is a flat monthly
#          sum and not a tax multiplier; a multiplier on a colony's tax is a multiplier on nothing.
#
#   Levy — NOT substitutable. Every levy modifier in the game is either a multiplier (`levy_size` and
#          the per-terrain variants) or a vassal-contribution add (`*_levy_contribution_add`, which
#          governs what vassals hand up, not what your own domain raises). There is no additive levy
#          modifier for a holding you hold yourself, so there is no way to put men back into a
#          holding the engine has emptied.
#
# The practical warning, which is the reason this note is this long: a `levy_size` line here LOOKS
# like it works. It validates, it shows in the tooltip, and it changes nothing. One was here — see
# the note in settlement_overseen — and it read as a deliberate, tuned half-measure for as long as
# nobody counted the men. Do not add another.

# Character scope, on the lord standing in his own colony.
overseeing_the_settlement = {
	# There is no `county_modifier_economy_*` in the icon set, whatever its name suggests it ought to
	# contain — ck3-tiger catches the missing file, the game shows a blank square and says nothing.
	icon = county_modifier_control_positive

	# The colony renders something, because there is somebody there to render it to. Deliberately a
	# real sum rather than a token: this is the whole reward for a lord being absent from his court
	# for years, and if it does not beat sitting at home the choice is not a choice.
	#
	# 1.5 until the levy line came out of settlement_overseen. It is carrying both halves of presence
	# now — the colony's coin and the men it cannot raise — because income is the only channel the
	# engine leaves open. That makes it the tuning knob for the whole mechanic: if overseeing feels
	# too weak or too strong, this is the number, and there is no second one to balance it against.
	monthly_income = 2.0

	# He is a lord in a palisade, not a lord in a hall. Nobody is impressed.
	monthly_prestige = -0.5
}

# County scope, on the colony being overseen.
settlement_overseen = {
	icon = county_modifier_development_positive

	# Exactly cancels settlement_01's -0.75. A colony with its lord in it develops like ordinary
	# land; a colony without one does not. That is the cleanest statement of the rule this file
	# exists to express, and it is why the number is a mirror rather than a round figure — if the
	# malus in 00_settlement_buildings.txt is ever retuned, this has to move with it.
	development_growth_factor = 0.15

	monthly_county_control_growth_add = 0.5
	county_opinion_add = 20

	# Work goes faster when whoever is paying for it is watching it.
	build_speed = -0.15

	# ---- There is no levy line here, and there cannot be ----
	#
	# `levy_size = 0.5` was, with the comment "half again of very little is still not much, which is
	# the honest amount". It was half again of nothing. A colony held by a feudal lord raises exactly
	# zero levies — see GOVERNMENT_WRONG_HOLDING_TYPE in the note at the top of this file — so the
	# multiplier had nothing to multiply, and the tooltip said +50% Levy Size on a county that was
	# producing no men at all.
	#
	# Nothing replaces it, because nothing can: there is no additive levy modifier for a holding you
	# hold yourself. Presence is worth coin and it is not worth soldiers, and
	# `overseeing_the_settlement.monthly_income` now carries what this was pretending to be worth.
}
